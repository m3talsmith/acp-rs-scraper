6/// Media URL referenced by a post, before download.
7pub struct MediaRef {
8    /// Page / media URL.
9    pub url: String,
10    /// Media kind (`image`, `video`, `thumbnail`, ...).
11    pub kind: String,
12}
13
14/// Verified (PostMedia-*.js): `post.media` = [{id, type, url, thumb_url, width, height}],
15/// `type == "video"` renders `<video src=url poster=thumb_url>`; other types render `<img>`.
16/// Body (ProseMirror) has no image node in the editor bundle; nodes found: embed
17/// {provider,id,kind} (third-party iframes, not downloaded), postRef, mention.
18/// GUESSED: any other node attr named src/url/href pointing at the site's media hosts.
19pub fn collect_media(v: &Value, out: &mut Vec<MediaRef>) {
20    if let Some(ms) = v.get("media").and_then(Value::as_array) {
21        for m in ms {
22            let kind = m.get("type").and_then(Value::as_str).unwrap_or("image").to_string();
23            if let Some(u) = m.get("url").and_then(Value::as_str) {
24                out.push(MediaRef { url: u.into(), kind: kind.clone() });
25            }
26            if let Some(u) = m.get("thumb_url").and_then(Value::as_str) {
27                out.push(MediaRef { url: u.into(), kind: "thumbnail".into() });
28            }
29        }
30    }
31    for k in ["attachments", "files", "images"] {
32        if let Some(ms) = v.get(k).and_then(Value::as_array) {
33            for m in ms {
34                if let Some(u) = m.get("url").or(m.get("src")).and_then(Value::as_str) {
35                    let kind = m.get("type").and_then(Value::as_str).unwrap_or("file").to_string();
36                    out.push(MediaRef { url: u.into(), kind });
37                }
38            }
39        }
40    }
41    fn walk(n: &Value, out: &mut Vec<MediaRef>) {
42        if let Some(a) = n.get("attrs").and_then(Value::as_object) {
43            for k in ["src", "url", "href"] {
44                if let Some(u) = a.get(k).and_then(Value::as_str)
45                    && MEDIA_HOSTS.iter().any(|h| u.contains(h))
46                {
47                    let kind = n.get("type").and_then(Value::as_str).unwrap_or("file").to_string();
48                    out.push(MediaRef { url: u.into(), kind });
49                }
50            }
51        }
52        for key in ["content", "marks"] {
53            if let Some(kids) = n.get(key).and_then(Value::as_array) { kids.iter().for_each(|k| walk(k, out)); }
54        }
55    }
56    if let Some(b) = v.get("body") {
57        let b = match b.as_str() { Some(r) => serde_json::from_str(r).unwrap_or(Value::Null), None => b.clone() };
58        walk(&b, out);
59    }
60}

63/// A downloaded attachment of a post.
64pub struct Attachment {

66    pub post_id: String,

77/// File extension for a known media MIME type.
78pub fn ext_from_mime(m: &str) -> Option<&'static str> {
79    Some(match m.split(';').next()?.trim() {
80        "image/jpeg" => "jpg",
81        "image/png" => "png",
82        "image/gif" => "gif",
83        "image/webp" => "webp",
84        "image/avif" => "avif",
85        "image/svg+xml" => "svg",
86        "video/mp4" => "mp4",
87        "video/webm" => "webm",
88        "video/quicktime" => "mov",
89        "audio/mpeg" => "mp3",
90        "audio/ogg" => "ogg",
91        "audio/wav" | "audio/x-wav" => "wav",
92        "audio/mp4" => "m4a",
93        "application/pdf" => "pdf",
94        _ => return None,
95    })
96}

98/// MIME type for a file extension (fallback `application/octet-stream`).
99pub fn mime_from_ext(e: &str) -> &'static str {
100    match e {
101        "jpg" | "jpeg" => "image/jpeg",
102        "png" => "image/png",
103        "gif" => "image/gif",
104        "webp" => "image/webp",
105        "avif" => "image/avif",
106        "svg" => "image/svg+xml",
107        "mp4" | "m4v" => "video/mp4",
108        "webm" => "video/webm",
109        "mov" => "video/quicktime",
110        "mp3" => "audio/mpeg",
111        "ogg" | "oga" => "audio/ogg",
112        "wav" => "audio/wav",
113        "m4a" => "audio/mp4",
114        "pdf" => "application/pdf",
115        _ => "application/octet-stream",
116    }
117}

119/// File extension from a URL path, if it has a short alphanumeric one.
120pub fn ext_from_url(u: &str) -> Option<String> {
121    let path = url::Url::parse(u).ok()?.path().to_string();
122    let e = path.rsplit('/').next()?.rsplit_once('.')?.1.to_ascii_lowercase();
123    (e.len() <= 5 && e.chars().all(|c| c.is_ascii_alphanumeric())).then_some(e)
124}

126/// A media URL without its query string or fragment. The site serves media
127/// through presigned links whose `X-Amz-Date`/`X-Amz-Signature` change on every
128/// page load, so the bare path is what identifies the file.
129pub fn canonical_url(u: &str) -> &str {
130    u.split(['?', '#']).next().unwrap_or(u)
131}
132
133/// Stable hex hash of a URL without its query string, used to name downloaded files.
134pub fn url_key(u: &str) -> String {
135    let mut h = DefaultHasher::new();
136    canonical_url(u).hash(&mut h);
137    format!("{:016x}", h.finish())
138}

140/// Download every media URL once (dedupe by URL), skipping failures.
141pub async fn download_all(http: &reqwest::Client, posts: &[Post], dir: &Path) -> Vec<Attachment> {
142    let _ = std::fs::create_dir_all(dir);
143    let mut done: HashMap<String, Option<(PathBuf, String)>> = HashMap::new();
144    let mut out = Vec::new();
145    let mut seen_pair = HashSet::new();
146    for p in posts {
147        for m in &p.media {
148            if !seen_pair.insert((p.post_id.clone(), canonical_url(&m.url).to_string())) { continue; }
149            if !done.contains_key(&m.url) {
150                let r = fetch_one_pub(http, &m.url, dir).await;
151                if let Err(e) = &r { log_error(format_args!("warning: attachment {} skipped: {e:#}", m.url)); }
152                done.insert(m.url.clone(), r.ok());
153            }
154            if let Some(Some((path, mime))) = done.get(&m.url) {
155                out.push(Attachment {
156                    post_id: p.post_id.clone(), url: m.url.clone(), kind: m.kind.clone(),
157                    local_path: path.clone(), mime_type: mime.clone(),
158                });
159            }
160        }
161    }
162    log_info(format_args!("attachments: {} files for {} references", done.values().flatten().count(), out.len()));
163    out
164}

166/// Download one URL into `dir` (named by [`url_key`] + extension), reusing an existing file; returns the path and MIME type.
167pub async fn fetch_one_pub(http: &reqwest::Client, u: &str, dir: &Path) -> Result<(PathBuf, String)> {
168    let key = url_key(u);
169    // Reuse an existing download.
170    if let Ok(rd) = std::fs::read_dir(dir) {
171        for e in rd.flatten() {
172            let n = e.file_name().to_string_lossy().to_string();
173            if n.starts_with(&key) {
174                let ext = n.rsplit_once('.').map(|x| x.1).unwrap_or("");
175                return Ok((e.path(), mime_from_ext(ext).to_string()));
176            }
177        }
178    }
179    let resp = http.get(u)
180        .header("User-Agent", FIREFOX_UA)
181        .header("Accept", "image/avif,image/webp,video/*,audio/*,*/*;q=0.8")
182        .header("Sec-Fetch-Dest", "image")
183        .header("Sec-Fetch-Mode", "no-cors")
184        .header("Referer", "https://acp.us/")
185        .send().await?.error_for_status()?;
186    let ct = resp.headers().get(reqwest::header::CONTENT_TYPE)
187        .and_then(|v| v.to_str().ok()).unwrap_or("").to_string();
188    let ext = ext_from_mime(&ct).map(str::to_string).or_else(|| ext_from_url(u)).unwrap_or_else(|| "bin".into());
189    let mime = if ct.is_empty() { mime_from_ext(&ext).to_string() } else { ct.split(';').next().unwrap_or("").trim().to_string() };
190    let bytes = resp.bytes().await?;
191    let path = dir.join(format!("{key}.{ext}"));
192    std::fs::write(&path, &bytes)?;
193    Ok((path, mime))
194}

196/// Download a nick's profile image to `{attachments}/profiles/{nick}.{ext}`
197/// (replacing any older copy) and return its path.
198pub async fn fetch_avatar(http: &reqwest::Client, u: &str, att_dir: &Path, nick: &str) -> Result<PathBuf> {
199    let dir = att_dir.join(PROFILE_IMAGES_DIR);
200    std::fs::create_dir_all(&dir)?;
201    let tmp = dir.join(format!(".{nick}.download"));
202    std::fs::create_dir_all(&tmp)?;
203    let res = fetch_one_pub(http, u, &tmp).await;
204    let out = match res {
205        Ok((p, _)) => {
206            let ext = p.extension().and_then(|x| x.to_str()).unwrap_or("bin").to_string();
207            for e in std::fs::read_dir(&dir)?.flatten() {
208                let n = e.file_name().to_string_lossy().to_string();
209                if n.rsplit_once('.').is_some_and(|(stem, _)| stem == nick) { let _ = std::fs::remove_file(e.path()); }
210            }
211            let dst = dir.join(format!("{nick}.{ext}"));
212            std::fs::rename(&p, &dst)?;
213            Ok(dst)
214        }
215        Err(e) => Err(e),
216    };
217    let _ = std::fs::remove_dir_all(&tmp);
218    out
219}

221#[cfg(test)]
222mod url_tests {
223    use super::*;
224    #[test]
225    fn presigned_links_share_a_key() {
226        let a = "https://media.acp.us/uploads/x.avif?X-Amz-Date=20261015T104332Z&X-Amz-Signature=aa";
227        let b = "https://media.acp.us/uploads/x.avif?X-Amz-Date=20261016T114716Z&X-Amz-Signature=bb";
228        assert_eq!(canonical_url(a), "https://media.acp.us/uploads/x.avif");
229        assert_eq!(url_key(a), url_key(b));
230        assert_ne!(url_key(a), url_key("https://media.acp.us/uploads/x-thumb.avif?X-Amz-Date=1"));
231    }
