5/// Run whichever `--research-*` flag was given.
6pub async fn run_research(args: &Args) -> Result<()> {
7    let db = home()?.join(DB_REL_PATH);
8    let json_out = std::env::args().any(|a| a == "--format" || a.starts_with("--format=")) && args.formats().contains(&Format::Json);
9    if args.research_list {
10        let c = research_conn(&db)?;
11        let ps = list_projects(&c)?;
12        if json_out {
13            let v: Vec<Value> = ps.iter().map(|p| json!({"name": p.name, "description": p.description, "created_at": p.created_at, "items": p.items})).collect();
14            println!("{}", serde_json::to_string_pretty(&v)?);
15        } else if ps.is_empty() {
16            println!("No research projects yet (--research-create NAME).");
17        } else {
18            println!("{:<30} {:>6}  {:<10}  DESCRIPTION", "PROJECT", "ITEMS", "CREATED");
19            for p in &ps { println!("{:<30} {:>6}  {:<10}  {}", p.name, p.items, p.created_at.get(..10).unwrap_or(""), p.description); }
20        }
21        return Ok(());
22    }
23    if let Some(name) = &args.research_create {
24        let c = research_conn(&db)?;
25        create_project(&c, name, args.research_description.as_deref().unwrap_or(""))?;
26        info!("research project {:?} created", name.trim());
27        return Ok(());
28    }
29    if let Some(name) = &args.research_show {
30        let c = research_conn(&db)?;
31        let p = get_project(&c, name)?.ok_or_else(|| anyhow!("no research project {name:?} (--research-list shows them)"))?;
32        let items = project_items(&c, p.id)?;
33        if json_out {
34            println!("{}", serde_json::to_string_pretty(&research_json(&p, &items, &|i| (!i.local_path.is_empty()).then(|| i.local_path.clone())))?);
35        } else {
36            println!("{} ({} items){}", p.name, items.len(), if p.description.is_empty() { String::new() } else { format!(" — {}", p.description) });
37            for i in &items {
38                println!("  [{}] {:<13} {}  {}", i.id, i.kind.as_str(), i.title, i.r#ref);
39                if !i.note.is_empty() { println!("        note: {}", i.note); }
40            }
41        }
42        return Ok(());
43    }
44    if let Some(name) = &args.research_add {
45        let note = args.note.clone().unwrap_or_default();
46        let mut items = Vec::new();
47        if let Some(ids) = &args.post_ids {
48            let (v, missing) = post_items(&db, &ids.0, None, json!({}), &note)?;
49            for m in &missing { err!("warning: post {m} is not in the database; not added"); }
50            items.extend(v);
51        }
52        if let Some(q) = &args.search {
53            let nicks = args.nick_list()?;
54            let (posts, _, n) = search_db(&db, Some(q), &nicks)?;
55            let ids: Vec<String> = posts.iter().take(n).map(|p| p.post_id.clone()).collect();
56            let (v, _) = post_items(&db, &ids, Some(ItemKind::SearchResult), json!({"query": q, "nicks": nicks}), &note)?;
57            info!("search {q:?}: {} matches", v.len());
58            items.extend(v);
59        }
60        if let Some(f) = &args.file {
61            let p = expand_tilde(f);
62            items.push(file_item(name, &p, !args.no_copy, &note)?);
63        }
64        if let Some(u) = &args.url { items.push(web_item(name, u, &note).await?); }
65        if items.is_empty() && args.post_ids.is_none() && args.search.is_none() {
66            bail!("--research-add needs something to add: --post-ids, --search, --file or --url");
67        }
68        let c = research_conn(&db)?;
69        let (pid, created) = get_or_create(&c, name, args.research_description.as_deref())?;
70        if created { info!("research project {:?} created", name.trim()); }
71        let mut added = 0;
72        for it in &items { if add_item(&c, pid, it)? { added += 1; } }
73        info!("research {:?}: {added} added, {} already there", name.trim(), items.len() - added);
74        return Ok(());
75    }
76    if let Some(name) = &args.research_export {
77        let c = research_conn(&db)?;
78        let p = get_project(&c, name)?.ok_or_else(|| anyhow!("no research project {name:?} (--research-list shows them)"))?;
79        let items = project_items(&c, p.id)?;
80        drop(c);
81        let ids: Vec<String> = items.iter().filter(|i| i.kind.is_post()).map(|i| i.r#ref.clone()).collect();
82        let (posts, mut atts, missing) = posts_by_ids(&db, &ids)?;
83        for m in &missing { err!("warning: post {m} of the project is no longer in the database"); }
84        atts.extend(item_attachments(&items));
85        let mut nicks: Vec<String> = posts.iter().map(|p| p.nick.clone()).collect();
86        nicks.sort();
87        nicks.dedup();
88        let (profiles, hits) = if nicks.is_empty() { (Vec::new(), Vec::new()) } else { load_profiles_mentions(&db, &nicks)? };
89        let mut a = args.clone();
90        a.output = args.export.clone();
91        let stamp = chrono::Local::now().format("%Y%m%d-%H%M%S").to_string();
92        let ename = args.name.clone().unwrap_or_else(|| format!("research-{}-{stamp}", slug(&p.name))).replace(['/', '\\'], "_");
93        info!("research export {:?}: {} items, {} posts, {} files", p.name, items.len(), posts.len(), atts.len());
94        return export_with(&a, &ename, &format!("research: {}", p.name), &posts, &atts, &profiles, &hits,
95            &home()?.join(EXPORTS_REL_PATH), Some((&p, &items)));
96    }
97    Ok(())
98}

100/// `~/x` → `$HOME/x`.
101pub fn expand_tilde(p: &Path) -> PathBuf {
102    match p.to_str().and_then(|s| s.strip_prefix("~/")) {
103        Some(rest) => home().map(|h| h.join(rest)).unwrap_or_else(|_| p.to_path_buf()),
104        None => p.to_path_buf(),
105    }
