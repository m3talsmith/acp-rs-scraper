1//! Research projects: named collections of posts, replies, search results,
2//! local files and web pages, each with an optional note, exported together
3//! (JSON + HTML, or a bundle with the files and page snapshots).
4//!
5//! Stored in the `research_projects` / `research_items` tables of the main
6//! database. Local files are copied to `~/.acp/research/<project>/files/`
7//! (unless `--no-copy`); web pages are fetched once and kept as an HTML
8//! snapshot in `~/.acp/research/<project>/web/` plus their readable text.

15/// What a research item points at.
16#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
17pub enum ItemKind {
18    /// A thread root post (ref = post id).
19    Post,
20    /// A reply (ref = post id).
21    Reply,
22    /// A post or reply added from search results (ref = post id; meta has the query).
23    SearchResult,
24    /// A file on the local drive (ref = its absolute path).
25    LocalFile,
26    /// A web page (ref = its URL).
27    WebUrl,
28}
29
30impl ItemKind {
31    /// Name stored in the database.
32    pub fn as_str(self) -> &'static str {
33        match self { ItemKind::Post => "post", ItemKind::Reply => "reply", ItemKind::SearchResult => "search-result",
34            ItemKind::LocalFile => "local-file", ItemKind::WebUrl => "web-url" }
35    }
36    /// Parse a stored name.
37    pub fn parse(s: &str) -> Option<Self> {
38        Some(match s { "post" => ItemKind::Post, "reply" => ItemKind::Reply, "search-result" => ItemKind::SearchResult,
39            "local-file" => ItemKind::LocalFile, "web-url" => ItemKind::WebUrl, _ => return None })
40    }
41    /// Whether the ref is a post id.
42    pub fn is_post(self) -> bool { matches!(self, ItemKind::Post | ItemKind::Reply | ItemKind::SearchResult) }
43}

112/// Check a project name: 1–80 characters, no control characters.
113pub fn check_name(name: &str) -> Result<String> {
114    let n = name.trim();
115    if n.is_empty() || n.chars().count() > 80 || n.chars().any(char::is_control) {
116        bail!("research project names need 1-80 printable characters (got {name:?})");
117    }
118    Ok(n.to_string())
119}

121/// Create a project; an existing name is an error.
122pub fn create_project(c: &Connection, name: &str, description: &str) -> Result<i64> {
123    let name = check_name(name)?;
124    if project_id(c, &name)?.is_some() { bail!("research project {name:?} already exists"); }
125    c.execute("INSERT INTO research_projects (name, description, created_at) VALUES (?1, ?2, ?3)", params![name, description.trim(), now()])?;
126    Ok(c.last_insert_rowid())
127}

129/// The id of project `name` (case-insensitive).
130pub fn project_id(c: &Connection, name: &str) -> Result<Option<i64>> {
131    if !crate::maintenance::table_exists(c, "research_projects")? { return Ok(None); }
132    Ok(c.query_row("SELECT id FROM research_projects WHERE name = ?1 COLLATE NOCASE", [name.trim()], |r| r.get(0)).optional()?)
133}

135/// The project `name`, created (with `description`) when missing; true when created.
136pub fn get_or_create(c: &Connection, name: &str, description: Option<&str>) -> Result<(i64, bool)> {
137    match project_id(c, name)? {
138        Some(id) => {
139            if let Some(d) = description.filter(|d| !d.trim().is_empty()) {
140                c.execute("UPDATE research_projects SET description = ?1 WHERE id = ?2", params![d.trim(), id])?;
141            }
142            Ok((id, false))
143        }
144        None => Ok((create_project(c, name, description.unwrap_or(""))?, true)),
145    }
146}

148/// Every project, newest first, with item counts.
149pub fn list_projects(c: &Connection) -> Result<Vec<ResearchProject>> {
150    if !crate::maintenance::table_exists(c, "research_projects")? { return Ok(Vec::new()); }
151    let mut st = c.prepare("SELECT p.id, p.name, p.description, p.created_at,
152            (SELECT count(*) FROM research_items i WHERE i.project_id = p.id)
153        FROM research_projects p ORDER BY p.created_at DESC, p.id DESC")?;
154    let v = st.query_map([], |r| Ok(ResearchProject { id: r.get(0)?, name: r.get(1)?, description: r.get(2)?, created_at: r.get(3)?,
155        items: r.get::<_, i64>(4)? as usize }))?.collect::<rusqlite::Result<Vec<_>>>()?;
156    Ok(v)
157}

159/// Project `name` (case-insensitive).

164/// Add an item; false when the project already has it (its note is updated if a new one is given).
165pub fn add_item(c: &Connection, project: i64, it: &NewItem) -> Result<bool> {
166    let kind = it.kind.ok_or_else(|| anyhow!("research item without a kind"))?;
167    let meta = if it.meta.is_object() { it.meta.to_string() } else { "{}".into() };
168    let n = c.execute("INSERT OR IGNORE INTO research_items (project_id, kind, ref, title, note, meta, content, local_path, added_at)
169        VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7, ?8, ?9)",
170        params![project, kind.as_str(), it.r#ref, it.title, it.note, meta, it.content, it.local_path, now()])?;
171    if n == 0 && !it.note.trim().is_empty() {
172        c.execute("UPDATE research_items SET note = ?1 WHERE project_id = ?2 AND kind = ?3 AND ref = ?4",
173            params![it.note, project, kind.as_str(), it.r#ref])?;
174    }
175    Ok(n > 0)
176}

178/// A project's items, in the order added.
179pub fn project_items(c: &Connection, project: i64) -> Result<Vec<ResearchItem>> {
180    if !crate::maintenance::table_exists(c, "research_items")? { return Ok(Vec::new()); }
181    let mut st = c.prepare("SELECT id, kind, ref, title, note, meta, content, local_path, added_at
182        FROM research_items WHERE project_id = ?1 ORDER BY added_at, id")?;
183    let v = st.query_map([project], |r| Ok((r.get::<_, i64>(0)?, r.get::<_, String>(1)?, r.get(2)?, r.get(3)?, r.get(4)?,
184        r.get::<_, String>(5)?, r.get(6)?, r.get(7)?, r.get(8)?)))?
185        .filter_map(|x| x.ok())
186        .filter_map(|(id, k, r#ref, title, note, meta, content, local_path, added_at)| Some(ResearchItem {
187            id, kind: ItemKind::parse(&k)?, r#ref, title, note, meta: serde_json::from_str(&meta).unwrap_or_else(|_| json!({})),
188            content, local_path, added_at }))
189        .collect();
190    Ok(v)
191}

196/// Short single-line title for a post.
197pub fn post_title(p: &Post) -> String {
198    let t = if p.title.trim().is_empty() { p.content.as_str() } else { p.title.as_str() };
199    let one: String = t.split_whitespace().collect::<Vec<_>>().join(" ");
200    let mut s: String = one.chars().take(90).collect();
201    if one.chars().count() > 90 { s.push('…'); }
202    format!("@{}: {s}", p.nick)
203}

205/// Items for posts already in the database (`kind` None = post/reply by the post itself).
206pub fn post_items(db: &Path, ids: &[String], kind: Option<ItemKind>, meta: Value, note: &str) -> Result<(Vec<NewItem>, Vec<String>)> {
207    let (posts, _, missing) = posts_by_ids(db, ids)?;
208    let v = posts.iter().map(|p| NewItem {
209        kind: Some(kind.unwrap_or(if p.parent_post_id.is_empty() { ItemKind::Post } else { ItemKind::Reply })),
210        r#ref: p.post_id.clone(), title: post_title(p), note: note.into(), meta: meta.clone(), ..NewItem::default()
211    }).collect();
212    Ok((v, missing))
213}

215/// MIME type from a file extension (enough for exports and opening).
216pub fn mime_for(p: &Path) -> &'static str {
217    match p.extension().map(|e| e.to_string_lossy().to_ascii_lowercase()).as_deref() {
218        Some("txt" | "log") => "text/plain", Some("md" | "markdown") => "text/markdown", Some("csv") => "text/csv",
219        Some("json") => "application/json", Some("html" | "htm") => "text/html", Some("xml") => "application/xml",
220        Some("pdf") => "application/pdf", Some("png") => "image/png", Some("jpg" | "jpeg") => "image/jpeg",
221        Some("gif") => "image/gif", Some("webp") => "image/webp", Some("avif") => "image/avif", Some("svg") => "image/svg+xml",
222        Some("mp4") => "video/mp4", Some("webm") => "video/webm", Some("mp3") => "audio/mpeg", Some("ogg") => "audio/ogg",
223        Some("doc") => "application/msword", Some("docx") => "application/vnd.openxmlformats-officedocument.wordprocessingml.document",
224        Some("odt") => "application/vnd.oasis.opendocument.text", Some("epub") => "application/epub+zip",
225        Some("zip") => "application/zip", Some("gz" | "tgz") => "application/gzip",
226        _ => "application/octet-stream",
227    }
228}

230/// Text files small enough to keep their content searchable in the export.
231const TEXT_MAX: u64 = 2 * 1024 * 1024;
232
233/// An item for a local file: metadata, its text (for text files), and a copy
234/// under the project's `files/` unless `copy` is false.
235pub fn file_item(project: &str, path: &Path, copy: bool, note: &str) -> Result<NewItem> {
236    let abs = std::fs::canonicalize(path).with_context(|| format!("{}", path.display()))?;
237    let md = std::fs::metadata(&abs)?;
238    if !md.is_file() { bail!("{} is not a file", abs.display()); }
239    let name = abs.file_name().map(|n| n.to_string_lossy().into_owned()).unwrap_or_default();
240    let mime = mime_for(&abs);
241    let modified = md.modified().ok().map(|t| chrono::DateTime::<chrono::Local>::from(t).to_rfc3339());
242    let content = if (mime.starts_with("text/") || mime == "application/json" || mime == "application/xml") && md.len() <= TEXT_MAX {
243        let raw = String::from_utf8_lossy(&std::fs::read(&abs)?).into_owned();
244        if mime == "text/html" { readable(&raw).1 } else { raw }
245    } else { String::new() };
246    let local = if copy {
247        let mut h = DefaultHasher::new();
248        abs.hash(&mut h);
249        let dir = project_dir(project)?.join("files");
250        std::fs::create_dir_all(&dir)?;
251        let dst = dir.join(format!("{:08x}-{name}", h.finish() as u32));
252        std::fs::copy(&abs, &dst).with_context(|| format!("copy {} to {}", abs.display(), dst.display()))?;
253        dst
254    } else { abs.clone() };
255    Ok(NewItem { kind: Some(ItemKind::LocalFile), r#ref: abs.display().to_string(), title: name, note: note.into(),
256        meta: json!({ "size": md.len(), "mime_type": mime, "modified": modified, "copied": copy, "original_path": abs.display().to_string() }),
257        content, local_path: local.display().to_string() })
258}

260/// Largest web page or file fetched for a research item.
261pub const WEB_MAX: usize = 20 * 1024 * 1024;
262
263/// Fetch `url` (no acp.us cookies are sent) into an item: the page title, its
264/// readable text and an HTML snapshot (other content types are saved as a file).
265pub async fn web_item(project: &str, url: &str, note: &str) -> Result<NewItem> {
266    let u = url::Url::parse(url.trim()).with_context(|| format!("not a URL: {url}"))?;
267    if !matches!(u.scheme(), "http" | "https") { bail!("only http and https URLs can be added (got {})", u.scheme()); }
268    let http = client(Arc::new(Jar::default()))?;
269    let resp = http.get(u.clone()).send().await.with_context(|| format!("fetching {u}"))?;
270    let status = resp.status();
271    let final_url = resp.url().to_string();
272    let ctype = resp.headers().get(reqwest::header::CONTENT_TYPE).and_then(|v| v.to_str().ok()).unwrap_or("").to_string();
273    if !status.is_success() { bail!("{u}: HTTP {status}"); }
274    let bytes = resp.bytes().await?;
275    if bytes.len() > WEB_MAX { bail!("{u}: {} bytes is more than the {WEB_MAX}-byte limit", bytes.len()); }
276    let mut h = DefaultHasher::new();
277    u.as_str().hash(&mut h);
278    let dir = project_dir(project)?.join("web");
279    std::fs::create_dir_all(&dir)?;
280    let is_html = ctype.contains("html") || (ctype.is_empty() && bytes.starts_with(b"<"));
281    let (title, content, file) = if is_html {
282        let raw = String::from_utf8_lossy(&bytes).into_owned();
283        let (t, text) = readable(&raw);
284        let f = dir.join(format!("{:016x}.html", h.finish()));
285        std::fs::write(&f, raw.as_bytes())?;
286        (t, text, f)
287    } else {
288        let name = u.path_segments().and_then(|mut s| s.next_back()).filter(|s| !s.is_empty()).unwrap_or("download").to_string();
289        let f = dir.join(format!("{:08x}-{name}", h.finish() as u32));
290        std::fs::write(&f, &bytes)?;
291        let text = if ctype.starts_with("text/") { String::from_utf8_lossy(&bytes).into_owned() } else { String::new() };
292        (name, text, f)
293    };
294    let title = if title.trim().is_empty() { u.host_str().unwrap_or("web page").to_string() } else { title };
295    Ok(NewItem { kind: Some(ItemKind::WebUrl), r#ref: u.to_string(), title, note: note.into(),
296        meta: json!({ "status": status.as_u16(), "content_type": ctype, "final_url": final_url, "bytes": bytes.len(), "fetched_at": now() }),
297        content, local_path: file.display().to_string() })
298}

300/// Title and readable text of an HTML page: headings (as `#` Markdown),
301/// paragraphs, list items, quotes and code from `<article>`, else `<main>`,
302/// else `<body>`, skipping navigation, headers, footers, scripts and forms.
303pub fn readable(html: &str) -> (String, String) {
304    let doc = Html::parse_document(html);
305    let sel = |s: &str| Selector::parse(s).expect("selector");
306    // Inline text runs join as written (`in <b>X</b>.` stays `in X.`), then whitespace collapses.
307    let text_of = |e: scraper::ElementRef| e.text().collect::<String>().split_whitespace().collect::<Vec<_>>().join(" ");
308    let title = doc.select(&sel("meta[property='og:title']")).next().and_then(|m| m.value().attr("content").map(str::to_string))
309        .or_else(|| doc.select(&sel("title")).next().map(text_of))
310        .or_else(|| doc.select(&sel("h1")).next().map(text_of))
311        .unwrap_or_default();
312    let root = doc.select(&sel("article")).next().or_else(|| doc.select(&sel("main")).next()).or_else(|| doc.select(&sel("body")).next());
313    let Some(root) = root else { return (title, String::new()) };
314    let blocks = ["h1", "h2", "h3", "h4", "h5", "h6", "p", "li", "blockquote", "pre"];
315    let skip = ["nav", "header", "footer", "aside", "script", "style", "form", "noscript", "template"];
316    let mut out: Vec<String> = Vec::new();
317    for n in root.descendants() {
318        let Some(e) = scraper::ElementRef::wrap(n) else { continue };
319        let name = e.value().name();
320        if !blocks.contains(&name) { continue; }
321        let mut skipped = false;
322        let mut inside_block = false;
323        for a in e.ancestors() {
324            let Some(ae) = scraper::ElementRef::wrap(a) else { continue };
325            if a.id() == root.id() { break; }
326            let an = ae.value().name();
327            if skip.contains(&an) { skipped = true; break; }
328            if blocks.contains(&an) { inside_block = true; }
329        }
330        if skipped || inside_block { continue; }
331        let t = if name == "pre" { e.text().collect::<String>() } else { text_of(e) };
332        if t.trim().is_empty() { continue; }
333        out.push(match name {
334            h if h.starts_with('h') => format!("{} {}", "#".repeat(h[1..].parse().unwrap_or(2)), t),
335            "li" => format!("- {t}"),
336            "blockquote" => format!("> {t}"),
337            "pre" => format!("```\n{}\n```", t.trim_end()),
338            _ => t,
339        });
340    }
341    (title.trim().to_string(), out.join("\n\n"))
342}

344/// The export document's `research` section.
345pub fn research_json(p: &ResearchProject, items: &[ResearchItem], rel: &dyn Fn(&ResearchItem) -> Option<String>) -> Value {
346    json!({
347        "name": p.name, "description": p.description, "created_at": p.created_at,
348        "items": items.iter().map(|i| json!({
349            "id": i.id, "kind": i.kind.as_str(), "ref": i.r#ref, "title": i.title, "note": i.note,
350            "meta": i.meta, "content": i.content, "file": rel(i), "added_at": i.added_at,
351        })).collect::<Vec<_>>(),
352    })
353}

355/// Attachment entries (`post_id` `research:<id>`) for items with a stored file, so exports copy them.
356pub fn item_attachments(items: &[ResearchItem]) -> Vec<Attachment> {
357    items.iter().filter(|i| !i.local_path.is_empty() && Path::new(&i.local_path).exists()).map(|i| Attachment {
358        post_id: format!("research:{}", i.id),
359        url: if i.kind == ItemKind::WebUrl { i.r#ref.clone() } else { format!("file://{}", i.r#ref) },
360        kind: "file".into(), local_path: PathBuf::from(&i.local_path),
361        mime_type: i.meta["mime_type"].as_str().or(i.meta["content_type"].as_str()).unwrap_or("application/octet-stream").to_string(),
362    }).collect()
363}
