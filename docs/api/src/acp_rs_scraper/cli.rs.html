5/// Print `msg` on stderr and read one trimmed line from stdin.
6pub fn prompt(msg: &str) -> Result<String> {
7    eprint!("{msg}");
8    io::stderr().flush()?;
9    let mut s = String::new();
10    io::stdin().read_line(&mut s)?;
11    Ok(s.trim().to_string())
12}

260/// Write the man page or a completion script for [`Args`] to `out`.
261pub fn generate(what: Generate, out: &mut dyn Write) -> Result<()> {
262    use clap::CommandFactory;
263    let mut cmd = Args::command();
264    let name = cmd.get_name().to_string();
265    match what {
266        Generate::Man => clap_mangen::Man::new(cmd).render(out)?,
267        Generate::Bash => clap_complete::generate(clap_complete::Shell::Bash, &mut cmd, name, out),
268        Generate::Zsh => clap_complete::generate(clap_complete::Shell::Zsh, &mut cmd, name, out),
269        Generate::Fish => clap_complete::generate(clap_complete::Shell::Fish, &mut cmd, name, out),
270    }
271    Ok(())
272}

275/// Which site-wide feed the mentions crawl walks (`--latest` / `--trending`).
276pub enum FeedMode {
277    /// `/rs/latest`: chronological, so incremental early-stop works (default).
278    Latest,
279    /// `/rs`: the site's Trending tab; not chronological, no early stop.
280    Trending,
281}

283impl FeedMode {
284    /// Site-wide feed URL crawled for mentions in this mode.
285    pub fn site_feed_url(self) -> String {
286        match self {
287            FeedMode::Latest => feed_latest_url(),
288            FeedMode::Trending => feed_trending_url(),
289        }
290    }
291    /// Whether paging may stop early at an all-known page (only for chronological feeds).
292    pub fn allow_incremental_stop(self) -> bool {
293        matches!(self, FeedMode::Latest)
294    }
295}

307/// A parsed `--nicks` list: `@` stripped, trimmed, deduplicated case-insensitively.
308pub struct NickList(pub Vec<String>);
309
310/// "@a, b,,@A" -> ["a", "b"]: strip @, trim, drop empties, dedupe (case-insensitive, first wins).
311pub fn parse_nicks(s: &str) -> std::result::Result<NickList, String> {
312    let mut seen = HashSet::new();
313    let v: Vec<String> = s.split(',')
314        .map(|n| n.trim().trim_start_matches('@').trim().to_string())
315        .filter(|n| !n.is_empty() && seen.insert(n.to_lowercase()))
316        .collect();
317    if v.is_empty() { return Err("expected at least one nick".into()); }
318    if let Some(bad) = v.iter().find(|n| n.contains(char::is_whitespace) || n.contains('/')) {
319        return Err(format!("invalid nick {bad:?}"));
320    }
321    Ok(NickList(v))
322}

324/// Default export name for a set of nicks.
325pub fn nicks_name(nicks: &[String], stamp: &str) -> String {
326    let j = nicks.join("+");
327    if nicks.len() > 4 || j.len() > 40 { format!("multi-{stamp}") } else { format!("{j}-{stamp}") }
328}

330/// Whether these parsed arguments open the terminal UI: `--tui`, or no flag at all
331/// other than the logging ones in [`TUI_NEUTRAL_ARGS`].
332pub fn wants_tui(m: &clap::ArgMatches) -> bool {
333    use clap::CommandFactory;
334    if m.get_flag("tui") { return true; }
335    let cmd = Args::command();
336    !cmd.get_arguments().map(|a| a.get_id().as_str())
337        .filter(|id| !TUI_NEUTRAL_ARGS.contains(id))
338        .any(|id| m.value_source(id) == Some(clap::parser::ValueSource::CommandLine))
339}
340
341/// Parse `argv` into [`Args`] plus whether it opens the terminal UI ([`wants_tui`]).
342pub fn parse_with_mode<I, T>(argv: I) -> std::result::Result<(Args, bool), clap::Error>
343where I: IntoIterator<Item = T>, T: Into<std::ffi::OsString> + Clone {
344    use clap::{CommandFactory, FromArgMatches};
345    let m = Args::command().try_get_matches_from(argv)?;
346    let tui = wants_tui(&m);
347    Ok((Args::from_arg_matches(&m)?, tui))
348}

350impl Args {
351    /// --nick / --nicks merged (empty = none given).
352    pub fn nick_list(&self) -> Result<Vec<String>> {
353        if let Some(n) = &self.nicks { return Ok(n.0.clone()); }
354        match &self.nick { Some(n) => Ok(parse_nicks(n).map_err(|e| anyhow!(e))?.0), None => Ok(Vec::new()) }
355    }
356
357    /// Selected archive format (clap guarantees --compress <=> exactly one format flag).
358    pub fn compression(&self) -> Option<Compress> {
359        if !self.compress { return None; }
360        if self.tar_gz { Some(Compress::TarGz) } else { None }
361    }
362    /// Selected export formats, deduplicated in command-line order (never empty).
363    pub fn formats(&self) -> Vec<Format> {
364        let mut v: Vec<Format> = Vec::new();
365        for f in &self.format { if !v.contains(f) { v.push(*f); } }
366        if v.is_empty() { v.push(Format::Json); }
367        v
368    }
369
370    /// Whether the mentions crawl should run (`--no-mentions` turns it off).
371    pub fn want_mentions(&self) -> bool { self.mentions && !self.said_by }
372    /// Default is Latest (our default), even though the site lands on Trending.
373    pub fn feed_mode(&self) -> FeedMode {
374        if self.trending { FeedMode::Trending } else { FeedMode::Latest }
375    }
