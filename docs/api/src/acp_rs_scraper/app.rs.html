6/// `--remove-nick`: preview (rolled-back transaction), confirm, then commit.
7pub fn run_remove_nick(args: &Args, nicks: &[String]) -> Result<()> {
8    let db = home()?.join(DB_REL_PATH);
9    let att = home()?.join(ATTACH_REL_PATH);
10    let preview = remove_nicks(&db, &att, nicks, args.keep_replies, false)?;
11    let who = nicks.iter().map(|n| format!("@{n}")).collect::<Vec<_>>().join(", ");
12    eprintln!("{}", preview.summary(&format!("remove {who} would delete")));
13    if args.keep_replies { eprintln!("--keep-replies: other people's replies in their threads are kept"); }
14    if args.dry_run {
15        for f in &preview.files { eprintln!("  file: {}", f.display()); }
16        eprintln!("dry run: nothing changed");
17        return Ok(());
18    }
19    if preview.is_empty() {
20        eprintln!("nothing to remove");
21        return Ok(());
22    }
23    if !confirm("Proceed?", false, args.yes)? {
24        eprintln!("aborted: nothing changed");
25        return Ok(());
26    }
27    let done = remove_nicks(&db, &att, nicks, args.keep_replies, true)?;
28    eprintln!("{}", done.summary("removed"));
29    Ok(())
30}

32/// `--clear-all`: delete the database and attachments dir; prefs and exports stay.
33pub fn run_clear_all(args: &Args) -> Result<()> {
34    let db = home()?.join(DB_REL_PATH);
35    let att = home()?.join(ATTACH_REL_PATH);
36    let targets = clear_all_targets(&db, &att);
37    let (n, bytes) = dir_usage(&att);
38    if targets.is_empty() {
39        eprintln!("nothing to clear ({} and {} do not exist)", db.display(), att.display());
40        return Ok(());
41    }
42    eprintln!("clear-all will delete:");
43    for t in &targets {
44        if t.is_dir() { eprintln!("  {}/ ({n} files, {bytes} bytes)", t.display()); } else { eprintln!("  {}", t.display()); }
45    }
46    eprintln!("kept: {} and {}", home()?.join(PREFS_REL_PATH).display(), home()?.join(EXPORTS_REL_PATH).display());
47    if args.dry_run {
48        eprintln!("dry run: nothing changed");
49        return Ok(());
50    }
51    if !confirm("Delete all scraped data?", true, args.yes)? {
52        eprintln!("aborted: nothing changed");
53        return Ok(());
54    }
55    let gone = clear_all(&db, &att)?;
56    eprintln!("cleared: {} paths removed", gone.len());
57    Ok(())
58}

60/// Run the program for the current process arguments.
61pub async fn run() -> Result<()> {
62    use clap::Parser;
63    let args = Args::parse();
64    if let Some(what) = args.generate {
65        return generate(what, &mut io::stdout().lock());
66    }
67    let level = if args.quiet {
68        LogLevel::Quiet
69    } else if args.debug || std::env::var("ACP_DEBUG").is_ok_and(|v| v == "1") {
70        LogLevel::Debug
71    } else if args.verbose {
72        LogLevel::Verbose
73    } else {
74        LogLevel::Normal
75    };
76    set_log_level(level);
77    let stamp = chrono::Local::now().format("%Y%m%d-%H%M%S").to_string();
78    let clean = |n: String| n.replace(['/', '\\'], "_");
79    if let Some(list) = &args.remove_nick {
80        return run_remove_nick(&args, &list.0);
81    }
82    if args.clear_all {
83        return run_clear_all(&args);
84    }
85    if args.list_nicks {
86        let infos = list_nicks_db(&home()?.join(DB_REL_PATH))?;
87        let argv: Vec<String> = std::env::args().collect();
88        let want_json = argv.windows(2).any(|w| w[0] == "--format" && w[1] == "json")
89            || argv.iter().any(|a| a == "--format=json");
90        if want_json {
91            let v: Vec<_> = infos.iter().map(|i| json!({
92                "nick": i.nick, "display_name": i.display_name, "post_count": i.post_count,
93                "latest": i.latest, "profile_scraped": i.profile_scraped, "scraped_at": i.scraped_at,
94            })).collect();
95            println!("{}", serde_json::to_string_pretty(&v)?);
96        } else {
97            print_nicks_table(&infos);
98        }
99        return Ok(());
100    }
101    if let Some(file) = &args.import {
102        let (posts, atts, profiles, hits) = import(file, args.compression(), &home()?.join(ATTACH_REL_PATH), &home()?.join(EXPORTS_REL_PATH))?;
103        let n = save(&posts)?;
104        save_attachments(&home()?.join(DB_REL_PATH), &atts)?;
105        for p in &profiles { save_profile(&home()?.join(DB_REL_PATH), p)?; }
106        save_mentions(&home()?.join(DB_REL_PATH), &hits)?;
107        info!("import: {} posts read, {n} new; {} attachments; {} profiles; {} mentions",
108            posts.len(), atts.len(), profiles.len(), hits.len());
109        return Ok(());
110    }
111    // Offline DB export: --export FILE, or --search (to stdout/--output).
112    if args.export.is_some() || args.search.is_some() {
113        let mut a = args.clone();
114        if let Some(f) = &args.export { a.output = Some(f.clone()); }
115        let q = args.search.as_deref();
116        let nicks = args.nick_list()?;
117        let (posts, atts, n) = search_db(&home()?.join(DB_REL_PATH), q, &nicks)?;
118        info!("db export: {n} matches, {} posts total, {} attachments", posts.len(), atts.len());
119        let name = clean(args.name.clone().unwrap_or_else(|| match q {
120            Some(q) => format!("search-{}-{stamp}", slug(q)),
121            None if !nicks.is_empty() => nicks_name(&nicks, &stamp),
122            None => format!("db-{stamp}"),
123        }));
124        let who = nicks.iter().map(|n| format!("@{n}")).collect::<Vec<_>>().join(", ");
125        let label = match (q, who.is_empty()) {
126            (Some(q), true) => format!("search: {q}"),
127            (Some(q), false) => format!("search: {q} ({who})"),
128            (None, false) => who,
129            (None, true) => "all posts".into(),
130        };
131        let (profiles, hits) = load_profiles_mentions(&home()?.join(DB_REL_PATH), &nicks)?;
132        return export(&a, &name, &label, &posts, &atts, &profiles, &hits, &home()?.join(EXPORTS_REL_PATH));
133    }
134    let jar = Arc::new(Jar::default());
135    let http = client(jar.clone())?;
136    login(&http, &jar).await?;
137    let mut nicks = args.nick_list()?;
138    if nicks.is_empty() {
139        nicks = parse_nicks(&prompt("Nick(s) to scrape (comma-separated): ")?).map_err(|e| anyhow!(e))?.0;
140    }
141    let db = home()?.join(DB_REL_PATH);
142    let att_dir = home()?.join(ATTACH_REL_PATH);
143    let known = db_known_ids(&db)?;
144    let reply_counts = db_reply_counts(&db)?;
145    let known_infos = list_nicks_db(&db)?;
146    info!("== mode: {} (site crawl); profiles always use Latest", args.feed_mode());
147    if args.said_by { info!("== --said-by: only the nicks' own posts and replies; no thread crawl, no mentions"); }
148    info!("== targets");
149    for nick in &nicks {
150        if let Some(i) = known_infos.iter().find(|i| i.nick.eq_ignore_ascii_case(nick)) {
151            let dn = i.display_name.as_deref().unwrap_or("-");
152            info!("  @{nick}: known ({dn}, {} posts{})",
153                i.post_count,
154                if i.profile_scraped { ", profile scraped" } else { "" });
155        } else {
156            info!("  @{nick}: new");
157        }
158    }
159    let mut posts: Vec<Post> = Vec::new();
160    let mut profiles: Vec<Profile> = Vec::new();
161    let mut mention_hits: Vec<MentionHit> = Vec::new();
162    let mut targets: Vec<MentionTarget> = Vec::new();
163    let mut seen = HashSet::new();
164    let mut summary = Vec::new();
165    let mut ok_nicks = Vec::new();
166    for nick in &nicks {
167        info!("== scraping @{nick}");
168        match scrape(&http, &jar, nick, &known, &reply_counts, args.full, args.said_by).await {
169            Ok((ps, html, st)) => {
170                info!("  @{nick}: feed {} items ({} from posts and replies), ~{} new, {} skipped known, threads fetched {} / skipped {}",
171                    st.feed_items, st.replies_tab_items, st.new_posts, st.skipped_known, st.threads_fetched, st.threads_skipped);
172                let mut profile = Profile {
173                    nick: nick.clone(),
174                    scraped_at: chrono::Local::now().to_rfc3339(),
175                    ..Default::default()
176                };
177                // The feed's author object carries the real profile photo; read it
178                // before the page's og: tags, which fall back to the site banner.
179                // Reply-only nicks have an empty Posts feed, so also look at
180                // "Posts and replies" when the Posts feed had no author data.
181                for url in [profile_url(nick), profile_replies_url(nick)] {
182                    if profile.avatar_url.is_some() && profile.display_name.is_some() { break; }
183                    match fetch_feed(&http, &jar, &url, Some(10), None, true).await {
184                        Ok(feed) => {
185                            enrich_from_feed(&mut profile, &feed.items, nick);
186                            if profile.avatar_url.is_none() && let Some(it) = feed.items.first() {
187                                dbg_log("profile", &format!("@{nick} no photo in {url}; first item: {it}"));
188                            }
189                        }
190                        Err(e) => dbg_log("profile", &format!("@{nick} {url}: {e:#}")),
191                    }
192                }
193                enrich_from_html(&mut profile, &html);
194                if profile.avatar_url.as_deref().is_some_and(is_site_image) { profile.avatar_url = None; }
195                dbg_log("profile", &format!("@{nick} show-info unused event={PROFILE_SHOW_INFO_EVENT} fields={profile:?}"));
196                if !args.no_attachments {
197                    if let Some(u) = &profile.avatar_url {
198                        match fetch_avatar(&http, u, &att_dir, nick).await {
199                            Ok(path) => {
200                                info!("  @{nick}: profile image {}", path.display());
201                                profile.avatar_path = Some(path.to_string_lossy().into());
202                            }
203                            Err(e) => err!("warning: @{nick} profile image: {e:#}"),
204                        }
205                    }
206                    if let Some(u) = &profile.banner_url
207                        && let Ok((path, _)) = fetch_one_pub(&http, u, &att_dir).await {
208                            profile.banner_path = Some(path.to_string_lossy().into());
209                        }
210                }
211                profile.raw_json = serde_json::to_string(&profile).unwrap_or_default();
212                save_profile(&db, &profile)?;
213                targets.push(MentionTarget {
214                    nick: nick.clone(),
215                    user_id: profile.user_id.clone(),
216                    display_name: profile.display_name.clone(),
217                });
218                profiles.push(profile);
219
220                let roots = ps.iter().filter(|p| p.parent_post_id.is_empty()).count();
221                let total = ps.len();
222                let new: Vec<Post> = ps.into_iter().filter(|p| seen.insert(p.post_id.clone())).collect();
223                let inserted = save(&new)?;
224                summary.push(format!(
225                    "@{nick}: {total} items ({roots} root posts, {} replies), {inserted} new DB rows",
226                    total - roots
227                ));
228                posts.extend(new);
229                ok_nicks.push(nick.clone());
230            }
231            Err(e) => {
232                err!("warning: @{nick} failed: {e:#}");
233                summary.push(format!("@{nick}: FAILED ({e})"));
234            }
235        }
236    }
237    if args.want_mentions() && !ok_nicks.is_empty() {
238        info!("== mentions crawl (Latest, limit {})", args.crawl_limit);
239        match crawl_mentions(&http, &jar, &targets, args.crawl_limit, &mut seen, &known, args.full, args.feed_mode()).await {
240            Ok((extra, hits)) => {
241                let inserted = save(&extra)?;
242                save_mentions(&db, &hits)?;
243                info!("mentions: {} posts, {} hits, {inserted} new DB rows", extra.len(), hits.len());
244                posts.extend(extra);
245                mention_hits.extend(hits);
246            }
247            Err(e) => err!("warning: mentions crawl failed: {e:#}"),
248        }
249    }
250    info!("== summary");
251    for l in &summary { info!("  {l}"); }
252    if ok_nicks.is_empty() { bail!("all nicks failed"); }
253    let atts = if args.no_attachments { Vec::new() } else {
254        let for_dl: Vec<Post> = if args.full {
255            posts.clone()
256        } else {
257            // Only download media for posts we didn't already have.
258            posts.iter().filter(|p| !known.contains(&p.post_id)).cloned().collect()
259        };
260        info!("attachments: downloading for {} posts ({} skipped as known)",
261            for_dl.len(), posts.len().saturating_sub(for_dl.len()));
262        let a = download_all(&http, &for_dl, &att_dir).await;
263        save_attachments(&db, &a)?;
264        a
265    };
266    let name = clean(args.name.clone().unwrap_or_else(|| nicks_name(&ok_nicks, &stamp)));
267    let label = ok_nicks.iter().map(|n| format!("@{n}")).collect::<Vec<_>>().join(", ");
268    export(&args, &name, &label, &posts, &atts, &profiles, &mention_hits, &home()?.join(EXPORTS_REL_PATH))?;
269
270    Ok(())
