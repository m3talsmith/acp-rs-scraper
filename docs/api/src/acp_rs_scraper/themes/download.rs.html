1//! Downloading theme files from the repo's `themes/` folder at this version's tag
2//! ([`THEME_DOWNLOAD_BASE`], or [`THEME_URL_ENV`] in tests). The first run with an empty
3//! `~/.acp/themes` fetches `default.toml` in the background; `--theme-init` and
4//! *Theme: init* fetch on request. Each download is capped at [`THEME_MAX_BYTES`] and
5//! [`THEME_DOWNLOAD_TIMEOUT`], must parse as a theme, sends no cookies, and is written
6//! atomically (mode 0600) without replacing an existing file.
7
8use super::*;
9
10/// What a download did.
11#[derive(Clone, Debug, PartialEq, Eq)]
12pub enum Fetched {
13    /// The file was written.
14    Written(PathBuf),
15    /// A file of that name was already there; it was kept and nothing was replaced.
16    Kept(PathBuf),
17}
18
19impl Fetched {
20    /// The theme file.
21    pub fn path(&self) -> &Path { match self { Fetched::Written(p) | Fetched::Kept(p) => p } }
22}
23
24/// Whether `ACP_THEME_DOWNLOAD` turns the automatic download off.
25pub fn env_off(v: Option<&str>) -> bool {
26    matches!(v.map(|v| v.trim().to_ascii_lowercase()).as_deref(), Some("off" | "0" | "false" | "no"))
27}

29/// Whether the first run may download the default theme: not `ACP_THEME_DOWNLOAD=off`
30/// and not `[ui] theme_download = false`.
31pub fn auto_enabled(env: Option<&str>, prefs: &Prefs) -> bool {
32    !env_off(env) && prefs.ui.as_ref().and_then(|u| u.theme_download) != Some(false)
33}

35/// [`auto_enabled`] for this process.
36pub fn auto_enabled_now(prefs: &Prefs) -> bool { auto_enabled(std::env::var(THEME_DOWNLOAD_ENV).ok().as_deref(), prefs) }

38/// Fetch `url` with the program's HTTP client and an empty cookie jar (no cookies are
39/// sent), refusing bodies over [`THEME_MAX_BYTES`] and giving up after `timeout`.
40pub async fn fetch(url: &str, timeout: std::time::Duration) -> Result<Vec<u8>> {
41    let http = crate::http::client(Arc::new(Jar::default()))?;
42    let go = async {
43        let mut r = http.get(url).header(reqwest::header::ACCEPT, "text/plain, application/toml, */*").send().await
44            .with_context(|| format!("fetching {url}"))?;
45        if !r.status().is_success() { bail!("{url} answered {}", r.status()); }
46        let too_big = || anyhow!("{url} is over {} KiB; theme files are at most that", THEME_MAX_BYTES / 1024);
47        if r.content_length().is_some_and(|n| n > THEME_MAX_BYTES) { return Err(too_big()); }
48        let mut buf = Vec::new();
49        while let Some(c) = r.chunk().await.with_context(|| format!("reading {url}"))? {
50            buf.extend_from_slice(&c);
51            if buf.len() as u64 > THEME_MAX_BYTES { return Err(too_big()); }
52        }
53        Ok(buf)
54    };
55    match tokio::time::timeout(timeout, go).await {
56        Ok(r) => r,
57        Err(_) => bail!("no answer from {url} within {} s (timed out)", timeout.as_secs_f64()),
58    }
59}

61/// Check that `bytes` is a usable theme and write it as `dir/<name>.toml` (atomic,
62/// mode 0600), keeping a file that is already there.
63pub fn store(dir: &Path, name: &str, bytes: &[u8]) -> Result<Fetched> {
64    let path = theme_path(dir, name).map_err(|e| anyhow!(e))?;
65    if bytes.len() as u64 > THEME_MAX_BYTES { bail!("the downloaded {name}.toml is over {} KiB", THEME_MAX_BYTES / 1024); }
66    let src = std::str::from_utf8(bytes).map_err(|_| anyhow!("the downloaded {name}.toml is not UTF-8 text"))?;
67    parse(src, name).map_err(|e| anyhow!("the downloaded {name}.toml is not a valid theme: {e}"))?;
68    if atomic_write_new(&path, bytes).with_context(|| format!("writing {}", path.display()))? { Ok(Fetched::Written(path)) } else { Ok(Fetched::Kept(path)) }
69}

71/// Download theme `name` from `url` into `dir`. An existing file is kept without
72/// asking the network.
73pub async fn download_from(dir: &Path, name: &str, url: &str, timeout: std::time::Duration) -> Result<Fetched> {
74    let path = theme_path(dir, name).map_err(|e| anyhow!(e))?;
75    if path.exists() { return Ok(Fetched::Kept(path)); }
76    let bytes = fetch(url, timeout).await?;
77    store(dir, name, &bytes)
78}

80/// Download theme `name` from [`theme_download_url`] into `dir`.
81pub async fn download(dir: &Path, name: &str) -> Result<Fetched> {
82    download_from(dir, name, &theme_download_url(name), THEME_DOWNLOAD_TIMEOUT).await
83}

85/// What the user sees when theme `name` could not be downloaded, and how to fix it.
86pub fn failure_notice(dir: &Path, name: &str, err: &str) -> String {
87    format!("could not download the {name} theme: {err}. The terminal UI uses your terminal's own colours until a theme \
88        is there. To fix it, check the connection and run `acp-rs-scraper --theme-init` (or Theme: init in the command \
89        palette), or save {} as {}.", theme_download_url(name), dir.join(format!("{name}.toml")).display())
90}

92/// What the user sees when the first run did not download the default because it is turned off.
93pub fn off_notice(dir: &Path) -> String {
94    format!("no theme files in {} and the automatic theme download is off ({THEME_DOWNLOAD_ENV}=off or [ui] theme_download = false); \
95        using the terminal's own colours. Run `acp-rs-scraper --theme-init` to download the default theme, or create one with Shift-F5.",
96        dir.display())
97}

99/// What the user sees after a theme was downloaded.
100pub fn done_notice(name: &str, f: &Fetched) -> String {
101    match f {
102        Fetched::Written(p) => format!("downloaded the {name} theme into {}", p.display()),
103        Fetched::Kept(p) => format!("{} already exists; it was kept (nothing downloaded)", p.display()),
104    }
