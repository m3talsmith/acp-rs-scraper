1//! Downloading theme files from the repo's `themes/` folder at this version's tag
2//! ([`THEME_DOWNLOAD_BASE`], or [`THEME_URL_ENV`] in tests). The first run with an empty
3//! `~/.acp/themes` fetches `default.toml` in the background; `--theme-init` and
4//! *Theme: init* fetch on request. Each download is capped at [`THEME_MAX_BYTES`] and
5//! [`THEME_DOWNLOAD_TIMEOUT`], must parse as a theme, sends no cookies, and is written
6//! atomically (mode 0600) without replacing an existing file.
7
8use super::*;
9use ratatui::style::Color;
10
11/// What a download did.
12#[derive(Clone, Debug, PartialEq, Eq)]
13pub enum Fetched {
14    /// The file was written.
15    Written(PathBuf),
16    /// A file of that name was already there; it was kept and nothing was replaced.
17    Kept(PathBuf),
18}
19
20impl Fetched {
21    /// The theme file.
22    pub fn path(&self) -> &Path { match self { Fetched::Written(p) | Fetched::Kept(p) => p } }
23}

25/// Whether `ACP_THEME_DOWNLOAD` turns the automatic download off.
26pub fn env_off(v: Option<&str>) -> bool {
27    matches!(v.map(|v| v.trim().to_ascii_lowercase()).as_deref(), Some("off" | "0" | "false" | "no"))
28}

30/// Whether the first run may download the default theme: not `ACP_THEME_DOWNLOAD=off`
31/// and not `[ui] theme_download = false`.
32pub fn auto_enabled(env: Option<&str>, prefs: &Prefs) -> bool {
33    !env_off(env) && prefs.ui.as_ref().and_then(|u| u.theme_download) != Some(false)
34}

36/// [`auto_enabled`] for this process.

39/// Fetch `url` with the program's HTTP client and an empty cookie jar (no cookies are
40/// sent), refusing bodies over [`THEME_MAX_BYTES`] and giving up after `timeout`.
41pub async fn fetch(url: &str, timeout: std::time::Duration) -> Result<Vec<u8>> {
42    let http = crate::http::client(Arc::new(Jar::default()))?;
43    let go = async {
44        let mut r = http.get(url).header(reqwest::header::ACCEPT, "text/plain, application/toml, */*").send().await
45            .with_context(|| format!("fetching {url}"))?;
46        if !r.status().is_success() { bail!("{url} answered {}", r.status()); }
47        let too_big = || anyhow!("{url} is over {} KiB; theme files are at most that", THEME_MAX_BYTES / 1024);
48        if r.content_length().is_some_and(|n| n > THEME_MAX_BYTES) { return Err(too_big()); }
49        let mut buf = Vec::new();
50        while let Some(c) = r.chunk().await.with_context(|| format!("reading {url}"))? {
51            buf.extend_from_slice(&c);
52            if buf.len() as u64 > THEME_MAX_BYTES { return Err(too_big()); }
53        }
54        Ok(buf)
55    };
56    match tokio::time::timeout(timeout, go).await {
57        Ok(r) => r,
58        Err(_) => bail!("no answer from {url} within {} s (timed out)", timeout.as_secs_f64()),
59    }
60}

62/// Check that `bytes` is a usable theme and write it as `dir/<name>.toml` (atomic,
63/// mode 0600), keeping a file that is already there.
64pub fn store(dir: &Path, name: &str, bytes: &[u8]) -> Result<Fetched> {
65    let path = theme_path(dir, name).map_err(|e| anyhow!(e))?;
66    if bytes.len() as u64 > THEME_MAX_BYTES { bail!("the downloaded {name}.toml is over {} KiB", THEME_MAX_BYTES / 1024); }
67    let src = std::str::from_utf8(bytes).map_err(|_| anyhow!("the downloaded {name}.toml is not UTF-8 text"))?;
68    parse(src, name).map_err(|e| anyhow!("the downloaded {name}.toml is not a valid theme: {e}"))?;
69    if atomic_write_new(&path, bytes).with_context(|| format!("writing {}", path.display()))? { Ok(Fetched::Written(path)) } else { Ok(Fetched::Kept(path)) }
70}

72/// Download theme `name` from `url` into `dir`. An existing file is kept without
73/// asking the network.
74pub async fn download_from(dir: &Path, name: &str, url: &str, timeout: std::time::Duration) -> Result<Fetched> {
75    let path = theme_path(dir, name).map_err(|e| anyhow!(e))?;
76    if path.exists() { return Ok(Fetched::Kept(path)); }
77    let bytes = fetch(url, timeout).await?;
78    store(dir, name, &bytes)
79}

81/// Download theme `name` from [`theme_download_url`] into `dir`.
82pub async fn download(dir: &Path, name: &str) -> Result<Fetched> {
83    download_from(dir, name, &theme_download_url(name), THEME_DOWNLOAD_TIMEOUT).await
84}

101/// What the user sees after a theme was downloaded.

109/// The minimal starter theme: every palette colour set to a plain terminal colour
110/// (the 16 ANSI colours and `reset`), no element overrides, one variant (light uses
111/// it too). Generated here, not a copy of any shipped theme; edit it with Shift-F5.
112pub fn starter(name: &str) -> ThemeFile {
113    let mut f = ThemeFile::empty(name);
114    f.description = "Starter theme: plain terminal colours (edit it with Shift-F5)".into();
115    for (k, v) in [("accent", Color::Red), ("muted", Color::DarkGray), ("border", Color::DarkGray), ("on_accent", Color::White),
116        ("fg", Color::Reset), ("bg", Color::Reset)] {
117        f.dark.set(k, ColorSpec::Value(v));
118    }
119    f
120}

122/// Write [`starter`] as `dir/<name>.toml` (atomic, mode 0600), keeping a file that is
123/// already there. Never touches the network.
124pub fn write_starter(dir: &Path, name: &str) -> Result<Fetched> {
125    let path = theme_path(dir, name).map_err(|e| anyhow!(e))?;
126    let text = to_toml(&starter(name));
127    if atomic_write_new(&path, text.as_bytes()).with_context(|| format!("writing {}", path.display()))? { Ok(Fetched::Written(path)) } else { Ok(Fetched::Kept(path)) }
128}

130/// A free name for a starter theme: `base`, else `base-2`, `base-3`…
131pub fn free_name(dir: &Path, base: &str) -> String {
132    (1..1000).map(|i| if i == 1 { base.to_string() } else { format!("{base}-{i}") })
133        .find(|n| !dir.join(format!("{n}.toml")).exists()).unwrap_or_else(|| base.to_string())
134}
