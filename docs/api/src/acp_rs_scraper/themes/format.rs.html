1//! The theme file format, v1 (TOML):
2//!
3//! ```toml
4//! version = 1
5//! name = "default"
6//! description = "ACP red on your terminal's colours"
7//!
8//! [dark.palette]            # colours; any extra name can be referenced as "@name"
9//! accent = "#dc2626"
10//! muted = "#9ca3af"
11//!
12//! [dark.styles.link]        # optional per-element overrides
13//! fg = "@accent"
14//! underline = true
15//!
16//! [light.palette]           # optional; missing keys fall back to [dark]
17//! ```
18//!
19//! A colour is `0`–`255` (the 256-colour palette), `#rrggbb` / `#rgb`, one of
20//! the 16 ANSI names (`black` … `white`, `bright_black` … `bright_white`),
21//! `reset` (the terminal's own colour), or `@name` for a palette entry.
22//! Borders and padding are fixed in code; a style sets colours and attributes.
23

40macro_rules! elements {
41    ($($v:ident = $n:literal, $box:literal, $about:literal;)*) => {
42        /// Every styled UI element a theme can override.
43        #[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
44        #[allow(missing_docs)]
45        pub enum Element { $($v),* }
46        impl Element {
47            /// All of them, in editor order.
48            pub const ALL: &'static [Element] = &[$(Element::$v),*];
49            /// The key under `[dark.styles]`.
50            pub fn name(self) -> &'static str { match self { $(Element::$v => $n),* } }
51            /// A box (its `border` key colours the border).
52            pub fn is_box(self) -> bool { match self { $(Element::$v => $box),* } }
53            /// What it is.
54            pub fn about(self) -> &'static str { match self { $(Element::$v => $about),* } }
55        }
56    };
57}

105/// Overrides for one element; `None` keeps the built-in look.
106#[derive(Clone, Debug, Default, PartialEq, Eq)]
107pub struct StyleSpec {
108    /// Text colour.
109    pub fg: Option<ColorSpec>,
110    /// Background.
111    pub bg: Option<ColorSpec>,
112    /// Border colour (boxes only).
113    pub border: Option<ColorSpec>,
114    /// Bold.
115    pub bold: Option<bool>,
116    /// Italic.
117    pub italic: Option<bool>,
118    /// Underlined.
119    pub underline: Option<bool>,
120    /// Foreground and background swapped.
121    pub reverse: Option<bool>,
122    /// Dim.
123    pub dim: Option<bool>,
124}
125
126impl StyleSpec {
127    /// Nothing set.
128    pub fn is_empty(&self) -> bool { *self == StyleSpec::default() }
129    /// A colour key (`fg`, `bg`, `border`).
130    pub fn color(&self, k: &str) -> Option<&ColorSpec> {
131        match k { "fg" => self.fg.as_ref(), "bg" => self.bg.as_ref(), "border" => self.border.as_ref(), _ => None }
132    }
133    /// Set a colour key.
134    pub fn set_color(&mut self, k: &str, c: Option<ColorSpec>) {
135        match k { "fg" => self.fg = c, "bg" => self.bg = c, "border" => self.border = c, _ => {} }
136    }
137    /// An attribute key (`bold` … `dim`).
138    pub fn flag(&self, k: &str) -> Option<bool> {
139        match k { "bold" => self.bold, "italic" => self.italic, "underline" => self.underline, "reverse" => self.reverse, "dim" => self.dim, _ => None }
140    }
141    /// Set an attribute key.
142    pub fn set_flag(&mut self, k: &str, v: Option<bool>) {
143        match k { "bold" => self.bold = v, "italic" => self.italic = v, "underline" => self.underline = v, "reverse" => self.reverse = v, "dim" => self.dim = v, _ => {} }
144    }
145}

147/// `[dark]` or `[light]`.
148#[derive(Clone, Debug, Default, PartialEq, Eq)]
149pub struct Section {
150    /// Colours, in file order.
151    pub palette: Vec<(String, ColorSpec)>,
152    /// Element overrides, in file order.
153    pub styles: Vec<(String, StyleSpec)>,
154}
155
156impl Section {
157    /// A palette entry as written.
158    pub fn get(&self, name: &str) -> Option<&ColorSpec> { self.palette.iter().find(|(n, _)| n == name).map(|(_, c)| c) }
159    /// Set (or add) a palette entry.
160    pub fn set(&mut self, name: &str, c: ColorSpec) {
161        match self.palette.iter_mut().find(|(n, _)| n == name) { Some(e) => e.1 = c, None => self.palette.push((name.into(), c)) }
162    }
163    /// An element's overrides (empty when none).
164    pub fn style(&self, e: Element) -> StyleSpec {
165        self.styles.iter().find(|(n, _)| n == e.name()).map(|(_, s)| s.clone()).unwrap_or_default()
166    }
167    /// Replace an element's overrides (removed when empty).
168    pub fn set_style(&mut self, e: Element, s: StyleSpec) {
169        self.styles.retain(|(n, _)| n != e.name());
170        if !s.is_empty() { self.styles.push((e.name().into(), s)); self.styles.sort_by_key(|(n, _)| Element::by_name(n).map(Element::index)); }
171    }
172    /// Follow `@` references to a colour; unknown or circular ones give `Color::Reset`.
173    pub fn resolve(&self, c: &ColorSpec) -> Color {
174        let mut cur = c;
175        for _ in 0..8 {
176            match cur {
177                ColorSpec::Value(v) => return *v,
178                ColorSpec::Ref(r) => match self.get(r) { Some(n) => cur = n, None => return Color::Reset },
179            }
180        }
181        Color::Reset
182    }
183    /// A palette entry resolved (`Color::Reset` when missing).
184    pub fn color(&self, name: &str) -> Color { self.get(name).map(|c| self.resolve(c)).unwrap_or(Color::Reset) }
185}

203/// A parsed theme file.
204#[derive(Clone, Debug, PartialEq, Eq)]
205pub struct ThemeFile {
206    /// Format version (1).
207    pub version: i64,
208    /// Theme name (the file stem).
209    pub name: String,
210    /// One line about it.
211    pub description: String,
212    /// The dark variant.
213    pub dark: Section,
214    /// The light variant (`None`: same as dark).
215    pub light: Option<Section>,
216}
217
218impl ThemeFile {
219    /// An empty theme (every colour the terminal's own).
220    pub fn empty(name: &str) -> Self {
221        ThemeFile { version: THEME_FORMAT_VERSION, name: name.into(), description: String::new(), dark: Section::default(), light: None }
222    }
223    /// The variant as it applies: `[light]` with missing entries taken from `[dark]`.
224    pub fn section(&self, v: Variant) -> Section {
225        match (v, &self.light) {
226            (Variant::Light, Some(l)) => {
227                let mut s = self.dark.clone();
228                for (n, c) in &l.palette { s.set(n, c.clone()); }
229                for (n, st) in &l.styles { if let Some(e) = Element::by_name(n) { s.set_style(e, st.clone()); } }
230                s
231            }
232            _ => self.dark.clone(),
233        }
234    }
235    /// The variant to edit (`[light]` is created from `[dark]` on first use).
236    pub fn section_mut(&mut self, v: Variant) -> &mut Section {
237        match v {
238            Variant::Dark => &mut self.dark,
239            Variant::Light => {
240                if self.light.is_none() { self.light = Some(self.dark.clone()); }
241                self.light.as_mut().expect("just set")
242            }
243        }
244    }
245}

247/// A theme name: 1–32 of `a-z`, `0-9`, `-`, `_` (so it can't leave the themes folder).
248pub fn valid_name(s: &str) -> bool {
249    !s.is_empty() && s.len() <= 32 && s.bytes().all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-' || b == b'_')
250}

263/// Parse a colour value (see the module docs).
264pub fn parse_color(v: &toml::Value) -> std::result::Result<ColorSpec, String> {
265    match v {
266        toml::Value::Integer(n) if (0..=255).contains(n) => Ok(ColorSpec::Value(Color::Indexed(*n as u8))),
267        toml::Value::Integer(n) => Err(format!("{n} is not 0–255")),
268        toml::Value::String(s) => parse_color_str(s),
269        other => Err(format!("expected a colour, found {}", other.type_str())),
270    }
271}

273/// Parse a colour written as text (also what the editor's hex field takes).
274pub fn parse_color_str(s: &str) -> std::result::Result<ColorSpec, String> {
275    let t = s.trim().to_ascii_lowercase().replace('-', "_");
276    if t == "reset" || t == "default" || t == "none" { return Ok(ColorSpec::Value(Color::Reset)); }
277    if let Some(r) = t.strip_prefix('@') {
278        return if valid_name(r) { Ok(ColorSpec::Ref(r.into())) } else { Err(format!("bad palette reference {s:?}")) };
279    }
280    if let Some(h) = t.strip_prefix('#') {
281        let ok = h.chars().all(|c| c.is_ascii_hexdigit());
282        let p = |x: &str| u8::from_str_radix(x, 16).ok();
283        return match (h.len(), ok) {
284            (6, true) => Ok(ColorSpec::Value(Color::Rgb(p(&h[0..2]).unwrap_or(0), p(&h[2..4]).unwrap_or(0), p(&h[4..6]).unwrap_or(0)))),
285            (3, true) => {
286                let d = |i: usize| p(&h[i..i + 1]).unwrap_or(0) * 17;
287                Ok(ColorSpec::Value(Color::Rgb(d(0), d(1), d(2))))
288            }
289            _ => Err(format!("bad hex colour {s:?} (use #rrggbb or #rgb)")),
290        };
291    }
292    if !t.is_empty() && t.chars().all(|c| c.is_ascii_digit()) {
293        return t.parse::<u8>().map(|n| ColorSpec::Value(Color::Indexed(n))).map_err(|_| format!("{s} is not 0–255"));
294    }
295    let key = match t.as_str() {
296        "gray" | "grey" => "white".to_string(),
297        "dark_gray" | "dark_grey" => "bright_black".into(),
298        "light_gray" | "light_grey" => "bright_white".into(),
299        x => x.strip_prefix("light_").map(|r| format!("bright_{r}")).unwrap_or_else(|| x.to_string()),
300    };
301    ANSI.iter().find(|(n, _)| *n == key).map(|(_, c)| ColorSpec::Value(*c))
302        .ok_or_else(|| format!("unknown colour {s:?} (use 0-255, #rrggbb, an ANSI name, reset or @name)"))
303}

305/// How a colour is written back.
306pub fn color_text(c: &ColorSpec) -> String {
307    match c {
308        ColorSpec::Ref(r) => format!("@{r}"),
309        ColorSpec::Value(v) => match v {
310            Color::Reset => "reset".into(),
311            Color::Rgb(r, g, b) => format!("#{r:02x}{g:02x}{b:02x}"),
312            Color::Indexed(n) => n.to_string(),
313            other => ANSI.iter().find(|(_, c)| c == other).map(|(n, _)| n.to_string()).unwrap_or_else(|| "reset".into()),
314        },
315    }
316}

318fn color_toml(c: &ColorSpec) -> String {
319    match c {
320        ColorSpec::Value(Color::Indexed(n)) => n.to_string(),
321        other => toml::Value::String(color_text(other)).to_string(),
322    }

325/// Parse a theme file. `stem` is the file name without `.toml` (the name when the
326/// file has none). Errors refuse the whole file; warnings name keys that were skipped.
327pub fn parse(src: &str, stem: &str) -> std::result::Result<(ThemeFile, Vec<String>), String> {
328    let table: toml::Table = toml::from_str(src).map_err(|e| format!("not valid TOML: {}", e.to_string().trim()))?;
329    let mut warn = Vec::new();
330    let version = match table.get("version") {
331        None => { warn.push(format!("no version; reading it as version {THEME_FORMAT_VERSION}")); THEME_FORMAT_VERSION }
332        Some(toml::Value::Integer(v)) if *v > THEME_FORMAT_VERSION => return Err(format!(
333            "format version {v} is newer than this acp-rs-scraper reads ({THEME_FORMAT_VERSION}); update acp-rs-scraper")),
334        Some(toml::Value::Integer(v)) if *v >= 1 => *v,
335        Some(v) => return Err(format!("version must be {THEME_FORMAT_VERSION}, found {v}")),
336    };
337    let text = |k: &str, w: &mut Vec<String>| match table.get(k) {
338        None => None,
339        Some(toml::Value::String(s)) => Some(s.clone()),
340        Some(v) => { w.push(format!("{k}: expected text, found {} (ignored)", v.type_str())); None }
341    };
342    let name = text("name", &mut warn).unwrap_or_else(|| stem.to_string());
343    let description = text("description", &mut warn).unwrap_or_default();
344    let sec = |k: &str, w: &mut Vec<String>| -> Option<Section> {
345        match table.get(k) {
346            None => None,
347            Some(toml::Value::Table(t)) => Some(parse_section(k, t, w)),
348            Some(v) => { w.push(format!("[{k}]: expected a table, found {} (ignored)", v.type_str())); None }
349        }
350    };
351    let dark = sec("dark", &mut warn);
352    let light = sec("light", &mut warn);
353    if dark.is_none() { warn.push("no [dark] section: using the terminal's colours".into()); }
354    for k in table.keys() {
355        if !["version", "name", "description", "dark", "light"].contains(&k.as_str()) { warn.push(format!("unknown key {k:?} (ignored)")); }
356    }
357    let f = ThemeFile { version, name, description, dark: dark.unwrap_or_default(), light };
358    for v in [Variant::Dark, Variant::Light] {
359        if v == Variant::Light && f.light.is_none() { continue; }
360        let s = f.section(v);
361        let refs = s.palette.iter().map(|(n, c)| (format!("palette.{n}"), c))
362            .chain(s.styles.iter().flat_map(|(n, st)| ["fg", "bg", "border"].into_iter().filter_map(move |k| st.color(k).map(|c| (format!("styles.{n}.{k}"), c)))));
363        for (at, c) in refs {
364            if let ColorSpec::Ref(r) = c && !is_reset_chain(&s, r) {
365                warn.push(format!("[{}.{at}]: @{r} is not in the palette (or loops); using the terminal's colour", v.name()));
366            }
367        }
368    }
369    Ok((f, warn))
370}

372/// Whether `@r` resolves through the palette to a value (possibly `reset`) without a loop.
373fn is_reset_chain(s: &Section, r: &str) -> bool {
374    let mut cur = r.to_string();
375    for _ in 0..8 {
376        match s.get(&cur) { Some(ColorSpec::Value(_)) => return true, Some(ColorSpec::Ref(n)) => cur = n.clone(), None => return false }
377    }
378    false
379}

381fn parse_section(k: &str, t: &toml::Table, w: &mut Vec<String>) -> Section {
382    let mut s = Section::default();
383    for (key, v) in t {
384        match (key.as_str(), v) {
385            ("palette", toml::Value::Table(p)) => for (n, c) in p {
386                if !valid_name(n) { w.push(format!("[{k}.palette] {n:?}: names are a-z, 0-9, - and _ (ignored)")); continue; }
387                match parse_color(c) {
388                    Ok(c) => s.palette.push((n.clone(), c)),
389                    Err(e) => w.push(format!("[{k}.palette] {n}: {e} (ignored)")),
390                }
391            },
392            ("styles", toml::Value::Table(st)) => for (n, v) in st {
393                let Some(e) = Element::by_name(n) else { w.push(format!("[{k}.styles] unknown element {n:?} (ignored)")); continue };
394                let toml::Value::Table(fields) = v else { w.push(format!("[{k}.styles.{n}]: expected a table (ignored)")); continue };
395                let mut spec = StyleSpec::default();
396                for (f, v) in fields {
397                    match (f.as_str(), v) {
398                        ("fg" | "bg" | "border", c) => match parse_color(c) {
399                            Ok(c) => spec.set_color(f, Some(c)),
400                            Err(er) => w.push(format!("[{k}.styles.{n}] {f}: {er} (ignored)")),
401                        },
402                        ("bold" | "italic" | "underline" | "reverse" | "dim", toml::Value::Boolean(b)) => spec.set_flag(f, Some(*b)),
403                        ("bold" | "italic" | "underline" | "reverse" | "dim", _) => w.push(format!("[{k}.styles.{n}] {f}: expected true or false (ignored)")),
404                        _ => w.push(format!("[{k}.styles.{n}] unknown key {f:?} (ignored)")),
405                    }
406                }
407                if spec.border.is_some() && !e.is_box() { w.push(format!("[{k}.styles.{n}] border: {n} has no border (ignored)")); spec.border = None; }
408                s.set_style(e, spec);
409            },
410            ("palette" | "styles", v) => w.push(format!("[{k}.{key}]: expected a table, found {} (ignored)", v.type_str())),
411            _ => w.push(format!("[{k}] unknown key {key:?} (ignored)")),
412        }
413    }
414    s
415}

417/// The file text for `f` (canonical order: core palette keys first).
418pub fn to_toml(f: &ThemeFile) -> String {
419    let q = |s: &str| toml::Value::String(s.into()).to_string();
420    let mut o = format!("# acp-rs-scraper theme, format v{THEME_FORMAT_VERSION}. See docs/USAGE.md (Themes).\nversion = {}\nname = {}\ndescription = {}\n",
421        f.version, q(&f.name), q(&f.description));
422    let mut sec = |k: &str, s: &Section| {
423        o.push_str(&format!("\n[{k}.palette]\n"));
424        let mut pal: Vec<&(String, ColorSpec)> = s.palette.iter().collect();
425        pal.sort_by_key(|(n, _)| PALETTE_KEYS.iter().position(|(p, _)| p == n).unwrap_or(usize::MAX));
426        for (n, c) in pal { o.push_str(&format!("{n} = {}\n", color_toml(c))); }
427        for (n, st) in &s.styles {
428            o.push_str(&format!("\n[{k}.styles.{n}]\n"));
429            for key in STYLE_KEYS {
430                if let Some(c) = st.color(key) { o.push_str(&format!("{key} = {}\n", color_toml(c))); }
431                if let Some(b) = st.flag(key) { o.push_str(&format!("{key} = {b}\n")); }
432            }
433        }
434    };
435    sec("dark", &f.dark);
436    if let Some(l) = &f.light { sec("light", l); }
437    o
438}

440/// WCAG contrast ratio of two colours (1–21); `None` when either is the terminal's own.
441pub fn contrast(a: Color, b: Color) -> Option<f64> {
442    let (la, lb) = (luminance(a)?, luminance(b)?);
443    let (hi, lo) = if la > lb { (la, lb) } else { (lb, la) };
444    Some((hi + 0.05) / (lo + 0.05))
445}

447/// sRGB of a colour (the xterm 256-colour palette for indexed ones).
448pub fn rgb(c: Color) -> Option<(u8, u8, u8)> {
449    const BASE: [(u8, u8, u8); 16] = [(0, 0, 0), (205, 0, 0), (0, 205, 0), (205, 205, 0), (0, 0, 238), (205, 0, 205), (0, 205, 205), (229, 229, 229),
450        (127, 127, 127), (255, 0, 0), (0, 255, 0), (255, 255, 0), (92, 92, 255), (255, 0, 255), (0, 255, 255), (255, 255, 255)];
451    let idx = |n: u8| -> (u8, u8, u8) {
452        match n {
453            0..=15 => BASE[n as usize],
454            16..=231 => { let i = n - 16; let s = |v: u8| if v == 0 { 0 } else { 55 + v * 40 }; (s(i / 36), s((i / 6) % 6), s(i % 6)) }
455            _ => { let g = 8 + (n - 232) * 10; (g, g, g) }
456        }
457    };
458    match c {
459        Color::Reset => None,
460        Color::Rgb(r, g, b) => Some((r, g, b)),
461        Color::Indexed(n) => Some(idx(n)),
462        other => ANSI.iter().position(|(_, a)| *a == other).map(|i| BASE[i]),
463    }
464}

466fn luminance(c: Color) -> Option<f64> {
467    let (r, g, b) = rgb(c)?;
468    let ch = |v: u8| { let s = v as f64 / 255.0; if s <= 0.03928 { s / 12.92 } else { ((s + 0.055) / 1.055).powf(2.4) } };
469    Some(0.2126 * ch(r) + 0.7152 * ch(g) + 0.0722 * ch(b))
