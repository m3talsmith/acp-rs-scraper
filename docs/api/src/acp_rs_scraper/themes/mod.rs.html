1//! Theme files: every theme, the default included, is `~/.acp/themes/<name>.toml`
2//! ([`format`] describes the file). Themes are read from that folder only: nothing
3//! is built into the binary or installed by the packages. The first run with an empty
4//! folder downloads `default.toml` from the repo ([`download`]). A missing or broken
5//! theme falls back to the terminal's own colours (`Color::Reset`) with a notice;
6//! broken files are never overwritten.

17/// `dir/<name>.toml`, refusing names that are not [`valid_name`] (no path tricks).
18pub fn theme_path(dir: &Path, name: &str) -> std::result::Result<PathBuf, String> {
19    if !valid_name(name) { return Err(format!("invalid theme name {name:?}: use 1–32 of a-z, 0-9, - and _")); }
20    Ok(dir.join(format!("{name}.toml")))
21}

23/// Why a theme could not be used.
24#[derive(Clone, Debug, PartialEq, Eq)]
25pub enum LoadError {
26    /// The name is not allowed.
27    BadName(String),
28    /// No such file.
29    Missing(PathBuf),
30    /// Unreadable, too big, not TOML, or a newer format.
31    Broken(PathBuf, String),
32}
33
34impl LoadError {
35    /// What went wrong and how to fix it, naming the file.
36    pub fn notice(&self) -> String {
37        match self {
38            LoadError::BadName(e) => e.clone(),
39            LoadError::Missing(p) => format!("theme file {} is missing; using the terminal's colours. Pick a theme with F5, \
40                create one in the theme editor (Shift-F5), or download the default with `acp-rs-scraper --theme-init` \
41                (Theme: init in the command palette).", p.display()),
42            LoadError::Broken(p, e) => format!("theme file {} can't be used: {e}. Using the terminal's colours; fix the file \
43                (or move it away and run `acp-rs-scraper --theme-init` to download the default again), or pick another theme with F5. It is never overwritten.", p.display()),
44        }
45    }
46}

48/// A theme read from disk.
49#[derive(Clone, Debug, PartialEq)]
50pub struct Loaded {
51    /// The file.
52    pub path: PathBuf,
53    /// Its contents.
54    pub file: ThemeFile,
55    /// Keys that were skipped.
56    pub warnings: Vec<String>,
57}
58
59/// Read and check one file (size limit, TOML, format version).
60pub fn load_path(path: &Path) -> std::result::Result<Loaded, LoadError> {
61    let stem = path.file_stem().map(|s| s.to_string_lossy().to_string()).unwrap_or_default();
62    let meta = match std::fs::metadata(path) {
63        Ok(m) => m,
64        Err(e) if e.kind() == io::ErrorKind::NotFound => return Err(LoadError::Missing(path.into())),
65        Err(e) => return Err(LoadError::Broken(path.into(), e.to_string())),
66    };
67    if meta.len() > THEME_MAX_BYTES { return Err(LoadError::Broken(path.into(), format!("it is {} bytes; theme files are at most {} KiB", meta.len(), THEME_MAX_BYTES / 1024))); }
68    let src = std::fs::read_to_string(path).map_err(|e| LoadError::Broken(path.into(), e.to_string()))?;
69    let (mut file, warnings) = parse(&src, &stem).map_err(|e| LoadError::Broken(path.into(), e))?;
70    file.name = stem;
71    Ok(Loaded { path: path.into(), file, warnings })
72}

74/// Read theme `name` from `dir`.
75pub fn load(dir: &Path, name: &str) -> std::result::Result<Loaded, LoadError> {
76    load_path(&theme_path(dir, name).map_err(LoadError::BadName)?)
77}

79/// One theme file, for the picker and `--list-themes`.
80#[derive(Clone, Debug, PartialEq, Eq)]
81pub struct Entry {
82    /// Name (the file stem).
83    pub name: String,
84    /// Its description.
85    pub description: String,
86    /// Why it can't be used, if so.
87    pub error: Option<String>,
88    /// How many keys were skipped.
89    pub warnings: usize,
90}

92/// Every `*.toml` in `dir` with a valid name, sorted by name.
93pub fn list(dir: &Path) -> Vec<Entry> {
94    let mut v: Vec<Entry> = std::fs::read_dir(dir).into_iter().flatten().flatten()
95        .filter_map(|e| {
96            let p = e.path();
97            let name = p.file_name()?.to_str()?.strip_suffix(".toml")?.to_string();
98            valid_name(&name).then(|| match load_path(&p) {
99                Ok(l) => Entry { name, description: l.file.description, error: None, warnings: l.warnings.len() },
100                Err(LoadError::Broken(_, e)) => Entry { name, description: String::new(), error: Some(e), warnings: 0 },
101                Err(e) => Entry { name, description: String::new(), error: Some(e.notice()), warnings: 0 },
102            })
103        }).collect();
104    v.sort_by(|a, b| a.name.cmp(&b.name));
105    v
106}

108/// Write `bytes` to `path` atomically (a temp file in the same folder, renamed over it), mode 0600.
109pub fn atomic_write(path: &Path, bytes: &[u8]) -> io::Result<()> {
110    let tmp = write_tmp(path, bytes)?;
111    std::fs::rename(&tmp, path).inspect_err(|_| { let _ = std::fs::remove_file(&tmp); })
112}
113
114/// Like [`atomic_write`] but never replaces an existing file; `Ok(false)` when one is there.
115pub fn atomic_write_new(path: &Path, bytes: &[u8]) -> io::Result<bool> {
116    if path.exists() { return Ok(false); }
117    let tmp = write_tmp(path, bytes)?;
118    // A hard link fails when the target exists, so a file that appeared meanwhile is kept.
119    let r = match std::fs::hard_link(&tmp, path) {
120        Ok(()) => Ok(true),
121        Err(e) if e.kind() == io::ErrorKind::AlreadyExists => Ok(false),
122        Err(_) if !path.exists() => std::fs::rename(&tmp, path).map(|_| true),
123        Err(e) => Err(e),
124    };
125    let _ = std::fs::remove_file(&tmp);
126    r
127}

129fn write_tmp(path: &Path, bytes: &[u8]) -> io::Result<PathBuf> {
130    let dir = path.parent().ok_or_else(|| io::Error::other("no parent folder"))?;
131    create_dir_private(dir)?;
132    let name = path.file_name().map(|n| n.to_string_lossy().to_string()).unwrap_or_default();
133    let tmp = dir.join(format!(".{name}.{}.tmp", std::process::id()));
134    let mut o = std::fs::OpenOptions::new();
135    o.write(true).create(true).truncate(true);
136    #[cfg(unix)]
137    { use std::os::unix::fs::OpenOptionsExt; o.mode(0o600); }
138    let mut f = o.open(&tmp)?;
139    f.write_all(bytes)?;
140    f.sync_all()?;
141    #[cfg(unix)]
142    { use std::os::unix::fs::PermissionsExt; std::fs::set_permissions(&tmp, std::fs::Permissions::from_mode(0o600))?; }
143    Ok(tmp)
144}
145
146fn create_dir_private(dir: &Path) -> io::Result<()> {
147    if dir.is_dir() { return Ok(()); }
148    std::fs::create_dir_all(dir)?;
149    #[cfg(unix)]
150    { use std::os::unix::fs::PermissionsExt; std::fs::set_permissions(dir, std::fs::Permissions::from_mode(0o700))?; }
151    Ok(())
152}

154/// Save `f` as `dir/<f.name>.toml` (atomic, 0600).
155pub fn save(dir: &Path, f: &ThemeFile) -> Result<PathBuf> {
156    let p = theme_path(dir, &f.name).map_err(|e| anyhow!(e))?;
157    atomic_write(&p, to_toml(f).as_bytes()).with_context(|| format!("writing {}", p.display()))?;
158    Ok(p)
159}

163    let p = theme_path(dir, name).map_err(|e| anyhow!(e))?;
164    std::fs::remove_file(&p).with_context(|| format!("deleting {}", p.display()))

167fn toml_files(dir: &Path) -> Vec<PathBuf> {
168    let mut v: Vec<PathBuf> = std::fs::read_dir(dir).into_iter().flatten().flatten().map(|e| e.path())
169        .filter(|p| p.extension().is_some_and(|x| x == "toml") && p.file_stem().and_then(|s| s.to_str()).is_some_and(valid_name))
170        .collect();
171    v.sort();
172    v
173}

175/// The warnings of one file as a single notice (`None` when there are none).
176pub fn warning_notice(path: &Path, warnings: &[String]) -> Option<String> {
177    match warnings {
178        [] => None,
179        [w] => Some(format!("warning: {}: {w}", path.display())),
180        ws => Some(format!("warning: {}: {} keys skipped: {}", path.display(), ws.len(), ws.join("; "))),
181    }

187/// Whether the old `[theme]` table of preferences.toml sets anything.

190/// The old `[theme]` colours as a theme: `base` (normally the default theme) with
191/// them applied to both variants. Invalid colours are reported and skipped.
192pub fn from_legacy(name: &str, legacy: &ThemePrefs, base: Option<&ThemeFile>) -> (ThemeFile, Vec<String>) {
193    let mut f = base.cloned().unwrap_or_else(|| ThemeFile::empty(name));
194    f.name = name.into();
195    f.description = "Converted from the [theme] colours in ~/.acp/preferences.toml".into();
196    let mut warn = Vec::new();
197    for (k, v) in [("accent", &legacy.accent), ("muted", &legacy.muted), ("border", &legacy.border), ("on_accent", &legacy.on_accent)] {
198        let Some(v) = v else { continue };
199        // The old table took `#rrggbb` or bare `rrggbb`.
200        let bare = v.len() == 6 && v.chars().all(|c| c.is_ascii_hexdigit());
201        match parse_color_str(&if bare { format!("#{v}") } else { v.clone() }) {
202            Ok(ColorSpec::Value(c)) => {
203                f.dark.set(k, ColorSpec::Value(c));
204                if let Some(l) = f.light.as_mut() { l.set(k, ColorSpec::Value(c)); }
205            }
206            _ => warn.push(format!("[theme] {k} = {v:?} is not a colour (skipped)")),
207        }
208    }
209    (f, warn)
210}

212/// Save the old `[theme]` colours as a new theme file in `dir`: [`MIGRATED_THEME`],
213/// or `custom-2`, `custom-3`… when that name is taken. Returns the name and warnings.
214pub fn migrate_legacy(dir: &Path, legacy: &ThemePrefs) -> Result<(String, Vec<String>)> {
215    let base = load(dir, DEFAULT_THEME).ok().map(|l| l.file);
216    for i in 1..100 {
217        let name = if i == 1 { MIGRATED_THEME.to_string() } else { format!("{MIGRATED_THEME}-{i}") };
218        let (f, warn) = from_legacy(&name, legacy, base.as_ref());
219        let p = theme_path(dir, &name).map_err(|e| anyhow!(e))?;
220        if atomic_write_new(&p, to_toml(&f).as_bytes()).with_context(|| format!("writing {}", p.display()))? { return Ok((name, warn)); }
221    }
222    bail!("no free name for the migrated theme in {}", dir.display())
223}

225/// Theme mode from `[ui] theme_mode`.
226#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
227pub enum Mode {
228    /// Always the dark variant.
229    Dark,
230    /// Always the light variant.
231    Light,
232    /// From the terminal's background (`COLORFGBG`, or asking the terminal); dark if unknown.
233    #[default]
234    System,
235}
236
237impl Mode {
238    /// `dark` / `light` / `system` (anything else: system).
239    pub fn parse(s: Option<&str>) -> Mode {
240        match s.map(|s| s.trim().to_ascii_lowercase()).as_deref() { Some("dark") => Mode::Dark, Some("light") => Mode::Light, _ => Mode::System }
241    }
242    /// The variant, given what the terminal's background looks like (`Some(true)`: dark).
243    pub fn variant(self, dark_bg: Option<bool>) -> Variant {
244        match self { Mode::Dark => Variant::Dark, Mode::Light => Variant::Light, Mode::System => if dark_bg == Some(false) { Variant::Light } else { Variant::Dark } }
245    }
246}

248/// `COLORFGBG` ("fg;bg", set by rxvt, Konsole and others): `Some(true)` for a dark background.
249pub fn colorfgbg_dark(v: Option<&str>) -> Option<bool> {
250    let bg: u8 = v?.rsplit(';').next()?.trim().parse().ok()?;
251    Some(!matches!(bg, 7 | 9..=15))
252}
253
254/// An OSC 11 reply (`ESC ] 11 ; rgb:rrrr/gggg/bbbb`): `Some(true)` for a dark background.
255pub fn osc11_dark(reply: &str) -> Option<bool> {
256    let i = reply.find("]11;rgb:")?;
257    let rest = &reply[i + 8..];
258    let mut it = rest.split(['/', '\x07', '\x1b']);
259    let mut ch = || -> Option<f64> {
260        let h = it.next()?;
261        let h = &h[..h.len().min(4)];
262        let v = u32::from_str_radix(h, 16).ok()? as f64;
263        Some(v / ((1u32 << (4 * h.len())) - 1) as f64)
264    };
265    let (r, g, b) = (ch()?, ch()?, ch()?);
266    Some(0.2126 * r + 0.7152 * g + 0.0722 * b < 0.5)
267}

269/// The startup work: turn an old `[theme]` table into a theme file once (when
270/// `[ui] theme` is unset). Returns the migrated theme's name (to select and save) and
271/// notices for the user. Downloading the default is separate ([`download`]).
272pub fn prepare(dir: &Path, prefs: &Prefs) -> (Option<String>, Vec<String>) {
273    let mut notes = Vec::new();
274    let ui_theme = prefs.ui.as_ref().and_then(|u| u.theme.clone());
275    let mut migrated = None;
276    if ui_theme.is_none() && let Some(old) = prefs.theme.as_ref().filter(|t| legacy_set(t)) {
277        match migrate_legacy(dir, old) {
278            Ok((name, warn)) => {
279                notes.push(format!("Your [theme] colours in ~/.acp/preferences.toml were converted into {} and selected; \
280                    the old [theme] keys are kept but no longer used (edit the theme with Shift-F5 instead).", dir.join(format!("{name}.toml")).display()));
281                if let Some(n) = warning_notice(&dir.join(format!("{name}.toml")), &warn) { notes.push(n); }
282                migrated = Some(name);
283            }
284            Err(e) => notes.push(format!("warning: could not convert the [theme] colours: {e:#}")),
285        }
286    }
287    (migrated, notes)
288}
