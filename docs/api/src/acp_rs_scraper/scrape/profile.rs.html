1//! Profile "Posts and replies" feed filter.
2//!
3//! Verified live on the logged-in site: the profile page has a Filter dialog
4//! (`<dialog id="js-configureFeed">`) with `<form phx-submit="configure-feed">`
5//! and radios `name="replies"` (`0` = Posts, the default; `1` = Posts and
6//! replies). Submitting patches the URL to `/rs/@{nick}?replies=1`, and the feed
7//! then also lists the nick's replies ("Replying to …"). The scraper loads that
8//! URL directly and pages it with `page-next`; if the URL has no feed, it falls
9//! back to pushing the `configure-feed` form on the profile LiveView.
10
11use crate::prelude::*;
12
13/// Fetch the nick's "Posts and replies" feed, paged to the end (or until a page
14/// is all known when `known` is given). Items are not yet deduplicated against
15/// the Posts feed; see [`merge_feed_items`].
16pub async fn fetch_replies_feed(
17    http: &reqwest::Client, jar: &Jar, nick: &str, profile_html: &str,
18    known: Option<&HashSet<String>>,
19) -> Result<Vec<Value>> {
20    let url = profile_replies_url(nick);
21    match fetch_feed(http, jar, &url, None, known, true).await {
22        Ok(feed) => {
23            verbose!("profile: posts and replies via {url}: {} items", feed.items.len());
24            Ok(feed.items)
25        }
26        Err(e) => {
27            verbose!("profile: {url} failed ({e:#}); falling back to {PROFILE_FILTER_EVENT}");
28            replies_via_form(jar, nick, profile_html, known).await
29        }
30    }
31}
32
33/// Fallback: submit the Filter form (`configure-feed`, `replies=1`) on the
34/// profile LiveView, read the re-rendered Feed props, then page as usual.
35async fn replies_via_form(
36    jar: &Jar, nick: &str, profile_html: &str, known: Option<&HashSet<String>>,
37) -> Result<Vec<Value>> {
38    let page = parse_lv(&profile_url(nick), profile_html)?;
39    let mut lv = Lv::connect(&page, jar).await?;
40    let r = lv.form(PROFILE_FILTER_EVENT, PROFILE_REPLIES_FIELD, "1", None).await?;
41    if r["status"] != "ok" { bail!("{PROFILE_FILTER_EVENT} failed: {r}"); }
42    let body = reply_body(&r);
43    let feed = if body["posts"].is_array() { body.clone() }
44        else { props(&lv.html(), FEED_PROPS_SEL).context("no forum/Feed props after configure-feed")? };
45    let mut items: Vec<Value> = feed["posts"].as_array().cloned().unwrap_or_default();
46    let at_end = feed["at_end"].as_bool().unwrap_or(true);
47    let cursor = feed["next_cursor"].as_str().map(str::to_string);
48    if known.is_some_and(|k| page_all_known(&items, k)) {
49        info!("feed: initial page all known — stopping (use --full to re-crawl)");
50    } else if !at_end && cursor.is_some() {
51        page_rest(&mut lv, &mut items, cursor, None, known, true).await?;
52    }
53    Ok(items)
54}
