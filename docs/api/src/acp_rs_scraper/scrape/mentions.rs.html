6/// Crawl the site feed (Latest or Trending) up to `limit` posts, confirm candidate mentions against full threads, and return the new posts and mention hits.
7pub async fn crawl_mentions(
8    http: &reqwest::Client, jar: &Jar, targets: &[MentionTarget],
9    limit: usize, seen: &mut HashSet<String>, known: &HashSet<String>, full: bool,
10    mode: FeedMode,
11) -> Result<(Vec<Post>, Vec<MentionHit>)> {
12    let allow_inc = mode.allow_incremental_stop() && !full;
13    let known_opt = if allow_inc { Some(known) } else { None };
14    let url = mode.site_feed_url();
15    info!("mentions feed: {url} (mode={mode})");
16    let feed = fetch_feed(http, jar, &url, Some(limit), known_opt, allow_inc).await?;
17    let mut posts = Vec::new();
18    let mut hits = Vec::new();
19    let mut checked = HashSet::new();
20    for it in &feed.items {
21        let id = s(it, "id");
22        let tid = { let t = s(it, "thread_id"); if t.is_empty() { id.clone() } else { t } };
23        let mut any = false;
24        for t in targets {
25            if !match_post(it, t).is_empty() { any = true; break; }
26        }
27        if !any { continue; }
28        if !checked.insert(tid.clone()) { continue; }
29        let (th, replies) = match fetch_thread(http, jar, &tid).await {
30            Ok(x) => x,
31            Err(e) => { err!("warning: mention thread {tid}: {e:#}"); continue; }
32        };
33        let root = th.get("root").cloned().unwrap_or(th.clone());
34        let mut thread_posts = vec![root];
35        thread_posts.extend(replies);
36        for tp in &thread_posts {
37            let pid = s(tp, "id");
38            if pid.is_empty() { continue; }
39            for t in targets {
40                for mt in match_post(tp, t) {
41                    hits.push(MentionHit {
42                        post_id: pid.clone(), mentioned_nick: t.nick.clone(), match_type: mt,
43                    });
44                }
45            }
46            if !seen.insert(pid.clone()) { continue; }
47            let parent = {
48                let thid = s(tp, "thread_id");
49                if thid.is_empty() || thid == pid { String::new() } else { thid }
50            };
51            let author = tp.pointer("/author/username").and_then(Value::as_str).unwrap_or("");
52            posts.push(to_post(tp, &parent, author));
53        }
54    }
55    Ok((posts, hits))
56}

59/// How a post mentions a nick.
60pub enum MatchType {
61    /// An @mention node pointing at the nick's user id or label.
62    Mention,
63    /// The bare nick as a whole word.
64    Nick,
65    /// The profile display name as a whole word (3+ characters).
66    DisplayName,
67}
68
69impl MatchType {
70    /// Database / export spelling of the match type.
71    pub fn as_str(&self) -> &'static str {
72        match self {
73            MatchType::Mention => "mention",
74            MatchType::Nick => "nick",
75            MatchType::DisplayName => "display_name",
76        }
77    }
78    /// Parse the database / export spelling of a match type.
79    pub fn parse(s: &str) -> Option<Self> {
80        match s {
81            "mention" => Some(Self::Mention),
82            "nick" => Some(Self::Nick),
83            "display_name" => Some(Self::DisplayName),
84            _ => None,
85        }
86    }
87}

100/// Who we're looking for. `user_id` makes ProseMirror mention nodes definitive.
101#[derive(Debug, Clone)]
102pub struct MentionTarget {
103    /// Nick (username, without `@`).
104    pub nick: String,
105    /// Site user id.
106    pub user_id: Option<String>,
107    /// Display name.
108    pub display_name: Option<String>,
109}

111/// Walk a ProseMirror doc for `mention` nodes whose attrs.id (user_id) or
112/// attrs.label (@username / username) match the target.
113pub fn mention_nodes(body: &Value, target: &MentionTarget, out: &mut HashSet<MatchType>) {
114    let ty = body.get("type").and_then(Value::as_str).unwrap_or("");
115    if ty == "mention" {
116        let id = body.pointer("/attrs/id").and_then(Value::as_str).unwrap_or("");
117        let label = body.pointer("/attrs/label").and_then(Value::as_str).unwrap_or("");
118        let label = label.trim_start_matches('@');
119        let by_id = target.user_id.as_deref().is_some_and(|u| !u.is_empty() && u == id);
120        let by_nick = !label.is_empty() && label.eq_ignore_ascii_case(&target.nick);
121        if by_id || by_nick {
122            out.insert(MatchType::Mention);
123        }
124    }
125    for key in ["content", "marks"] {
126        if let Some(kids) = body.get(key).and_then(Value::as_array) {
127            for k in kids { mention_nodes(k, target, out); }
128        }
129    }
130}

132/// Case-insensitive whole-word match of `needle` in `hay`. Word chars = ASCII
133/// alphanumeric + underscore (so "haz" does not match "hazard").
134pub fn word_match(hay: &str, needle: &str) -> bool {
135    if needle.is_empty() { return false; }
136    let h: Vec<char> = hay.chars().collect();
137    let n: Vec<char> = needle.chars().collect();
138    let nl = n.len();
139    if nl > h.len() { return false; }
140    let eq = |a: char, b: char| a.eq_ignore_ascii_case(&b);
141    let word = |c: char| c.is_ascii_alphanumeric() || c == '_';
142    for i in 0..=(h.len() - nl) {
143        if !(0..nl).all(|j| eq(h[i + j], n[j])) { continue; }
144        let before_ok = i == 0 || !word(h[i - 1]);
145        let after_ok = i + nl == h.len() || !word(h[i + nl]);
146        if before_ok && after_ok { return true; }
147    }
148    false
149}

151/// Flatten ProseMirror text (same idea as main::prosemirror_text) for local matching.
152pub fn flatten_text(v: &Value, out: &mut String) {
153    if let Some(t) = v.get("text").and_then(Value::as_str) { out.push_str(t); }
154    let ty = v.get("type").and_then(Value::as_str).unwrap_or("");
155    if matches!(ty, "hard_break" | "hardBreak") { out.push('\n'); }
156    if let Some(kids) = v.get("content").and_then(Value::as_array) {
157        for k in kids { flatten_text(k, out); }
158    }
159    if matches!(ty, "paragraph" | "heading" | "blockquote" | "list_item" | "listItem" | "code_block" | "codeBlock") {
160        out.push('\n');
161    }
162}

164/// A post's ProseMirror body as JSON (the site sends it as an object or a JSON string).
165pub fn body_of(v: &Value) -> Value {
166    match v.get("body") {
167        Some(Value::String(raw)) => serde_json::from_str(raw).unwrap_or(Value::Null),
168        Some(b) => b.clone(),
169        None => Value::Null,
170    }
171}

173/// All match types that apply to this post JSON for `target`.
174pub fn match_post(v: &Value, target: &MentionTarget) -> HashSet<MatchType> {
175    let mut out = HashSet::new();
176    let body = body_of(v);
177    mention_nodes(&body, target, &mut out);
178
179    let mut text = String::new();
180    flatten_text(&body, &mut text);
181    if text.trim().is_empty() {
182        if let Some(s) = v.get("body_text_summary").and_then(Value::as_str) { text = s.to_string(); }
183        if let Some(s) = v.get("title").and_then(Value::as_str) {
184            text.push(' ');
185            text.push_str(s);
186        }
187        if let Some(s) = v.get("content").and_then(Value::as_str) {
188            text.push(' ');
189            text.push_str(s);
190        }
191    } else if let Some(s) = v.get("title").and_then(Value::as_str) {
192        text.push(' ');
193        text.push_str(s);
194    }
195
196    if word_match(&text, &target.nick) {
197        out.insert(MatchType::Nick);
198    }
199    if let Some(dn) = &target.display_name {
200        let dn = dn.trim();
201        // Avoid matching very short / empty display names (common-name hazard).
202        if dn.chars().count() >= 3 && word_match(&text, dn) {
203            out.insert(MatchType::DisplayName);
204        }
205    }
206    out
207}

246/// Pull whatever profile-like fields we can find in a JSON blob (feed author,
247/// profile props, show-info reply, etc.). Prefers the first non-empty value.
248pub fn enrich_profile(p: &mut Profile, v: &Value) {
249    let g = |keys: &[&str]| -> Option<String> {
250        for k in keys {
251            if let Some(s) = v.pointer(k).and_then(Value::as_str).filter(|s| !s.is_empty()) {
252                return Some(s.to_string());
253            }
254        }
255        None
256    };
257    let n = |keys: &[&str]| -> Option<i64> {
258        for k in keys {
259            if let Some(x) = v.pointer(k).and_then(Value::as_i64) { return Some(x); }
260            if let Some(x) = v.pointer(k).and_then(Value::as_f64) { return Some(x as i64); }
261        }
262        None
263    };
264    if p.user_id.is_none() {
265        p.user_id = g(&["/user_id", "/id", "/author/user_id", "/user/id", "/profile/user_id"]);
266    }
267    if p.display_name.is_none() {
268        p.display_name = g(&["/display_name", "/author/display_name", "/user/display_name", "/profile/display_name", "/name"]);
269    }
270    let un = g(&["/username", "/author/username", "/user/username", "/profile/username", "/nick"]);
271    if let Some(u) = un.filter(|u| !u.is_empty()) { p.nick = u.trim_start_matches('@').to_string(); }
272    if p.bio.is_none() { p.bio = g(&["/bio", "/about", "/profile/bio", "/description"]); }
273    if p.avatar_url.is_none() {
274        p.avatar_url = g(&["/profile_photo_url", "/avatar_url", "/avatar", "/author/profile_photo_url", "/user/profile_photo_url", "/image"]);
275    }
276    if p.banner_url.is_none() {
277        p.banner_url = g(&["/banner_url", "/banner", "/cover_url", "/cover"]);
278    }
279    if p.location.is_none() {
280        p.location = g(&["/location", "/author_location", "/chapter", "/author/location"]);
281    }
282    if p.role.is_none() {
283        p.role = g(&["/role", "/author_role", "/author/role"]);
284    }
285    if p.joined_at.is_none() {
286        p.joined_at = g(&["/joined_at", "/inserted_at", "/created_at", "/member_since"]);
287    }
288    if p.followers.is_none() { p.followers = n(&["/followers_count", "/followers", "/stats/followers"]); }
289    if p.following.is_none() { p.following = n(&["/following_count", "/following", "/stats/following"]); }
290    if p.posts_count.is_none() { p.posts_count = n(&["/posts_count", "/post_count", "/stats/posts"]); }
291}

293/// Harvest any author-like objects from a feed items list into `p`.
294pub fn enrich_from_feed(p: &mut Profile, items: &[Value], nick: &str) {
295    for it in items {
296        let author = it.get("author").unwrap_or(it);
297        let un = author.get("username").and_then(Value::as_str).unwrap_or("");
298        if un.eq_ignore_ascii_case(nick) || un.is_empty() {
299            enrich_profile(p, author);
300            enrich_profile(p, it);
301            if p.user_id.is_some() && p.display_name.is_some() { break; }
302        }
303    }
304}

306/// Walk an HTML document looking for Svelte `data-props` JSON that looks like a profile.
307pub fn enrich_from_html(p: &mut Profile, html: &str) {
308    // Any data-props blob that mentions username / display_name.
309    for cap in regex::Regex::new(r#"data-props="([^"]+)""#).unwrap().captures_iter(html) {
310        let raw = html_unescape(&cap[1]);
311        if let Ok(v) = serde_json::from_str::<Value>(&raw)
312            && (v.get("username").is_some() || v.get("display_name").is_some()
313                || v.pointer("/user/username").is_some() || v.pointer("/profile/username").is_some())
314            {
315                enrich_profile(p, &v);
316            }
317    }
318    // og: tags as a last resort.
319    if p.display_name.is_none()
320        && let Some(t) = regex::Regex::new(r#"property="og:title"\s+content="([^"]+)""#)
321            .unwrap().captures(html).or_else(|| regex::Regex::new(r#"content="([^"]+)"\s+property="og:title""#).unwrap().captures(html))
322        {
323            let t = html_unescape(&t[1]);
324            // "Display Name (@nick)" or similar.
325            if let Some((dn, _)) = t.split_once("(@") {
326                p.display_name = Some(dn.trim().to_string());
327            } else if !t.is_empty() {
328                p.display_name = Some(t);
329            }
330        }
331    if p.bio.is_none()
332        && let Some(t) = regex::Regex::new(r#"property="og:description"\s+content="([^"]+)""#)
333            .unwrap().captures(html).or_else(|| regex::Regex::new(r#"content="([^"]+)"\s+property="og:description""#).unwrap().captures(html))
334        {
335            p.bio = Some(html_unescape(&t[1]));
336        }
337    if p.avatar_url.is_none()
338        && let Some(t) = regex::Regex::new(r#"property="og:image"\s+content="([^"]+)""#)
339            .unwrap().captures(html).or_else(|| regex::Regex::new(r#"content="([^"]+)"\s+property="og:image""#).unwrap().captures(html))
340        {
341            p.avatar_url = Some(html_unescape(&t[1]));
342        }
343}

345/// Undo the HTML entity escaping used in `data-props` attributes.
346pub fn html_unescape(s: &str) -> String {
347    s.replace("&quot;", "\"").replace("&amp;", "&").replace("&lt;", "<").replace("&gt;", ">").replace("&#39;", "'")
348}
