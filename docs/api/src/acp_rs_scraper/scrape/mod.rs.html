12/// One stored post (the `posts` table row), plus the media it references.
13pub struct Post {
14    /// Site post id (22-character base64url).
15    pub post_id: String,
16    /// Thread root id for replies; `""` for root posts.
17    pub parent_post_id: String,
18    /// Nick (username, without `@`).
19    pub nick: String,
20    /// Title (`""` for replies).
21    pub title: String,
22    /// Body flattened to plain text.
23    pub content: String,
24    /// `inserted_at` timestamp (ISO 8601).
25    pub posted_at: String,
26    /// Media referenced by the post (url, kind).
27    pub media: Vec<MediaRef>,
28}

30/// Flatten a ProseMirror doc (`{type, text?, content:[...]}`) to plain text.
31pub fn prosemirror_text(v: &Value, out: &mut String) {
32    if let Some(t) = v.get("text").and_then(Value::as_str) {
33        out.push_str(t);
34    }
35    let ty = v.get("type").and_then(Value::as_str).unwrap_or("");
36    if ty == "hard_break" || ty == "hardBreak" {
37        out.push('\n');
38    }
39    if let Some(kids) = v.get("content").and_then(Value::as_array) {
40        for k in kids {
41            prosemirror_text(k, out);
42        }
43    }
44    if matches!(ty, "paragraph" | "heading" | "blockquote" | "list_item" | "listItem" | "code_block" | "codeBlock") {
45        out.push('\n');
46    }
47}

49/// String field `k` of a JSON object, or `""`.
50pub fn s(v: &Value, k: &str) -> String {
51    v.get(k).and_then(Value::as_str).unwrap_or_default().to_string()
52}

54/// Convert site post JSON to a [`Post`] with the given parent (`""` for roots), flattening the body to text.
55pub fn to_post(v: &Value, parent: &str, fallback_nick: &str) -> Post {
56    let mut content = String::new();
57    if let Some(b) = v.get("body") {
58        let b = match b.as_str() {
59            Some(raw) => serde_json::from_str(raw).unwrap_or(Value::Null),
60            None => b.clone(),
61        };
62        prosemirror_text(&b, &mut content);
63    }
64    let mut content = content.trim().to_string();
65    if content.is_empty() {
66        content = s(v, "body_text_summary");
67    }
68    let nick = v.pointer("/author/username").and_then(Value::as_str)
69        .map(str::to_string).unwrap_or_else(|| fallback_nick.to_string());
70    Post {
71        post_id: s(v, "id"),
72        parent_post_id: parent.to_string(),
73        nick,
74        title: s(v, "title"),
75        content,
76        posted_at: s(v, "inserted_at"),
77        media: { let mut m = Vec::new(); collect_media(v, &mut m); m },
78    }
79}

81/// Scrape a nick: their feed (root posts + their replies elsewhere) and all
82/// replies in their own threads. Also returns the profile page HTML.
83#[derive(Default)]
84pub struct ScrapeStats {
85    /// Items on the Posts feed plus the "Posts and replies" filter.
86    pub feed_items: usize,
87    /// Items not stored before (approximate).
88    pub new_posts: usize,
89    /// Items skipped as already stored.
90    pub skipped_known: usize,
91    /// Threads fetched for replies.
92    pub threads_fetched: usize,
93    /// Threads skipped as unchanged.
94    pub threads_skipped: usize,
95    /// Items only found via the profile's "Posts and replies" filter.
96    pub replies_tab_items: usize,
97}
98
99/// Scrape a nick: the Posts feed and the "Posts and replies" filter (merged by id), then replies in every thread the nick started that changed since the last run. Returns the posts, the profile HTML and counts.
100pub async fn scrape(
101    http: &reqwest::Client, jar: &Jar, nick: &str,
102    known: &HashSet<String>, reply_counts: &HashMap<String, usize>, full: bool,
103) -> Result<(Vec<Post>, String, ScrapeStats)> {
104    let nick = nick.trim_start_matches('@');
105    let known_opt = if full { None } else { Some(known) };
106    // Profile feed is always Latest (/rs/@nick); Trending does not apply to profiles.
107    let profile_url = profile_url(nick);
108    let mut feed = fetch_feed(http, jar, &profile_url, None, known_opt, true).await?;
109    // Also walk the "Posts and replies" filter (?replies=1) (the nick's replies in other threads).
110    let mut replies_tab_added = 0;
111    match fetch_replies_feed(http, jar, nick, &feed.html, known_opt).await {
112        Ok(extra) => {
113            replies_tab_added = merge_feed_items(&mut feed.items, extra);
114            info!("@{nick}: posts and replies added {replies_tab_added} items");
115        }
116        Err(e) => err!("warning: posts and replies feed: {e:#}"),
117    }
118    let mut seen = HashSet::new();
119    let mut out = Vec::new();
120    let mut roots: Vec<(String, usize)> = Vec::new(); // (id, replies_count from feed)
121    let mut stats = ScrapeStats { feed_items: feed.items.len(), replies_tab_items: replies_tab_added, ..Default::default() };
122    for it in &feed.items {
123        let id = s(it, "id");
124        if id.is_empty() || !seen.insert(id.clone()) { continue; }
125        let thread = s(it, "thread_id");
126        let is_root = thread.is_empty() || thread == id;
127        let already = known.contains(&id);
128        if already && !full {
129            stats.skipped_known += 1;
130        } else {
131            stats.new_posts += 1;
132        }
133        if is_root {
134            let rc = it.get("replies_count").and_then(Value::as_u64).unwrap_or(0) as usize;
135            roots.push((id.clone(), rc));
136            if !already || full {
137                out.push(to_post(it, "", nick));
138            }
139        } else if !already || full {
140            out.push(to_post(it, &thread, nick));
141        }
142    }
143    for (tid, feed_rc) in roots {
144        let db_rc = reply_counts.get(&tid).copied().unwrap_or(0);
145        let need = full || !known.contains(&tid) || feed_rc > db_rc;
146        if !need {
147            stats.threads_skipped += 1;
148            continue;
149        }
150        stats.threads_fetched += 1;
151        let (_, replies) = match fetch_thread(http, jar, &tid).await {
152            Ok(x) => x,
153            Err(e) => { err!("warning: {e:#}"); continue; }
154        };
155        let mut n = 0;
156        for rp in &replies {
157            let id = s(rp, "id");
158            if id.is_empty() || id == tid || !seen.insert(id.clone()) { continue; }
159            if known.contains(&id) && !full {
160                stats.skipped_known += 1;
161                continue;
162            }
163            out.push(to_post(rp, &tid, nick));
164            stats.new_posts += 1;
165            n += 1;
166        }
167        verbose!("thread {tid}: {n} new replies (feed says {feed_rc}, db had {db_rc})");
168    }
169    Ok((out, feed.html, stats))
170}
