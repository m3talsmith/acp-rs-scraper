1//! Backfill for bodies stored truncated (`--refetch-truncated`).
2//!
3//! Profile feeds send long posts and comments shortened: the item has
4//! `body_truncated: true`, a cut `body` doc and a `body_text_summary`. The
5//! thread page (`/rs/post/{thread_id}`, what "show more" opens) renders the
6//! root and every reply in full, so one GET per thread completes all of its
7//! stored rows. Rows stored before 1.26.0 have `body_state = 'unknown'`
8//! and are checked the same way.
9
10use crate::prelude::*;
11
12/// Counts from [`refetch_truncated`].
13#[derive(Debug, Default, Clone, PartialEq, Eq)]
14pub struct RefetchStats {
15    /// Rows that were candidates (`body_state` not `full`).
16    pub candidates: usize,
17    /// Thread pages fetched.
18    pub threads: usize,
19    /// Rows whose stored text was shorter than the full body and was replaced.
20    pub fixed: usize,
21    /// Rows already complete (now marked `full`).
22    pub already_full: usize,
23    /// Rows not found on their thread page (deleted, or thread failed).
24    pub missing: usize,
25}
26
27/// A stored row waiting for its full body.
28#[derive(Debug, Clone)]
29pub struct Pending {
30    /// Site post id.
31    pub post_id: String,
32    /// Thread root id (`post_id` for roots).
33    pub thread_id: String,
34    /// Stored text.
35    pub content: String,
36}
37
38/// Rows whose body is not known to be complete, optionally only for `nicks`
39/// (case-insensitive). Ordered by thread so each thread is fetched once.
40pub fn pending_bodies(db: &Path, nicks: &[String]) -> Result<Vec<Pending>> {
41    if !db.exists() { return Ok(Vec::new()); }
42    let c = Connection::open(db)?;
43    ensure_posts_schema(&c)?;
44    let want: HashSet<String> = nicks.iter().map(|n| n.trim_start_matches('@').to_lowercase()).collect();
45    let mut st = c.prepare(
46        "SELECT post_id, CASE WHEN parent_post_id = '' THEN post_id ELSE parent_post_id END, content, nick
47         FROM posts WHERE body_state <> 'full' ORDER BY 2, posted_at")?;
48    let rows = st.query_map([], |r| Ok((Pending { post_id: r.get(0)?, thread_id: r.get(1)?, content: r.get(2)? }, r.get::<_, String>(3)?)))?;
49    let mut out = Vec::new();
50    for row in rows {
51        let (p, nick) = row?;
52        if want.is_empty() || want.contains(&nick.to_lowercase()) { out.push(p); }
53    }
54    Ok(out)
55}
56
57/// Decide what to store for a row given its full thread-page item: `Some(text)`
58/// when the full body differs from the stored one, `None` when it already matches.
59/// Whitespace-only differences (old single-newline paragraph joins) count as
60/// a change too, so paragraph breaks are restored.
61pub fn completed_body(stored: &str, full: &str) -> Option<String> {
62    (!full.is_empty() && full != stored).then(|| full.to_string())
63}
64
65/// Fetch the thread page of every pending row (sequentially, pausing
66/// [`THREAD_FETCH_PAUSE_MS`] between threads) and store the full bodies.
67pub async fn refetch_truncated(http: &reqwest::Client, jar: &Jar, db: &Path, nicks: &[String]) -> Result<RefetchStats> {
68    let pending = pending_bodies(db, nicks)?;
69    let mut stats = RefetchStats { candidates: pending.len(), ..Default::default() };
70    let mut by_thread: Vec<(String, Vec<Pending>)> = Vec::new();
71    for p in pending {
72        match by_thread.last_mut() {
73            Some((t, v)) if *t == p.thread_id => v.push(p),
74            _ => by_thread.push((p.thread_id.clone(), vec![p])),
75        }
76    }
77    info!("refetch: {} rows to check in {} threads", stats.candidates, by_thread.len());
78    let c = Connection::open(db)?;
79    for (i, (tid, rows)) in by_thread.iter().enumerate() {
80        if i > 0 { polite_pause().await; }
81        stats.threads += 1;
82        let items = match fetch_thread(http, jar, tid).await {
83            Ok((th, mut r)) => { if let Some(root) = th.get("root") { r.push(root.clone()); } r }
84            Err(e) => { err!("warning: refetch thread {tid}: {e:#}"); stats.missing += rows.len(); continue; }
85        };
86        for row in rows {
87            let Some(item) = items.iter().find(|x| s(x, "id") == row.post_id) else {
88                verbose!("refetch: {} not on thread {tid}", row.post_id);
89                stats.missing += 1;
90                continue;
91            };
92            let parent = if row.thread_id == row.post_id { "" } else { row.thread_id.as_str() };
93            let fp = to_post(item, parent, "");
94            if fp.truncated { stats.missing += 1; continue; }
95            match completed_body(&row.content, &fp.content) {
96                Some(text) => {
97                    c.execute("UPDATE posts SET content = ?1, body_state = 'full' WHERE post_id = ?2", params![text, row.post_id])?;
98                    stats.fixed += 1;
99                }
100                None => {
101                    c.execute("UPDATE posts SET body_state = 'full' WHERE post_id = ?1", params![row.post_id])?;
102                    stats.already_full += 1;
103                }
104            }
105        }
106        if (i + 1) % 25 == 0 { info!("refetch: {}/{} threads", i + 1, by_thread.len()); }
107    }
108    info!("refetch: {} rows checked, {} fixed, {} already full, {} not found ({} threads)",
109        stats.candidates, stats.fixed, stats.already_full, stats.missing, stats.threads);
110    Ok(stats)
111}
