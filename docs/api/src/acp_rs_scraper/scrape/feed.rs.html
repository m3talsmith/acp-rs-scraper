5/// One feed (profile, /rs, tag...) fully paged via `page-next {cursor}`.
6pub struct FeedPage {
7    /// Feed posts in order, deduplicated by id.
8    pub items: Vec<Value>,
9    /// HTML of the first page (used for profile parsing).
10    pub html: String,
11    /// Paging stopped early at an all-known page.
12    pub stopped_incremental: bool,
13    /// Pages fetched.
14    pub pages: u32,
15}
16
17/// True when every id in `page` is already known (and the page is non-empty).
18pub fn page_all_known(page: &[Value], known: &HashSet<String>) -> bool {
19    !page.is_empty() && page.iter().all(|x| {
20        let id = s(x, "id");
21        !id.is_empty() && known.contains(&id)
22    })
23}

25/// Nick from a `/rs/@{nick}` URL path, if it is a profile URL.
26pub fn profile_nick_of(url: &str) -> Option<String> {
27    let u = Url::parse(url).ok()?;
28    let rest = u.path().strip_prefix("/rs/@")?;
29    let nick = rest.split('/').next().unwrap_or_default();
30    (!nick.is_empty()).then(|| nick.to_string())
31}
32
33/// Fail when a profile URL (`/rs/@nick`) did not land on that profile.
34/// ACP answers an unknown nick with a 302 to `/rs` (Trending) and a flash
35/// "No profile exists with the username @nick"; reqwest follows redirects,
36/// so without this check the Trending feed would be scraped as the profile.
37pub fn check_profile_landing(requested: &str, final_url: &str, html: &str) -> Result<()> {
38    let Some(want) = profile_nick_of(requested) else { return Ok(()) };
39    let got = profile_nick_of(final_url);
40    if !got.as_deref().is_some_and(|g| g.eq_ignore_ascii_case(&want)) {
41        let to = Url::parse(final_url).map(|u| u.path().to_string()).unwrap_or_else(|_| final_url.to_string());
42        bail!("no profile @{want}: {requested} redirected to {to} (no such user? check the nick)");
43    }
44    if html.contains("No profile exists with the username") {
45        bail!("no profile @{want}: site says \"No profile exists with the username @{want}\"");
46    }
47    Ok(())
48}

50/// Fetch a feed page (profile, site or filter URL), then page it to the end with `page-next`, honouring `limit` and incremental early-stop against `known`.
51pub async fn fetch_feed(
52    http: &reqwest::Client, jar: &Jar, url: &str, limit: Option<usize>,
53    known: Option<&HashSet<String>>,
54    allow_incremental: bool, // false for Trending: no early stop on all-known pages
55) -> Result<FeedPage> {
56    let resp = http.get(url).send().await?.error_for_status()?;
57    let final_url = resp.url().to_string();
58    let html = resp.text().await?;
59    check_profile_landing(url, &final_url, &html)?;
60    let feed = props(&html, FEED_PROPS_SEL).with_context(|| format!("no forum/Feed data-props on {url}"))?;
61    let mut items: Vec<Value> = feed["posts"].as_array().cloned().unwrap_or_default();
62    let mut at_end = feed["at_end"].as_bool().unwrap_or(true);
63    let cursor = feed["next_cursor"].as_str().map(str::to_string);
64    let hit_limit = |n: usize| limit.is_some_and(|l| n >= l);
65    let mut stopped_incremental = false;
66    let mut pages: u32 = 1;
67    info!("feed {url}: {} posts from initial page", items.len());
68    if allow_incremental
69        && let Some(k) = known
70        && page_all_known(&items, k) {
71            info!("feed: initial page all known — stopping (use --full to re-crawl)");
72            stopped_incremental = true;
73            at_end = true;
74        }
75
76    if !at_end && cursor.is_some() && !hit_limit(items.len()) && !stopped_incremental {
77        let page = parse_lv(&final_url, &html)?;
78        let mut lv = Lv::connect(&page, jar).await?;
79        let (stop, extra) = page_rest(&mut lv, &mut items, cursor, limit, known, allow_incremental).await?;
80        stopped_incremental = stop;
81        pages += extra;
82    }
83    if let Some(l) = limit { items.truncate(l); }
84    Ok(FeedPage { items, html, stopped_incremental, pages })
85}

87/// Page a feed that is open on `lv` to its end with `page-next {cursor}`,
88/// appending new (id-deduped) posts to `items`. Returns
89/// `(stopped_incremental, extra_pages)`. Shared by profile, site and
90/// "Posts and replies" feeds so incremental/limit rules are identical.
91pub async fn page_rest(
92    lv: &mut Lv, items: &mut Vec<Value>, mut cursor: Option<String>, limit: Option<usize>,
93    known: Option<&HashSet<String>>, allow_incremental: bool,
94) -> Result<(bool, u32)> {
95    let hit_limit = |n: usize| limit.is_some_and(|l| n >= l);
96    let mut ids: HashSet<String> = items.iter().map(|x| s(x, "id")).collect();
97    let mut stale = 0;
98    let mut pages = 0;
99    let mut at_end = false;
100    let mut stopped_incremental = false;
101    for _ in 0..MAX_PAGES {
102        let Some(c) = cursor.clone() else { at_end = true; break };
103        let r = lv.push("event", json!({"type": "hook", "event": FEED_NEXT_EVENT, "value": {"cursor": c}})).await?;
104        if r["status"] != "ok" { err!("warning: page-next failed: {r}"); break; }
105        let body = reply_body(&r);
106        let body = if body["posts"].is_array() { body.clone() }
107            else { props(&lv.html(), FEED_PROPS_SEL).unwrap_or(Value::Null) };
108        let ps = body["posts"].as_array().cloned().unwrap_or_default();
109        pages += 1;
110        if allow_incremental
111            && let Some(k) = known
112            && page_all_known(&ps, k) {
113                info!("feed: page {} all known — stopping (use --full to re-crawl)", pages + 1);
114                stopped_incremental = true;
115                break;
116            }
117        let new: Vec<Value> = ps.into_iter().filter(|x| ids.insert(s(x, "id"))).collect();
118        verbose!("feed: +{} new posts", new.len());
119        items.extend(new.iter().cloned());
120        at_end = body["at_end"].as_bool().unwrap_or(new.is_empty());
121        cursor = body["next_cursor"].as_str().map(str::to_string).or(cursor);
122        if new.is_empty() { stale += 1; } else { stale = 0; }
123        if hit_limit(items.len()) { info!("feed: reached limit {}", limit.unwrap_or(0)); break; }
124        if at_end || stale >= 2 || (new.is_empty() && cursor.as_deref() == Some(c.as_str())) { break; }
125    }
126    if !at_end && !hit_limit(items.len()) && !stopped_incremental {
127        err!("warning: feed paging stopped before at_end (run with --debug)");
128    }
129    Ok((stopped_incremental, pages))
130}

132/// Append `extra` feed items whose `id` isn't already in `items`.
133/// Returns how many were added.
134pub fn merge_feed_items(items: &mut Vec<Value>, extra: Vec<Value>) -> usize {
135    let mut ids: HashSet<String> = items.iter().map(|x| s(x, "id")).collect();
136    let before = items.len();
137    items.extend(extra.into_iter().filter(|x| { let id = s(x, "id"); !id.is_empty() && ids.insert(id) }));
138    items.len() - before
139}
