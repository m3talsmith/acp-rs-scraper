5/// Collect every object having an `id` and `inserted_at` from any `replies`
6/// array in a JSON value, including JSON hidden in strings (diffs / data-props).
7pub fn harvest_replies(v: &Value, out: &mut Vec<Value>) {
8    match v {
9        Value::Object(m) => {
10            if let Some(Value::Array(rs)) = m.get("replies") {
11                out.extend(rs.iter().filter(|r| r.get("inserted_at").is_some()).cloned());
12            }
13            m.values().for_each(|x| harvest_replies(x, out));
14        }
15        Value::Array(a) => a.iter().for_each(|x| harvest_replies(x, out)),
16        Value::String(st) if st.contains("replies") => {
17            let st = st.replace("&quot;", "\"").replace("&amp;", "&");
18            if let Some(i) = st.find('{')
19                && let Ok(j) = serde_json::from_str::<Value>(&st[i..st.rfind('}').unwrap_or(i) + 1]) {
20                    harvest_replies(&j, out);
21                }
22        }
23        _ => {}
24    }
25}

27/// Thread page props plus all replies (paged via the InfiniteScroll hook).
28pub async fn fetch_thread(http: &reqwest::Client, jar: &Jar, tid: &str) -> Result<(Value, Vec<Value>)> {
29    let url = thread_url(tid);
30    let resp = http.get(&url).send().await?;
31    let final_url = resp.url().to_string();
32    let html = resp.text().await?;
33    let th = props(&html, THREAD_PROPS_SEL).with_context(|| format!("no forum/Thread props for {tid}"))?;
34    let mut replies: Vec<Value> = th["replies"].as_array().cloned().unwrap_or_default();
35    // Thread.js has no load-more event; more replies come from the
36    // InfiniteScroll hook (`pushEventTo(this.el, "next-page", {})`),
37    // whose target cid is the element's phx-target (none = root view).
38    // New replies arrive as a diff re-rendering forum/Thread data-props.
39    let (hook, cid) = {
40        let doc = Html::parse_document(&html);
41        let sel = Selector::parse("[phx-hook=InfiniteScroll],[data-phx-hook=InfiniteScroll]").unwrap();
42        let el = doc.select(&sel).next();
43        (el.is_some(), el.and_then(|e| e.value().attr("phx-target").and_then(|t| t.parse::<u64>().ok())))
44    };
45    let total = th.pointer("/root/replies_count").and_then(Value::as_u64).unwrap_or(0) as usize;
46    if (hook || total > replies.len())
47        && let Ok(page) = parse_lv(&final_url, &html)
48        && let Ok(mut lv) = Lv::connect(&page, jar).await
49    {
50        let mut ids: HashSet<String> = replies.iter().map(|x| s(x, "id")).collect();
51        for _ in 0..MAX_PAGES {
52            let mut ev = json!({"type": "hook", "event": THREAD_NEXT_EVENT, "value": {}});
53            if let Some(c) = cid { ev["cid"] = json!(c); }
54            let Ok(r) = lv.push("event", ev).await else { break };
55            let mut got = Vec::new();
56            harvest_replies(reply_body(&r), &mut got);
57            if let Some(p) = props(&lv.html(), THREAD_PROPS_SEL) { harvest_replies(&p, &mut got); }
58            let new: Vec<Value> = got.into_iter().filter(|x| ids.insert(s(x, "id"))).collect();
59            if new.is_empty() { break; }
60            replies.extend(new);
61        }
62    }
63    Ok((th, replies))
64}
