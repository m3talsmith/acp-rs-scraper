1//! SQL schema for every table, kept in one place.
2
3/// `posts` table (the original spec schema) plus the unique `post_id` index.
4pub const POSTS_SCHEMA: &str = "CREATE TABLE IF NOT EXISTS posts (
5            id INTEGER PRIMARY KEY AUTOINCREMENT,
6            post_id TEXT NOT NULL,
7            parent_post_id TEXT NOT NULL,
8            nick TEXT NOT NULL,
9            title TEXT NOT NULL,
10            content TEXT NOT NULL,
11            posted_at TEXT NOT NULL
12        );
13        CREATE UNIQUE INDEX IF NOT EXISTS posts_post_id_uq ON posts(post_id);";
14
15/// `profiles` table: one row per scraped nick, upserted every scrape.
16pub const PROFILES_SCHEMA: &str = "CREATE TABLE IF NOT EXISTS profiles (
17            nick TEXT PRIMARY KEY,
18            user_id TEXT,
19            display_name TEXT,
20            bio TEXT,
21            avatar_url TEXT,
22            avatar_path TEXT,
23            banner_url TEXT,
24            banner_path TEXT,
25            location TEXT,
26            role TEXT,
27            joined_at TEXT,
28            followers INTEGER,
29            following INTEGER,
30            posts_count INTEGER,
31            raw_json TEXT NOT NULL,
32            scraped_at TEXT NOT NULL
33        );";
34
35/// `mentions` table: which posts mention which nick, and how.
36pub const MENTIONS_SCHEMA: &str = "CREATE TABLE IF NOT EXISTS mentions (
37            id INTEGER PRIMARY KEY AUTOINCREMENT,
38            post_id TEXT NOT NULL,
39            mentioned_nick TEXT NOT NULL,
40            match_type TEXT NOT NULL
41        );
42        CREATE UNIQUE INDEX IF NOT EXISTS mentions_uq ON mentions(post_id, mentioned_nick, match_type);";
43
44/// `attachments` table: downloaded media per post, unique per (post_id, url).
45pub const ATTACHMENTS_SCHEMA: &str = "CREATE TABLE IF NOT EXISTS attachments (
46            id INTEGER PRIMARY KEY AUTOINCREMENT,
47            post_id TEXT NOT NULL,
48            url TEXT NOT NULL,
49            local_path TEXT NOT NULL,
50            mime_type TEXT NOT NULL
51        );
52        CREATE UNIQUE INDEX IF NOT EXISTS attachments_post_url_uq ON attachments(post_id, url);";
