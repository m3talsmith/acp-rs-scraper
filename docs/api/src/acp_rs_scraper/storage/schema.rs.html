3/// `posts` table (the original spec schema) plus the unique `post_id` index.
4///
5/// Reply relationships:
6/// - `parent_post_id`: the thread root (`""` for root posts). Unchanged since 1.0.
7/// - `in_reply_to_id`: the immediate parent, i.e. the comment a reply answers
8///   (its first `postRef`), else the thread root; `""` for root posts.
9///
10/// Older databases gain `in_reply_to_id` through [`POSTS_MIGRATIONS`].
11pub const POSTS_SCHEMA: &str = "CREATE TABLE IF NOT EXISTS posts (
12            id INTEGER PRIMARY KEY AUTOINCREMENT,
13            post_id TEXT NOT NULL,
14            parent_post_id TEXT NOT NULL,
15            in_reply_to_id TEXT NOT NULL DEFAULT '',
16            nick TEXT NOT NULL,
17            title TEXT NOT NULL,
18            content TEXT NOT NULL,
19            posted_at TEXT NOT NULL,
20            body_state TEXT NOT NULL DEFAULT 'unknown'
21        );
22        CREATE UNIQUE INDEX IF NOT EXISTS posts_post_id_uq ON posts(post_id);";
23
24/// Run after [`POSTS_SCHEMA`] when `posts` lacks `in_reply_to_id`: add the
25/// column and point existing replies at their thread root (the best known parent).
26pub const POSTS_ADD_IN_REPLY_TO: &str = "ALTER TABLE posts ADD COLUMN in_reply_to_id TEXT NOT NULL DEFAULT '';
27        UPDATE posts SET in_reply_to_id = parent_post_id WHERE in_reply_to_id = '' AND parent_post_id <> '';";
28
29/// Run when `posts` lacks `body_state` (databases before 1.26.0). Values:
30/// `full` (complete body), `truncated` (only the feed's shortened preview
31/// was available) and `unknown` (stored before 1.26.0, could be either;
32/// `--refetch-truncated` checks these against their thread page).
33pub const POSTS_ADD_BODY_STATE: &str = "ALTER TABLE posts ADD COLUMN body_state TEXT NOT NULL DEFAULT 'unknown';";
34
35/// Indexes on the relationship columns (safe to re-run).
36pub const POSTS_MIGRATIONS: &str = "CREATE INDEX IF NOT EXISTS posts_parent_idx ON posts(parent_post_id);
37        CREATE INDEX IF NOT EXISTS posts_in_reply_to_idx ON posts(in_reply_to_id);
38        CREATE INDEX IF NOT EXISTS posts_body_state_idx ON posts(body_state);";
39
40/// `profiles` table: one row per scraped nick, upserted every scrape.
41pub const PROFILES_SCHEMA: &str = "CREATE TABLE IF NOT EXISTS profiles (
42            nick TEXT PRIMARY KEY,
43            user_id TEXT,
44            display_name TEXT,
45            bio TEXT,
46            avatar_url TEXT,

60/// `mentions` table: which posts mention which nick, and how.
61pub const MENTIONS_SCHEMA: &str = "CREATE TABLE IF NOT EXISTS mentions (
62            id INTEGER PRIMARY KEY AUTOINCREMENT,
63            post_id TEXT NOT NULL,
64            mentioned_nick TEXT NOT NULL,
65            match_type TEXT NOT NULL
66        );
67        CREATE UNIQUE INDEX IF NOT EXISTS mentions_uq ON mentions(post_id, mentioned_nick, match_type);";
68
69/// `attachments` table: downloaded media per post, unique per (post_id, url).
70pub const ATTACHMENTS_SCHEMA: &str = "CREATE TABLE IF NOT EXISTS attachments (
71            id INTEGER PRIMARY KEY AUTOINCREMENT,
72            post_id TEXT NOT NULL,
73            url TEXT NOT NULL,
74            local_path TEXT NOT NULL,
75            mime_type TEXT NOT NULL
76        );
77        CREATE UNIQUE INDEX IF NOT EXISTS attachments_post_url_uq ON attachments(post_id, url);";

79/// `bookmarks` table (2.1.0). One row per bookmarked post:
80/// - `remote`: the site has it bookmarked too (seen in its bookmarks feed, or pushed).
81/// - `deleted`: removed locally but still bookmarked on the site; the next
82///   `--sync-bookmarks` removes it there and drops the row.
83///
84/// Rows are added by `--bookmark` / the TUI (local, `remote = 0`) and by
85/// `--sync-bookmarks` (from the site, `remote = 1`). Created on first use.
86pub const BOOKMARKS_SCHEMA: &str = "CREATE TABLE IF NOT EXISTS bookmarks (
87            post_id TEXT PRIMARY KEY,
88            created_at TEXT NOT NULL,
89            remote INTEGER NOT NULL DEFAULT 0,
90            deleted INTEGER NOT NULL DEFAULT 0
91        );";
92
93/// Research projects and their items (see `research`). Items are unique per
94/// project, kind and ref; deleting a project deletes its items.
95pub const RESEARCH_SCHEMA: &str = "CREATE TABLE IF NOT EXISTS research_projects (
96            id INTEGER PRIMARY KEY AUTOINCREMENT,
97            name TEXT NOT NULL UNIQUE COLLATE NOCASE,
98            description TEXT NOT NULL DEFAULT '',
99            created_at TEXT NOT NULL
100        );
101        CREATE TABLE IF NOT EXISTS research_items (
102            id INTEGER PRIMARY KEY AUTOINCREMENT,
103            project_id INTEGER NOT NULL REFERENCES research_projects(id) ON DELETE CASCADE,
104            kind TEXT NOT NULL CHECK (kind IN ('post', 'reply', 'search-result', 'local-file', 'web-url')),
105            ref TEXT NOT NULL,
106            title TEXT NOT NULL DEFAULT '',
107            note TEXT NOT NULL DEFAULT '',
108            meta TEXT NOT NULL DEFAULT '{}',
109            content TEXT NOT NULL DEFAULT '',
110            local_path TEXT NOT NULL DEFAULT '',
111            added_at TEXT NOT NULL,
112            UNIQUE (project_id, kind, ref)
113        );
114        CREATE INDEX IF NOT EXISTS research_items_project_idx ON research_items(project_id);";
