1//! Post bodies as Markdown → styled, word-wrapped ratatui lines: headings,
2//! bold / italic / strikethrough, inline and block code, bullet and numbered
3//! lists, block quotes and links. Single line breaks are kept (bodies are
4//! stored as plain text whose newlines matter), and paragraphs stay separated
5//! by a blank line.
6
7use crate::tui::theme::{Gloss, Theme};
8use pulldown_cmark::{Event, HeadingLevel, Options, Parser, Tag, TagEnd};
9use ratatui::style::Style;
10use ratatui::text::{Line, Span};
11use unicode_width::UnicodeWidthStr;
12
13/// Styled words of one logical line before wrapping.
14#[derive(Default)]
15struct Block {
16    /// Prefix on the first visual line (bullet, quote bar) and on the rest (indent, quote bar).
17    first: Vec<Span<'static>>,
18    rest: Vec<Span<'static>>,
19    /// (text, style) pieces.
20    parts: Vec<(String, Style)>,
21}
22
23/// Word-wrap styled `parts` to `width`, with the given prefixes.
24fn wrap_parts(b: Block, width: usize, out: &mut Vec<Line<'static>>) {
25    let pw = |p: &[Span]| p.iter().map(|s| s.content.width()).sum::<usize>();
26    let mut line: Vec<Span<'static>> = b.first.clone();
27    let mut w = pw(&b.first);
28    let mut started = false;
29    let rest_w = pw(&b.rest);
30    for (text, st) in b.parts {
31        // Split into words keeping single spaces as separators.
32        for (k, word) in text.split(' ').enumerate() {
33            if k > 0 && started && w < width { line.push(Span::styled(" ", st)); w += 1; }
34            if word.is_empty() { continue; }
35            let ww = word.width();
36            if started && w + ww > width {
37                // Trim a trailing space before breaking.
38                if line.last().is_some_and(|s| s.content == " ") { line.pop(); }
39                out.push(Line::from(std::mem::take(&mut line)));
40                line = b.rest.clone();
41                w = rest_w;
42            }
43            if ww > width.saturating_sub(w) && ww > width / 2 {
44                // Hard-break very long words (URLs).
45                let mut chunk = String::new();
46                let mut cw = 0;
47                for ch in word.chars() {
48                    let c = unicode_width::UnicodeWidthChar::width(ch).unwrap_or(0);
49                    if w + cw + c > width && (cw > 0 || started) {
50                        if !chunk.is_empty() { line.push(Span::styled(std::mem::take(&mut chunk), st)); }
51                        out.push(Line::from(std::mem::take(&mut line)));
52                        line = b.rest.clone();
53                        w = rest_w;
54                        cw = 0;
55                    }
56                    chunk.push(ch);
57                    cw += c;
58                }
59                w += cw;
60                line.push(Span::styled(chunk, st));
61            } else {
62                line.push(Span::styled(word.to_string(), st));
63                w += ww;
64            }
65            started = true;
66        }
67    }
68    out.push(Line::from(line));
69}

71/// Render `text` as Markdown wrapped to `width` columns.
72pub fn render(text: &str, width: usize, t: &Theme) -> Vec<Line<'static>> {
73    let width = width.max(8);
74    let mut out: Vec<Line<'static>> = Vec::new();
75    let mut opts = Options::empty();
76    opts.insert(Options::ENABLE_STRIKETHROUGH);
77    let mut style_stack: Vec<Gloss> = vec![t.plain()];
78    let mut cur = Block::default();
79    let mut lists: Vec<Option<u64>> = Vec::new();
80    let mut quote = 0usize;
81    let mut in_code = false;
82    let mut link: Option<String> = None;
83    let mut pending_item = false;
84    let blank = |out: &mut Vec<Line<'static>>| if out.last().is_some_and(|l: &Line| !l.spans.is_empty() && l.width() > 0) { out.push(Line::raw("")); };
85    let quote_prefix = |q: usize, t: &Theme| -> Vec<Span<'static>> { (0..q).map(|_| Span::styled("│ ", t.muted())).collect() };
86    let flush = |cur: &mut Block, out: &mut Vec<Line<'static>>| {
87        let b = std::mem::take(cur);
88        if b.parts.iter().any(|(s, _)| !s.trim().is_empty()) { wrap_parts(b, width, out); }
89    };
90    let start_block = |cur: &mut Block, lists: &[Option<u64>], quote: usize, pending: &mut bool, t: &Theme| {
91        let indent = "  ".repeat(lists.len().saturating_sub(1));
92        let mut first = quote_prefix(quote, t);
93        let mut rest = quote_prefix(quote, t);
94        if let Some(l) = lists.last() {
95            let bullet = if *pending { match l { Some(n) => format!("{indent}{n}. "), None => format!("{indent}• ") } } else { format!("{indent}  ") };
96            let pad = " ".repeat(bullet.width());
97            first.push(Span::styled(bullet, t.accent()));
98            rest.push(Span::raw(pad));
99            *pending = false;
100        }
101        cur.first = first;
102        cur.rest = rest;
103    };
104    // Open a block (with the item's bullet when one is due) before the first inline content.
105    let ensure_block = |cur: &mut Block, lists: &[Option<u64>], quote: usize, pending: &mut bool| {
106        if *pending || (cur.first.is_empty() && cur.rest.is_empty() && cur.parts.is_empty()) { start_block(cur, lists, quote, pending, t); }
107    };
108    for ev in Parser::new_ext(text, opts) {
109        let g = style_stack.last().cloned().unwrap_or_else(|| t.plain());
110        let st = Style::from(&g);
111        match ev {
112            Event::Start(tag) => match tag {
113                Tag::Paragraph => { if lists.is_empty() { blank(&mut out); } start_block(&mut cur, &lists, quote, &mut pending_item, t); }
114                Tag::Heading { level, .. } => {
115                    blank(&mut out);
116                    start_block(&mut cur, &lists, quote, &mut pending_item, t);
117                    style_stack.push(if level == HeadingLevel::H1 { t.accent_bold().underline(true) } else { t.accent_bold() });
118                }
119                Tag::BlockQuote(_) => { blank(&mut out); quote += 1; }
120                Tag::CodeBlock(_) => { blank(&mut out); in_code = true; }
121                Tag::List(start) => {
122                    if lists.is_empty() { blank(&mut out); }
123                    // An item that opens straight with a sublist still shows its own bullet.
124                    if pending_item { start_block(&mut cur, &lists, quote, &mut pending_item, t); cur.parts.push((String::new(), st)); wrap_parts(std::mem::take(&mut cur), width, &mut out); }
125                    flush(&mut cur, &mut out);
126                    lists.push(start);
127                }
128                // The bullet goes on whatever block opens the item: its text in a tight list, its first
129                // paragraph in a loose one (so the paragraph's own start can't replace it).
130                Tag::Item => { flush(&mut cur, &mut out); pending_item = true; }
131                Tag::Emphasis => { ensure_block(&mut cur, &lists, quote, &mut pending_item); style_stack.push(g.italic(true)) }
132                Tag::Strong => { ensure_block(&mut cur, &lists, quote, &mut pending_item); style_stack.push(g.bold(true)) }
133                Tag::Strikethrough => { ensure_block(&mut cur, &lists, quote, &mut pending_item); style_stack.push(g.strikethrough(true)) }
134                Tag::Link { dest_url, .. } => { ensure_block(&mut cur, &lists, quote, &mut pending_item); link = Some(dest_url.to_string()); style_stack.push(t.link()); }
135                _ => {}
136            },
137            Event::End(tag) => match tag {
138                TagEnd::Paragraph => flush(&mut cur, &mut out),
139                TagEnd::Heading(_) => { flush(&mut cur, &mut out); style_stack.pop(); }
140                TagEnd::BlockQuote(_) => { flush(&mut cur, &mut out); quote = quote.saturating_sub(1); }
141                TagEnd::CodeBlock => in_code = false,
142                TagEnd::List(_) => { flush(&mut cur, &mut out); lists.pop(); }
143                TagEnd::Item => {
144                    // An empty item still shows its bullet.
145                    if pending_item { start_block(&mut cur, &lists, quote, &mut pending_item, t); cur.parts.push((String::new(), st)); wrap_parts(std::mem::take(&mut cur), width, &mut out); }
146                    flush(&mut cur, &mut out);
147                    if let Some(Some(n)) = lists.last_mut() { *n += 1; }
148                }
149                TagEnd::Emphasis | TagEnd::Strong | TagEnd::Strikethrough => { style_stack.pop(); }
150                TagEnd::Link => {
151                    style_stack.pop();
152                    if let Some(u) = link.take() {
153                        let shown: String = cur.parts.iter().rev().take(1).map(|(s, _)| s.clone()).collect();
154                        if !u.is_empty() && !shown.contains(u.as_str()) { cur.parts.push((format!(" <{u}>"), t.muted().into())); }
155                    }
156                }
157                _ => {}
158            },
159            Event::Text(s) if in_code => {
160                for l in s.trim_end_matches('\n').split('\n') {
161                    let mut first = quote_prefix(quote, t);
162                    first.push(Span::styled("  ", t.muted()));
163                    wrap_parts(Block { rest: first.clone(), first, parts: vec![(l.to_string(), t.muted().into())] }, width, &mut out);
164                }
165            }
166            Event::Text(s) => {
167                ensure_block(&mut cur, &lists, quote, &mut pending_item);
168                cur.parts.push((s.to_string(), st));
169            }
170            Event::Code(s) => { ensure_block(&mut cur, &lists, quote, &mut pending_item); cur.parts.push((s.to_string(), t.code().into())) }
171            Event::SoftBreak | Event::HardBreak => {
172                // Keep the author's line breaks.
173                let (first, rest) = (cur.first.clone(), cur.rest.clone());
174                flush(&mut cur, &mut out);
175                cur.first = rest.clone();
176                cur.rest = rest;
177                let _ = first;
178            }
179            Event::Rule => { blank(&mut out); out.push(Line::styled("─".repeat(width.min(40)), t.muted())); }
180            Event::Html(s) | Event::InlineHtml(s) => { ensure_block(&mut cur, &lists, quote, &mut pending_item); cur.parts.push((s.to_string(), st)) }
181            _ => {}
182        }
183    }
184    flush(&mut cur, &mut out);
185    while out.first().is_some_and(|l| l.width() == 0) { out.remove(0); }
186    while out.last().is_some_and(|l| l.width() == 0) { out.pop(); }
187    out
188}

190/// Plain text of rendered lines (tests, card snippets).
191pub fn plain(lines: &[Line]) -> Vec<String> {
192    lines.iter().map(|l| l.spans.iter().map(|s| s.content.as_ref()).collect::<String>()).collect()
193}
