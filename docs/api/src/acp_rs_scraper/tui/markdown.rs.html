1//! Post bodies as Markdown → styled, word-wrapped ratatui lines.
2//!
3//! CommonMark plus the GitHub extensions (tables, task lists, strikethrough,
4//! footnotes) parsed by `pulldown-cmark` into a small block tree, then laid
5//! out to a width: headings H1–H6 (a `━` / `─` rule under H1 / H2), bold,
6//! italic, strikethrough, inline code, boxed code blocks, block quotes with a
7//! `┃` gutter, bullet / numbered / task lists (nested, tight or loose) with
8//! hanging indents, box-drawn tables with aligned, wrapping columns, rules,
9//! links and autolinks (bare `https://` URLs too) with their URL shown,
10//! images as links, and footnotes listed at the end. Every style comes from
11//! the [`Theme`] (the markdown elements `h1`…`footnote`).
12//!
13//! Single line breaks are kept: post bodies are stored as plain text whose
14//! newlines matter. [`normalize`] first repairs what the site's plain text
15//! does to Markdown (typed `•` bullets, a lone indented line used to centre
16//! text), and simple inline HTML (`<b>`, `<br>`, `<a href>`, …) is honoured.
17

67static URL: LazyLock<Regex> = LazyLock::new(|| Regex::new(r#"https?://[^\s<>"]+"#).expect("url regex"));
68static TAG: LazyLock<Regex> = LazyLock::new(|| Regex::new(r#"(?s)<(/?)([A-Za-z][A-Za-z0-9]*)([^>]*)>|[^<]+|<"#).expect("tag regex"));
69static HREF: LazyLock<Regex> = LazyLock::new(|| Regex::new(r#"(?i)href\s*=\s*["']([^"']*)["']"#).expect("href regex"));

71/// A bare URL without the punctuation that usually follows it in prose.
72fn trim_url(u: &str) -> &str {
73    let mut u = u;
74    loop {
75        let Some(c) = u.chars().last() else { return u };
76        let unbalanced = c == ')' && u.matches('(').count() < u.matches(')').count();
77        if ".,;:!?'\"*_".contains(c) || unbalanced || c == ']' && !u.contains('[') { u = &u[..u.len() - c.len_utf8()]; } else { return u; }
78    }
79}

81fn entities(s: &str) -> String {
82    s.replace("&nbsp;", " ").replace("&lt;", "<").replace("&gt;", ">").replace("&quot;", "\"").replace("&#39;", "'").replace("&apos;", "'").replace("&amp;", "&")
83}

85struct Builder {
86    stack: Vec<Frame>,
87    inl: Vec<Inl>,
88    sink: Option<Sink>,
89    bold: u32,
90    italic: u32,
91    strike: u32,
92    code: u32,
93    links: Vec<(String, usize)>,
94    block_code: Option<(String, String)>,
95    notes: Vec<(String, Vec<Blk>)>,
96    note_ids: Vec<String>,
97}

99impl Builder {
100    fn new() -> Self {
101        Builder { stack: vec![Frame::Root(Vec::new())], inl: Vec::new(), sink: None, bold: 0, italic: 0, strike: 0, code: 0,
102            links: Vec::new(), block_code: None, notes: Vec::new(), note_ids: Vec::new() }
103    }
104
105    fn fmt(&self) -> Fmt {
106        Fmt { bold: self.bold > 0, italic: self.italic > 0, strike: self.strike > 0, code: self.code > 0, link: !self.links.is_empty(), ..Fmt::default() }
107    }
108    fn url(&self) -> Option<String> { self.links.last().map(|(u, _)| u.clone()) }
109
110    fn ensure_inline(&mut self) { if self.sink.is_none() { self.sink = Some(Sink::Implicit); } }
111
112    /// End text that came straight inside a container (a tight list item).
113    fn close_implicit(&mut self) {
114        if self.sink == Some(Sink::Implicit) {
115            self.sink = None;
116            let inl = std::mem::take(&mut self.inl);
117            self.push_block(Blk::Para(inl));
118        }
119    }
120
121    fn push_block(&mut self, b: Blk) {
122        if let Blk::Para(i) = &b && i.is_empty() { return; }
123        for f in self.stack.iter_mut().rev() {
124            match f {
125                Frame::Root(v) | Frame::Quote(v) | Frame::Note(_, v) => { v.push(b); return; }
126                Frame::Item(it) => { it.blocks.push(b); return; }
127                _ => {}
128            }
129        }
130    }
131
132    fn text(&mut self, s: &str) {
133        self.ensure_inline();
134        let f = self.fmt();
135        if f.link || f.code {
136            self.inl.push(Inl::Text(s.to_string(), f, self.url()));
137            return;
138        }
139        // Bare URLs become links.
140        let mut at = 0;
141        for m in URL.find_iter(s) {
142            let u = trim_url(m.as_str());
143            if u.len() < 10 { continue; }
144            if m.start() > at { self.inl.push(Inl::Text(s[at..m.start()].to_string(), f, None)); }
145            self.inl.push(Inl::Text(u.to_string(), Fmt { link: true, ..f }, Some(u.to_string())));
146            at = m.start() + u.len();
147        }
148        if at < s.len() { self.inl.push(Inl::Text(s[at..].to_string(), f, None)); }
149    }
150
151    fn end_link(&mut self) {
152        let Some((url, start)) = self.links.pop() else { return };
153        let shown: String = self.inl[start.min(self.inl.len())..].iter().filter_map(|i| match i { Inl::Text(s, ..) => Some(s.as_str()), Inl::Break => None }).collect();
154        if !url.is_empty() && !shown.contains(url.as_str()) && !url.starts_with('#') {
155            self.inl.push(Inl::Text(format!(" <{url}>"), Fmt { muted: true, ..Fmt::default() }, None));
156        }
157    }
158
159    /// Simple HTML as the formatting it stands for; other tags are dropped.
160    fn html(&mut self, s: &str) {
161        for m in TAG.captures_iter(s) {
162            let whole = m.get(0).map_or("", |x| x.as_str());
163            let Some(name) = m.get(2) else {
164                let t = entities(whole);
165                if !t.trim().is_empty() || self.sink.is_some() { self.text(&t.replace('\n', " ")); }
166                continue;
167            };
168            let close = m.get(1).is_some_and(|c| !c.as_str().is_empty());
169            let dec = |n: &mut u32| *n = n.saturating_sub(1);
170            match (name.as_str().to_ascii_lowercase().as_str(), close) {
171                ("br", _) => { self.ensure_inline(); self.inl.push(Inl::Break); }
172                ("p" | "div", true) => { self.ensure_inline(); self.inl.push(Inl::Break); }
173                ("li", false) => { self.ensure_inline(); if !self.inl.is_empty() { self.inl.push(Inl::Break); } self.text("• "); }
174                ("b" | "strong", false) => self.bold += 1,
175                ("b" | "strong", true) => dec(&mut self.bold),
176                ("i" | "em", false) => self.italic += 1,
177                ("i" | "em", true) => dec(&mut self.italic),
178                ("s" | "del" | "strike", false) => self.strike += 1,
179                ("s" | "del" | "strike", true) => dec(&mut self.strike),
180                ("code" | "kbd" | "tt", false) => self.code += 1,
181                ("code" | "kbd" | "tt", true) => dec(&mut self.code),
182                ("a", false) => {
183                    let href = m.get(3).and_then(|a| HREF.captures(a.as_str())).and_then(|c| c.get(1)).map(|h| entities(h.as_str())).unwrap_or_default();
184                    self.ensure_inline();
185                    let at = self.inl.len();
186                    self.links.push((href, at));
187                }
188                ("a", true) => self.end_link(),
189                _ => {}
190            }
191        }
192    }

194    fn event(&mut self, ev: Event) {
195        match ev {
196            Event::Start(tag) => match tag {
197                Tag::Paragraph => {
198                    self.close_implicit();
199                    self.sink = Some(Sink::Para);
200                    // A paragraph directly inside an item makes the list loose.
201                    let n = self.stack.len();
202                    if n >= 2 && matches!(self.stack[n - 1], Frame::Item(_))
203                        && let Frame::List { loose, .. } = &mut self.stack[n - 2] { *loose = true; }
204                }
205                Tag::Heading { level, .. } => {
206                    self.close_implicit();
207                    let n = match level { HeadingLevel::H1 => 1, HeadingLevel::H2 => 2, HeadingLevel::H3 => 3, HeadingLevel::H4 => 4, HeadingLevel::H5 => 5, HeadingLevel::H6 => 6 };
208                    self.sink = Some(Sink::Heading(n));
209                }
210                Tag::BlockQuote(_) => { self.close_implicit(); self.stack.push(Frame::Quote(Vec::new())); }
211                Tag::CodeBlock(kind) => {
212                    self.close_implicit();
213                    let lang = match kind { CodeBlockKind::Fenced(l) => l.split_whitespace().next().unwrap_or("").to_string(), CodeBlockKind::Indented => String::new() };
214                    self.block_code = Some((lang, String::new()));
215                }
216                Tag::HtmlBlock => { self.close_implicit(); self.sink = Some(Sink::Para); }
217                Tag::List(start) => { self.close_implicit(); self.stack.push(Frame::List { start, loose: false, items: Vec::new() }); }
218                Tag::Item => { self.close_implicit(); self.stack.push(Frame::Item(Item::default())); }
219                Tag::Table(aligns) => { self.close_implicit(); self.stack.push(Frame::Table(TableData { aligns, ..TableData::default() })); }
220                Tag::TableHead | Tag::TableRow => { if let Some(Frame::Table(t)) = self.stack.last_mut() { t.row.clear(); } }
221                Tag::TableCell => { self.inl.clear(); self.sink = Some(Sink::Cell); }
222                Tag::FootnoteDefinition(label) => { self.close_implicit(); self.stack.push(Frame::Note(label.to_string(), Vec::new())); }
223                Tag::Emphasis => self.italic += 1,
224                Tag::Strong => self.bold += 1,
225                Tag::Strikethrough => self.strike += 1,
226                Tag::Link { dest_url, .. } => { self.ensure_inline(); let at = self.inl.len(); self.links.push((dest_url.to_string(), at)); }
227                Tag::Image { dest_url, .. } => {
228                    self.ensure_inline();
229                    let at = self.inl.len();
230                    self.links.push((dest_url.to_string(), at));
231                    self.inl.push(Inl::Text("🖼 ".into(), self.fmt(), self.url()));
232                }
233                _ => {}
234            },
235            Event::End(tag) => match tag {
236                TagEnd::Paragraph | TagEnd::HtmlBlock => {
237                    self.sink = None;
238                    while matches!(self.inl.last(), Some(Inl::Break)) { self.inl.pop(); }
239                    let inl = std::mem::take(&mut self.inl);
240                    self.push_block(Blk::Para(inl));
241                }
242                TagEnd::Heading(_) => {
243                    let n = match self.sink { Some(Sink::Heading(n)) => n, _ => 1 };
244                    self.sink = None;
245                    let inl = std::mem::take(&mut self.inl);
246                    self.push_block(Blk::Heading(n, inl));
247                }
248                TagEnd::BlockQuote(_) => {
249                    self.close_implicit();
250                    if let Some(Frame::Quote(v)) = self.stack.pop() { self.push_block(Blk::Quote(v)); }
251                }
252                TagEnd::CodeBlock => {
253                    if let Some((lang, mut text)) = self.block_code.take() {
254                        if text.ends_with('\n') { text.pop(); }
255                        self.push_block(Blk::Code(lang, text));
256                    }
257                }
258                TagEnd::List(_) => {
259                    self.close_implicit();
260                    if let Some(Frame::List { start, loose, items }) = self.stack.pop() { self.push_block(Blk::List { start, loose, items }); }
261                }
262                TagEnd::Item => {
263                    self.close_implicit();
264                    if let Some(Frame::Item(it)) = self.stack.pop()
265                        && let Some(Frame::List { items, .. }) = self.stack.last_mut() { items.push(it); }
266                }
267                TagEnd::TableCell => {
268                    self.sink = None;
269                    let inl = std::mem::take(&mut self.inl);
270                    if let Some(Frame::Table(t)) = self.stack.last_mut() { t.row.push(inl); }
271                }
272                TagEnd::TableHead => { if let Some(Frame::Table(t)) = self.stack.last_mut() { t.head = std::mem::take(&mut t.row); } }
273                TagEnd::TableRow => { if let Some(Frame::Table(t)) = self.stack.last_mut() { let r = std::mem::take(&mut t.row); t.rows.push(r); } }
274                TagEnd::Table => { if let Some(Frame::Table(t)) = self.stack.pop() { self.push_block(Blk::Table(t)); } }
275                TagEnd::FootnoteDefinition => {
276                    self.close_implicit();
277                    if let Some(Frame::Note(l, v)) = self.stack.pop() { self.notes.push((l, v)); }
278                }
279                TagEnd::Emphasis => self.italic = self.italic.saturating_sub(1),
280                TagEnd::Strong => self.bold = self.bold.saturating_sub(1),
281                TagEnd::Strikethrough => self.strike = self.strike.saturating_sub(1),
282                TagEnd::Link | TagEnd::Image => self.end_link(),
283                _ => {}
284            },
285            Event::Text(s) => match &mut self.block_code { Some((_, c)) => c.push_str(&s), None => self.text(&s) },
286            Event::Code(s) => {
287                self.ensure_inline();
288                let f = Fmt { code: true, ..self.fmt() };
289                self.inl.push(Inl::Text(s.to_string(), f, self.url()));
290            }
291            Event::SoftBreak | Event::HardBreak => { self.ensure_inline(); self.inl.push(Inl::Break); }
292            Event::Rule => { self.close_implicit(); self.push_block(Blk::Rule); }
293            Event::TaskListMarker(done) => {
294                for f in self.stack.iter_mut().rev() { if let Frame::Item(it) = f { it.task = Some(done); break; } }
295            }
296            Event::FootnoteReference(label) => {
297                self.ensure_inline();
298                let n = match self.note_ids.iter().position(|l| l.as_str() == label.as_ref()) {
299                    Some(i) => i + 1,
300                    None => { self.note_ids.push(label.to_string()); self.note_ids.len() }
301                };
302                self.inl.push(Inl::Text(format!("[{n}]"), Fmt { note: true, ..Fmt::default() }, None));
303            }
304            Event::Html(s) | Event::InlineHtml(s) => self.html(&s),
305            _ => {}
306        }
307    }
308
309    fn finish(mut self) -> (Vec<Blk>, Vec<(usize, Vec<Blk>)>) {
310        self.close_implicit();
311        while self.stack.len() > 1 {
312            // Unbalanced input: fold open containers into their parents.
313            match self.stack.pop() {
314                Some(Frame::Quote(v)) => self.push_block(Blk::Quote(v)),
315                Some(Frame::Item(it)) => { if let Some(Frame::List { items, .. }) = self.stack.last_mut() { items.push(it); } }
316                Some(Frame::List { start, loose, items }) => self.push_block(Blk::List { start, loose, items }),
317                Some(Frame::Table(t)) => self.push_block(Blk::Table(t)),
318                Some(Frame::Note(l, v)) => self.notes.push((l, v)),
319                _ => {}
320            }
321        }
322        let root = match self.stack.pop() { Some(Frame::Root(v)) => v, _ => Vec::new() };
323        // Notes in the order they are first referenced, then any never referenced.
324        let mut notes = Vec::new();
325        let mut defs = self.notes;
326        for (i, id) in self.note_ids.iter().enumerate() {
327            if let Some(p) = defs.iter().position(|(l, _)| l == id) { notes.push((i + 1, defs.remove(p).1)); }
328        }
329        let mut n = self.note_ids.len();
330        for (_, v) in defs { n += 1; notes.push((n, v)); }
331        (root, notes)
332    }
333}

337fn bullet_char(t: &str) -> Option<&str> {
338    let c = t.chars().next()?;
339    if !"•●▪◦‣".contains(c) { return None; }
340    let rest = &t[c.len_utf8()..];
341    rest.starts_with([' ', '\t']).then_some(rest)
342}

344fn is_list_line(t: &str) -> bool {
345    let t = t.trim_start();
346    if t.starts_with("- ") || t.starts_with("* ") || t.starts_with("+ ") { return true; }
347    let d = t.chars().take_while(|c| c.is_ascii_digit()).count();
348    d > 0 && (t[d..].starts_with(". ") || t[d..].starts_with(") "))
349}

351/// Repairs for Markdown as the site stores it (plain text, newlines kept):
352/// lines starting with a typed bullet (`•`, `●`, `▪`, `◦`, `‣`) become list
353/// items, and a lone indented line between blank lines (text centred with
354/// spaces) is un-indented instead of turning into a code block. Fenced code
355/// is left alone.
356pub fn normalize(src: &str) -> String {
357    let src = src.replace("\r\n", "\n");
358    let lines: Vec<&str> = src.split('\n').collect();
359    let mut out: Vec<String> = Vec::with_capacity(lines.len());
360    let mut fence: Option<&str> = None;
361    for (i, l) in lines.iter().enumerate() {
362        let t = l.trim_start();
363        if let Some(f) = fence {
364            if t.starts_with(f) { fence = None; }
365            out.push(l.to_string());
366            continue;
367        }
368        if t.starts_with("```") || t.starts_with("~~~") {
369            fence = Some(&t[..3]);
370            out.push(l.to_string());
371            continue;
372        }
373        if let Some(rest) = bullet_char(t) {
374            out.push(format!("- {}", rest.trim_start()));
375            continue;
376        }
377        let indented = l.starts_with("    ") || l.starts_with('\t');
378        if indented && !t.is_empty() {
379            let prev_blank = i == 0 || lines[i - 1].trim().is_empty();
380            let next_blank = i + 1 >= lines.len() || lines[i + 1].trim().is_empty();
381            let prev = lines[..i].iter().rev().find(|p| !p.trim().is_empty());
382            let after_list = prev.is_some_and(|p| is_list_line(p) || p.starts_with("    ") || p.starts_with('\t') || bullet_char(p.trim_start()).is_some());
383            if prev_blank && next_blank && !after_list { out.push(t.to_string()); continue; }
384        }
385        out.push(l.to_string());
386    }
387    out.join("\n")
388}

392/// A laid-out line and the links on it: (column, text, URL).
393#[derive(Clone, Default)]
394struct RLine { spans: Vec<Span<'static>>, links: Vec<(usize, String, String)> }
395
396impl RLine {
397    fn width(&self) -> usize { self.spans.iter().map(|s| s.content.width()).sum() }
398    fn styled(s: String, st: Style) -> RLine { RLine { spans: vec![Span::styled(s, st)], links: Vec::new() } }
399    fn prefixed(mut self, p: &[Span<'static>]) -> RLine {
400        let pw: usize = p.iter().map(|s| s.content.width()).sum();
401        let mut spans = p.to_vec();
402        spans.append(&mut self.spans);
403        RLine { spans, links: self.links.into_iter().map(|(c, t, u)| (c + pw, t, u)).collect() }
404    }
405    fn push(&mut self, s: String, st: Style, url: Option<&String>, at: usize) {
406        if let Some(u) = url {
407            match self.links.last_mut() {
408                Some((c, t, lu)) if lu == u && *c + t.width() + 1 >= at => { while *c + t.width() < at { t.push(' '); } t.push_str(&s); }
409                _ => self.links.push((at, s.clone(), u.clone())),
410            }
411        }
412        self.spans.push(Span::styled(s, st));
413    }
414}

418/// Display width of inline content on one line (the widest line when it has breaks).
419fn inl_width(inl: &[Inl]) -> usize {
420    let mut best = 0;
421    let mut w = 0;
422    for i in inl {
423        match i { Inl::Text(s, ..) => w += s.width(), Inl::Break => { best = best.max(w); w = 0; } }
424    }
425    best.max(w)
426}

428struct Layout<'a> { t: &'a Theme, depth: usize }

430impl Layout<'_> {
431    fn style(&self, base: &Gloss, f: Fmt) -> Style {
432        let t = self.t;
433        let mut g = if f.code { t.code() } else if f.link { t.link() } else if f.note { t.footnote() } else if f.muted { t.muted() } else { base.clone() };
434        if f.bold { g = g.bold(true); }
435        if f.italic { g = g.italic(true); }
436        if f.strike { g = g.strikethrough(true); }
437        Style::from(&g)
438    }
439
440    /// Word-wrap inline content to `width`.
441    fn wrap(&self, inl: &[Inl], width: usize, base: &Gloss) -> Vec<RLine> {
442        let width = width.max(1);
443        let mut out = Vec::new();
444        let mut line = RLine::default();
445        let mut w = 0usize;
446        let mut started = false;
447        for piece in inl {
448            let (text, f, url) = match piece {
449                Inl::Break => { out.push(std::mem::take(&mut line)); w = 0; started = false; continue; }
450                Inl::Text(s, f, u) => (s, *f, u.as_ref()),
451            };
452            let st = self.style(base, f);
453            for (k, word) in text.split(' ').enumerate() {
454                if k > 0 && started && w < width {
455                    // The space keeps the link's underline across words of one link.
456                    line.spans.push(Span::styled(" ", if url.is_some() { st } else { self.style(base, Fmt { strike: false, ..f }) }));
457                    w += 1;
458                }
459                if word.is_empty() { continue; }
460                let ww = word.width();
461                if started && w + ww > width {
462                    if line.spans.last().is_some_and(|s| s.content == " ") { line.spans.pop(); }
463                    out.push(std::mem::take(&mut line));
464                    w = 0;
465                }
466                if ww > width.saturating_sub(w) {
467                    // Hard-break words longer than the line (URLs).
468                    let mut chunk = String::new();
469                    let mut cw = 0;
470                    let mut at = w;
471                    for ch in word.chars() {
472                        let c = ch.width().unwrap_or(0);
473                        if w + cw + c > width && cw > 0 {
474                            line.push(std::mem::take(&mut chunk), st, url, at);
475                            out.push(std::mem::take(&mut line));
476                            w = 0;
477                            cw = 0;
478                            at = 0;
479                        }
480                        chunk.push(ch);
481                        cw += c;
482                    }
483                    line.push(chunk, st, url, at);
484                    w += cw;
485                } else {
486                    line.push(word.to_string(), st, url, w);
487                    w += ww;
488                }
489                started = true;
490            }
491        }
492        out.push(line);
493        out
494    }
495
496    fn blocks(&mut self, bs: &[Blk], width: usize, base: &Gloss, tight: bool) -> Vec<RLine> {
497        let mut out = Vec::new();
498        for (i, b) in bs.iter().enumerate() {
499            if i > 0 && !tight { out.push(RLine::default()); }
500            out.extend(self.block(b, width, base));
501        }
502        out
503    }
504
505    fn block(&mut self, b: &Blk, width: usize, base: &Gloss) -> Vec<RLine> {
506        let t = self.t;
507        match b {
508            Blk::Para(inl) => self.wrap(inl, width, base),
509            Blk::Heading(n, inl) => {
510                let g = t.heading(*n);
511                let mut v = if *n == 1 {
512                    // H1 is a banner: one cell of its background either side.
513                    let st = Style::from(&g);
514                    self.wrap(inl, width.saturating_sub(2).max(1), &g).into_iter().map(|l| {
515                        let mut l = l.prefixed(&[Span::styled(" ", st)]);
516                        l.spans.push(Span::styled(" ", st));
517                        l
518                    }).collect()
519                } else { self.wrap(inl, width, &g) };
520                if *n <= 2 {
521                    let w = v.iter().map(RLine::width).max().unwrap_or(0).max(3);
522                    v.push(RLine::styled((if *n == 1 { "━" } else { "─" }).repeat(w), Style::from(&t.heading_rule(*n))));
523                }
524                v
525            }
526            Blk::Quote(bs) => {
527                let bar = Span::styled("┃ ", Style::from(&t.quote_bar()));
528                self.blocks(bs, width.saturating_sub(2).max(1), &t.quote(), false).into_iter().map(|l| l.prefixed(std::slice::from_ref(&bar))).collect()
529            }
530            Blk::Code(lang, text) => self.code(lang, text, width),
531            Blk::List { start, loose, items } => self.list(*start, *loose, items, width, base),
532            Blk::Table(td) => self.table(td, width),
533            Blk::Rule => vec![RLine::styled("─".repeat(width), Style::from(&t.rule()))],
534        }
535    }
536
537    fn code(&self, lang: &str, text: &str, width: usize) -> Vec<RLine> {
538        let t = self.t;
539        let bs = Style::from(&t.code_block_border());
540        let ts = Style::from(&t.code_block());
541        let text = text.replace('\t', "    ");
542        if width < 6 {
543            return text.split('\n').map(|l| RLine::styled(l.to_string(), ts)).collect();
544        }
545        let inner = width - 4;
546        let mut out = Vec::new();
547        let label = if lang.is_empty() || lang.width() + 4 > width - 2 { String::new() } else { format!(" {lang} ") };
548        let mut top = RLine::styled("╭─".into(), bs);
549        if !label.is_empty() { top.spans.push(Span::styled(label.clone(), Style::from(&t.muted()))); }
550        top.spans.push(Span::styled(format!("{}╮", "─".repeat(width - 3 - label.width())), bs));
551        out.push(top);
552        for l in text.split('\n') {
553            let mut chunks: Vec<(String, usize)> = vec![(String::new(), 0)];
554            for ch in l.chars() {
555                let c = ch.width().unwrap_or(0);
556                if chunks.last().is_some_and(|(_, w)| w + c > inner) { chunks.push((String::new(), 0)); }
557                let last = chunks.last_mut().expect("one chunk");
558                last.0.push(ch);
559                last.1 += c;
560            }
561            for (s, w) in chunks {
562                out.push(RLine { spans: vec![Span::styled("│ ", bs), Span::styled(s, ts), pad(inner - w), Span::styled(" │", bs)], links: Vec::new() });
563            }
564        }
565        out.push(RLine::styled(format!("╰{}╯", "─".repeat(width - 2)), bs));
566        out
567    }
568
569    fn list(&mut self, start: Option<u64>, loose: bool, items: &[Item], width: usize, base: &Gloss) -> Vec<RLine> {
570        let t = self.t;
571        let ms = Style::from(&t.list_marker());
572        let ts = Style::from(&t.task());
573        let numw = start.map(|s| format!("{}.", s + items.len().saturating_sub(1) as u64).len()).unwrap_or(0);
574        let bullet = ["•", "◦", "▪"][self.depth % 3];
575        let mut out = Vec::new();
576        self.depth += 1;
577        for (i, it) in items.iter().enumerate() {
578            if loose && i > 0 { out.push(RLine::default()); }
579            let mut marker: Vec<Span<'static>> = Vec::new();
580            match start {
581                Some(s) => marker.push(Span::styled(format!("{:>numw$} ", format!("{}.", s + i as u64)), ms)),
582                None if it.task.is_none() => marker.push(Span::styled(format!("{bullet} "), ms)),
583                None => {}
584            }
585            if let Some(done) = it.task { marker.push(Span::styled(if done { "☑ " } else { "☐ " }, ts)); }
586            let mw: usize = marker.iter().map(|s| s.content.width()).sum();
587            let mut body = self.blocks(&it.blocks, width.saturating_sub(mw).max(1), base, !loose);
588            if body.is_empty() { body.push(RLine::default()); }
589            let indent = [pad(mw)];
590            for (k, l) in body.into_iter().enumerate() {
591                out.push(if k == 0 { l.prefixed(&marker) } else if l.spans.is_empty() { l } else { l.prefixed(&indent) });
592            }
593        }
594        self.depth -= 1;
595        out
596    }
597
598    fn table(&self, td: &TableData, width: usize) -> Vec<RLine> {
599        let t = self.t;
600        let bs = Style::from(&t.table_border());
601        let ncols = td.rows.iter().map(Vec::len).chain(std::iter::once(td.head.len())).max().unwrap_or(0);
602        if ncols == 0 { return Vec::new(); }
603        fn cell(r: &[Vec<Inl>], c: usize) -> &[Inl] { r.get(c).map(Vec::as_slice).unwrap_or(&[]) }
604        let mut w: Vec<usize> = (0..ncols).map(|c| {
605            std::iter::once(&td.head).chain(td.rows.iter()).map(|r| inl_width(cell(r, c))).max().unwrap_or(0).max(1)
606        }).collect();
607        let avail = width.saturating_sub(3 * ncols + 1).max(ncols);
608        while w.iter().sum::<usize>() > avail {
609            let (c, _) = w.iter().enumerate().max_by_key(|(_, x)| **x).expect("columns");
610            if w[c] <= 1 { break; }
611            w[c] -= 1;
612        }
613        let rule = |l: &str, m: &str, r: &str| RLine::styled(format!("{l}{}{r}", w.iter().map(|x| "─".repeat(x + 2)).collect::<Vec<_>>().join(m)), bs);
614        let row_lines = |r: &Vec<Vec<Inl>>, base: &Gloss| -> Vec<RLine> {
615            let cells: Vec<Vec<RLine>> = (0..ncols).map(|c| self.wrap(cell(r, c), w[c], base)).collect();
616            let h = cells.iter().map(Vec::len).max().unwrap_or(1);
617            (0..h).map(|y| {
618                let mut line = RLine::styled("│".into(), bs);
619                let mut col = 1;
620                for c in 0..ncols {
621                    let content = cells[c].get(y).cloned().unwrap_or_default();
622                    let cw = content.width().min(w[c]);
623                    let left = match td.aligns.get(c) { Some(Alignment::Right) => w[c] - cw, Some(Alignment::Center) => (w[c] - cw) / 2, _ => 0 };
624                    line.spans.push(pad(1 + left));
625                    let shifted = content.prefixed(&[]);
626                    for (lc, lt, lu) in shifted.links { line.links.push((col + 1 + left + lc, lt, lu)); }
627                    line.spans.extend(shifted.spans);
628                    line.spans.push(pad(w[c] - cw - left + 1));
629                    line.spans.push(Span::styled("│", bs));
630                    col += w[c] + 3;
631                }
632                line
633            }).collect()
634        };
635        let mut out = vec![rule("┌", "┬", "┐")];
636        if !td.head.is_empty() {
637            out.extend(row_lines(&td.head, &t.table_header()));
638            out.push(rule("├", "┼", "┤"));
639        }
640        for r in &td.rows { out.extend(row_lines(r, &t.plain())); }
641        out.push(rule("└", "┴", "┘"));
642        out
643    }
644}

648/// A link drawn at `line`, starting at display column `col`.
649#[derive(Clone, Debug, PartialEq)]
650pub struct MdLink {
651    /// Line index in [`Rendered::lines`].
652    pub line: usize,
653    /// Display column where the link text starts.
654    pub col: usize,
655    /// The link text as shown on that line.
656    pub text: String,
657    /// Its target.
658    pub url: String,
659}
660
661/// Rendered Markdown: the lines and where their links are (for OSC 8 hyperlinks).
662#[derive(Clone, Debug, Default)]
663pub struct Rendered {
664    /// Styled, wrapped lines.
665    pub lines: Vec<Line<'static>>,
666    /// Links on those lines.
667    pub links: Vec<MdLink>,
668}
669
670/// Render `text` as Markdown wrapped to `width` columns, with link positions.
671pub fn render_doc(text: &str, width: usize, t: &Theme) -> Rendered {
672    let width = width.max(8);
673    let mut opts = Options::empty();
674    opts.insert(Options::ENABLE_STRIKETHROUGH);
675    opts.insert(Options::ENABLE_TABLES);
676    opts.insert(Options::ENABLE_TASKLISTS);
677    opts.insert(Options::ENABLE_FOOTNOTES);
678    let src = normalize(text);
679    let mut b = Builder::new();
680    for ev in Parser::new_ext(&src, opts) { b.event(ev); }
681    let (root, notes) = b.finish();
682    let mut lay = Layout { t, depth: 0 };
683    let mut out = lay.blocks(&root, width, &t.plain(), false);
684    if !notes.is_empty() {
685        let fs = Style::from(&t.footnote());
686        out.push(RLine::default());
687        out.push(RLine::styled("─".repeat(width.min(24)), fs));
688        let mw = notes.iter().map(|(n, _)| format!("[{n}] ").len()).max().unwrap_or(4);
689        for (n, bs) in &notes {
690            let body = lay.blocks(bs, width.saturating_sub(mw).max(1), &t.footnote(), true);
691            let marker = [Span::styled(format!("{:<mw$}", format!("[{n}]")), fs)];
692            let indent = [pad(mw)];
693            for (k, l) in body.into_iter().enumerate() { out.push(if k == 0 { l.prefixed(&marker) } else { l.prefixed(&indent) }); }
694        }
695    }
696    while out.first().is_some_and(|l| l.width() == 0) { out.remove(0); }
697    while out.last().is_some_and(|l| l.width() == 0) { out.pop(); }
698    let mut r = Rendered::default();
699    for (i, l) in out.into_iter().enumerate() {
700        for (col, text, url) in l.links { r.links.push(MdLink { line: i, col, text, url }); }
701        r.lines.push(Line::from(l.spans));
702    }
703    r
704}
705
706/// Render `text` as Markdown wrapped to `width` columns.
707pub fn render(text: &str, width: usize, t: &Theme) -> Vec<Line<'static>> { render_doc(text, width, t).lines }
708
709/// Plain text of rendered lines (tests, card snippets).
710pub fn plain(lines: &[Line]) -> Vec<String> {
711    lines.iter().map(|l| l.spans.iter().map(|s| s.content.as_ref()).collect::<String>()).collect()
712}
