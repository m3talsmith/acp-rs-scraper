1//! The theme editor (Shift-F5): full screen, the palette and every element on the
2//! left (swatch, value, `*` when changed, `⚠` below 3:1 contrast), a live preview
3//! on the right drawn by the real UI code over sample posts. `Tab` flips dark /
4//! light; colours come from a hex field or a grid (16 colours, the 216 cube, 24
5//! greys, `@` palette references). `^S` saves, `^R` reverts, `^N` new, `^D`
6//! duplicate, `^X` delete, `[` `]` other themes, `Esc` closes (asking first when
7//! there are unsaved changes).
8

17/// A row of the left list.
18#[derive(Clone, Debug, PartialEq, Eq)]
19pub enum Row {
20    /// A section heading.
21    Heading(&'static str),
22    /// A palette colour.
23    Palette(String),
24    /// An element (Enter shows its keys).
25    Element(Element),
26    /// One key of the expanded element.
27    Field(Element, &'static str),
28}

30/// What a colour pick changes.
31#[derive(Clone, Debug, PartialEq, Eq)]
32pub enum Target {
33    /// A palette entry.
34    Palette(String),
35    /// `fg` / `bg` / `border` of an element.
36    Field(Element, &'static str),
37}

131    /// Edit theme `name` (a new empty one when it can't be read; a broken file is
132    /// opened as a new theme and only replaced if you save it).
133    pub fn open(app: &App, name: &str) -> Self {
134        let dir = app.themes_dir.clone().unwrap_or_default();
135        let (file, saved, message) = match themes::load(&dir, name) {
136            Ok(l) => (l.file.clone(), Some(l.file), None),
137            Err(e) => (ThemeFile::empty(name), None, Some((true, e.notice()))),
138        };
139        let t = Theme::from_file(&file, app.variant());
140        let mut ed = ThemeEditor { file, saved, variant: app.variant(), cursor: 1, scroll: 0, expanded: None, picker: None, prompt: None, message, preview: preview_app(t) };
141        ed.sync_preview();
142        ed
143    }

153    /// Palette names in the shown variant: the six used ones, then any others.
154    pub fn palette_names(&self) -> Vec<String> {
155        let s = self.file.section(self.variant);
156        let mut v: Vec<String> = PALETTE_KEYS.iter().map(|(k, _)| k.to_string()).collect();
157        for (n, _) in &s.palette { if !v.contains(n) { v.push(n.clone()); } }
158        v
159    }

161    /// The list rows.
162    pub fn rows(&self) -> Vec<Row> {
163        let mut v = vec![Row::Heading("Palette")];
164        v.extend(self.palette_names().into_iter().map(Row::Palette));
165        v.push(Row::Heading("Elements"));
166        for e in Element::ALL {
167            v.push(Row::Element(*e));
168            if self.expanded == Some(*e) {
169                v.extend(STYLE_KEYS.iter().filter(|k| **k != "border" || e.is_box()).map(|k| Row::Field(*e, k)));
170            }
171        }
172        v
173    }

175    /// The colour a row shows (resolved), if it has one.
176    pub fn row_color(&self, r: &Row) -> Option<Color> {
177        let s = self.file.section(self.variant);
178        match r {
179            Row::Palette(n) => Some(s.color(n)),
180            Row::Field(e, k) => s.style(*e).color(k).map(|c| s.resolve(c)),
181            _ => None,

185    /// The value a row shows, as written.
186    pub fn row_value(&self, r: &Row) -> String {
187        let s = self.file.section(self.variant);
188        match r {
189            Row::Palette(n) => s.get(n).map(themes::color_text).unwrap_or_else(|| "(unset)".into()),
190            Row::Element(e) => {
191                let st = s.style(*e);
192                let mut parts: Vec<String> = Vec::new();
193                for k in STYLE_KEYS {
194                    if let Some(c) = st.color(k) { parts.push(format!("{k}={}", themes::color_text(c))); }
195                    if let Some(b) = st.flag(k) { parts.push(if b { k.to_string() } else { format!("no {k}") }); }
196                }
197                if parts.is_empty() { "built-in".into() } else { parts.join(" ") }
198            }
199            Row::Field(e, k) => {
200                let st = s.style(*e);
201                st.color(k).map(themes::color_text).or_else(|| st.flag(k).map(|b| b.to_string())).unwrap_or_else(|| "–".into())
202            }
203            Row::Heading(_) => String::new(),
204        }
205    }

207    /// Changed since the last save.
208    pub fn row_changed(&self, r: &Row) -> bool {
209        let Some(saved) = &self.saved else { return !matches!(r, Row::Heading(_)) };
210        let (a, b) = (self.file.section(self.variant), saved.section(self.variant));
211        match r {
212            Row::Palette(n) => a.get(n) != b.get(n),
213            Row::Element(e) => a.style(*e) != b.style(*e),
214            Row::Field(e, k) => { let (x, y) = (a.style(*e), b.style(*e)); x.color(k) != y.color(k) || x.flag(k) != y.flag(k) }
215            Row::Heading(_) => false,
216        }
217    }

222    /// Keep the cursor off headings.
223    fn settle(&mut self, d: i32) {
224        let rows = self.rows();
225        let n = rows.len();
226        self.cursor = self.cursor.min(n - 1);
227        let mut guard = 0;
228        while matches!(rows.get(self.cursor), Some(Row::Heading(_))) && guard < n {
229            self.cursor = if d < 0 { if self.cursor == 0 { 1 } else { self.cursor - 1 } } else { (self.cursor + 1).min(n - 1) };
230            guard += 1;
231        }
232    }

234    fn move_cursor(&mut self, d: i32) {
235        let n = self.rows().len() as i32;
236        self.cursor = (self.cursor as i32 + d).clamp(0, n - 1) as usize;
237        self.settle(d);
238    }

248    /// The colour cells of a picker tab.
249    pub fn grid(&self, tab: usize) -> Vec<(String, ColorSpec)> {
250        match tab {
251            0 => themes::ansi_colors().iter().map(|(n, c)| (n.to_string(), ColorSpec::Value(*c))).collect(),
252            1 => (16u8..=231).map(|i| (i.to_string(), ColorSpec::Value(Color::Indexed(i)))).collect(),
253            2 => (232u8..=255).map(|i| (i.to_string(), ColorSpec::Value(Color::Indexed(i)))).collect(),
254            _ => {
255                let own = match self.picker.as_ref().map(|p| &p.target) { Some(Target::Palette(n)) => Some(n.clone()), _ => None };
256                let mut v: Vec<(String, ColorSpec)> = self.palette_names().into_iter().filter(|n| Some(n) != own.as_ref())
257                    .map(|n| (format!("@{n}"), ColorSpec::Ref(n))).collect();
258                v.push(("reset".into(), ColorSpec::Value(Color::Reset)));
259                v
260            }
261        }
262    }

264    /// Cells per grid row of a picker tab.
265    pub fn grid_cols(tab: usize) -> usize { match tab { 0 => 8, 1 => 18, 2 => 12, _ => 1 } }

267    fn open_picker(&mut self, target: Target) {
268        let cur = match &target { Target::Palette(n) => self.file.section(self.variant).get(n).cloned(), Target::Field(e, k) => self.file.section(self.variant).style(*e).color(k).cloned() };
269        let mut p = ColorPicker { target, tab: 0, sel: 0, hex: TextField::default(), error: None };
270        if let Some(c) = &cur { p.hex = TextField::new(&themes::color_text(c)); }
271        self.picker = Some(p);
272        if let Some(c) = cur {
273            for tab in 0..PICKER_TABS.len() {
274                if let Some(i) = self.grid(tab).iter().position(|(_, g)| *g == c) {
275                    if let Some(p) = self.picker.as_mut() { p.tab = tab; p.sel = i; }
276                    break;
277                }
278            }
279        }
280    }

282    /// Contrast of colour `c` for the picker's target, when below [`THEME_MIN_CONTRAST`].
283    pub fn picker_warning(&self, c: &ColorSpec) -> Option<f64> {
284        let p = self.picker.as_ref()?;
285        let mut f = self.file.clone();
286        set_target(&mut f, self.variant, &p.target, Some(c.clone()));
287        let row = match &p.target { Target::Palette(n) => Row::Palette(n.clone()), Target::Field(e, k) => Row::Field(*e, k) };
288        warning(&f, self.variant, &row)
289    }

291    fn picker_key(&mut self, k: KeyEvent) {
292        let Some(mut p) = self.picker.take() else { return };
293        let cells = self.grid(p.tab).len();
294        let cols = Self::grid_cols(p.tab);
295        match k.code {
296            KeyCode::Esc => return,
297            KeyCode::Tab => { p.tab = (p.tab + 1) % PICKER_TABS.len(); p.sel = 0; }
298            KeyCode::BackTab => { p.tab = (p.tab + PICKER_TABS.len() - 1) % PICKER_TABS.len(); p.sel = 0; }
299            KeyCode::Left => p.sel = p.sel.saturating_sub(1),
300            KeyCode::Right => p.sel = (p.sel + 1).min(cells.saturating_sub(1)),
301            KeyCode::Up => p.sel = p.sel.saturating_sub(cols),
302            KeyCode::Down => p.sel = (p.sel + cols).min(cells.saturating_sub(1)),
303            KeyCode::Enter => {
304                let typed = p.hex.value.trim().to_string();
305                let pick = if !typed.is_empty() && Some(typed.as_str()) != self.value_text(&p.target).as_deref() {
306                    match themes::parse_color_str(&typed) {
307                        Ok(c) => c,
308                        Err(e) => { p.error = Some(e); self.picker = Some(p); return; }
309                    }
310                } else {
311                    let tab = p.tab;
312                    self.picker = Some(p.clone());
313                    let g = self.grid(tab);
314                    self.picker = None;
315                    match g.get(p.sel) { Some((_, c)) => c.clone(), None => return }
316                };
317                if let ColorSpec::Ref(r) = &pick && !self.palette_names().contains(r) {
318                    p.error = Some(format!("@{r} is not in the palette"));
319                    self.picker = Some(p);
320                    return;
321                }
322                self.set_target(&p.target, Some(pick));
323                return;
324            }
325            _ => { if p.hex.on_key(k) { p.error = None; } }
326        }
327        self.picker = Some(p);
328    }

339    fn activate(&mut self) {
340        match self.current() {
341            Row::Palette(n) => self.open_picker(Target::Palette(n)),
342            Row::Element(e) => { self.expanded = if self.expanded == Some(e) { None } else { Some(e) }; }
343            Row::Field(e, k) if ["fg", "bg", "border"].contains(&k) => self.open_picker(Target::Field(e, k)),
344            Row::Field(e, k) => {
345                let v = self.variant;
346                let mut st = self.file.section(v).style(e);
347                st.set_flag(k, match st.flag(k) { None => Some(true), Some(true) => Some(false), Some(false) => None });
348                self.file.section_mut(v).set_style(e, st);
349                self.sync_preview();
350            }
351            Row::Heading(_) => {}
352        }
353    }
354
355    fn clear(&mut self) {
356        let v = self.variant;
357        match self.current() {
358            Row::Palette(n) => {
359                let core = PALETTE_KEYS.iter().any(|(k, _)| *k == n);
360                let sec = self.file.section_mut(v);
361                if core { sec.set(&n, ColorSpec::Value(Color::Reset)); } else { sec.palette.retain(|(x, _)| *x != n); }
362            }
363            Row::Element(e) => self.file.section_mut(v).set_style(e, Default::default()),
364            Row::Field(e, k) => {
365                let mut st = self.file.section(v).style(e);
366                st.set_color(k, None);
367                st.set_flag(k, None);
368                self.file.section_mut(v).set_style(e, st);
369            }
370            Row::Heading(_) => {}
371        }
372        self.settle(1);
373        self.sync_preview();
374    }

378    /// Save to the themes folder (and use it at once when it is the selected theme).
379    pub fn save(&mut self, app: &mut App) -> bool {
380        let Some(dir) = app.themes_dir.clone() else { self.message = Some((true, "no themes folder".into())); return false };
381        match themes::save(&dir, &self.file) {
382            Ok(p) => {
383                self.saved = Some(self.file.clone());
384                self.message = Some((false, format!("saved {}", p.display())));
385                app.push_status(format!("theme saved: {}", p.display()));
386                if app.theme_name == self.file.name { let n = self.file.name.clone(); app.use_theme(&n, false); }
387                true
388            }
389            Err(e) => { self.message = Some((true, format!("{e:#}"))); false }
390        }
391    }

410    fn step_theme(&mut self, app: &App, d: i32) {
411        let names = Self::names(app);
412        if names.is_empty() { return; }
413        let i = names.iter().position(|n| *n == self.file.name).map(|i| (i as i32 + d).rem_euclid(names.len() as i32) as usize).unwrap_or(0);
414        let next = names[i].clone();
415        if self.dirty() { self.ask(Ask::UnsavedOpen(next)); } else { self.open_other(app, &next); }
416    }

418    fn prompt_key(&mut self, app: &mut App, k: KeyEvent) -> Outcome {
419        let Some(mut p) = self.prompt.take() else { return Outcome::Stay };
420        let named = matches!(p.ask, Ask::NewName | Ask::DuplicateName | Ask::PaletteName);
421        if named {
422            match k.code {
423                KeyCode::Esc => return Outcome::Stay,
424                KeyCode::Enter => {
425                    let name = p.field.value.trim().to_string();
426                    if !themes::valid_name(&name) { p.error = Some("use 1–32 of a-z, 0-9, - and _".into()); self.prompt = Some(p); return Outcome::Stay; }
427                    match p.ask {
428                        Ask::PaletteName => {
429                            if self.palette_names().contains(&name) { p.error = Some(format!("{name} is already in the palette")); self.prompt = Some(p); return Outcome::Stay; }
430                            self.file.section_mut(self.variant).set(&name, ColorSpec::Value(Color::Reset));
431                            if let Some(i) = self.rows().iter().position(|r| *r == Row::Palette(name.clone())) { self.cursor = i; }
432                            self.open_picker(Target::Palette(name));
433                        }
434                        _ => {
435                            if Self::names(app).contains(&name) { p.error = Some(format!("{name}.toml already exists")); self.prompt = Some(p); return Outcome::Stay; }
436                            let mut f = if p.ask == Ask::NewName {
437                                let dir = app.themes_dir.clone().unwrap_or_default();
438                                themes::load(&dir, DEFAULT_THEME).map(|l| l.file).unwrap_or_else(|_| ThemeFile::empty(&name))
439                            } else { self.file.clone() };
440                            f.name = name.clone();
441                            f.description = if p.ask == Ask::NewName { "A new theme".into() } else { format!("A copy of {}", self.file.name) };
442                            self.file = f;
443                            self.saved = None;
444                            self.message = Some((false, format!("{name}: not saved yet (^S saves)")));
445                            self.sync_preview();
446                        }
447                    }
448                    return Outcome::Stay;
449                }
450                _ => { p.field.on_key(k); p.error = None; }
451            }
452            self.prompt = Some(p);
453            return Outcome::Stay;
454        }
455        match (&p.ask, k.code) {
456            (_, KeyCode::Esc) | (_, KeyCode::Char('c')) | (Ask::Delete(_), KeyCode::Char('n')) => Outcome::Stay,
457            (Ask::Delete(name), KeyCode::Char('y')) => {
458                let name = name.clone();
459                if name == app.theme_name { self.message = Some((true, format!("{name} is the theme in use; switch to another (F5) first"))); return Outcome::Stay; }
460                let dir = app.themes_dir.clone().unwrap_or_default();
461                match themes::delete(&dir, &name) {
462                    Ok(()) => {
463                        app.push_status(format!("theme {name} deleted"));
464                        let cur = app.theme_name.clone();
465                        self.open_other(app, &cur);
466                        self.message = Some((false, format!("deleted {name}.toml")));
467                    }
468                    Err(e) => self.message = Some((true, format!("{e:#}"))),
469                }
470                Outcome::Stay
471            }
472            (Ask::UnsavedClose | Ask::UnsavedOpen(_), KeyCode::Char('s') | KeyCode::Enter) => {
473                if !self.save(app) { return Outcome::Stay; }
474                self.after_unsaved(app, p.ask)
475            }
476            (Ask::UnsavedClose | Ask::UnsavedOpen(_), KeyCode::Char('d')) => self.after_unsaved(app, p.ask),
477            _ => { self.prompt = Some(p); Outcome::Stay }
478        }
479    }

488    /// Handle a key.
489    pub fn on_key(&mut self, app: &mut App, k: KeyEvent) -> Outcome {
490        if self.prompt.is_some() { return self.prompt_key(app, k); }
491        if self.picker.is_some() { self.picker_key(k); return Outcome::Stay; }
492        let ctrl = k.modifiers.contains(KeyModifiers::CONTROL);
493        self.message = None;
494        match k.code {
495            KeyCode::Char('s') if ctrl => { self.save(app); }
496            KeyCode::Char('r') if ctrl => self.revert(),
497            KeyCode::Char('n') if ctrl => self.ask(Ask::NewName),
498            KeyCode::Char('d') if ctrl => self.ask(Ask::DuplicateName),
499            KeyCode::Char('x') if ctrl => { let n = self.file.name.clone(); self.ask(Ask::Delete(n)); }
500            KeyCode::Char('c') if ctrl => return self.close_or_ask(),
501            KeyCode::Esc | KeyCode::F(17) => return self.close_or_ask(),
502            KeyCode::F(5) if k.modifiers.contains(KeyModifiers::SHIFT) => return self.close_or_ask(),
503            KeyCode::Tab | KeyCode::BackTab => { self.variant = self.variant.flip(); self.settle(1); self.sync_preview(); }
504            KeyCode::Up | KeyCode::Char('k') => self.move_cursor(-1),
505            KeyCode::Down | KeyCode::Char('j') => self.move_cursor(1),
506            KeyCode::PageUp => self.move_cursor(-10),
507            KeyCode::PageDown => self.move_cursor(10),
508            KeyCode::Home => { self.cursor = 0; self.settle(1); }
509            KeyCode::End => { self.cursor = self.rows().len() - 1; self.settle(-1); }
510            KeyCode::Enter | KeyCode::Char(' ') => self.activate(),
511            KeyCode::Right | KeyCode::Char('l') => if let Row::Element(e) = self.current() { self.expanded = Some(e); },
512            KeyCode::Left | KeyCode::Char('h') => {
513                if let Row::Field(e, _) = self.current() && let Some(i) = self.rows().iter().position(|r| *r == Row::Element(e)) { self.cursor = i; }
514                self.expanded = None;
515            }
516            KeyCode::Delete | KeyCode::Backspace => self.clear(),
517            KeyCode::Char('a') | KeyCode::Char('+') => self.ask(Ask::PaletteName),
518            KeyCode::Char('[') => self.step_theme(app, -1),
519            KeyCode::Char(']') => self.step_theme(app, 1),
520            _ => {}
521        }
522        Outcome::Stay
523    }
524
525    fn close_or_ask(&mut self) -> Outcome {
526        if self.dirty() { self.ask(Ask::UnsavedClose); Outcome::Stay } else { Outcome::Close }
527    }
528
529    /// A click on list row `i` selects it (a second click activates it).
530    pub fn click_row(&mut self, i: usize) {
531        if self.prompt.is_some() || self.picker.is_some() { return; }
532        if matches!(self.rows().get(i), Some(Row::Heading(_)) | None) { return; }
533        if self.cursor == i { self.activate(); } else { self.cursor = i; }
534    }
535}

537fn set_target(f: &mut ThemeFile, v: Variant, t: &Target, c: Option<ColorSpec>) {
538    match t {
539        Target::Palette(n) => f.section_mut(v).set(n, c.unwrap_or(ColorSpec::Value(Color::Reset))),
540        Target::Field(e, k) => {
541            let mut st = f.section(v).style(*e);
542            st.set_color(k, c);
543            f.section_mut(v).set_style(*e, st);
544        }
545    }
546}

548/// The background text sits on when a theme leaves it to the terminal (assumed
549/// black for dark, white for light), and the matching text colour.
550fn assumed(t: &Theme, v: Variant) -> (Color, Color) {
551    let dark = v == Variant::Dark;
552    let bg = if t.bg != Color::Reset { t.bg } else if dark { Color::Black } else { Color::White };
553    let fg = if t.fg != Color::Reset { t.fg } else if dark { Color::White } else { Color::Black };
554    (fg, bg)
555}

557/// Every palette and element row of `f`'s `v` variant (nothing expanded), e.g. to check all
558/// of a theme's contrast warnings at once.
559pub fn all_rows(f: &ThemeFile, v: Variant) -> Vec<Row> {
560    let s = f.section(v);
561    let mut names: Vec<String> = PALETTE_KEYS.iter().map(|(k, _)| k.to_string()).collect();
562    for (n, _) in &s.palette { if !names.contains(n) { names.push(n.clone()); } }
563    names.into_iter().map(Row::Palette).chain(Element::ALL.iter().map(|e| Row::Element(*e))).collect()
564}

566/// A row's text contrast in `f` when below [`THEME_MIN_CONTRAST`].
567pub fn warning(f: &ThemeFile, v: Variant, r: &Row) -> Option<f64> {
568    let t = Theme::from_file(f, v);
569    let (fg, bg) = assumed(&t, v);
570    let ratio = match r {
571        Row::Palette(n) => match n.as_str() {
572            "on_accent" => contrast(t.on_accent, t.accent),
573            "accent" => contrast(t.on_accent, t.accent).into_iter().chain(contrast(t.accent, bg)).reduce(f64::min),
574            "muted" => contrast(t.muted, bg),
575            "fg" | "bg" => contrast(fg, bg),
576            _ => None,
577        },
578        Row::Element(e) | Row::Field(e, _) => {
579            if matches!(e, Element::Dimmed | Element::Border | Element::BorderFocused) || e.is_box() { return None; }
580            let st = ratatui::style::Style::from(&t.element(*e));
581            let (mut a, mut b) = (st.fg.filter(|c| *c != Color::Reset).unwrap_or(fg), st.bg.filter(|c| *c != Color::Reset).unwrap_or(bg));
582            if st.add_modifier.contains(ratatui::style::Modifier::REVERSED) { std::mem::swap(&mut a, &mut b); }
583            contrast(a, b)
584        }
585        Row::Heading(_) => None,
586    };
587    ratio.filter(|x| *x < THEME_MIN_CONTRAST)
588}
