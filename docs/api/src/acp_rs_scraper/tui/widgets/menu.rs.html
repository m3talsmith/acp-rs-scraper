12/// Label of the pinned scan button.
13pub fn scan_label(app: &App) -> &'static str { if app.job.is_some() { " ■ Cancel " } else { " ▶ Start Scanning " } }
14
15/// The scan button on a narrow menu bar.
16pub fn short_scan_label(app: &App) -> &'static str { if app.job.is_some() { " ■ Cancel " } else { " ▶ Scan " } }

18/// Draw the menu bar into `r` (one row); the dropdown is drawn by [`draw_dropdown`].
19pub fn draw(f: &mut Frame, app: &mut App, r: Rect) {
20    let t = app.theme;
21    let mut x = r.x;
22    // Wide: " Label " │ " Label "; narrow: drop the separators, then the padding,
23    // then shorten the scan button so every command (Quit included) still shows.
24    let labels: u16 = app.menu.menu.items.iter().map(|c| c.label.chars().count() as u16).sum();
25    let n = app.menu.menu.items.len() as u16;
26    let full = scan_label(app);
27    let scan = if labels + 2 * n + full.chars().count() as u16 + 1 > r.width { short_scan_label(app) } else { full };
28    let scan_w = scan.chars().count() as u16;
29    let limit = r.x + r.width.saturating_sub(scan_w + 1);
30    let room = limit.saturating_sub(r.x);
31    let (pad, sep) = if labels + 3 * n <= room { (1, true) } else if labels + 2 * n <= room { (1, false) } else { (0, false) };
32    for (i, c) in app.menu.menu.items.iter().enumerate() {
33        let label = c.label;
34        let w = label.chars().count() as u16 + 2 * pad + u16::from(pad == 0);
35        if x + w > limit { break; }
36        let active = app.menu.open == Some(i) || (app.focus == Focus::Menu && app.menu.item == i);
37        let base = if active { t.selected() } else { Style::default() };
38        let hint = label.chars().position(|ch| ch.to_ascii_lowercase() == c.hint).unwrap_or(0);
39        let mut spans = vec![Span::styled(if pad == 1 { " " } else { "" }, base)];
40        for (k, ch) in label.chars().enumerate() {
41            let st = if k == hint { base.add_modifier(Modifier::UNDERLINED) } else { base };
42            spans.push(Span::styled(ch.to_string(), st));
43        }
44        spans.push(Span::styled(" ", base));
45        let rr = Rect { x, y: r.y, width: w, height: 1 };
46        f.render_widget(Paragraph::new(Line::from(spans)), rr);
47        app.hits.push((rr, Hit::MenuItem(i)));
48        x += w;
49        if sep && x < limit { f.render_widget(Paragraph::new(Span::styled("│", t.muted())), Rect { x, y: r.y, width: 1, height: 1 }); x += 1; }
50    }
51    let sr = Rect { x: r.x + r.width.saturating_sub(scan_w), y: r.y, width: scan_w.min(r.width), height: 1 };
52    let st = Style::default().bg(t.accent).fg(t.on_accent).add_modifier(Modifier::BOLD);
53    f.render_widget(Paragraph::new(Span::styled(scan, st)), sr);
54    app.hits.push((sr, Hit::StartScan));
55}

57/// Widget text right of an option label.
58pub fn widget_text(w: &Widget, editing: bool) -> String {
59    match w {
60        Widget::Check(_) => String::new(),
61        Widget::Choice { choices, idx } => format!("‹ {} ›", choices[*idx]),
62        Widget::Text(tf) => {
63            let mut s = tf.display();
64            if editing { let b = s.char_indices().nth(tf.cursor).map(|(b, _)| b).unwrap_or(s.len()); s.insert(b, '▏'); }
65            format!("[{s:<12}]")
66        }
67        Widget::Button(_) => String::new(),
68    }
69}

71/// Draw the open dropdown under its item (on top of the layout).
72pub fn draw_dropdown(f: &mut Frame, app: &mut App, screen: Rect) {
73    let Some(i) = app.menu.open else { return };
74    let t = app.theme;
75    let x = app.hits.iter().find(|(_, h)| *h == Hit::MenuItem(i)).map(|(r, _)| r.x).unwrap_or(screen.x);
76    let c = &app.menu.menu.items[i];
77    let rows: Vec<(Line, bool)> = c.opts.iter().enumerate().map(|(k, o)| {
78        let cur = k == app.menu.opt;
79        let editing = cur && app.menu.editing;
80        let req = if o.required { "*" } else { " " };
81        let base = if cur { t.selected() } else { Style::default() };
82        let line = match &o.widget {
83            Widget::Check(v) => Line::from(vec![Span::styled(req, t.accent()), Span::styled(format!(" [{}] {}", if *v { "x" } else { " " }, o.label), base)]),
84            Widget::Button(_) => Line::from(vec![Span::styled(req, t.accent()), Span::styled(format!("  < {} >", o.label), if cur { base } else { t.accent_bold() })]),
85            w => Line::from(vec![Span::styled(req, t.accent()), Span::styled(format!("     {} ", o.label), base),
86                Span::styled(widget_text(w, editing), if editing { t.accent_bold() } else { base })]),
87        };
88        (line, cur)
89    }).collect();
90    let hint = c.opts.get(app.menu.opt).map(|o| o.flag).unwrap_or("");
91    let w = rows.iter().map(|(l, _)| l.width()).max().unwrap_or(10).max(hint.chars().count() + 2).max(c.label.len() + 4) as u16 + 4;
92    let h = rows.len() as u16 + 3;
93    let w = w.min(screen.width);
94    let x = x.min(screen.x + screen.width - w);
95    let r = Rect { x, y: screen.y + 2, width: w, height: h.min(screen.height.saturating_sub(2)) };
96    f.render_widget(Clear, r);
97    let b = Block::default().borders(Borders::ALL).border_type(BorderType::Double).border_style(t.accent()).title(format!(" {} ", c.label));
98    let inner = b.inner(r);
99    f.render_widget(b, r);
100    for (k, (line, _)) in rows.into_iter().enumerate() {
101        if k as u16 >= inner.height { break; }
102        let rr = Rect { x: inner.x, y: inner.y + k as u16, width: inner.width, height: 1 };
103        f.render_widget(Paragraph::new(line), rr);
104        app.hits.push((rr, Hit::MenuOpt(k)));
105    }
106    if inner.height as usize > c.opts.len() {
107        f.render_widget(Paragraph::new(Span::styled(format!(" {hint}"), t.muted())),
108            Rect { x: inner.x, y: inner.y + c.opts.len() as u16, width: inner.width, height: 1 });
109    }
