1//! Single-line text field state (cursor editing), shared by menu options and modals.
2
3use crossterm::event::{KeyCode, KeyEvent, KeyModifiers};
4
5/// An editable line of text with a cursor (in chars).
6#[derive(Clone, Debug, Default, PartialEq)]
7pub struct TextField {
8    /// Current text.
9    pub value: String,
10    /// Cursor position, in chars.
11    pub cursor: usize,
12    /// Show `•` instead of the characters.
13    pub masked: bool,
14}
15
16impl TextField {
17    /// A field holding `v`, cursor at the end.
18    pub fn new(v: &str) -> Self { TextField { value: v.into(), cursor: v.chars().count(), masked: false } }
19    /// Replace the text, cursor at the end.
20    pub fn set(&mut self, v: &str) { *self = TextField { masked: self.masked, ..TextField::new(v) }; }
21    fn byte(&self, i: usize) -> usize { self.value.char_indices().nth(i).map(|(b, _)| b).unwrap_or(self.value.len()) }
22    /// Insert a char at the cursor.
23    pub fn insert(&mut self, c: char) { let b = self.byte(self.cursor); self.value.insert(b, c); self.cursor += 1; }
24    /// Apply an editing key; returns whether it was consumed.
25    pub fn on_key(&mut self, k: KeyEvent) -> bool {
26        let n = self.value.chars().count();
27        match k.code {
28            KeyCode::Char('u') if k.modifiers.contains(KeyModifiers::CONTROL) => { self.set(""); }
29            KeyCode::Char(c) if !k.modifiers.contains(KeyModifiers::CONTROL) && !k.modifiers.contains(KeyModifiers::ALT) => self.insert(c),
30            KeyCode::Backspace if self.cursor > 0 => {
31                let b = self.byte(self.cursor - 1);
32                self.value.remove(b);
33                self.cursor -= 1;
34            }
35            KeyCode::Delete if self.cursor < n => { let b = self.byte(self.cursor); self.value.remove(b); }
36            KeyCode::Left => self.cursor = self.cursor.saturating_sub(1),
37            KeyCode::Right => self.cursor = (self.cursor + 1).min(n),
38            KeyCode::Home => self.cursor = 0,
39            KeyCode::End => self.cursor = n,
40            KeyCode::Backspace | KeyCode::Delete => {}
41            _ => return false,
42        }
43        true
44    }
45    /// Text as displayed (masked or not).
46    pub fn display(&self) -> String {
47        if self.masked { "•".repeat(self.value.chars().count()) } else { self.value.clone() }
48    }
49}
