1//! Post detail content, at parity with the HTML export: Post/Reply badge,
2//! `@nick · posted_at · post_id`, "Replying to … · thread …", title, body with
3//! its paragraphs, attachments (path + open hint), and every cross-link.
4
5use crate::prelude::*;
6use crate::tui::layout::{ellipsize, wrap};
7use crate::tui::state::{DetailFrame, LinkKind};
8use crate::tui::theme::Theme;
9use ratatui::text::{Line, Span};
10
11/// `Post` / `Reply` badge.
12pub fn badge(p: &Post, t: &Theme) -> Span<'static> {
13    if p.parent_post_id.is_empty() {
14        Span::styled(" Post ", t.post_tag())
15    } else {
16        Span::styled(" Reply ", t.reply_tag())
17    }
18}
19
20/// `[badge] @nick · posted_at · post_id`.
21pub fn meta_line(p: &Post, t: &Theme) -> Line<'static> {
22    let mut v = vec![badge(p, t), Span::raw(" "),
23        Span::styled(format!("@{}", p.nick), t.bold()),
24        Span::styled(format!(" · {} · {}", p.posted_at, p.post_id), t.muted())];
25    if p.truncated { v.push(Span::styled(" · shortened (run --refetch-truncated)", t.muted())); }
26    Line::from(v)
27}

29/// First non-empty line of the title or body.
30pub fn snippet(p: &Post) -> String {
31    let s = if p.title.trim().is_empty() { &p.content } else { &p.title };
32    s.lines().map(str::trim).find(|l| !l.is_empty()).unwrap_or("(no text)").to_string()
33}
34
35/// Two-row header of a collapsed frame.
36pub fn header_lines(f: &DetailFrame, width: usize, t: &Theme) -> Vec<Line<'static>> {
37    vec![meta_line(&f.post, t), Line::styled(ellipsize(&snippet(&f.post), width), t.muted())]
38}

40fn link_label(kind: LinkKind) -> &'static str {
41    match kind { LinkKind::ReplyTo => "↳ Replying to", LinkKind::Thread => "⌂ Thread", LinkKind::Mentioned => "→ Linked post",
42        LinkKind::Reply => "↩ Reply", LinkKind::Attachment => "▣ Open" }
43}
44
45/// Full detail lines for `width` columns, and (line index, link index) for each link row.
46pub fn detail_lines(f: &DetailFrame, width: usize, t: &Theme) -> (Vec<Line<'static>>, Vec<(usize, usize)>) {
47    let (lines, links, _) = detail_layout(f, width, t, 0);
48    (lines, links)
49}

51/// Rows reserved for inline pictures: (first line index, image file).
52pub type ImageSlots = Vec<(usize, std::path::PathBuf)>;
53
54/// [`detail_lines`] plus `img_rows` blank rows under the text for each stored
55/// picture (0 = no pictures), and where those rows start.
56pub fn detail_layout(f: &DetailFrame, width: usize, t: &Theme, img_rows: u16)
57    -> (Vec<Line<'static>>, Vec<(usize, usize)>, ImageSlots) {
58    let mut slots = Vec::new();
59    let mut lines = vec![meta_line(&f.post, t)];
60    let mut links = Vec::new();
61    let link_line = |i: usize, lines: &mut Vec<Line<'static>>, links: &mut Vec<(usize, usize)>| {
62        let l = &f.links[i];
63        let sel = i == f.link_cursor;
64        if l.kind == LinkKind::Attachment {
65            let a = f.attachments.iter().find(|a| a.local_path.display().to_string() == l.id);
66            let mime = a.map(|a| a.mime_type.as_str()).unwrap_or("file");
67            let exists = std::path::Path::new(&l.id).exists();
68            links.push((lines.len(), i));
69            let name = std::path::Path::new(&l.id).file_name().map(|n| n.to_string_lossy().into_owned()).unwrap_or_else(|| l.id.clone());
70            lines.push(Line::from(vec![
71                Span::styled(format!("{} {mime}", link_label(l.kind)), if sel { t.selected() } else { t.accent() }),
72                Span::raw("  "),
73                Span::styled(name, if exists { t.accent().underline(true) } else { t.muted().strikethrough(true) }),
74                Span::styled(if exists { "" } else { "  (file missing)" }, t.muted()),
75            ]));
76            return;
77        }
78        let who = match &l.target {
79            Some(p) => format!("@{} · {} · {}", p.nick, p.posted_at, ellipsize(&snippet(p), width.saturating_sub(40).max(10))),
80            None => "not in database".into(),
81        };
82        let st = if sel { t.selected() } else { t.accent() };
83        links.push((lines.len(), i));
84        lines.push(Line::from(vec![
85            Span::styled(format!("{} {}", link_label(l.kind), l.id), st.underline(true)),
86            Span::styled(format!("  {who}"), if l.target.is_some() { t.muted() } else { t.muted().italic(true) }),
87        ]));
88    };
89    // "Replying to … · thread …" right under the meta line, like the HTML export.
90    for i in 0..f.links.len() {
91        if matches!(f.links[i].kind, LinkKind::ReplyTo | LinkKind::Thread) { link_line(i, &mut lines, &mut links); }
92    }
93    if let Some(n) = &f.notice { lines.push(Line::styled(format!("⚠ {n}"), t.accent_bold())); }
94    lines.push(Line::raw(""));
95    if !f.post.title.trim().is_empty() {
96        for l in wrap(f.post.title.trim(), width) { lines.push(Line::styled(l, t.bold())); }
97        lines.push(Line::raw(""));
98    }
99    lines.extend(crate::tui::markdown::render(f.post.content.trim_end(), width, t));
100    if img_rows > 0 {
101        for a in &f.attachments {
102            if !crate::tui::images::is_image(&a.mime_type, &a.local_path) || !a.local_path.exists() { continue; }
103            lines.push(Line::raw(""));
104            let name = a.local_path.file_name().map(|n| n.to_string_lossy().into_owned()).unwrap_or_default();
105            lines.push(Line::styled(format!("🖼 {name}"), t.muted()));
106            slots.push((lines.len(), a.local_path.clone()));
107            for _ in 0..img_rows { lines.push(Line::raw("")); }
108        }
109    }
110    let atts: Vec<usize> = (0..f.links.len()).filter(|i| f.links[*i].kind == LinkKind::Attachment).collect();
111    if !atts.is_empty() {
112        lines.push(Line::raw(""));
113        lines.push(Line::styled(format!("Attachments ({}) · Enter / click opens", atts.len()), t.bold()));
114        for i in atts { link_line(i, &mut lines, &mut links); }
115    }
116    let others: Vec<usize> = (0..f.links.len()).filter(|i| matches!(f.links[*i].kind, LinkKind::Mentioned)).collect();
117    if !others.is_empty() {
118        lines.push(Line::raw(""));
119        lines.push(Line::styled("Linked posts", t.bold()));
120        for i in others { link_line(i, &mut lines, &mut links); }
121    }
122    let replies: Vec<usize> = (0..f.links.len()).filter(|i| matches!(f.links[*i].kind, LinkKind::Reply)).collect();
123    if !replies.is_empty() {
124        lines.push(Line::raw(""));
125        lines.push(Line::styled(format!("Replies ({})", replies.len()), t.bold()));
126        for i in replies { link_line(i, &mut lines, &mut links); }
127    }
128    if !f.links.is_empty() {
129        lines.push(Line::raw(""));
130        lines.push(Line::styled("←/→ choose a link · Enter opens it · Esc closes", t.muted()));
131    }
132    (lines, links, slots)
133}
