1//! Left nick bar: `@nick` and post count; multi-select.
2
3use crate::tui::state::{App, Focus, Hit};
4use ratatui::Frame;
5use ratatui::layout::Rect;
6use ratatui::text::{Line, Span};
7use ratatui::widgets::Paragraph;
8
9/// Draw the nick bar into `r`.
10pub fn draw(f: &mut Frame, app: &mut App, r: Rect) {
11    let t = app.theme;
12    let count = if app.nick_filter_active() { format!("{}/{}", app.nicks.len(), app.all_nicks.len()) } else { app.nicks.len().to_string() };
13    let title = if app.selected.len() > 1 { format!(" Nicks ({count}) · {} selected ", app.selected.len()) } else { format!(" Nicks ({count}) ") };
14    let b = crate::tui::theme::frame(&t.pane(app.focus == Focus::Nicks)).title(title);
15    let mut inner = b.inner(r);
16    f.render_widget(b, r);
17    app.hits.push((r, Hit::NickPane));
18    // "★ Bookmarks (n)" row on top: opens the bookmarks view.
19    if inner.height > 1 {
20        let br = Rect { height: 1, ..inner };
21        let on = app.search.as_ref().is_some_and(|v| v.bookmarks);
22        let st = if on { t.selected() } else { t.accent_bold() };
23        f.render_widget(Paragraph::new(Line::from(vec![Span::styled(format!(" ★ Bookmarks ({})", app.marks.len()), st)])), br);
24        app.hits.push((br, Hit::Bookmarks));
25        inner = Rect { y: inner.y + 1, height: inner.height - 1, ..inner };
26    }
27    // Filter field: "/ text▏  3/390 x".
28    if inner.height > 1 && !app.all_nicks.is_empty() {
29        let fr = Rect { height: 1, ..inner };
30        let active = app.nick_filter_active();
31        let tail = if active { format!(" {}/{} x ", app.nicks.len(), app.all_nicks.len()) } else { String::new() };
32        let tw = unicode_width::UnicodeWidthStr::width(tail.as_str()) as u16;
33        let fw = fr.width.saturating_sub(tw + 3) as usize;
34        let mut text = app.nick_filter.display();
35        if app.filter_editing { let b = text.char_indices().nth(app.nick_filter.cursor).map(|(b, _)| b).unwrap_or(text.len()); text.insert(b, '▏'); }
36        let shown = if text.is_empty() { Span::styled(crate::tui::layout::ellipsize("filter nicks (/)", fw), t.muted()) }
37            else { Span::styled(crate::tui::layout::ellipsize(&text, fw), if app.filter_editing { t.selected() } else { t.plain() }) };
38        let lead = Span::styled(" / ", if app.filter_editing { t.accent_bold() } else { t.muted() });
39        f.render_widget(Paragraph::new(Line::from(vec![lead, shown])), fr);
40        app.hits.push((Rect { width: fr.width.saturating_sub(tw), ..fr }, Hit::NickFilter));
41        if active {
42            let cr = Rect { x: fr.x + fr.width - tw, width: tw, ..fr };
43            f.render_widget(Paragraph::new(Span::styled(tail, t.muted())), cr);
44            // Only the "x" clears; the count is part of the field.
45            app.hits.push((Rect { x: cr.x + tw.saturating_sub(2), width: 2.min(tw), ..cr }, Hit::NickFilterClear));
46        }
47        inner = Rect { y: inner.y + 1, height: inner.height - 1, ..inner };
48    }
49    if app.nicks.is_empty() {
50        let msg = if app.all_nicks.is_empty() { vec![Line::raw("No nicks yet —"), Line::raw("use Scrape or Import.")] }
51            else { vec![Line::raw("No nick matches."), Line::raw("Esc or x clears the filter.")] };
52        f.render_widget(Paragraph::new(msg).style(t.muted()), inner);
53        return;
54    }
55    let h = inner.height as usize;
56    if app.nick_cursor < app.nick_offset { app.nick_offset = app.nick_cursor; }
57    if h > 0 && app.nick_cursor >= app.nick_offset + h { app.nick_offset = app.nick_cursor + 1 - h; }
58    for (row, i) in (app.nick_offset..app.nicks.len()).take(h).enumerate() {
59        let n = &app.nicks[i];
60        let sel = app.selected.iter().any(|s| s == &n.nick);
61        let cur = i == app.nick_cursor;
62        let rr = Rect { x: inner.x, y: inner.y + row as u16, width: inner.width, height: 1 };
63        let count = format!("{} ", n.post_count);
64        let mark = if sel { "▌" } else { " " };
65        let w = (inner.width as usize).saturating_sub(count.len() + 2);
66        let name = crate::tui::layout::ellipsize(&format!("@{}", n.nick), w);
67        let pad = w.saturating_sub(unicode_width::UnicodeWidthStr::width(name.as_str()));
68        let mut st = if sel { t.accent_bold() } else { t.plain() };
69        if cur && app.focus == Focus::Nicks { st = t.selected(); } else if cur { st = st.reverse(true); }
70        let line = Line::from(vec![Span::styled(mark, t.accent()), Span::styled(format!("{name}{}", " ".repeat(pad)), &st),
71            Span::styled(format!(" {count}"), if cur && app.focus == Focus::Nicks { st } else { t.muted() })]);
72        f.render_widget(Paragraph::new(line), rr);
73        app.hits.push((rr, Hit::Nick(i)));
74    }
75}
