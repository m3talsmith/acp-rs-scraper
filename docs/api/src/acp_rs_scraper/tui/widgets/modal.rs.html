1//! Modal rendering: dims the layout, then draws the modal with a small margin.

14fn dim(f: &mut Frame, area: Rect) {
15    let buf = f.buffer_mut();
16    for y in area.y..area.y + area.height {
17        for x in area.x..area.x + area.width {
18            if let Some(c) = buf.cell_mut((x, y)) { c.set_style(Style::default().add_modifier(Modifier::DIM)); }
19        }
20    }
21}

23fn frame(f: &mut Frame, t: &Theme, r: Rect, title: &str) -> Rect {
24    f.render_widget(Clear, r);
25    let b = Block::default().borders(Borders::ALL).border_type(BorderType::Double).border_style(t.accent()).title(format!(" {title} "));
26    let inner = b.inner(r);
27    f.render_widget(b, r);
28    inner
29}

31fn buttons(f: &mut Frame, app: &mut App, r: Rect, labels: &[String], focus: Option<usize>) {
32    let t = app.theme;
33    let mut x = r.x;
34    for (i, l) in labels.iter().enumerate() {
35        let s = format!("[ {l} ]");
36        let w = s.chars().count() as u16;
37        if x + w > r.x + r.width { break; }
38        let st = if focus == Some(i) { t.selected() } else { t.accent_bold() };
39        let br = Rect { x, y: r.y, width: w, height: 1 };
40        f.render_widget(Paragraph::new(Span::styled(s, st)), br);
41        app.hits.push((br, Hit::Modal(ModalHit::Button(i))));
42        x += w + 2;
43    }
44}

46fn field_line(label: &str, tf: &TextField, focused: bool, t: &Theme) -> Line<'static> {
47    let mut v = tf.display();
48    if focused { let b = v.char_indices().nth(tf.cursor).map(|(b, _)| b).unwrap_or(v.len()); v.insert(b, '▏'); }
49    Line::from(vec![Span::styled(format!("{label:>18}: "), if focused { t.accent_bold() } else { Style::default() }),
50        Span::styled(format!("{v:<30}"), Style::default().add_modifier(Modifier::UNDERLINED))])
51}

53/// Draw the open modal.
54pub fn draw(f: &mut Frame, app: &mut App, area: Rect) {
55    dim(f, area);
56    app.hits.push((area, Hit::Backdrop));
57    let t = app.theme;
58    let Some(m) = app.modal.take() else { return };
59    match &m {
60        Modal::Help(h) => {
61            let r = modal_rect(area, area.width, area.height);
62            let inner = frame(f, &t, r, "Help · ↑↓ PgUp PgDn scroll · Esc closes");
63            let body = Rect { height: inner.height.saturating_sub(1), ..inner };
64            let lines: Vec<Line> = h.lines.iter().skip(h.scroll).take(body.height as usize).map(|l| Line::raw(l.clone())).collect();
65            f.render_widget(Paragraph::new(lines), body);
66            buttons(f, app, Rect { y: inner.y + inner.height.saturating_sub(1), height: 1, ..inner }, &["Close".into()], Some(0));
67        }
68        Modal::Message(mm) => {
69            let w = 76;
70            let text: Vec<String> = mm.lines.iter().flat_map(|l| wrap(l, w as usize - 4)).collect();
71            let r = modal_rect(area, w, text.len() as u16 + 5);
72            let inner = frame(f, &t, r, &mm.title);
73            let st = if mm.error { t.accent() } else { Style::default() };
74            f.render_widget(Paragraph::new(text.into_iter().map(|l| Line::styled(l, st)).collect::<Vec<_>>()),
75                Rect { height: inner.height.saturating_sub(2), ..inner });
76            buttons(f, app, Rect { y: inner.y + inner.height.saturating_sub(1), height: 1, ..inner }, &["OK".into()], Some(0));
77        }
78        Modal::Confirm(c) => {
79            let w = 76;
80            let text: Vec<String> = c.lines.iter().flat_map(|l| wrap(l, w as usize - 4)).collect();
81            let extra = if c.strict { 2 } else { 0 };
82            let r = modal_rect(area, w, text.len() as u16 + 5 + extra);
83            let inner = frame(f, &t, r, &c.title);
84            let mut lines: Vec<Line> = text.into_iter().map(Line::raw).collect();
85            if c.strict { lines.push(Line::raw("")); lines.push(field_line("Type yes", &c.typed, true, &t)); }
86            if let Some(e) = &c.error { lines.push(Line::styled(e.clone(), t.accent())); }
87            f.render_widget(Paragraph::new(lines), Rect { height: inner.height.saturating_sub(2), ..inner });
88            buttons(f, app, Rect { y: inner.y + inner.height.saturating_sub(1), height: 1, ..inner }, &c.buttons, Some(c.focus));
89        }
90        Modal::Login(l) => {
91            let r = modal_rect(area, area.width, area.height);
92            let inner = frame(f, &t, r, &l.title());
93            let mut lines = vec![Line::raw("")];
94            if let Some(why) = &l.reason { lines.push(Line::styled(why.clone(), t.accent_bold())); lines.push(Line::raw("")); }
95            lines.push(Line::raw("acp.us logs you in with a one-time code sent by email."));
96            lines.push(Line::raw(""));
97            match &l.step {
98                LoginStep::Email => {
99                    lines.push(field_line("Email", &l.email, true, &t));
100                    lines.push(Line::raw(""));
101                    lines.push(Line::styled("Enter sends the code.", t.muted()));
102                }
103                LoginStep::Code => {
104                    lines.push(Line::styled(format!("{:>18}: {}", "Email", l.email.value), t.muted()));
105                    lines.push(field_line("Code or login link", &l.code, true, &t));
106                    lines.push(Line::raw(""));
107                    lines.push(Line::styled("Paste the code (or the whole login link) from the email; Enter verifies.", t.muted()));
108                }
109                LoginStep::Busy(s) => lines.push(Line::styled(format!("⟳ {s}"), t.accent())),
110            }
111            if let Some(e) = &l.error { lines.push(Line::raw("")); lines.push(Line::styled(format!("✗ {e}"), t.accent_bold())); }
112            lines.push(Line::raw(""));
113            lines.push(Line::styled("The session is saved in ~/.acp/session.json (mode 0600); browsing works offline without it.", t.muted()));
114            f.render_widget(Paragraph::new(lines), Rect { height: inner.height.saturating_sub(2), ..inner });
115            let submit = if matches!(l.step, LoginStep::Code) { "Verify" } else { "Send code" };
116            buttons(f, app, Rect { y: inner.y + inner.height.saturating_sub(1), height: 1, ..inner }, &[submit.into(), "Cancel".into()], None);
117        }
118        Modal::Search(s) => {
119            let r = modal_rect(area, 72, SEARCH_FIELDS.len() as u16 + 8);
120            let inner = frame(f, &t, r, "Search · Tab moves · Enter searches");
121            let nf = SEARCH_FIELDS.len();
122            for (i, label) in SEARCH_FIELDS.iter().enumerate() {
123                let rr = Rect { y: inner.y + 1 + i as u16, height: 1, ..inner };
124                let line = if i < nf - 1 { field_line(label, &s.fields[i], s.focus == i, &t) } else {
125                    let k = match s.kind { crate::tui::data::KindFilter::Any => "Posts and replies", crate::tui::data::KindFilter::Posts => "Posts only", crate::tui::data::KindFilter::Replies => "Replies only" };
126                    Line::from(vec![Span::styled(format!("{label:>18}: "), if s.focus == i { t.accent_bold() } else { Style::default() }), Span::raw(format!("‹ {k} ›"))])
127                };
128                f.render_widget(Paragraph::new(line), rr);
129                app.hits.push((rr, Hit::Modal(ModalHit::Field(i))));
130            }
131            if let Some(e) = &s.error {
132                f.render_widget(Paragraph::new(Span::styled(e.clone(), t.accent())), Rect { y: inner.y + nf as u16 + 2, height: 1, ..inner });
133            }
134            let labels: Vec<String> = SEARCH_BUTTONS.iter().map(|b| b.to_string()).collect();
135            let focus = s.focus.checked_sub(nf);
136            buttons(f, app, Rect { y: inner.y + inner.height.saturating_sub(1), height: 1, ..inner }, &labels, focus);
137        }
138        Modal::Research(m) => {
139            let list_rows = (m.projects.len() as u16).clamp(1, 10);
140            let add = m.mode == ResearchMode::Add;
141            let h = list_rows + if add { 13 } else { 10 };
142            let r = modal_rect(area, 84, h.min(area.height));
143            let inner = frame(f, &t, r, &format!("{} · Tab moves · Enter {}", m.title(), if add { "adds" } else { "opens" }));
144            let mut y = inner.y + 1;
145            let row = |y: u16| Rect { y, height: 1, ..inner };
146            let lab = |s: &str, on: bool| Span::styled(format!("{s:>18}: "), if on { t.accent_bold() } else { Style::default() });
147            if add {
148                let what = m.what.get(m.what_idx).map(|w| w.0.clone()).unwrap_or_else(|| "nothing here".into());
149                let n = if m.what.len() > 1 { format!("  ({}/{} · ←/→)", m.what_idx + 1, m.what.len()) } else { String::new() };
150                f.render_widget(Paragraph::new(Line::from(vec![lab("Add", m.focus == RField::What), Span::raw(format!("‹ {what} ›")),
151                    Span::styled(n, t.muted())])), row(y));
152                app.hits.push((row(y), Hit::Modal(ModalHit::Field(0))));
153                y += 1;
154                f.render_widget(Paragraph::new(field_line("Note", &m.note, m.focus == RField::Note, &t)), row(y));
155                app.hits.push((row(y), Hit::Modal(ModalHit::Field(1))));
156                y += 2;
157            }
158            f.render_widget(Paragraph::new(Span::styled(format!("Projects ({})", m.projects.len()),
159                if m.focus == RField::List { t.accent_bold() } else { Style::default().add_modifier(Modifier::BOLD) })), row(y));
160            y += 1;
161            if m.projects.is_empty() {
162                f.render_widget(Paragraph::new(Span::styled("  none yet — type a name below to create one", t.muted())), row(y));
163            }
164            let skip = m.sel.saturating_sub(list_rows as usize - 1);
165            for (k, (i, (name, n, desc))) in m.projects.iter().enumerate().skip(skip).take(list_rows as usize).enumerate() {
166                let rr = row(y + k as u16);
167                let sel = i == m.sel;
168                let st = if sel && m.focus == RField::List { t.selected() } else if sel { t.accent_bold() } else { Style::default() };
169                let text = format!(" {} {name}  ({n} items){}", if sel { "▸" } else { " " }, if desc.is_empty() { String::new() } else { format!(" — {desc}") });
170                f.render_widget(Paragraph::new(Span::styled(text, st)), rr);
171                app.hits.push((rr, Hit::Modal(ModalHit::Item(i))));
172            }
173            y += list_rows + 1;
174            f.render_widget(Paragraph::new(Span::styled("…or a new project", Style::default().add_modifier(Modifier::BOLD))), row(y));
175            y += 1;
176            f.render_widget(Paragraph::new(field_line("Name", &m.name, m.focus == RField::Name, &t)), row(y));
177            app.hits.push((row(y), Hit::Modal(ModalHit::Field(2))));
178            y += 1;
179            f.render_widget(Paragraph::new(field_line("Description", &m.desc, m.focus == RField::Desc, &t)), row(y));
180            app.hits.push((row(y), Hit::Modal(ModalHit::Field(3))));
181            y += 1;
182            if let Some(e) = &m.error { f.render_widget(Paragraph::new(Span::styled(e.clone(), t.accent())), row(y)); }
183            let focus = match m.focus { RField::Button(b) => Some(b), _ => None };
184            buttons(f, app, Rect { y: inner.y + inner.height.saturating_sub(1), height: 1, ..inner }, &m.buttons(), focus);
185        }
186        Modal::Palette(p) => {
187            let r = modal_rect(area, 70, area.height.saturating_sub(4));
188            let inner = frame(f, &t, r, "Commands · type to filter · Enter runs");
189            f.render_widget(Paragraph::new(field_line("›", &p.query, true, &t)), Rect { height: 1, ..inner });
190            let list = Rect { y: inner.y + 2, height: inner.height.saturating_sub(2), ..inner };
191            let items = p.filtered();
192            let skip = p.cursor.saturating_sub(list.height.saturating_sub(1) as usize);
193            for (row, (i, (label, _))) in items.iter().enumerate().skip(skip).take(list.height as usize).enumerate() {
194                let rr = Rect { y: list.y + row as u16, height: 1, ..list };
195                let st = if i == p.cursor { t.selected() } else { Style::default() };
196                f.render_widget(Paragraph::new(Span::styled(format!(" {label}"), st)), rr);
197                app.hits.push((rr, Hit::Modal(ModalHit::Item(i))));
198            }
199        }
200    }
201    app.modal = Some(m);
