1//! Modal rendering: dims the layout, then draws the modal with a small margin.

3use crate::tui::layout::{modal_rect, wrap};
4use crate::tui::modals::*;

6use crate::tui::theme::Theme;

10use ratatui::text::{Line, Span};
11use crate::tui::theme::frame as boxed;

14fn dim(f: &mut Frame, t: &Theme, area: Rect) {
15    let st = ratatui::style::Style::from(t.dimmed());
16    let buf = f.buffer_mut();
17    for y in area.y..area.y + area.height {
18        for x in area.x..area.x + area.width {
19            if let Some(c) = buf.cell_mut((x, y)) { c.set_style(st); }
20        }
21    }
22}

24fn frame(f: &mut Frame, t: &Theme, r: Rect, title: &str) -> Rect {
25    f.render_widget(Clear, r);
26    let b = boxed(&t.dialog()).title(format!(" {title} "));
27    let inner = b.inner(r);
28    f.render_widget(b, r);
29    inner
30}

32fn buttons(f: &mut Frame, app: &mut App, r: Rect, labels: &[String], focus: Option<usize>) {
33    let t = app.theme;
34    let mut x = r.x;
35    for (i, l) in labels.iter().enumerate() {
36        let s = format!("[ {l} ]");
37        let w = s.chars().count() as u16;
38        if x + w > r.x + r.width { break; }
39        let st = if focus == Some(i) { t.selected() } else { t.accent_bold() };
40        let br = Rect { x, y: r.y, width: w, height: 1 };
41        f.render_widget(Paragraph::new(Span::styled(s, st)), br);
42        app.hits.push((br, Hit::Modal(ModalHit::Button(i))));
43        x += w + 2;
44    }
45}

47fn field_line(label: &str, tf: &TextField, focused: bool, t: &Theme) -> Line<'static> {
48    let mut v = tf.display();
49    if focused { let b = v.char_indices().nth(tf.cursor).map(|(b, _)| b).unwrap_or(v.len()); v.insert(b, '▏'); }
50    Line::from(vec![Span::styled(format!("{label:>18}: "), if focused { t.accent_bold() } else { t.plain() }),
51        Span::styled(format!("{v:<30}"), t.field())])
52}

54/// Draw the open modal.
55pub fn draw(f: &mut Frame, app: &mut App, area: Rect) {
56    let theme = app.theme;
57    dim(f, &theme, area);
58    app.hits.push((area, Hit::Backdrop));
59    let t = app.theme;
60    let Some(m) = app.modal.take() else { return };
61    match &m {
62        Modal::Help(h) => {
63            let r = modal_rect(area, area.width, area.height);
64            let inner = frame(f, &t, r, "Help · ↑↓ PgUp PgDn scroll · Esc closes");
65            let body = Rect { height: inner.height.saturating_sub(1), ..inner };
66            let lines: Vec<Line> = h.lines.iter().skip(h.scroll).take(body.height as usize).map(|l| Line::raw(l.clone())).collect();
67            f.render_widget(Paragraph::new(lines), body);
68            buttons(f, app, Rect { y: inner.y + inner.height.saturating_sub(1), height: 1, ..inner }, &["Close".into()], Some(0));
69        }
70        Modal::Message(mm) => {
71            let w = 76;
72            let text: Vec<String> = mm.lines.iter().flat_map(|l| wrap(l, w as usize - 4)).collect();
73            let r = modal_rect(area, w, text.len() as u16 + 5);
74            let inner = frame(f, &t, r, &mm.title);
75            let st = if mm.error { t.accent() } else { t.plain() };
76            f.render_widget(Paragraph::new(text.into_iter().map(|l| Line::styled(l, &st)).collect::<Vec<_>>()),
77                Rect { height: inner.height.saturating_sub(2), ..inner });
78            buttons(f, app, Rect { y: inner.y + inner.height.saturating_sub(1), height: 1, ..inner }, &["OK".into()], Some(0));
79        }
80        Modal::Confirm(c) => {
81            let w = 76;
82            let text: Vec<String> = c.lines.iter().flat_map(|l| wrap(l, w as usize - 4)).collect();
83            let extra = if c.strict { 2 } else { 0 };
84            let r = modal_rect(area, w, text.len() as u16 + 5 + extra);
85            let inner = frame(f, &t, r, &c.title);
86            let mut lines: Vec<Line> = text.into_iter().map(Line::raw).collect();
87            if c.strict { lines.push(Line::raw("")); lines.push(field_line("Type yes", &c.typed, true, &t)); }
88            if let Some(e) = &c.error { lines.push(Line::styled(e.clone(), t.accent())); }
89            f.render_widget(Paragraph::new(lines), Rect { height: inner.height.saturating_sub(2), ..inner });
90            buttons(f, app, Rect { y: inner.y + inner.height.saturating_sub(1), height: 1, ..inner }, &c.buttons, Some(c.focus));
91        }
92        Modal::Login(l) => {
93            let r = modal_rect(area, area.width, area.height);
94            let inner = frame(f, &t, r, &l.title());
95            let mut lines = vec![Line::raw("")];
96            if let Some(why) = &l.reason { lines.push(Line::styled(why.clone(), t.accent_bold())); lines.push(Line::raw("")); }
97            lines.push(Line::raw("acp.us logs you in with a one-time code sent by email."));
98            lines.push(Line::raw(""));
99            match &l.step {
100                LoginStep::Email => {
101                    lines.push(field_line("Email", &l.email, true, &t));
102                    lines.push(Line::raw(""));
103                    lines.push(Line::styled("Enter sends the code.", t.muted()));
104                }
105                LoginStep::Code => {
106                    lines.push(Line::styled(format!("{:>18}: {}", "Email", l.email.value), t.muted()));
107                    lines.push(field_line("Code or login link", &l.code, true, &t));
108                    lines.push(Line::raw(""));
109                    lines.push(Line::styled("Paste the code (or the whole login link) from the email; Enter verifies.", t.muted()));
110                }
111                LoginStep::Busy(s) => lines.push(Line::styled(format!("⟳ {s}"), t.accent())),
112            }
113            if let Some(e) = &l.error { lines.push(Line::raw("")); lines.push(Line::styled(format!("✗ {e}"), t.accent_bold())); }
114            lines.push(Line::raw(""));
115            lines.push(Line::styled("The session is saved in ~/.acp/session.json (mode 0600); browsing works offline without it.", t.muted()));
116            f.render_widget(Paragraph::new(lines), Rect { height: inner.height.saturating_sub(2), ..inner });
117            let submit = if matches!(l.step, LoginStep::Code) { "Verify" } else { "Send code" };
118            buttons(f, app, Rect { y: inner.y + inner.height.saturating_sub(1), height: 1, ..inner }, &[submit.into(), "Cancel".into()], None);
119        }
120        Modal::Search(s) => {
121            let r = modal_rect(area, 72, SEARCH_FIELDS.len() as u16 + 8);
122            let inner = frame(f, &t, r, "Search · Tab moves · Enter searches");
123            let nf = SEARCH_FIELDS.len();
124            for (i, label) in SEARCH_FIELDS.iter().enumerate() {
125                let rr = Rect { y: inner.y + 1 + i as u16, height: 1, ..inner };
126                let line = if i < nf - 1 { field_line(label, &s.fields[i], s.focus == i, &t) } else {
127                    let k = match s.kind { crate::tui::data::KindFilter::Any => "Posts and replies", crate::tui::data::KindFilter::Posts => "Posts only", crate::tui::data::KindFilter::Replies => "Replies only" };
128                    Line::from(vec![Span::styled(format!("{label:>18}: "), if s.focus == i { t.accent_bold() } else { t.plain() }), Span::raw(format!("‹ {k} ›"))])
129                };
130                f.render_widget(Paragraph::new(line), rr);
131                app.hits.push((rr, Hit::Modal(ModalHit::Field(i))));
132            }
133            if let Some(e) = &s.error {
134                f.render_widget(Paragraph::new(Span::styled(e.clone(), t.accent())), Rect { y: inner.y + nf as u16 + 2, height: 1, ..inner });
135            }
136            let labels: Vec<String> = SEARCH_BUTTONS.iter().map(|b| b.to_string()).collect();
137            let focus = s.focus.checked_sub(nf);
138            buttons(f, app, Rect { y: inner.y + inner.height.saturating_sub(1), height: 1, ..inner }, &labels, focus);
139        }
140        Modal::Research(m) => {
141            let list_rows = (m.projects.len() as u16).clamp(1, 10);
142            let add = m.mode == ResearchMode::Add;
143            let h = list_rows + if add { 13 } else { 10 };
144            let r = modal_rect(area, 84, h.min(area.height));
145            let inner = frame(f, &t, r, &format!("{} · Tab moves · Enter {}", m.title(), if add { "adds" } else { "opens" }));
146            let mut y = inner.y + 1;
147            let row = |y: u16| Rect { y, height: 1, ..inner };
148            let lab = |s: &str, on: bool| Span::styled(format!("{s:>18}: "), if on { t.accent_bold() } else { t.plain() });
149            if add {
150                let what = m.what.get(m.what_idx).map(|w| w.0.clone()).unwrap_or_else(|| "nothing here".into());
151                let n = if m.what.len() > 1 { format!("  ({}/{} · ←/→)", m.what_idx + 1, m.what.len()) } else { String::new() };
152                f.render_widget(Paragraph::new(Line::from(vec![lab("Add", m.focus == RField::What), Span::raw(format!("‹ {what} ›")),
153                    Span::styled(n, t.muted())])), row(y));
154                app.hits.push((row(y), Hit::Modal(ModalHit::Field(0))));
155                y += 1;
156                f.render_widget(Paragraph::new(field_line("Note", &m.note, m.focus == RField::Note, &t)), row(y));
157                app.hits.push((row(y), Hit::Modal(ModalHit::Field(1))));
158                y += 2;
159            }
160            f.render_widget(Paragraph::new(Span::styled(format!("Projects ({})", m.projects.len()),
161                if m.focus == RField::List { t.accent_bold() } else { t.bold() })), row(y));
162            y += 1;
163            if m.projects.is_empty() {
164                f.render_widget(Paragraph::new(Span::styled("  none yet — type a name below to create one", t.muted())), row(y));
165            }
166            let skip = m.sel.saturating_sub(list_rows as usize - 1);
167            for (k, (i, (name, n, desc))) in m.projects.iter().enumerate().skip(skip).take(list_rows as usize).enumerate() {
168                let rr = row(y + k as u16);
169                let sel = i == m.sel;
170                let st = if sel && m.focus == RField::List { t.selected() } else if sel { t.accent_bold() } else { t.plain() };
171                let text = format!(" {} {name}  ({n} items){}", if sel { "▸" } else { " " }, if desc.is_empty() { String::new() } else { format!(" — {desc}") });
172                f.render_widget(Paragraph::new(Span::styled(text, st)), rr);
173                app.hits.push((rr, Hit::Modal(ModalHit::Item(i))));
174            }
175            y += list_rows + 1;
176            f.render_widget(Paragraph::new(Span::styled("…or a new project", t.bold())), row(y));
177            y += 1;
178            f.render_widget(Paragraph::new(field_line("Name", &m.name, m.focus == RField::Name, &t)), row(y));
179            app.hits.push((row(y), Hit::Modal(ModalHit::Field(2))));
180            y += 1;
181            f.render_widget(Paragraph::new(field_line("Description", &m.desc, m.focus == RField::Desc, &t)), row(y));
182            app.hits.push((row(y), Hit::Modal(ModalHit::Field(3))));
183            y += 1;
184            if let Some(e) = &m.error { f.render_widget(Paragraph::new(Span::styled(e.clone(), t.accent())), row(y)); }
185            let focus = match m.focus { RField::Button(b) => Some(b), _ => None };
186            buttons(f, app, Rect { y: inner.y + inner.height.saturating_sub(1), height: 1, ..inner }, &m.buttons(), focus);
187        }
188        Modal::ThemePicker(p) => {
189            let rows = p.rows() as u16;
190            let r = modal_rect(area, 76, (rows + 4).min(area.height.saturating_sub(2)));
191            let inner = frame(f, &t, r, &p.title());
192            let list = Rect { y: inner.y + 1, height: inner.height.saturating_sub(1), ..inner };
193            let skip = p.cursor.saturating_sub(list.height.saturating_sub(1) as usize);
194            for (row, i) in (0..p.rows()).skip(skip).take(list.height as usize).enumerate() {
195                let rr = Rect { y: list.y + row as u16, height: 1, ..list };
196                let sel = i == p.cursor;
197                let (text, bad) = match p.entries.get(i) {
198                    Some(e) => {
199                        let cur = if e.name == p.current { "●" } else { " " };
200                        match &e.error {
201                            Some(err) => (format!(" {cur} {:<16} ⚠ {err}", e.name), true),
202                            None => (format!(" {cur} {:<16} {}", e.name, e.description), false),
203                        }
204                    }
205                    None => ("   Edit themes…  (Shift-F5)".to_string(), false),
206                };
207                let st = if sel { t.selected() } else if bad { t.muted() } else { t.plain() };
208                let w = rr.width as usize;
209                let text: String = if text.chars().count() > w { text.chars().take(w.saturating_sub(1)).chain(['…']).collect() } else { text };
210                f.render_widget(Paragraph::new(Span::styled(text, st)), rr);
211                app.hits.push((rr, Hit::Modal(ModalHit::Item(i))));
212            }
213        }
214        Modal::Palette(p) => {
215            let r = modal_rect(area, 70, area.height.saturating_sub(4));
216            let inner = frame(f, &t, r, "Commands · type to filter · Enter runs");
217            f.render_widget(Paragraph::new(field_line("›", &p.query, true, &t)), Rect { height: 1, ..inner });
218            let list = Rect { y: inner.y + 2, height: inner.height.saturating_sub(2), ..inner };
219            let items = p.filtered();
220            let skip = p.cursor.saturating_sub(list.height.saturating_sub(1) as usize);
221            for (row, (i, (label, _))) in items.iter().enumerate().skip(skip).take(list.height as usize).enumerate() {
222                let rr = Rect { y: list.y + row as u16, height: 1, ..list };
223                let st = if i == p.cursor { t.selected() } else { t.plain() };
224                f.render_widget(Paragraph::new(Span::styled(format!(" {label}"), st)), rr);
225                app.hits.push((rr, Hit::Modal(ModalHit::Item(i))));
226            }
227        }
228    }
229    app.modal = Some(m);
