21/// Draw the whole screen.
22pub fn draw(f: &mut Frame, app: &mut App, now: Instant) {
23    app.hits.clear();
24    app.osc_links.clear();
25    if let Some(im) = app.images.as_mut() { im.frame.clear(); }
26    let area = f.area();
27    if area.width < TUI_MIN_SIZE.0 || area.height < TUI_MIN_SIZE.1 {
28        f.render_widget(Paragraph::new(format!("Terminal too small: {}×{} (need {}×{}). q quits.",
29            area.width, area.height, TUI_MIN_SIZE.0, TUI_MIN_SIZE.1)), area);
30        return;
31    }
32    // The whole screen in the theme's base colours (the terminal's when unset).
33    f.buffer_mut().set_style(area, ratatui::style::Style::from(&app.theme.base()));
34    if let Some(mut ed) = app.theme_editor.take() {
35        editor::draw(f, app, &mut ed, now);
36        app.theme_editor = Some(ed);
37        if app.modal.is_some() { modal::draw(f, app, area); }
38        return;
39    }
40    let log_rows = if app.show_log { 10 } else { 1 };
41    let a = areas(area, log_rows);
42    topbar(f, app, a.top);
43    menu::draw(f, app, a.menu);
44    nicks::draw(f, app, a.nicks);
45    body::draw(f, app, a.body, now);
46    status(f, app, a.status);
47    menu::draw_dropdown(f, app, area);
48    if app.modal.is_some() { modal::draw(f, app, area); }
49}

51/// The header: the red star logo, a big "ACP", the title and the session, all
52/// vertically centred in [`TUI_HEADER_ROWS`] rows.
53fn topbar(f: &mut Frame, app: &mut App, r: Rect) {
54    let t = app.theme;
55    let bar = t.bar();
56    f.render_widget(Paragraph::new("").style(&bar), r);
57    // The big "ACP" is 3 rows: in the 4-row header it gets the spare row above it,
58    // and the title / session line sits on its middle row.
59    let big_top = r.y + r.height.saturating_sub(3).div_ceil(2);
60    let mid = Rect { x: r.x, y: (big_top + 1).min(r.y + r.height.saturating_sub(1)), width: r.width, height: 1.min(r.height) };
61    // Logo: square on screen (cells are about twice as tall as wide), the full
62    // header height, in on-accent over the red bar. Without terminal graphics it
63    // is a sextant silhouette of the same PNG.
64    let logo = Rect { x: r.x + 1, y: r.y, width: (r.height * 2).min(r.width.saturating_sub(1)), height: r.height };
65    let drawn = app.images.as_mut().is_some_and(|im| im.draw_logo(f.buffer_mut(), logo));
66    if !drawn { crate::tui::images::draw_logo_text(f.buffer_mut(), logo, t.on_accent, t.accent); }
67    let mut x = logo.x + logo.width + 1;
68    let big_w = 3 * 8;
69    if r.height >= 3 && x + big_w + 2 < r.width {
70        let big = tui_big_text::BigText::builder()
71            .pixel_size(tui_big_text::PixelSize::ThirdHeight)
72            .style(t.bar().bold(true))
73            .lines(vec![Line::from("ACP")])
74            .build();
75        f.render_widget(big, Rect { x, y: big_top, width: big_w, height: 3 });
76        x += big_w + 2;
77    } else {
78        f.render_widget(Paragraph::new(Span::styled("ACP", t.bar().bold(true))), Rect { x, ..mid });
79        x += 5;
80    }
81    let left = format!("{}  v{}", TUI_SUBTITLE, env!("CARGO_PKG_VERSION"));
82    let session = match app.session {
83        SessionState::LoggedOut => "○ logged out",
84        SessionState::Saved => "◐ session saved",
85        SessionState::Valid => "● logged in",
86        SessionState::Checking => "… checking session",
87    };
88    let job = app.job.as_ref().map(|j| format!("⟳ {} running (pid {})  ", j.kind.name(), j.pid)).unwrap_or_default();
89    // The theme switch: ‹name› ▾ (⚠ when its file can't be used); a click opens the picker.
90    let switch = format!("‹{}{}› ▾", app.theme_name,
91        if app.theme_downloading.is_some() { " · downloading…" } else if app.theme_error.is_some() { " ⚠" } else { "" });
92    let right = format!("  {session} ");
93    let rest = Rect { x, width: r.width.saturating_sub(x - r.x), ..mid };
94    let used = left.chars().count() + job.chars().count() + switch.chars().count() + right.chars().count();
95    let pad = (rest.width as usize).saturating_sub(used);
96    let sw_x = rest.x + (left.chars().count() + pad + job.chars().count()) as u16;
97    if (used as u16) < rest.width {
98        app.hits.push((Rect { x: sw_x, y: rest.y, width: switch.chars().count() as u16, height: 1 }, Hit::ThemeSwitch));
99    }
100    f.render_widget(Paragraph::new(Line::from(vec![
101        Span::styled(left, t.bar().bold(true)), Span::styled(" ".repeat(pad), &bar), Span::styled(job, &bar),
102        Span::styled(switch, t.bar().underline(true)), Span::styled(right, &bar),
103    ])), rest);
104}

106fn status(f: &mut Frame, app: &App, r: Rect) {
107    let t = app.theme;
108    let n = r.height as usize;
109    let lines: Vec<Line> = if app.status.is_empty() {
110        vec![Line::styled("? help · Tab focus · Space multi-select · Shift-S search · Ctrl-Space commands · F6 scan · F5 theme · q quit", t.muted())]
111    } else {
112        app.status.iter().rev().take(n).rev().map(|l| {
113            let st = if l.starts_with("error") || l.starts_with("warning") { t.accent() } else { t.muted() };
114            Line::styled(l.clone(), st)
115        }).collect()
116    };
117    f.render_widget(Paragraph::new(lines), r);
118}
