1//! Body: navigation bar (with `<` back from search), the post cards, and the
2//! detail stack that slides the list up and nests linked posts.

15/// Draw the Body into `r`.
16pub fn draw(f: &mut Frame, app: &mut App, r: Rect, now: Instant) {
17    let t = app.theme;
18    let focused = app.focus == Focus::Body;
19    let b = Block::default().borders(Borders::ALL).border_style(t.border(focused));
20    let inner = b.inner(r);
21    f.render_widget(b, r);
22    if inner.height < 2 { return; }
23    // Navigation bar.
24    let nav = Rect { height: 1, ..inner };
25    let content = Rect { y: inner.y + 1, height: inner.height - 1, ..inner };
26    let in_search = app.search.is_some();
27    if in_search { app.hits.push((Rect { width: 8.min(nav.width), ..nav }, Hit::Back)); }
28    let v = app.view();
29    let mut spans = Vec::new();
30    if in_search {
31        spans.push(Span::styled(" < Back ", t.selected()));
32        spans.push(Span::raw(" "));
33    }
34    let title = if v.title.is_empty() { "Select a nick on the left".to_string() } else { v.title.clone() };
35    spans.push(Span::styled(title, Style::default().add_modifier(Modifier::BOLD)));
36    spans.push(Span::styled(format!("  {} posts", v.posts.len()), t.muted()));
37    if !v.stack.is_empty() {
38        spans.push(Span::styled(format!("  · detail {}/{}", v.stack.len(), crate::config::MAX_DETAIL_DEPTH), t.muted()));
39    }
40    f.render_widget(Paragraph::new(Line::from(spans)), nav);
41
42    if v.posts.is_empty() {
43        let msg = if in_search { "No posts match this search. Press < to go back." }
44            else if app.nicks.is_empty() { "The database is empty. Open Scrape (or Import) in the menu bar to fill it." }
45            else { "Click a nick (Space / Ctrl-click to select several) to read their posts and replies." };
46        f.render_widget(Paragraph::new(msg).style(t.muted()), Rect { y: content.y + 1, height: content.height.saturating_sub(1), ..content });
47        return;
48    }
49    app.page_rows = (content.height / CARD_ROWS).max(1) as usize;
50    if app.view().stack.is_empty() { draw_cards(f, app, content); } else { draw_with_detail(f, app, content, now); }
51}

53fn draw_cards(f: &mut Frame, app: &mut App, r: Rect) {
54    let t = app.theme;
55    app.hits.push((r, Hit::ListPane));
56    let rows = app.page_rows;
57    let focused = app.focus == Focus::Body;
58    let v = app.view_mut();
59    if v.cursor < v.offset { v.offset = v.cursor; }
60    if v.cursor >= v.offset + rows { v.offset = v.cursor + 1 - rows; }
61    let (off, cur) = (v.offset, v.cursor);
62    let posts: Vec<_> = v.posts.iter().enumerate().skip(off).take(rows).map(|(i, p)| (i, p.clone())).collect();
63    for (k, (i, p)) in posts.into_iter().enumerate() {
64        let cr = Rect { x: r.x, y: r.y + k as u16 * CARD_ROWS, width: r.width, height: CARD_ROWS.min(r.height.saturating_sub(k as u16 * CARD_ROWS)) };
65        let sel = i == cur;
66        let b = Block::default().borders(Borders::ALL).title(meta_line(&p, &t))
67            .border_type(if sel { BorderType::Thick } else { BorderType::Plain })
68            .border_style(if sel && focused { t.accent() } else if sel { Style::default() } else { t.border(false) });
69        let inner = b.inner(cr);
70        f.render_widget(b, cr);
71        let text = if p.title.trim().is_empty() { p.content.clone() } else { format!("{}\n{}", p.title.trim(), p.content) };
72        let lines: Vec<Line> = wrap(text.trim(), inner.width as usize).into_iter().filter(|l| !l.trim().is_empty())
73            .take(inner.height as usize).map(Line::raw).collect();
74        f.render_widget(Paragraph::new(lines), inner);
75        app.hits.push((cr, Hit::Post(i)));
76    }
77}

79fn compact(f: &mut Frame, app: &mut App, i: usize, r: Rect) {
80    let t = app.theme;
81    let Some(p) = app.view().posts.get(i).cloned() else { return };
82    let w = r.width as usize;
83    let head = format!(" @{} · {} ", p.nick, p.posted_at.get(..16).unwrap_or(&p.posted_at));
84    let rest = ellipsize(&snippet(&p), w.saturating_sub(head.len() + 8));
85    f.render_widget(Paragraph::new(Line::from(vec![badge(&p, &t), Span::styled(head, t.muted()), Span::raw(rest)])), r);
86    app.hits.push((r, Hit::Post(i)));
87}

89fn draw_with_detail(f: &mut Frame, app: &mut App, r: Rect, now: Instant) {
90    let t = app.theme;
91    let (kind, p) = app.anim_progress(now);
92    let open_p = if kind == Some(AnimKind::Open) { p } else { 1.0 };
93    let push_p = if kind == Some(AnimKind::Push) { p } else { 1.0 };
94    let (n, cur) = (app.view().posts.len(), app.view().cursor);
95    let split = detail_split(r, n, cur, open_p);
96    for (i, rr) in split.above.clone() { compact(f, app, i, rr); }
97    for (i, rr) in split.below.clone() { compact(f, app, i, rr); }
98    let focused = app.focus == Focus::Body;
99    let b = Block::default().borders(Borders::ALL).border_type(BorderType::Double).border_style(if focused { t.accent() } else { t.border(false) });
100    let inner = b.inner(split.detail);
101    f.render_widget(b, split.detail);
102    let depth = app.view().stack.len();
103    let sl = stack_layout(inner, depth, push_p);
104    if let Some(nb) = sl.nested {
105        f.render_widget(Block::default().borders(Borders::ALL).border_type(BorderType::Double).border_style(t.accent()), nb);
106    }
107    for (d, fr, collapsed) in sl.frames {
108        if fr.height == 0 || fr.width < 3 { continue; }
109        let fr = Rect { x: fr.x + 1, width: fr.width - 1, ..fr };
110        if collapsed && fr.height <= DETAIL_HEADER_ROWS + 1 {
111            let lines = header_lines(&app.view().stack[d], fr.width as usize, &t);
112            f.render_widget(Paragraph::new(lines), fr);
113            app.hits.push((fr, Hit::Frame(d)));
114            continue;
115        }
116        // Full content (the top frame, or the previous one while it is being pushed up).
117        let w = fr.width.saturating_sub(1) as usize;
118        let (lines, links) = detail_lines(&app.view().stack[d], w, &t);
119        let h = fr.height as usize;
120        let top = d + 1 == depth;
121        let fm = &mut app.view_mut().stack[d];
122        if top && fm.follow_link
123            && let Some((li, _)) = links.iter().find(|(_, k)| *k == fm.link_cursor) {
124                let li = *li as u16;
125                if li < fm.scroll { fm.scroll = li; }
126                if li as usize >= fm.scroll as usize + h { fm.scroll = (li as usize + 1 - h) as u16; }
127                fm.follow_link = false;
128            }
129        let max = lines.len().saturating_sub(h) as u16;
130        if fm.scroll > max { fm.scroll = max; }
131        let scroll = fm.scroll;
132        f.render_widget(Paragraph::new(lines).scroll((scroll, 0)), fr);
133        if top {
134            app.hits.push((fr, Hit::DetailPane));
135            for (li, k) in links {
136                let y = li as i64 - scroll as i64;
137                if y >= 0 && (y as usize) < h {
138                    app.hits.push((Rect { x: fr.x, y: fr.y + y as u16, width: fr.width, height: 1 }, Hit::Link(k)));
139                }
140            }
141        } else {
142            app.hits.push((fr, Hit::Frame(d)));
143        }
144    }
