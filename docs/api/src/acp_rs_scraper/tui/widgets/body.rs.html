17/// Draw the Body into `r`.
18pub fn draw(f: &mut Frame, app: &mut App, r: Rect, now: Instant) {
19    let t = app.theme;
20    let focused = app.focus == Focus::Body;
21    let b = boxed(&t.pane(focused));
22    let inner = b.inner(r);
23    f.render_widget(b, r);
24    if inner.height < 2 { return; }
25    // Navigation bar.
26    let nav = Rect { height: 1, ..inner };
27    let content = Rect { y: inner.y + 1, height: inner.height - 1, ..inner };
28    let in_search = app.search.is_some();
29    let (pos, total) = app.stack_pos();
30    let mut spans = Vec::new();
31    let mut x = nav.x;
32    let mut button = |spans: &mut Vec<Span<'static>>, hits: &mut Vec<(Rect, Hit)>, label: &'static str, st: Gloss, hit: Option<Hit>| {
33        let w = unicode_width::UnicodeWidthStr::width(label) as u16;
34        if let Some(h) = hit { hits.push((Rect { x, width: w.min((nav.x + nav.width).saturating_sub(x)), ..nav }, h)); }
35        spans.push(Span::styled(label, st));
36        spans.push(Span::raw(" "));
37        x += w + 1;
38    };
39    if in_search { button(&mut spans, &mut app.hits, " < Back ", t.selected(), Some(Hit::Back)); }
40    if total > 0 {
41        let on = |b: bool| if b { t.selected() } else { t.muted() };
42        button(&mut spans, &mut app.hits, " ◂ prev ", on(pos > 0), Some(Hit::StackPrev));
43        button(&mut spans, &mut app.hits, " next ▸ ", on(pos < total), Some(Hit::StackNext));
44    }
45    let v = app.view();
46    if total > 0 { spans.push(Span::styled(format!("{pos}/{total}  "), t.muted())); }
47    let title = if v.title.is_empty() { "Select a nick on the left".to_string() } else { v.title.clone() };
48    spans.push(Span::styled(title, t.bold()));
49    spans.push(Span::styled(format!("  {} posts", v.posts.len()), t.muted()));
50    if !v.stack.is_empty() {
51        spans.push(Span::styled(format!("  · detail {}/{}", v.stack.len(), crate::config::MAX_DETAIL_DEPTH), t.muted()));
52    }
53    f.render_widget(Paragraph::new(Line::from(spans)), nav);
54    // View switch, right-aligned.
55    let vl = format!(" [View: {} ▸] ", app.view_mode.label());
56    let vw = (unicode_width::UnicodeWidthStr::width(vl.as_str()) as u16).min(nav.width);
57    let vr = Rect { x: nav.x + nav.width - vw, width: vw, ..nav };
58    f.render_widget(Paragraph::new(Span::styled(vl, t.accent_bold())), vr);
59    app.hits.push((vr, Hit::ViewMode));
60
61    if app.view().posts.is_empty() {
62        let msg = if in_search { "No posts match this search. Press < to go back." }
63            else if app.nicks.is_empty() { "The database is empty. Open Scrape (or Import) in the menu bar to fill it." }
64            else { "Click a nick (Space / Ctrl-click to select several) to read their posts and replies." };
65        f.render_widget(Paragraph::new(msg).style(t.muted()), Rect { y: content.y + 1, height: content.height.saturating_sub(1), ..content });
66        return;
67    }
68    let open = !app.view().stack.is_empty();
69    match app.view_mode {
70        ViewMode::Nested => {
71            app.page_rows = (content.height / CARD_ROWS).max(1) as usize;
72            if open { draw_with_detail(f, app, content, now); } else { draw_cards(f, app, content); }
73        }
74        ViewMode::Stack => if open { draw_flat(f, app, content); } else { draw_stack(f, app, content); },
75        ViewMode::Pane => {
76            let list_h = if open { (content.height * 2 / 5).max(4).min(content.height) } else { content.height };
77            draw_stack(f, app, Rect { height: list_h, ..content });
78            if open { draw_flat(f, app, Rect { y: content.y + list_h, height: content.height - list_h, ..content }); }
79        }
80        ViewMode::CarouselInfinite | ViewMode::CarouselFixed => if open { draw_flat(f, app, content); } else { draw_carousel(f, app, content); },
81        ViewMode::Grid => {
82            draw_grid(f, app, content);
83            if open { draw_grid_modal(f, app, r); }
84        }
85    }
86}

88fn post_text(p: &Post) -> String { if p.title.trim().is_empty() { p.content.clone() } else { format!("{}\n{}", p.title.trim(), p.content) } }

90/// One post card in `cr` (border with meta line and star, wrapped text, scrolled by `skip` lines).
91fn card(f: &mut Frame, app: &mut App, i: usize, cr: Rect, sel: bool) {
92    let t = app.theme;
93    let Some(p) = app.view().posts.get(i).cloned() else { return };
94    let focused = app.focus == Focus::Body;
95    let b = boxed(&t.card(sel, focused)).title(meta_line(&p, &t));
96    let inner = b.inner(cr);
97    f.render_widget(b, cr);
98    let lines: Vec<Line> = crate::tui::markdown::render(post_text(&p).trim(), inner.width as usize, &t).into_iter()
99        .filter(|l| l.width() > 0).take(inner.height as usize).collect();
100    f.render_widget(Paragraph::new(lines), inner);
101    app.hits.push((cr, Hit::Post(i)));
102    if cr.width > 8 {
103        let sr = Rect { x: cr.x + cr.width - 5, y: cr.y, width: 3, height: 1 };
104        f.render_widget(Paragraph::new(Span::styled(star(app.is_marked(&p.post_id)), t.accent_bold())), sr);
105        app.hits.push((sr, Hit::Star(i)));
106    }
107    if cr.width > 12 {
108        let br = Rect { x: cr.x + cr.width - 8, y: cr.y, width: 3, height: 1 };
109        f.render_widget(Paragraph::new(Span::styled(basket_icon(app.in_basket(&p.post_id)), t.accent_bold())), br);
110        app.hits.push((br, Hit::Basket(i)));
111    }
112    if cr.width > 16 && !p.post_id.starts_with("research:") {
113        let rr = Rect { x: cr.x + cr.width - 11, y: cr.y, width: 3, height: 1 };
114        f.render_widget(Paragraph::new(Span::styled(RESEARCH_ICON, t.accent_bold())), rr);
115        app.hits.push((rr, Hit::Research(i)));
116    }
117}

119/// Stack: one column of full posts (each card as tall as its text, up to the area).
120fn draw_stack(f: &mut Frame, app: &mut App, r: Rect) {
121    app.hits.push((r, Hit::ListPane));
122    let w = r.width;
123    let heights: Vec<u16> = app.view().posts.iter().map(|p| card_height(&post_text(p), w.saturating_sub(2), r.height)).collect();
124    let v = app.view_mut();
125    if v.cursor < v.offset { v.offset = v.cursor; }
126    // Keep the cursor's card fully on screen.
127    while v.offset < v.cursor && heights[v.offset..=v.cursor].iter().map(|h| *h as u32).sum::<u32>() > r.height as u32 { v.offset += 1; }
128    let (off, cur) = (v.offset, v.cursor);
129    let mut y = r.y;
130    let mut shown = 0;
131    for (i, h) in heights.iter().enumerate().skip(off) {
132        if y >= r.y + r.height { break; }
133        let h = (*h).min(r.y + r.height - y);
134        card(f, app, i, Rect { x: r.x, y, width: w, height: h }, i == cur);
135        y += h;
136        shown += 1;
137    }
138    app.page_rows = shown.max(1);
139}

141/// A detail drawn flat (no nesting): the top frame fills `r`.
142fn draw_flat(f: &mut Frame, app: &mut App, r: Rect) {
143    let t = app.theme;
144    let focused = app.focus == Focus::Body;
145    let depth = app.view().stack.len();
146    let title = if depth > 1 { format!(" {depth} deep · Esc back ") } else { " Esc closes ".into() };
147    let b = boxed(&t.detail(focused).padding_left(1)).title(title);
148    let inner = b.inner(r);
149    f.render_widget(b, r);
150    if inner.width < 2 || inner.height == 0 { return; }
151    full_frame(f, app, depth - 1, inner, true);
152}

154/// Carousel: the current post in the middle, its neighbours peeking at the sides.
155fn draw_carousel(f: &mut Frame, app: &mut App, r: Rect) {
156    let t = app.theme;
157    app.hits.push((r, Hit::ListPane));
158    app.page_rows = 1;
159    let n = app.view().posts.len();
160    let cur = app.view().cursor.min(n - 1);
161    let c = carousel(r);
162    let wrapv = app.view_mode == ViewMode::CarouselInfinite;
163    let prev = if cur > 0 { Some(cur - 1) } else if wrapv && n > 1 { Some(n - 1) } else { None };
164    let next = if cur + 1 < n { Some(cur + 1) } else if wrapv && n > 1 { Some(0) } else { None };
165    for (side, idx, arrow) in [(c.prev, prev, "◂"), (c.next, next, "▸")] {
166        let b = boxed(&t.pane(false));
167        let inner = b.inner(side);
168        f.render_widget(b, side);
169        if let Some(i) = idx {
170            let p = &app.view().posts[i];
171            let mut lines = vec![Line::styled(arrow, t.accent_bold()), Line::styled(ellipsize(&format!("@{}", p.nick), inner.width as usize), t.muted())];
172            lines.extend(wrap(&snippet(p), inner.width as usize).into_iter().take(inner.height.saturating_sub(2) as usize).map(|l| Line::styled(l, t.muted())));
173            f.render_widget(Paragraph::new(lines), inner);
174            app.hits.push((side, Hit::Post(i)));
175        }
176    }
177    card(f, app, cur, c.center, true);
178    let pos = format!(" {}/{n}{} ", cur + 1, if wrapv { " ∞" } else { "" });
179    let pr = Rect { x: c.center.x + 2, y: c.center.y + c.center.height.saturating_sub(1), width: (pos.chars().count() as u16).min(c.center.width.saturating_sub(4)), height: 1 };
180    f.render_widget(Paragraph::new(Span::styled(pos, t.muted())), pr);
181}

183/// Grid of cards; scrolls by rows to keep the cursor visible.
184fn draw_grid(f: &mut Frame, app: &mut App, r: Rect) {
185    app.hits.push((r, Hit::ListPane));
186    let n = app.view().posts.len();
187    let (cols, _) = grid_cells(r, n, 0);
188    let rows = grid_rows(r);
189    app.grid_cols = cols;
190    app.page_rows = rows * cols;
191    let v = app.view_mut();
192    let crow = v.cursor / cols;
193    let mut orow = v.offset / cols;
194    if crow < orow { orow = crow; }
195    if crow >= orow + rows { orow = crow + 1 - rows; }
196    v.offset = orow * cols;
197    let (first, cur) = (v.offset, v.cursor);
198    let (_, cells) = grid_cells(r, n, first);
199    for (i, cr) in cells { card(f, app, i, cr, i == cur); }
200}

202/// Grid: the open post as a modal over the Body, with a Close button.
203fn draw_grid_modal(f: &mut Frame, app: &mut App, body: Rect) {
204    let t = app.theme;
205    let mr = modal_rect(body, body.width.saturating_sub(6).max(40), body.height.saturating_sub(2));
206    f.render_widget(Clear, mr);
207    app.hits.push((mr, Hit::DetailPane));
208    let depth = app.view().stack.len();
209    let b = boxed(&t.dialog())
210        .title(if depth > 1 { format!(" Post · {depth} deep · Esc back ") } else { " Post ".into() });
211    let inner = b.inner(mr);
212    f.render_widget(b, mr);
213    if inner.height < 3 || inner.width < 12 { return; }
214    let close = " [ Close ] ";
215    let cw = close.chars().count() as u16;
216    let cr = Rect { x: inner.x + inner.width.saturating_sub(cw), y: inner.y + inner.height - 1, width: cw.min(inner.width), height: 1 };
217    full_frame(f, app, depth - 1, Rect { x: inner.x + 1, width: inner.width - 1, height: inner.height - 1, ..inner }, true);
218    f.render_widget(Paragraph::new(Span::styled(close, t.selected())), cr);
219    app.hits.push((cr, Hit::CloseDetail));
220}

222fn draw_cards(f: &mut Frame, app: &mut App, r: Rect) {
223    app.hits.push((r, Hit::ListPane));
224    let rows = app.page_rows;
225    let v = app.view_mut();
226    if v.cursor < v.offset { v.offset = v.cursor; }
227    if v.cursor >= v.offset + rows { v.offset = v.cursor + 1 - rows; }
228    let (off, cur, n) = (v.offset, v.cursor, v.posts.len());
229    for (k, i) in (off..n).take(rows).enumerate() {
230        let cr = Rect { x: r.x, y: r.y + k as u16 * CARD_ROWS, width: r.width, height: CARD_ROWS.min(r.height.saturating_sub(k as u16 * CARD_ROWS)) };
231        card(f, app, i, cr, i == cur);
232    }
233}

235/// Columns before the file name on an attachment row ("▣ Open mime  ").
236fn lines_label_width(atts: &[crate::prelude::Attachment], id: &str) -> usize {
237    let mime = atts.iter().find(|a| a.local_path.display().to_string() == id).map(|a| a.mime_type.as_str()).unwrap_or("file");
238    unicode_width::UnicodeWidthStr::width(format!("▣ Open {mime}  ").as_str())
239}

249fn compact(f: &mut Frame, app: &mut App, i: usize, r: Rect) {
250    let t = app.theme;
251    let Some(p) = app.view().posts.get(i).cloned() else { return };
252    let w = r.width as usize;
253    let head = format!(" @{} · {} ", p.nick, p.posted_at.get(..16).unwrap_or(&p.posted_at));
254    let rest = ellipsize(&snippet(&p), w.saturating_sub(head.len() + 8));
255    let mark = if app.is_marked(&p.post_id) { "★" } else { " " };
256    f.render_widget(Paragraph::new(Line::from(vec![Span::styled(mark, t.accent_bold()), badge(&p, &t), Span::styled(head, t.muted()), Span::raw(rest)])), r);
257    app.hits.push((r, Hit::Post(i)));
258}

260fn draw_with_detail(f: &mut Frame, app: &mut App, r: Rect, now: Instant) {
261    let t = app.theme;
262    let (kind, p) = app.anim_progress(now);
263    let open_p = if kind == Some(AnimKind::Open) { p } else { 1.0 };
264    let push_p = if kind == Some(AnimKind::Push) { p } else { 1.0 };
265    let (n, cur) = (app.view().posts.len(), app.view().cursor);
266    let split = detail_split(r, n, cur, open_p);
267    for (i, rr) in split.above.clone() { compact(f, app, i, rr); }
268    for (i, rr) in split.below.clone() { compact(f, app, i, rr); }
269    let focused = app.focus == Focus::Body;
270    let b = boxed(&t.detail(focused));
271    let inner = b.inner(split.detail);
272    f.render_widget(b, split.detail);
273    let depth = app.view().stack.len();
274    let sl = stack_layout(inner, depth, push_p);
275    if let Some(nb) = sl.nested {
276        f.render_widget(boxed(&t.dialog()), nb);
277    }
278    for (d, fr, collapsed) in sl.frames {
279        if fr.height == 0 || fr.width < 3 { continue; }
280        let fr = Rect { x: fr.x + 1, width: fr.width - 1, ..fr };
281        if collapsed && fr.height <= DETAIL_HEADER_ROWS + 1 {
282            let lines = header_lines(&app.view().stack[d], fr.width as usize, &t);
283            f.render_widget(Paragraph::new(lines), fr);
284            app.hits.push((fr, Hit::Frame(d)));
285            continue;
286        }
287        // Full content (the top frame, or the previous one while it is being pushed up).
288        full_frame(f, app, d, fr, d + 1 == depth);
289    }
290}

292/// Draw the pictures whose reserved rows are fully on screen; a picture cut by
293/// the frame edge shows a one-line hint instead (graphics can't be clipped).
294fn draw_images(f: &mut Frame, app: &mut App, slots: &[(usize, std::path::PathBuf)], rows: u16, scroll: u16, fr: Rect) {
295    let t = app.theme;
296    // Graphics go on top of everything, so under a modal or an open menu draw half blocks.
297    let graphics = app.modal.is_none() && app.menu.open.is_none();
298    let Some(im) = app.images.as_mut() else { return };
299    for (li, path) in slots {
300        let y = *li as i64 - scroll as i64;
301        let end = y + rows as i64;
302        if end <= 0 || y >= fr.height as i64 { continue; }
303        let full = y >= 0 && end <= fr.height as i64;
304        let ry = fr.y + y.max(0) as u16;
305        let hint = |f: &mut Frame, text: String| {
306            f.render_widget(Paragraph::new(Span::styled(format!("  {text}"), t.muted())), Rect { x: fr.x, y: ry, width: fr.width, height: 1 });
307        };
308        if !full { hint(f, "▣ picture — scroll to show it".into()); continue; }
309        let area = Rect { x: fr.x, y: ry, width: fr.width.saturating_sub(1), height: rows };
310        if let Err(e) = im.draw_file(f.buffer_mut(), area, path, graphics) { hint(f, e); }
311    }
312}

314/// Draw frame `d` of the detail stack in full into `fr`; the top one gets its link, star and wheel hits.
315fn full_frame(f: &mut Frame, app: &mut App, d: usize, fr: Rect, top: bool) {
316    let t = app.theme;
317    // One more column of padding on each side of the text.
318    let fr = if fr.width > 6 { Rect { x: fr.x + 1, width: fr.width - 2, ..fr } } else { fr };
319    let w = fr.width.saturating_sub(1) as usize;
320    let h = fr.height as usize;
321    let img_rows = if app.images.is_some() { crate::tui::images::IMAGE_ROWS.min(fr.height.saturating_sub(2)) } else { 0 };
322    let (lines, links, slots) = detail_layout(&app.view().stack[d], w, &t, img_rows);
323    let fm = &mut app.view_mut().stack[d];
324    if top && fm.follow_link
325        && let Some((li, _)) = links.iter().find(|(_, k)| *k == fm.link_cursor) {
326            let li = *li as u16;
327            if li < fm.scroll { fm.scroll = li; }
328            if li as usize >= fm.scroll as usize + h { fm.scroll = (li as usize + 1 - h) as u16; }
329            fm.follow_link = false;
330        }
331    let max = lines.len().saturating_sub(h) as u16;
332    if fm.scroll > max { fm.scroll = max; }
333    let scroll = fm.scroll;
334    f.render_widget(Paragraph::new(lines).scroll((scroll, 0)), fr);
335    draw_images(f, app, &slots, img_rows, scroll, fr);
336    if !top { app.hits.push((fr, Hit::Frame(d))); return; }
337    app.hits.push((fr, Hit::DetailPane));
338    for (li, k) in links {
339        let y = li as i64 - scroll as i64;
340        if y >= 0 && (y as usize) < h {
341            app.hits.push((Rect { x: fr.x, y: fr.y + y as u16, width: fr.width, height: 1 }, Hit::Link(k)));
342            // Attachment rows also become OSC 8 hyperlinks (file://) where the terminal supports them.
343            let l = &app.view().stack[d].links[k];
344            if l.kind == crate::tui::state::LinkKind::Attachment && std::path::Path::new(&l.id).exists() {
345                let label_w = lines_label_width(&app.view().stack[d].attachments, &l.id);
346                let name = std::path::Path::new(&l.id).file_name().map(|n| n.to_string_lossy().into_owned()).unwrap_or_default();
347                let x = fr.x + label_w as u16;
348                if x < fr.x + fr.width {
349                    let shown = ellipsize(&name, (fr.x + fr.width - x) as usize);
350                    app.osc_links.push((x, fr.y + y as u16, shown, format!("file://{}", l.id)));
351                }
352            }
353        }
354    }
355    if fr.width > 6 {
356        let marked = app.is_marked(&app.view().stack[d].post.post_id);
357        let sr = Rect { x: fr.x + fr.width - 4, y: fr.y, width: 3, height: 1 };
358        f.render_widget(Paragraph::new(Span::styled(star(marked), t.accent_bold())), sr);
359        app.hits.push((sr, Hit::StarTop));
360    }
361    if fr.width > 10 {
362        let inb = app.in_basket(&app.view().stack[d].post.post_id);
363        let br = Rect { x: fr.x + fr.width - 7, y: fr.y, width: 3, height: 1 };
364        f.render_widget(Paragraph::new(Span::styled(basket_icon(inb), t.accent_bold())), br);
365        app.hits.push((br, Hit::BasketTop));
366    }
367    if fr.width > 14 && !app.view().stack[d].post.post_id.starts_with("research:") {
368        let rr = Rect { x: fr.x + fr.width - 10, y: fr.y, width: 3, height: 1 };
369        f.render_widget(Paragraph::new(Span::styled(RESEARCH_ICON, t.accent_bold())), rr);
370        app.hits.push((rr, Hit::ResearchTop));
371    }
