16/// Draw the Body into `r`.
17pub fn draw(f: &mut Frame, app: &mut App, r: Rect, now: Instant) {
18    let t = app.theme;
19    let focused = app.focus == Focus::Body;
20    let b = Block::default().borders(Borders::ALL).border_style(t.border(focused));
21    let inner = b.inner(r);
22    f.render_widget(b, r);
23    if inner.height < 2 { return; }
24    // Navigation bar.
25    let nav = Rect { height: 1, ..inner };
26    let content = Rect { y: inner.y + 1, height: inner.height - 1, ..inner };
27    let in_search = app.search.is_some();
28    if in_search { app.hits.push((Rect { width: 8.min(nav.width), ..nav }, Hit::Back)); }
29    let v = app.view();
30    let mut spans = Vec::new();
31    if in_search {
32        spans.push(Span::styled(" < Back ", t.selected()));
33        spans.push(Span::raw(" "));
34    }
35    let title = if v.title.is_empty() { "Select a nick on the left".to_string() } else { v.title.clone() };
36    spans.push(Span::styled(title, Style::default().add_modifier(Modifier::BOLD)));
37    spans.push(Span::styled(format!("  {} posts", v.posts.len()), t.muted()));
38    if !v.stack.is_empty() {
39        spans.push(Span::styled(format!("  · detail {}/{}", v.stack.len(), crate::config::MAX_DETAIL_DEPTH), t.muted()));
40    }
41    f.render_widget(Paragraph::new(Line::from(spans)), nav);
42    // View switch, right-aligned.
43    let vl = format!(" [View: {} ▸] ", app.view_mode.label());
44    let vw = (unicode_width::UnicodeWidthStr::width(vl.as_str()) as u16).min(nav.width);
45    let vr = Rect { x: nav.x + nav.width - vw, width: vw, ..nav };
46    f.render_widget(Paragraph::new(Span::styled(vl, t.accent_bold())), vr);
47    app.hits.push((vr, Hit::ViewMode));
48
49    if app.view().posts.is_empty() {
50        let msg = if in_search { "No posts match this search. Press < to go back." }
51            else if app.nicks.is_empty() { "The database is empty. Open Scrape (or Import) in the menu bar to fill it." }
52            else { "Click a nick (Space / Ctrl-click to select several) to read their posts and replies." };
53        f.render_widget(Paragraph::new(msg).style(t.muted()), Rect { y: content.y + 1, height: content.height.saturating_sub(1), ..content });
54        return;
55    }
56    let open = !app.view().stack.is_empty();
57    match app.view_mode {
58        ViewMode::Nested => {
59            app.page_rows = (content.height / CARD_ROWS).max(1) as usize;
60            if open { draw_with_detail(f, app, content, now); } else { draw_cards(f, app, content); }
61        }
62        ViewMode::Stack => if open { draw_flat(f, app, content); } else { draw_stack(f, app, content); },
63        ViewMode::Pane => {
64            let list_h = if open { (content.height * 2 / 5).max(4).min(content.height) } else { content.height };
65            draw_stack(f, app, Rect { height: list_h, ..content });
66            if open { draw_flat(f, app, Rect { y: content.y + list_h, height: content.height - list_h, ..content }); }
67        }
68        ViewMode::CarouselInfinite | ViewMode::CarouselFixed => if open { draw_flat(f, app, content); } else { draw_carousel(f, app, content); },
69        ViewMode::Grid => {
70            draw_grid(f, app, content);
71            if open { draw_grid_modal(f, app, r); }
72        }
73    }
74}

76fn post_text(p: &Post) -> String { if p.title.trim().is_empty() { p.content.clone() } else { format!("{}\n{}", p.title.trim(), p.content) } }

78/// One post card in `cr` (border with meta line and star, wrapped text, scrolled by `skip` lines).
79fn card(f: &mut Frame, app: &mut App, i: usize, cr: Rect, sel: bool) {
80    let t = app.theme;
81    let Some(p) = app.view().posts.get(i).cloned() else { return };
82    let focused = app.focus == Focus::Body;
83    let b = Block::default().borders(Borders::ALL).title(meta_line(&p, &t))
84        .border_type(if sel { BorderType::Thick } else { BorderType::Plain })
85        .border_style(if sel && focused { t.accent() } else if sel { Style::default() } else { t.border(false) });
86    let inner = b.inner(cr);
87    f.render_widget(b, cr);
88    let lines: Vec<Line> = wrap(post_text(&p).trim(), inner.width as usize).into_iter().filter(|l| !l.trim().is_empty())
89        .take(inner.height as usize).map(Line::raw).collect();
90    f.render_widget(Paragraph::new(lines), inner);
91    app.hits.push((cr, Hit::Post(i)));
92    if cr.width > 8 {
93        let sr = Rect { x: cr.x + cr.width - 5, y: cr.y, width: 3, height: 1 };
94        f.render_widget(Paragraph::new(Span::styled(star(app.is_marked(&p.post_id)), t.accent_bold())), sr);
95        app.hits.push((sr, Hit::Star(i)));
96    }
97}

99/// Stack: one column of full posts (each card as tall as its text, up to the area).
100fn draw_stack(f: &mut Frame, app: &mut App, r: Rect) {
101    app.hits.push((r, Hit::ListPane));
102    let w = r.width;
103    let heights: Vec<u16> = app.view().posts.iter().map(|p| card_height(&post_text(p), w, r.height)).collect();
104    let v = app.view_mut();
105    if v.cursor < v.offset { v.offset = v.cursor; }
106    // Keep the cursor's card fully on screen.
107    while v.offset < v.cursor && heights[v.offset..=v.cursor].iter().map(|h| *h as u32).sum::<u32>() > r.height as u32 { v.offset += 1; }
108    let (off, cur) = (v.offset, v.cursor);
109    let mut y = r.y;
110    let mut shown = 0;
111    for (i, h) in heights.iter().enumerate().skip(off) {
112        if y >= r.y + r.height { break; }
113        let h = (*h).min(r.y + r.height - y);
114        card(f, app, i, Rect { x: r.x, y, width: w, height: h }, i == cur);
115        y += h;
116        shown += 1;
117    }
118    app.page_rows = shown.max(1);
119}

121/// A detail drawn flat (no nesting): the top frame fills `r`.
122fn draw_flat(f: &mut Frame, app: &mut App, r: Rect) {
123    let t = app.theme;
124    let focused = app.focus == Focus::Body;
125    let depth = app.view().stack.len();
126    let title = if depth > 1 { format!(" {depth} deep · Esc back ") } else { " Esc closes ".into() };
127    let b = Block::default().borders(Borders::ALL).border_type(BorderType::Double).title(title)
128        .border_style(if focused { t.accent() } else { t.border(false) });
129    let inner = b.inner(r);
130    f.render_widget(b, r);
131    if inner.width < 3 || inner.height == 0 { return; }
132    full_frame(f, app, depth - 1, Rect { x: inner.x + 1, width: inner.width - 1, ..inner }, true);
133}

135/// Carousel: the current post in the middle, its neighbours peeking at the sides.
136fn draw_carousel(f: &mut Frame, app: &mut App, r: Rect) {
137    let t = app.theme;
138    app.hits.push((r, Hit::ListPane));
139    app.page_rows = 1;
140    let n = app.view().posts.len();
141    let cur = app.view().cursor.min(n - 1);
142    let c = carousel(r);
143    let wrapv = app.view_mode == ViewMode::CarouselInfinite;
144    let prev = if cur > 0 { Some(cur - 1) } else if wrapv && n > 1 { Some(n - 1) } else { None };
145    let next = if cur + 1 < n { Some(cur + 1) } else if wrapv && n > 1 { Some(0) } else { None };
146    for (side, idx, arrow) in [(c.prev, prev, "◂"), (c.next, next, "▸")] {
147        let b = Block::default().borders(Borders::ALL).border_style(t.border(false));
148        let inner = b.inner(side);
149        f.render_widget(b, side);
150        if let Some(i) = idx {
151            let p = &app.view().posts[i];
152            let mut lines = vec![Line::styled(arrow, t.accent_bold()), Line::styled(ellipsize(&format!("@{}", p.nick), inner.width as usize), t.muted())];
153            lines.extend(wrap(&snippet(p), inner.width as usize).into_iter().take(inner.height.saturating_sub(2) as usize).map(|l| Line::styled(l, t.muted())));
154            f.render_widget(Paragraph::new(lines), inner);
155            app.hits.push((side, Hit::Post(i)));
156        }
157    }
158    card(f, app, cur, c.center, true);
159    let pos = format!(" {}/{n}{} ", cur + 1, if wrapv { " ∞" } else { "" });
160    let pr = Rect { x: c.center.x + 2, y: c.center.y + c.center.height.saturating_sub(1), width: (pos.chars().count() as u16).min(c.center.width.saturating_sub(4)), height: 1 };
161    f.render_widget(Paragraph::new(Span::styled(pos, t.muted())), pr);
162}

164/// Grid of cards; scrolls by rows to keep the cursor visible.
165fn draw_grid(f: &mut Frame, app: &mut App, r: Rect) {
166    app.hits.push((r, Hit::ListPane));
167    let n = app.view().posts.len();
168    let (cols, _) = grid_cells(r, n, 0);
169    let rows = grid_rows(r);
170    app.grid_cols = cols;
171    app.page_rows = rows * cols;
172    let v = app.view_mut();
173    let crow = v.cursor / cols;
174    let mut orow = v.offset / cols;
175    if crow < orow { orow = crow; }
176    if crow >= orow + rows { orow = crow + 1 - rows; }
177    v.offset = orow * cols;
178    let (first, cur) = (v.offset, v.cursor);
179    let (_, cells) = grid_cells(r, n, first);
180    for (i, cr) in cells { card(f, app, i, cr, i == cur); }
181}

183/// Grid: the open post as a modal over the Body, with a Close button.
184fn draw_grid_modal(f: &mut Frame, app: &mut App, body: Rect) {
185    let t = app.theme;
186    let mr = modal_rect(body, body.width.saturating_sub(6).max(40), body.height.saturating_sub(2));
187    f.render_widget(Clear, mr);
188    app.hits.push((mr, Hit::DetailPane));
189    let depth = app.view().stack.len();
190    let b = Block::default().borders(Borders::ALL).border_type(BorderType::Double).border_style(t.accent())
191        .title(if depth > 1 { format!(" Post · {depth} deep · Esc back ") } else { " Post ".into() });
192    let inner = b.inner(mr);
193    f.render_widget(b, mr);
194    if inner.height < 3 || inner.width < 12 { return; }
195    let close = " [ Close ] ";
196    let cw = close.chars().count() as u16;
197    let cr = Rect { x: inner.x + inner.width.saturating_sub(cw), y: inner.y + inner.height - 1, width: cw.min(inner.width), height: 1 };
198    full_frame(f, app, depth - 1, Rect { x: inner.x + 1, width: inner.width - 1, height: inner.height - 1, ..inner }, true);
199    f.render_widget(Paragraph::new(Span::styled(close, t.selected())), cr);
200    app.hits.push((cr, Hit::CloseDetail));
201}

203fn draw_cards(f: &mut Frame, app: &mut App, r: Rect) {
204    app.hits.push((r, Hit::ListPane));
205    let rows = app.page_rows;
206    let v = app.view_mut();
207    if v.cursor < v.offset { v.offset = v.cursor; }
208    if v.cursor >= v.offset + rows { v.offset = v.cursor + 1 - rows; }
209    let (off, cur, n) = (v.offset, v.cursor, v.posts.len());
210    for (k, i) in (off..n).take(rows).enumerate() {
211        let cr = Rect { x: r.x, y: r.y + k as u16 * CARD_ROWS, width: r.width, height: CARD_ROWS.min(r.height.saturating_sub(k as u16 * CARD_ROWS)) };
212        card(f, app, i, cr, i == cur);
213    }
214}

219fn compact(f: &mut Frame, app: &mut App, i: usize, r: Rect) {
220    let t = app.theme;
221    let Some(p) = app.view().posts.get(i).cloned() else { return };
222    let w = r.width as usize;
223    let head = format!(" @{} · {} ", p.nick, p.posted_at.get(..16).unwrap_or(&p.posted_at));
224    let rest = ellipsize(&snippet(&p), w.saturating_sub(head.len() + 8));
225    let mark = if app.is_marked(&p.post_id) { "★" } else { " " };
226    f.render_widget(Paragraph::new(Line::from(vec![Span::styled(mark, t.accent_bold()), badge(&p, &t), Span::styled(head, t.muted()), Span::raw(rest)])), r);
227    app.hits.push((r, Hit::Post(i)));
228}

230fn draw_with_detail(f: &mut Frame, app: &mut App, r: Rect, now: Instant) {
231    let t = app.theme;
232    let (kind, p) = app.anim_progress(now);
233    let open_p = if kind == Some(AnimKind::Open) { p } else { 1.0 };
234    let push_p = if kind == Some(AnimKind::Push) { p } else { 1.0 };
235    let (n, cur) = (app.view().posts.len(), app.view().cursor);
236    let split = detail_split(r, n, cur, open_p);
237    for (i, rr) in split.above.clone() { compact(f, app, i, rr); }
238    for (i, rr) in split.below.clone() { compact(f, app, i, rr); }
239    let focused = app.focus == Focus::Body;
240    let b = Block::default().borders(Borders::ALL).border_type(BorderType::Double).border_style(if focused { t.accent() } else { t.border(false) });
241    let inner = b.inner(split.detail);
242    f.render_widget(b, split.detail);
243    let depth = app.view().stack.len();
244    let sl = stack_layout(inner, depth, push_p);
245    if let Some(nb) = sl.nested {
246        f.render_widget(Block::default().borders(Borders::ALL).border_type(BorderType::Double).border_style(t.accent()), nb);
247    }
248    for (d, fr, collapsed) in sl.frames {
249        if fr.height == 0 || fr.width < 3 { continue; }
250        let fr = Rect { x: fr.x + 1, width: fr.width - 1, ..fr };
251        if collapsed && fr.height <= DETAIL_HEADER_ROWS + 1 {
252            let lines = header_lines(&app.view().stack[d], fr.width as usize, &t);
253            f.render_widget(Paragraph::new(lines), fr);
254            app.hits.push((fr, Hit::Frame(d)));
255            continue;
256        }
257        // Full content (the top frame, or the previous one while it is being pushed up).
258        full_frame(f, app, d, fr, d + 1 == depth);
259    }
260}

262/// Draw frame `d` of the detail stack in full into `fr`; the top one gets its link, star and wheel hits.
263fn full_frame(f: &mut Frame, app: &mut App, d: usize, fr: Rect, top: bool) {
264    let t = app.theme;
265    let w = fr.width.saturating_sub(1) as usize;
266    let (lines, links) = detail_lines(&app.view().stack[d], w, &t);
267    let h = fr.height as usize;
268    let fm = &mut app.view_mut().stack[d];
269    if top && fm.follow_link
270        && let Some((li, _)) = links.iter().find(|(_, k)| *k == fm.link_cursor) {
271            let li = *li as u16;
272            if li < fm.scroll { fm.scroll = li; }
273            if li as usize >= fm.scroll as usize + h { fm.scroll = (li as usize + 1 - h) as u16; }
274            fm.follow_link = false;
275        }
276    let max = lines.len().saturating_sub(h) as u16;
277    if fm.scroll > max { fm.scroll = max; }
278    let scroll = fm.scroll;
279    f.render_widget(Paragraph::new(lines).scroll((scroll, 0)), fr);
280    if !top { app.hits.push((fr, Hit::Frame(d))); return; }
281    app.hits.push((fr, Hit::DetailPane));
282    for (li, k) in links {
283        let y = li as i64 - scroll as i64;
284        if y >= 0 && (y as usize) < h {
285            app.hits.push((Rect { x: fr.x, y: fr.y + y as u16, width: fr.width, height: 1 }, Hit::Link(k)));
286        }
287    }
288    if fr.width > 6 {
289        let marked = app.is_marked(&app.view().stack[d].post.post_id);
290        let sr = Rect { x: fr.x + fr.width - 4, y: fr.y, width: 3, height: 1 };
291        f.render_widget(Paragraph::new(Span::styled(star(marked), t.accent_bold())), sr);
292        app.hits.push((sr, Hit::StarTop));
293    }
