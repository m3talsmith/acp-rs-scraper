17fn swatch(c: Option<Color>) -> Span<'static> {
18    match c {
19        Some(Color::Reset) => Span::raw("··"),
20        Some(c) => Span::styled("██", Gloss::new().foreground(c)),
21        None => Span::raw("  "),
22    }
23}

25/// Draw the editor over the whole screen.
26pub fn draw(f: &mut Frame, app: &mut App, ed: &mut ThemeEditor, now: Instant) {
27    let area = f.area();
28    let ui = app.theme;
29    let et = ed.theme();
30    f.render_widget(Clear, area);
31    let dirty = if ed.dirty() { " *" } else { "" };
32    let mode = match ed.variant { themes::Variant::Dark => "‹dark› light", themes::Variant::Light => "dark ‹light›" };
33    let title = format!(" Theme editor · {}{dirty} · {mode} (Tab) · ^S save ^R revert (r: row) ^N new ^D duplicate ^X delete · [ ] other theme · h highlight · Esc close",
34        ed.file.name);
35    f.render_widget(Paragraph::new(Span::styled(title, ui.bar().bold(true))).style(ui.bar()), Rect { height: 1, ..area });
36    let body = Rect { y: area.y + 1, height: area.height.saturating_sub(2), ..area };
37    let lw = (body.width / 2).clamp(30, 52).min(body.width);
38    let left = Rect { width: lw, ..body };
39    let right = Rect { x: body.x + lw, width: body.width.saturating_sub(lw), ..body };
40
41    // The list.
42    let lb = frame(&ui.pane(true)).title(format!(" {} ", if ed.file.description.is_empty() { "Theme" } else { &ed.file.description }));
43    let inner = lb.inner(left);
44    f.render_widget(lb, left);
45    let rows = ed.rows();
46    let h = inner.height as usize;
47    if ed.cursor < ed.scroll { ed.scroll = ed.cursor; }
48    if h > 0 && ed.cursor >= ed.scroll + h { ed.scroll = ed.cursor + 1 - h; }
49    for (k, (i, r)) in rows.iter().enumerate().skip(ed.scroll).take(h).enumerate() {
50        let rr = Rect { y: inner.y + k as u16, height: 1, ..inner };
51        let sel = i == ed.cursor;
52        let mark = if ed.row_changed(r) { "*" } else { " " };
53        let warn = ed.row_warning(r).map(|x| format!(" ⚠{x:.1}:1")).unwrap_or_default();
54        let base = if sel { ui.selected() } else { ui.plain() };
55        let line = match r {
56            Row::Heading(t) => Line::styled(format!(" {t}"), ui.accent_bold()),
57            Row::Palette(n) => Line::from(vec![Span::raw(" "), swatch(ed.row_color(r)), Span::styled(format!(" {n:<11} {:<10}{mark}", ed.row_value(r)), &base),
58                Span::styled(warn, ui.accent_bold())]),
59            Row::Element(e) => {
60                let open = if ed.expanded == Some(*e) { "▾" } else { "▸" };
61                let sample = if e.is_box() {
62                    let st = ratatui::style::Style::from(&et.element(*e));
63                    Span::styled("┌┐", Gloss::new().foreground(st.fg.unwrap_or(Color::Reset)).background(et.bg))
64                } else { Span::styled("Ab", et.element(*e)) };
65                let v = ed.row_value(r);
66                let room = (inner.width as usize).saturating_sub(24);
67                let v: String = if v.chars().count() > room { v.chars().take(room.saturating_sub(1)).chain(['…']).collect() } else { v };
68                Line::from(vec![Span::styled(format!(" {open} "), &base), sample, Span::styled(format!(" {:<15}{mark} ", e.name()), &base),
69                    Span::styled(v, if sel { base.clone() } else { ui.muted() }), Span::styled(warn, ui.accent_bold())])
70            }
71            Row::Field(_, k) => Line::from(vec![Span::raw("     "), swatch(ed.row_color(r)), Span::styled(format!(" {k:<9} {}{mark}", ed.row_value(r)), &base),
72                Span::styled(warn, ui.accent_bold())]),
73        };
74        f.render_widget(Paragraph::new(line), rr);
75        app.hits.push((rr, Hit::EditorRow(i)));
76    }
77
78    // The preview: the real UI, drawn off screen at a size it supports, then copied in.
79    // With highlighting on it is drawn a second time with the selected row's colours
80    // replaced by sentinels; the cells that differ are shown in reverse video.
81    let pin = frame(&ui.pane(false)).inner(right);
82    let (w, hh) = (pin.width.max(TUI_MIN_SIZE.0), pin.height.max(TUI_MIN_SIZE.1));
83    let mut shown = (pin.width > 0 && pin.height > 0).then(|| preview_buffer(&mut ed.preview, w, hh, now)).flatten();
84    let mut marked = 0;
85    let marked_theme = if ed.highlight { ed.marked_theme() } else { None };
86    if let (Some(s), Some(mt)) = (shown.as_mut(), marked_theme) {
87        let keep = ed.preview.theme;
88        ed.preview.theme = mt;
89        let m = preview_buffer(&mut ed.preview, w, hh, now);
90        ed.preview.theme = keep;
91        if let Some(m) = m { marked = highlight_changes(s, &m, Rect::new(0, 0, pin.width, pin.height)); }
92    }
93    let note = match (ed.highlight, ed.row_name()) {
94        (false, _) => " · h: highlight".to_string(),
95        (true, None) => String::new(),
96        (true, Some(n)) if marked == 0 => format!(" · ‹{n}› isn't on screen"),
97        (true, Some(n)) => format!(" · ‹{n}› highlighted (h)"),
98    };
99    let mut pb = frame(&ui.pane(false)).title(format!(" Preview (sample posts){note} "));
100    if let Some(n) = ed.truecolor_note() { pb = pb.title_bottom(Line::styled(format!(" ⚠ {n} "), ui.accent_bold())); }
101    f.render_widget(pb, right);
102    if let Some(src) = shown {
103        let buf = f.buffer_mut();
104        for y in 0..pin.height { for x in 0..pin.width {
105            if let (Some(c), Some(d)) = (src.cell((x, y)), buf.cell_mut((pin.x + x, pin.y + y))) { *d = c.clone(); }
106        } }
107    }
108
109    // Footer: the result of the last action, or what the row is.
110    let foot = Rect { y: area.y + area.height.saturating_sub(1), height: 1, ..area };
111    let (err, text) = ed.message.clone().unwrap_or_else(|| (false, match ed.current() {
112        Row::Palette(n) => PALETTE_KEYS.iter().find(|(k, _)| *k == n).map(|(_, a)| a.to_string()).unwrap_or_else(|| format!("@{n}: your own colour for references")) + " · Enter picks · Del resets · a adds a colour",
113        Row::Element(e) => format!("{} · Enter shows its keys · Del clears its overrides · r reverts", e.about()),
114        Row::Field(_, k) if ["fg", "bg", "border"].contains(&k) => "Enter picks a colour · Del clears (built-in)".into(),
115        Row::Field(..) => "Enter cycles on / off / built-in · Del clears".into(),
116        Row::Heading(_) => String::new(),
117    }));
118    f.render_widget(Paragraph::new(Span::styled(format!(" {text}"), if err { ui.accent_bold() } else { ui.muted() })), foot);
119
120    if ed.picker.is_some() { draw_picker(f, &ui, ed, area); }
121    if let Some(p) = &ed.prompt {
122        let (q, named) = match &p.ask {
123            Ask::UnsavedClose | Ask::UnsavedOpen(_) => (format!("{} has unsaved changes: [S]ave · [D]iscard · [C]ancel", ed.file.name), false),
124            Ask::Delete(n) => (format!("Delete {n}.toml? [y]es · [n]o"), false),
125            Ask::ChangedOnDisk(_) => (format!("{}.toml changed on disk since you opened it.\n[O]verwrite it · [L]oad it (drop your changes) · [C]ancel", ed.file.name), false),
126            Ask::OverwriteBroken(_) => (format!("{}.toml on disk is broken (it can't be read as a theme).\n[O]verwrite it · [C]ancel", ed.file.name), false),
127            Ask::NewName => ("Name of the new theme:".into(), true),
128            Ask::DuplicateName => (format!("Name of the copy of {}:", ed.file.name), true),
129            Ask::PaletteName => ("Name of the new palette colour (use it as @name):".into(), true),
130        };
131        let r = crate::tui::layout::modal_rect(area, 66.min(area.width), 6);
132        f.render_widget(Clear, r);
133        let b = frame(&ui.dialog()).title(" Theme editor ");
134        let i = b.inner(r);
135        f.render_widget(b, r);
136        let mut lines: Vec<Line> = q.lines().map(|l| Line::raw(l.to_string())).collect();
137        if named { lines.push(Line::styled(format!("› {}▏", p.field.value), ui.field())); }
138        if let Some(e) = &p.error { lines.push(Line::styled(e.clone(), ui.accent_bold())); }
139        f.render_widget(Paragraph::new(lines), i);
140    }
141}

143fn draw_picker(f: &mut Frame, ui: &Theme, ed: &ThemeEditor, area: Rect) {
144    let Some(p) = &ed.picker else { return };
145    let grid = ed.grid(p.tab);
146    let cols = ThemeEditor::grid_cols(p.tab);
147    let grid_rows = grid.len().div_ceil(cols) as u16;
148    let r = crate::tui::layout::modal_rect(area, 64.min(area.width), (grid_rows + 9).min(area.height));
149    f.render_widget(Clear, r);
150    let what = match &p.target { crate::tui::editor::Target::Palette(n) => n.clone(), crate::tui::editor::Target::Field(e, k) => format!("{}.{k}", e.name()) };
151    let b = frame(&ui.dialog()).title(format!(" Colour for {what} · Tab: grid · arrows · Enter sets · Esc "));
152    let i = b.inner(r);
153    f.render_widget(b, r);
154    let tabs: Vec<Span> = PICKER_TABS.iter().enumerate().map(|(k, t)| Span::styled(format!(" {t} "), if k == p.tab { ui.selected() } else { ui.plain() })).collect();
155    f.render_widget(Paragraph::new(Line::from(tabs)), Rect { height: 1, ..i });
156    f.render_widget(Paragraph::new(Line::from(vec![Span::raw("Type: "), Span::styled(format!("{}▏", p.hex.value), ui.field()),
157        Span::styled("  (#rrggbb, #rgb, 0-255, an ANSI name, @name, reset)", ui.muted())])), Rect { y: i.y + 1, height: 1, ..i });
158    let gy = i.y + 3;
159    for (k, (label, c)) in grid.iter().enumerate() {
160        let (row, col) = ((k / cols) as u16, (k % cols) as u16);
161        let sel = k == p.sel;
162        let span = if cols == 1 {
163            let v = ed.file.section(ed.variant).resolve(c);
164            Line::from(vec![swatch(Some(v)), Span::styled(format!(" {label}"), if sel { ui.selected() } else { ui.plain() })])
165        } else {
166            let ColorSpec::Value(v) = c else { continue };
167            let cell = if sel { "▐▌" } else { "██" };
168            let mut st = Gloss::new().foreground(*v);
169            if sel { st = st.background(ui.on_accent); }
170            Line::from(Span::styled(cell, st))
171        };
172        let w = if cols == 1 { i.width } else { 3 };
173        let rr = Rect { x: i.x + col * w, y: gy + row, width: w, height: 1 };
174        if rr.y < i.y + i.height.saturating_sub(2) { f.render_widget(Paragraph::new(span), rr); }
175    }
176    let chosen = grid.get(p.sel).map(|(l, c)| (l.clone(), c.clone()));
177    let mut info = vec![];
178    if let Some((l, c)) = chosen {
179        info.push(Span::raw(format!("Selected: {l} ")));
180        let v = ed.file.section(ed.variant).resolve(&c);
181        info.push(swatch(Some(v)));
182        if let Some(x) = ed.picker_warning(&c) { info.push(Span::styled(format!("  ⚠ contrast {x:.1}:1 (below {THEME_MIN_CONTRAST}:1)"), ui.accent_bold())); }
183    }
184    if let Some(e) = &p.error { info = vec![Span::styled(e.clone(), ui.accent_bold())]; }
185    f.render_widget(Paragraph::new(Line::from(info)), Rect { y: i.y + i.height.saturating_sub(1), height: 1, ..i });
186}

188/// The real UI for `app`, drawn off screen at `w`×`h`.
189fn preview_buffer(app: &mut App, w: u16, h: u16, now: Instant) -> Option<ratatui::buffer::Buffer> {
190    let mut term = ratatui::Terminal::new(ratatui::backend::TestBackend::new(w, h)).ok()?;
191    term.draw(|pf| { crate::tui::widgets::draw(pf, app, now); draw_samples(pf, &app.theme); }).ok()?;
192    Some(term.backend().buffer().clone())
193}

195/// Flip reverse video on every cell of `shown` inside `within` whose colours or
196/// attributes differ in `marked`; returns how many were flipped.
197pub fn highlight_changes(shown: &mut ratatui::buffer::Buffer, marked: &ratatui::buffer::Buffer, within: Rect) -> usize {
198    use ratatui::style::Modifier;
199    let mut n = 0;
200    for y in within.top()..within.bottom() { for x in within.left()..within.right() {
201        let Some(m) = marked.cell((x, y)).map(|m| (m.fg, m.bg, m.modifier)) else { continue };
202        if let Some(c) = shown.cell_mut((x, y)) && (c.fg, c.bg, c.modifier) != m {
203            c.modifier.toggle(Modifier::REVERSED);
204            n += 1;
205        }
206    } }
207    n
208}

210/// Sample frames over the preview's lower right corner, so the parts the post view
211/// doesn't show are there to look at: a dialog with a field, labels, a button, tags,
212/// deleted-post text, code and text dimmed behind a modal.
213pub fn draw_samples(f: &mut Frame, t: &Theme) {
214    let a = f.area();
215    let (w, h) = (40.min(a.width.saturating_sub(2)), 9.min(a.height.saturating_sub(4)));
216    if w < 34 || h < 8 { return; }
217    let r = Rect::new(a.right() - w - 1, a.bottom() - h - 2, w, h);
218    f.render_widget(Clear, r);
219    let b = frame(&t.dialog()).title(" Sample dialog ");
220    let i = b.inner(r);
221    f.render_widget(b, r);
222    let lines = vec![
223        Line::from(vec![Span::styled("Nick ", t.label(true)), Span::styled("[comrade     ]", t.field()), Span::raw(" "), Span::styled("Format", t.label(false))]),
224        Line::from(vec![Span::styled(" Post ", t.post_tag()), Span::raw(" "), Span::styled(" Reply ", t.reply_tag()), Span::raw(" "), Span::styled("link", t.link())]),
225        Line::from(vec![Span::styled("(post deleted)", t.gone()), Span::raw(" "), Span::styled("`--nick`", t.code())]),
226        Line::from(vec![Span::styled("text behind a modal", t.dimmed())]),
227        Line::from(vec![Span::styled("< OK >", t.button()), Span::raw(" "), Span::styled("selected", t.selected())]),
228    ];
229    f.render_widget(Paragraph::new(lines).style(t.dialog()), i);
