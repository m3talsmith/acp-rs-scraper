17fn swatch(c: Option<Color>) -> Span<'static> {
18    match c {
19        Some(Color::Reset) => Span::raw("··"),
20        Some(c) => Span::styled("██", Gloss::new().foreground(c)),
21        None => Span::raw("  "),
22    }
23}

25/// Draw the editor over the whole screen.
26pub fn draw(f: &mut Frame, app: &mut App, ed: &mut ThemeEditor, now: Instant) {
27    let area = f.area();
28    let ui = app.theme;
29    let et = ed.theme();
30    f.render_widget(Clear, area);
31    let dirty = if ed.dirty() { " *" } else { "" };
32    let mode = match ed.variant { themes::Variant::Dark => "‹dark› light", themes::Variant::Light => "dark ‹light›" };
33    let title = format!(" Theme editor · {}{dirty} · {mode} (Tab) · ^S save ^R revert ^N new ^D duplicate ^X delete · [ ] other theme · Esc close",
34        ed.file.name);
35    f.render_widget(Paragraph::new(Span::styled(title, ui.bar().bold(true))).style(ui.bar()), Rect { height: 1, ..area });
36    let body = Rect { y: area.y + 1, height: area.height.saturating_sub(2), ..area };
37    let lw = (body.width / 2).clamp(30, 52).min(body.width);
38    let left = Rect { width: lw, ..body };
39    let right = Rect { x: body.x + lw, width: body.width.saturating_sub(lw), ..body };
40
41    // The list.
42    let lb = frame(&ui.pane(true)).title(format!(" {} ", if ed.file.description.is_empty() { "Theme" } else { &ed.file.description }));
43    let inner = lb.inner(left);
44    f.render_widget(lb, left);
45    let rows = ed.rows();
46    let h = inner.height as usize;
47    if ed.cursor < ed.scroll { ed.scroll = ed.cursor; }
48    if h > 0 && ed.cursor >= ed.scroll + h { ed.scroll = ed.cursor + 1 - h; }
49    for (k, (i, r)) in rows.iter().enumerate().skip(ed.scroll).take(h).enumerate() {
50        let rr = Rect { y: inner.y + k as u16, height: 1, ..inner };
51        let sel = i == ed.cursor;
52        let mark = if ed.row_changed(r) { "*" } else { " " };
53        let warn = ed.row_warning(r).map(|x| format!(" ⚠{x:.1}:1")).unwrap_or_default();
54        let base = if sel { ui.selected() } else { ui.plain() };
55        let line = match r {
56            Row::Heading(t) => Line::styled(format!(" {t}"), ui.accent_bold()),
57            Row::Palette(n) => Line::from(vec![Span::raw(" "), swatch(ed.row_color(r)), Span::styled(format!(" {n:<11} {:<10}{mark}", ed.row_value(r)), &base),
58                Span::styled(warn, ui.accent_bold())]),
59            Row::Element(e) => {
60                let open = if ed.expanded == Some(*e) { "▾" } else { "▸" };
61                let sample = if e.is_box() {
62                    let st = ratatui::style::Style::from(&et.element(*e));
63                    Span::styled("┌┐", Gloss::new().foreground(st.fg.unwrap_or(Color::Reset)).background(et.bg))
64                } else { Span::styled("Ab", et.element(*e)) };
65                let v = ed.row_value(r);
66                let room = (inner.width as usize).saturating_sub(24);
67                let v: String = if v.chars().count() > room { v.chars().take(room.saturating_sub(1)).chain(['…']).collect() } else { v };
68                Line::from(vec![Span::styled(format!(" {open} "), &base), sample, Span::styled(format!(" {:<15}{mark} ", e.name()), &base),
69                    Span::styled(v, if sel { base.clone() } else { ui.muted() }), Span::styled(warn, ui.accent_bold())])
70            }
71            Row::Field(_, k) => Line::from(vec![Span::raw("     "), swatch(ed.row_color(r)), Span::styled(format!(" {k:<9} {}{mark}", ed.row_value(r)), &base),
72                Span::styled(warn, ui.accent_bold())]),
73        };
74        f.render_widget(Paragraph::new(line), rr);
75        app.hits.push((rr, Hit::EditorRow(i)));
76    }
77
78    // The preview: the real UI, drawn off screen at a size it supports, then copied in.
79    let pb = frame(&ui.pane(false)).title(" Preview (sample posts) ");
80    let pin = pb.inner(right);
81    f.render_widget(pb, right);
82    if pin.width > 0 && pin.height > 0 {
83        let (w, hh) = (pin.width.max(TUI_MIN_SIZE.0), pin.height.max(TUI_MIN_SIZE.1));
84        if let Ok(mut term) = ratatui::Terminal::new(ratatui::backend::TestBackend::new(w, hh))
85            && term.draw(|pf| crate::tui::widgets::draw(pf, &mut ed.preview, now)).is_ok() {
86            let src = term.backend().buffer().clone();
87            let buf = f.buffer_mut();
88            for y in 0..pin.height { for x in 0..pin.width {
89                if let (Some(c), Some(d)) = (src.cell((x, y)), buf.cell_mut((pin.x + x, pin.y + y))) { *d = c.clone(); }
90            } }
91        }
92    }
93
94    // Footer: the result of the last action, or what the row is.
95    let foot = Rect { y: area.y + area.height.saturating_sub(1), height: 1, ..area };
96    let (err, text) = ed.message.clone().unwrap_or_else(|| (false, match ed.current() {
97        Row::Palette(n) => PALETTE_KEYS.iter().find(|(k, _)| *k == n).map(|(_, a)| a.to_string()).unwrap_or_else(|| format!("@{n}: your own colour for references")) + " · Enter picks · Del resets · a adds a colour",
98        Row::Element(e) => format!("{} · Enter shows its keys · Del clears its overrides", e.about()),
99        Row::Field(_, k) if ["fg", "bg", "border"].contains(&k) => "Enter picks a colour · Del clears (built-in)".into(),
100        Row::Field(..) => "Enter cycles on / off / built-in · Del clears".into(),
101        Row::Heading(_) => String::new(),
102    }));
103    f.render_widget(Paragraph::new(Span::styled(format!(" {text}"), if err { ui.accent_bold() } else { ui.muted() })), foot);
104
105    if ed.picker.is_some() { draw_picker(f, &ui, ed, area); }
106    if let Some(p) = &ed.prompt {
107        let (q, named) = match &p.ask {
108            Ask::UnsavedClose | Ask::UnsavedOpen(_) => (format!("{} has unsaved changes: [S]ave · [D]iscard · [C]ancel", ed.file.name), false),
109            Ask::Delete(n) => (format!("Delete {n}.toml? [y]es · [n]o"), false),
110            Ask::NewName => ("Name of the new theme:".into(), true),
111            Ask::DuplicateName => (format!("Name of the copy of {}:", ed.file.name), true),
112            Ask::PaletteName => ("Name of the new palette colour (use it as @name):".into(), true),
113        };
114        let r = crate::tui::layout::modal_rect(area, 66.min(area.width), 6);
115        f.render_widget(Clear, r);
116        let b = frame(&ui.dialog()).title(" Theme editor ");
117        let i = b.inner(r);
118        f.render_widget(b, r);
119        let mut lines = vec![Line::raw(q)];
120        if named { lines.push(Line::styled(format!("› {}▏", p.field.value), ui.field())); }
121        if let Some(e) = &p.error { lines.push(Line::styled(e.clone(), ui.accent_bold())); }
122        f.render_widget(Paragraph::new(lines), i);
123    }
124}

126fn draw_picker(f: &mut Frame, ui: &Theme, ed: &ThemeEditor, area: Rect) {
127    let Some(p) = &ed.picker else { return };
128    let grid = ed.grid(p.tab);
129    let cols = ThemeEditor::grid_cols(p.tab);
130    let grid_rows = grid.len().div_ceil(cols) as u16;
131    let r = crate::tui::layout::modal_rect(area, 64.min(area.width), (grid_rows + 9).min(area.height));
132    f.render_widget(Clear, r);
133    let what = match &p.target { crate::tui::editor::Target::Palette(n) => n.clone(), crate::tui::editor::Target::Field(e, k) => format!("{}.{k}", e.name()) };
134    let b = frame(&ui.dialog()).title(format!(" Colour for {what} · Tab: grid · arrows · Enter sets · Esc "));
135    let i = b.inner(r);
136    f.render_widget(b, r);
137    let tabs: Vec<Span> = PICKER_TABS.iter().enumerate().map(|(k, t)| Span::styled(format!(" {t} "), if k == p.tab { ui.selected() } else { ui.plain() })).collect();
138    f.render_widget(Paragraph::new(Line::from(tabs)), Rect { height: 1, ..i });
139    f.render_widget(Paragraph::new(Line::from(vec![Span::raw("Type: "), Span::styled(format!("{}▏", p.hex.value), ui.field()),
140        Span::styled("  (#rrggbb, #rgb, 0-255, an ANSI name, @name, reset)", ui.muted())])), Rect { y: i.y + 1, height: 1, ..i });
141    let gy = i.y + 3;
142    for (k, (label, c)) in grid.iter().enumerate() {
143        let (row, col) = ((k / cols) as u16, (k % cols) as u16);
144        let sel = k == p.sel;
145        let span = if cols == 1 {
146            let v = ed.file.section(ed.variant).resolve(c);
147            Line::from(vec![swatch(Some(v)), Span::styled(format!(" {label}"), if sel { ui.selected() } else { ui.plain() })])
148        } else {
149            let ColorSpec::Value(v) = c else { continue };
150            let cell = if sel { "▐▌" } else { "██" };
151            let mut st = Gloss::new().foreground(*v);
152            if sel { st = st.background(ui.on_accent); }
153            Line::from(Span::styled(cell, st))
154        };
155        let w = if cols == 1 { i.width } else { 3 };
156        let rr = Rect { x: i.x + col * w, y: gy + row, width: w, height: 1 };
157        if rr.y < i.y + i.height.saturating_sub(2) { f.render_widget(Paragraph::new(span), rr); }
158    }
159    let chosen = grid.get(p.sel).map(|(l, c)| (l.clone(), c.clone()));
160    let mut info = vec![];
161    if let Some((l, c)) = chosen {
162        info.push(Span::raw(format!("Selected: {l} ")));
163        let v = ed.file.section(ed.variant).resolve(&c);
164        info.push(swatch(Some(v)));
165        if let Some(x) = ed.picker_warning(&c) { info.push(Span::styled(format!("  ⚠ contrast {x:.1}:1 (below {THEME_MIN_CONTRAST}:1)"), ui.accent_bold())); }
166    }
167    if let Some(e) = &p.error { info = vec![Span::styled(e.clone(), ui.accent_bold())]; }
168    f.render_widget(Paragraph::new(Line::from(info)), Rect { y: i.y + i.height.saturating_sub(1), height: 1, ..i });
169}
