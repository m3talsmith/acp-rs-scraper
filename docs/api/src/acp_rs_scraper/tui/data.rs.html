7/// One nick in the left bar.
8#[derive(Clone, Debug, PartialEq)]
9pub struct NickEntry {
10    /// Nick without `@`.
11    pub nick: String,
12    /// Profile display name, when scraped.
13    pub display_name: Option<String>,
14    /// Stored posts and replies by this nick.
15    pub post_count: i64,
16}
17
18/// Which rows a search keeps.
19#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
20pub enum KindFilter {
21    /// Posts and replies.
22    #[default]
23    Any,
24    /// Thread roots only.
25    Posts,
26    /// Replies only.
27    Replies,
28}

30/// Body search (Search menu or the Shift-S modal). Every field is optional; all combine.
31#[derive(Clone, Debug, Default, PartialEq)]
32pub struct SearchFilter {
33    /// Words in the title or body (case-insensitive substring).
34    pub keyword: String,
35    /// Only these nicks (empty: all).
36    pub nicks: Vec<String>,
37    /// A post id: that post and every reply in its thread.
38    pub post_id: String,
39    /// `posted_at >= from` (ISO prefix, e.g. `2026-10-01`).
40    pub from: String,
41    /// `posted_at <= to` (inclusive prefix, e.g. `2026-10-15` covers the whole day).
42    pub to: String,
43    /// Posts / replies / both.
44    pub kind: KindFilter,
45}
46
47impl SearchFilter {
48    /// No criterion set.
49    pub fn is_empty(&self) -> bool {
50        self.keyword.trim().is_empty() && self.nicks.is_empty() && self.post_id.trim().is_empty()
51            && self.from.trim().is_empty() && self.to.trim().is_empty() && self.kind == KindFilter::Any
52    }
53    /// Whether `p` passes the filter.
54    pub fn matches(&self, p: &Post) -> bool {
55        let kw = self.keyword.trim().to_lowercase();
56        if !kw.is_empty() && !p.title.to_lowercase().contains(&kw) && !p.content.to_lowercase().contains(&kw) { return false; }
57        if !self.nicks.is_empty() && !self.nicks.iter().any(|n| n.eq_ignore_ascii_case(&p.nick)) { return false; }
58        let id = self.post_id.trim();
59        if !id.is_empty() && p.post_id != id && p.parent_post_id != id && p.in_reply_to_id != id { return false; }
60        let (from, to) = (self.from.trim(), self.to.trim());
61        if !from.is_empty() && p.posted_at.as_str() < from { return false; }
62        if !to.is_empty() && p.posted_at.get(..to.len()).unwrap_or(&p.posted_at) > to { return false; }
63        match self.kind {
64            KindFilter::Any => true,
65            KindFilter::Posts => p.parent_post_id.is_empty(),
66            KindFilter::Replies => !p.parent_post_id.is_empty(),
67        }
68    }
69    /// One-line description for the Body title.
70    pub fn describe(&self) -> String {
71        let mut v = Vec::new();
72        if !self.keyword.trim().is_empty() { v.push(format!("\"{}\"", self.keyword.trim())); }
73        if !self.nicks.is_empty() { v.push(self.nicks.iter().map(|n| format!("@{n}")).collect::<Vec<_>>().join(",")); }
74        if !self.post_id.trim().is_empty() { v.push(format!("id {}", self.post_id.trim())); }
75        if !self.from.trim().is_empty() || !self.to.trim().is_empty() { v.push(format!("{}..{}", self.from.trim(), self.to.trim())); }
76        match self.kind { KindFilter::Any => {}, KindFilter::Posts => v.push("posts".into()), KindFilter::Replies => v.push("replies".into()) }
77        if v.is_empty() { "everything".into() } else { v.join(" · ") }
78    }
79}

81/// Read access to the scraped data.
82pub trait Store {
83    /// Nicks with their post counts, most posts first.
84    fn nicks(&self) -> Result<Vec<NickEntry>>;
85    /// Posts and replies by any of `nicks`, newest first.
86    fn posts_for(&self, nicks: &[String]) -> Result<Vec<Post>>;
87    /// Posts passing `f`, newest first.
88    fn search(&self, f: &SearchFilter) -> Result<Vec<Post>>;
89    /// One post by id.
90    fn post(&self, id: &str) -> Result<Option<Post>>;
91    /// Direct replies to `id` (its `in_reply_to_id`), oldest first.
92    fn replies(&self, id: &str) -> Result<Vec<Post>>;
93    /// Attachments stored for `id` (one per file).
94    fn attachments(&self, id: &str) -> Result<Vec<Attachment>>;
95}

97const COLS: &str = "post_id,parent_post_id,nick,title,content,posted_at,in_reply_to_id,body_state";
98
99fn row(r: &rusqlite::Row) -> rusqlite::Result<Post> {
100    Ok(Post { post_id: r.get(0)?, parent_post_id: r.get(1)?, nick: r.get(2)?, title: r.get(3)?,
101        content: r.get(4)?, posted_at: r.get(5)?, in_reply_to_id: r.get(6)?, media: Vec::new(),
102        truncated: r.get::<_, String>(7)? == "truncated" })
103}

105/// The real database, opened read-only. A missing database reads as empty.
106pub struct SqliteStore {
107    path: PathBuf,
108}
109
110impl SqliteStore {
111    /// Store for the database at `path` (opened per query, read-only).
112    pub fn new(path: PathBuf) -> Self { SqliteStore { path } }
113    fn conn(&self) -> Result<Option<Connection>> {
114        if !self.path.exists() { return Ok(None); }
115        let c = Connection::open_with_flags(&self.path,
116            rusqlite::OpenFlags::SQLITE_OPEN_READ_ONLY | rusqlite::OpenFlags::SQLITE_OPEN_NO_MUTEX)
117            .with_context(|| format!("open {} read-only", self.path.display()))?;
118        if !crate::maintenance::table_exists(&c, "posts")? { return Ok(None); }
119        Ok(Some(c))
120    }
121    fn has_col(c: &Connection, col: &str) -> bool {
122        c.prepare(&format!("SELECT {col} FROM posts LIMIT 0")).is_ok()
123    }
124    fn cols(c: &Connection) -> String {
125        let mut s = COLS.to_string();
126        if !Self::has_col(c, "in_reply_to_id") { s = s.replace("in_reply_to_id", "parent_post_id"); }
127        if !Self::has_col(c, "body_state") { s = s.replace("body_state", "'unknown'"); }
128        s
129    }
130    fn query(&self, where_: &str, order: &str, p: &[&dyn rusqlite::ToSql]) -> Result<Vec<Post>> {
131        let Some(c) = self.conn()? else { return Ok(Vec::new()) };
132        let mut st = c.prepare(&format!("SELECT {} FROM posts WHERE {where_} ORDER BY {order}", Self::cols(&c)))?;
133        let v = st.query_map(p, row)?.collect::<rusqlite::Result<Vec<_>>>()?;
134        Ok(v)
135    }
136}

138impl Store for SqliteStore {
139    fn nicks(&self) -> Result<Vec<NickEntry>> {
140        let Some(c) = self.conn()? else { return Ok(Vec::new()) };
141        let mut v: Vec<NickEntry> = list_nicks_conn(&c)?.into_iter()
142            .map(|i| NickEntry { nick: i.nick, display_name: i.display_name, post_count: i.post_count }).collect();
143        v.sort_by(|a, b| b.post_count.cmp(&a.post_count).then_with(|| a.nick.to_lowercase().cmp(&b.nick.to_lowercase())));
144        Ok(v)
145    }
146    fn posts_for(&self, nicks: &[String]) -> Result<Vec<Post>> {
147        if nicks.is_empty() { return Ok(Vec::new()); }
148        let want: Vec<String> = nicks.iter().map(|n| n.to_lowercase()).collect();
149        let j = serde_json::to_string(&want)?;
150        self.query("lower(nick) IN (SELECT value FROM json_each(?1))", "posted_at DESC", &[&j])
151    }
152    fn search(&self, f: &SearchFilter) -> Result<Vec<Post>> {
153        let kw = f.keyword.trim();
154        let like = format!("%{}%", kw.replace('\\', "\\\\").replace('%', "\\%").replace('_', "\\_"));
155        let pre = if kw.is_empty() { "1".to_string() } else { "(title LIKE ?1 ESCAPE '\\' OR content LIKE ?1 ESCAPE '\\')".into() };
156        let rows = if kw.is_empty() { self.query("1", "posted_at DESC", &[])? } else { self.query(&pre, "posted_at DESC", &[&like])? };
157        Ok(rows.into_iter().filter(|p| f.matches(p)).collect())
158    }
159    fn post(&self, id: &str) -> Result<Option<Post>> {
160        Ok(self.query("post_id = ?1", "id", &[&id])?.into_iter().next())
161    }
162    fn replies(&self, id: &str) -> Result<Vec<Post>> {
163        let Some(c) = self.conn()? else { return Ok(Vec::new()) };
164        let col = if Self::has_col(&c, "in_reply_to_id") { "in_reply_to_id" } else { "parent_post_id" };
165        self.query(&format!("{col} = ?1 AND post_id <> ?1"), "posted_at", &[&id])
166    }
167    fn attachments(&self, id: &str) -> Result<Vec<Attachment>> {
168        let Some(c) = self.conn()? else { return Ok(Vec::new()) };
169        if !crate::maintenance::table_exists(&c, "attachments")? { return Ok(Vec::new()); }
170        let mut st = c.prepare("SELECT post_id,url,local_path,mime_type FROM attachments WHERE post_id = ?1 ORDER BY id DESC")?;
171        let mut seen = HashSet::new();
172        let mut out = Vec::new();
173        for a in st.query_map(params![id], |r| Ok(Attachment {
174            post_id: r.get(0)?, url: r.get(1)?, kind: "file".into(),
175            local_path: PathBuf::from(r.get::<_, String>(2)?), mime_type: r.get(3)?,
176        }))? {
177            let a = a?;
178            if seen.insert(crate::media::canonical_url(&a.url).to_string()) { out.push(a); }
179        }
180        out.reverse();
181        Ok(out)
182    }
183}

185/// In-memory store (tests, headless renders).
186#[derive(Default, Clone)]
187pub struct MemStore {
188    /// Every post.
189    pub posts: Vec<Post>,
190    /// Every attachment.
191    pub atts: Vec<Attachment>,
192}
193
194impl Store for MemStore {
195    fn nicks(&self) -> Result<Vec<NickEntry>> {
196        let mut m: Vec<NickEntry> = Vec::new();
197        for p in &self.posts {
198            match m.iter_mut().find(|e| e.nick.eq_ignore_ascii_case(&p.nick)) {
199                Some(e) => e.post_count += 1,
200                None => m.push(NickEntry { nick: p.nick.clone(), display_name: None, post_count: 1 }),
201            }
202        }
203        m.sort_by(|a, b| b.post_count.cmp(&a.post_count).then_with(|| a.nick.cmp(&b.nick)));
204        Ok(m)
205    }
206    fn posts_for(&self, nicks: &[String]) -> Result<Vec<Post>> {
207        let mut v: Vec<Post> = self.posts.iter().filter(|p| nicks.iter().any(|n| n.eq_ignore_ascii_case(&p.nick))).cloned().collect();
208        v.sort_by(|a, b| b.posted_at.cmp(&a.posted_at));
209        Ok(v)
210    }
211    fn search(&self, f: &SearchFilter) -> Result<Vec<Post>> {
212        let mut v: Vec<Post> = self.posts.iter().filter(|p| f.matches(p)).cloned().collect();
213        v.sort_by(|a, b| b.posted_at.cmp(&a.posted_at));
214        Ok(v)
215    }
216    fn post(&self, id: &str) -> Result<Option<Post>> { Ok(self.posts.iter().find(|p| p.post_id == id).cloned()) }
217    fn replies(&self, id: &str) -> Result<Vec<Post>> {
218        let mut v: Vec<Post> = self.posts.iter().filter(|p| p.in_reply_to_id == id && p.post_id != id).cloned().collect();
219        v.sort_by(|a, b| a.posted_at.cmp(&b.posted_at));
220        Ok(v)
221    }
222    fn attachments(&self, id: &str) -> Result<Vec<Attachment>> {
223        Ok(self.atts.iter().filter(|a| a.post_id == id).cloned().collect())
224    }
225}

227/// Post ids linked from a body: `/rs/post/{root}` and `/rs/post/{root}/{reply}` URLs
228/// and `#post-{id}` anchors, in order of appearance, without duplicates.
229pub fn linked_ids(text: &str) -> Vec<String> {
230    use std::sync::LazyLock;
231    static RE: LazyLock<regex::Regex> = LazyLock::new(|| regex::Regex::new(
232        r"(?:/rs/post/([A-Za-z0-9_-]{8,})(?:/([A-Za-z0-9_-]{8,}))?)|(?:#post-([A-Za-z0-9_-]{8,}))").expect("link re"));
233    let mut seen = HashSet::new();
234    let mut out = Vec::new();
235    for c in RE.captures_iter(text) {
236        // The most specific id: the reply in /rs/post/{root}/{reply}, else the root / anchor.
237        let id = c.get(2).or(c.get(1)).or(c.get(3)).map(|m| m.as_str().to_string());
238        if let Some(id) = id && seen.insert(id.clone()) { out.push(id); }
239    }
240    out
