1//! Colours: ACP red (`#dc2626`) accent by default, overridable from the
2//! `[theme]` table of `~/.acp/preferences.toml`.
3
4use crate::prelude::*;
5use ratatui::style::{Color, Modifier, Style};
6
7/// Resolved colours used by every widget.
8#[derive(Clone, Copy, Debug, PartialEq)]
9pub struct Theme {
10    /// Accent: reply links, selection, focused borders, Start Scanning.
11    pub accent: Color,
12    /// Secondary text.
13    pub muted: Color,
14    /// Unfocused borders.
15    pub border: Color,
16    /// Text drawn on an accent background.
17    pub on_accent: Color,
18}
19
20fn rgb((r, g, b): (u8, u8, u8)) -> Color { Color::Rgb(r, g, b) }
21
22/// A colour as RGB bytes (named colours map to white, black, or grey).
23pub fn rgb_bytes(c: Color) -> [u8; 3] {
24    match c {
25        Color::Rgb(r, g, b) => [r, g, b],
26        Color::Black => [0, 0, 0],
27        Color::White => [255, 255, 255],
28        _ => [200, 200, 200],
29    }
30}
31
32/// `#rrggbb` (or `rrggbb`) to a colour.
33pub fn parse_hex(s: &str) -> Option<Color> {
34    let h = s.trim().trim_start_matches('#');
35    if h.len() != 6 || !h.chars().all(|c| c.is_ascii_hexdigit()) { return None; }
36    let p = |i: usize| u8::from_str_radix(&h[i..i + 2], 16).ok();
37    Some(Color::Rgb(p(0)?, p(2)?, p(4)?))
38}
39
40impl Default for Theme {
41    fn default() -> Self {
42        Theme { accent: rgb(ACP_RED), muted: rgb(TUI_MUTED), border: rgb(TUI_BORDER), on_accent: rgb(TUI_ON_ACCENT) }
43    }
44}
45
46impl Theme {
47    /// Defaults with any valid colour from `prefs` applied (invalid values are ignored).
48    pub fn from_prefs(prefs: Option<&ThemePrefs>) -> Self {
49        let mut t = Theme::default();
50        let Some(p) = prefs else { return t };
51        let pick = |v: &Option<String>, d: Color| v.as_deref().and_then(parse_hex).unwrap_or(d);
52        t.accent = pick(&p.accent, t.accent);
53        t.muted = pick(&p.muted, t.muted);
54        t.border = pick(&p.border, t.border);
55        t.on_accent = pick(&p.on_accent, t.on_accent);
56        t
57    }
58    /// Accent-coloured text.
59    pub fn accent(&self) -> Style { Style::default().fg(self.accent) }
60    /// Bold accent text.
61    pub fn accent_bold(&self) -> Style { self.accent().add_modifier(Modifier::BOLD) }
62    /// Muted text.
63    pub fn muted(&self) -> Style { Style::default().fg(self.muted) }
64    /// Highlighted (selected) row.
65    pub fn selected(&self) -> Style { Style::default().bg(self.accent).fg(self.on_accent).add_modifier(Modifier::BOLD) }
66    /// Border style for a pane, accent when focused.
67    pub fn border(&self, focused: bool) -> Style {
68        if focused { Style::default().fg(self.accent) } else { Style::default().fg(self.border) }
69    }
70}
