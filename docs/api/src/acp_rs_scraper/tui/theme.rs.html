1//! The single source of TUI styling. Colours: ACP red (`#dc2626`) accent by
2//! default, overridable from the `[theme]` table of `~/.acp/preferences.toml`.
3//!
4//! Every text style and every box (border kind, border colour, padding) is a
5//! [`ratatui_lipgloss`] [`Gloss`] style built here; widgets only pick one.
6//! `&Gloss` / `Gloss` convert into ratatui styles (`Span::styled(s, t.muted())`),
7//! and [`frame`] turns a box style into a ratatui `Block` so callers can add a
8//! border title (Lip Gloss has no border titles).
9
10use crate::prelude::*;
11use ratatui::style::Color;
12use ratatui::widgets::Block;
13pub use ratatui_lipgloss::{Border, Style as Gloss};
14
15/// Resolved colours used by every widget.
16#[derive(Clone, Copy, Debug, PartialEq)]
17pub struct Theme {
18    /// Accent: reply links, selection, focused borders, Start Scanning.
19    pub accent: Color,
20    /// Secondary text.
21    pub muted: Color,
22    /// Unfocused borders.
23    pub border: Color,
24    /// Text drawn on an accent background.
25    pub on_accent: Color,
26}
27
28fn rgb((r, g, b): (u8, u8, u8)) -> Color { Color::Rgb(r, g, b) }
29
30/// A colour as RGB bytes (named colours map to white, black, or grey).
31pub fn rgb_bytes(c: Color) -> [u8; 3] {
32    match c {
33        Color::Rgb(r, g, b) => [r, g, b],
34        Color::Black => [0, 0, 0],
35        Color::White => [255, 255, 255],
36        _ => [200, 200, 200],
37    }
38}
39
40/// `#rrggbb` (or `rrggbb`) to a colour.
41pub fn parse_hex(s: &str) -> Option<Color> {
42    let h = s.trim().trim_start_matches('#');
43    if h.len() != 6 || !h.chars().all(|c| c.is_ascii_hexdigit()) { return None; }
44    let p = |i: usize| u8::from_str_radix(&h[i..i + 2], 16).ok();
45    Some(Color::Rgb(p(0)?, p(2)?, p(4)?))
46}
47
48impl Default for Theme {
49    fn default() -> Self {
50        Theme { accent: rgb(ACP_RED), muted: rgb(TUI_MUTED), border: rgb(TUI_BORDER), on_accent: rgb(TUI_ON_ACCENT) }
51    }
52}
53
54impl Theme {
55    /// Defaults with any valid colour from `prefs` applied (invalid values are ignored).
56    pub fn from_prefs(prefs: Option<&ThemePrefs>) -> Self {
57        let mut t = Theme::default();
58        let Some(p) = prefs else { return t };
59        let pick = |v: &Option<String>, d: Color| v.as_deref().and_then(parse_hex).unwrap_or(d);
60        t.accent = pick(&p.accent, t.accent);
61        t.muted = pick(&p.muted, t.muted);
62        t.border = pick(&p.border, t.border);
63        t.on_accent = pick(&p.on_accent, t.on_accent);
64        t
65    }

67    // ----- text ------------------------------------------------------------
68    /// Unstyled text (the terminal's own colours).
69    pub fn plain(&self) -> Gloss { Gloss::new() }
70    /// Bold text (headings, titles, nicks).
71    pub fn bold(&self) -> Gloss { Gloss::new().bold(true) }
72    /// Accent-coloured text.
73    pub fn accent(&self) -> Gloss { Gloss::new().foreground(self.accent) }
74    /// Bold accent text.
75    pub fn accent_bold(&self) -> Gloss { self.accent().bold(true) }
76    /// A link: underlined accent.
77    pub fn link(&self) -> Gloss { self.accent().underline(true) }
78    /// Muted text.
79    pub fn muted(&self) -> Gloss { Gloss::new().foreground(self.muted) }
80    /// Something that is gone (a missing file): muted, struck through.
81    pub fn gone(&self) -> Gloss { self.muted().strikethrough(true) }
82    /// Inline code: bold muted.
83    pub fn code(&self) -> Gloss { self.muted().bold(true) }
84    /// Highlighted (selected) row.
85    pub fn selected(&self) -> Gloss { Gloss::new().background(self.accent).foreground(self.on_accent).bold(true) }
86    /// A text field's value.
87    pub fn field(&self) -> Gloss { Gloss::new().underline(true) }
88    /// A form label, accent when its field has focus.
89    pub fn label(&self, focused: bool) -> Gloss { if focused { self.accent_bold() } else { self.plain() } }
90    /// The header bar and its text: on-accent over accent.
91    pub fn bar(&self) -> Gloss { Gloss::new().background(self.accent).foreground(self.on_accent) }
92    /// A filled button (Start Scanning).
93    pub fn button(&self) -> Gloss { self.bar().bold(true) }
94    /// The "Post" badge of a card.
95    pub fn post_tag(&self) -> Gloss { Gloss::new().foreground(self.on_accent).background(self.border).bold(true) }
96    /// The "Reply" badge of a card.
97    pub fn reply_tag(&self) -> Gloss { Gloss::new().foreground(self.on_accent).background(self.accent).bold(true) }
98    /// The background under an open modal.
99    pub fn dimmed(&self) -> Gloss { Gloss::new().faint(true) }
100
101    // ----- boxes -----------------------------------------------------------
102    /// Border colour of a pane: accent when focused.
103    pub fn border(&self, focused: bool) -> Gloss { Gloss::new().foreground(if focused { self.accent } else { self.border }) }
104    /// A pane (Nicks, Body, carousel sides): plain border, accent when focused.
105    pub fn pane(&self, focused: bool) -> Gloss {
106        Gloss::new().border(Border::normal()).border_foreground(if focused { self.accent } else { self.border })
107    }
108    /// A post card with one column of padding: thick when selected (accent
109    /// when the Body has focus), plain and muted otherwise.
110    pub fn card(&self, selected: bool, focused: bool) -> Gloss {
111        let g = Gloss::new().border(if selected { Border::thick() } else { Border::normal() }).padding((0, 1));
112        match (selected, focused) {
113            (true, true) => g.border_foreground(self.accent),
114            (true, false) => g,
115            _ => g.border_foreground(self.border),
116        }
117    }
118    /// A detail / content pane: double border, accent when focused.
119    pub fn detail(&self, focused: bool) -> Gloss {
120        Gloss::new().border(Border::double()).border_foreground(if focused { self.accent } else { self.border })
121    }
122    /// Dialogs, dropdown menus and nested details: double accent border.
123    pub fn dialog(&self) -> Gloss { Gloss::new().border(Border::double()).border_foreground(self.accent) }
124}

126/// The ratatui `Block` for a box style (borders, border colour, padding), to
127/// render or to give a title.
128pub fn frame(g: &Gloss) -> Block<'static> { g.to_block() }
