1//! The animation in the empty Body: text frames played on a loop.
2//!
3//! Frames come from `~/.acp/animations/empty-body.acpanim(.gz)` (made with
4//! `tools/acp-anim`), `[ui] empty_animation_path`, or the built-in clip
5//! (`assets/empty-body.acpanim.gz`, see `assets/empty-body.NOTICE.md`); if that
6//! can't be read, or `[ui] empty_animation = "star"`, a procedural red star is
7//! drawn instead. `[ui] empty_animation = false` or `ACP_EMPTY_ANIMATION=off`
8//! turns it off.
9//!
10//! Format (gzip or plain UTF-8): `ACPANIM 1`, `fps N`, `size COLS ROWS`, then
11//! each frame as a `%%` line followed by up to ROWS lines.
12

17/// The built-in clip.
18const BUILTIN: &[u8] = include_bytes!("../../assets/empty-body.acpanim.gz");

20/// Decoded frames.
21#[derive(Clone, Debug, PartialEq)]
22pub struct Frames {
23    /// Frames per second (1–60).
24    pub fps: u16,
25    /// Columns.
26    pub width: u16,
27    /// Rows.
28    pub height: u16,
29    /// Each frame: `height` rows of chars, padded to `width`.
30    pub frames: Vec<Vec<Vec<char>>>,
31    /// Where they came from (shown nowhere; for logs and tests).
32    pub source: String,
33}
34
35/// Which animation to play.
36#[derive(Clone, Copy, Debug, PartialEq, Eq)]
37pub enum Choice {
38    /// None.
39    Off,
40    /// The local file, else the built-in clip, else the star.
41    Clip,
42    /// The procedural star only.
43    Star,
44}
45
46/// The setting: `ACP_EMPTY_ANIMATION` wins over `[ui] empty_animation`.
47pub fn choice(env: Option<&str>, pref: Option<&toml::Value>) -> Choice {
48    let word = |s: &str| match s.trim().to_ascii_lowercase().as_str() {
49        "off" | "false" | "no" | "0" | "none" => Some(Choice::Off),
50        "star" | "minimal" | "procedural" => Some(Choice::Star),
51        "on" | "true" | "yes" | "1" | "clip" | "" => Some(Choice::Clip),
52        _ => None,
53    };
54    if let Some(c) = env.and_then(word) { return c; }
55    match pref {
56        Some(toml::Value::Boolean(false)) => Choice::Off,
57        Some(toml::Value::String(s)) => word(s).unwrap_or(Choice::Clip),
58        _ => Choice::Clip,
59    }
60}

62/// Parse a frames file (gzip or plain).
63pub fn parse(bytes: &[u8], source: &str) -> Result<Frames> {
64    let text = if bytes.starts_with(&[0x1f, 0x8b]) {
65        let mut s = String::new();
66        flate2::read::GzDecoder::new(bytes).take(64 * 1024 * 1024).read_to_string(&mut s).context("gunzip")?;
67        s
68    } else {
69        String::from_utf8(bytes.to_vec()).context("not UTF-8")?
70    };
71    let mut lines = text.lines();
72    if lines.next().map(str::trim) != Some("ACPANIM 1") { bail!("not an ACPANIM 1 file"); }
73    let (mut fps, mut size) = (12u16, None);
74    let mut frames: Vec<Vec<Vec<char>>> = Vec::new();
75    for l in lines {
76        if l == "%%" { frames.push(Vec::new()); continue; }
77        match frames.last_mut() {
78            Some(f) => f.push(l.chars().collect()),
79            None => {
80                let mut it = l.split_whitespace();
81                match it.next() {
82                    Some("fps") => fps = it.next().and_then(|v| v.parse().ok()).filter(|v| (1..=60).contains(v)).context("bad fps")?,
83                    Some("size") => {
84                        let w: u16 = it.next().and_then(|v| v.parse().ok()).context("bad size")?;
85                        let h: u16 = it.next().and_then(|v| v.parse().ok()).context("bad size")?;
86                        if !(1..=1000).contains(&w) || !(1..=1000).contains(&h) { bail!("size {w}x{h} out of range"); }
87                        size = Some((w, h));
88                    }
89                    _ => {}
90                }
91            }
92        }
93    }
94    let (w, h) = size.context("missing size line")?;
95    if frames.is_empty() { bail!("no frames"); }
96    for f in &mut frames {
97        f.truncate(h as usize);
98        f.resize(h as usize, Vec::new());
99        for row in f.iter_mut() { row.truncate(w as usize); row.resize(w as usize, ' '); }
100    }
101    Ok(Frames { fps, width: w, height: h, frames, source: source.to_string() })
102}

104/// The built-in clip, or the star if it can't be decoded.
105pub fn builtin() -> Frames { parse(BUILTIN, "built-in").unwrap_or_else(|_| star()) }

107/// What to play, given the setting, an optional path override and `$HOME`.
108pub fn load(c: Choice, path: Option<&str>, home: Option<&Path>) -> Option<Frames> {
109    match c {
110        Choice::Off => return None,
111        Choice::Star => return Some(star()),
112        Choice::Clip => {}
113    }
114    let mut candidates: Vec<PathBuf> = Vec::new();
115    if let Some(p) = path.filter(|p| !p.trim().is_empty()) {
116        let p = p.trim();
117        candidates.push(match (p.strip_prefix("~/"), home) { (Some(rest), Some(h)) => h.join(rest), _ => PathBuf::from(p) });
118    } else if let Some(h) = home {
119        let base = h.join(ANIMATION_REL_PATH);
120        candidates.push(PathBuf::from(format!("{}.gz", base.display())));
121        candidates.push(base);
122    }
123    for p in candidates {
124        if let Ok(b) = std::fs::read(&p) {
125            match parse(&b, &p.display().to_string()) {
126                Ok(f) => return Some(f),
127                Err(e) => info!("warning: empty-Body animation {}: {e:#}; using the built-in one", p.display()),
128            }
129        }
130    }
131    Some(builtin())
132}

134/// From the environment and `[ui]` prefs.
135pub fn from_prefs(ui: Option<&UiPrefs>) -> (Option<Frames>, bool) {
136    let env = std::env::var("ACP_EMPTY_ANIMATION").ok();
137    let c = choice(env.as_deref(), ui.and_then(|u| u.empty_animation.as_ref()));
138    let accent = ui.and_then(|u| u.empty_animation_color.as_deref()).is_some_and(|s| s.eq_ignore_ascii_case("accent"));
139    (load(c, ui.and_then(|u| u.empty_animation_path.as_deref()), home().ok().as_deref()), accent)
140}

142impl Frames {
143    /// Duration of one frame.
144    pub fn frame_time(&self) -> Duration { Duration::from_micros(1_000_000 / self.fps.max(1) as u64) }
145    /// The frame shown `elapsed` after the start (loops).
146    pub fn index(&self, elapsed: Duration) -> usize {
147        ((elapsed.as_micros() * self.fps as u128 / 1_000_000) % self.frames.len().max(1) as u128) as usize
148    }
149    /// Time from `elapsed` until the next frame starts.
150    pub fn until_next(&self, elapsed: Duration) -> Duration {
151        let ft = self.frame_time().as_micros().max(1);
152        let into = elapsed.as_micros() % ft;
153        Duration::from_micros((ft - into) as u64)
154    }
155    /// Frame `i` fitted into `w`×`h`: shrunk (never enlarged) keeping its aspect,
156    /// by nearest-neighbour sampling. `None` when the area is too small (paused).
157    pub fn fit(&self, i: usize, w: u16, h: u16) -> Option<Vec<String>> {
158        if w < EMPTY_ANIM_MIN.0 || h < EMPTY_ANIM_MIN.1 || self.frames.is_empty() { return None; }
159        let f = &self.frames[i % self.frames.len()];
160        let s = (w as f64 / self.width as f64).min(h as f64 / self.height as f64).min(1.0);
161        let ow = ((self.width as f64 * s).round() as usize).clamp(1, w as usize);
162        let oh = ((self.height as f64 * s).round() as usize).clamp(1, h as usize);
163        Some((0..oh).map(|y| {
164            let sy = (y * self.height as usize / oh).min(self.height as usize - 1);
165            (0..ow).map(|x| f[sy][(x * self.width as usize / ow).min(self.width as usize - 1)]).collect::<String>().trim_end().to_string()
166        }).collect())
167    }
168}

170/// A five-pointed red star turning slowly about its vertical axis, over an
171/// "ACP" wordmark: 48 frames at 12 fps, 72×30, drawn from geometry alone.
172pub fn star() -> Frames {
173    const W: usize = 72;
174    const H: usize = 30;
175    const N: usize = 48;
176    const RAMP: &[char] = &['.', ':', '-', '=', '+', '*', '#', '%', '@'];
177    let word = ["  ███    ████  █████ ", " █   █  █      █    █", " █████  █      █████ ", " █   █  █      █     ", " █   █   ████  █     "];
178    let (cx, cy, r_out) = (W as f64 / 2.0, 11.0, 10.5);
179    let pts: Vec<(f64, f64)> = (0..10).map(|k| {
180        let a = -std::f64::consts::FRAC_PI_2 + k as f64 * std::f64::consts::PI / 5.0;
181        let r = if k % 2 == 0 { 1.0 } else { 0.382 };
182        (r * a.cos(), r * a.sin())
183    }).collect();
184    let inside = |x: f64, y: f64| {
185        let mut c = false;
186        let mut j = pts.len() - 1;
187        for i in 0..pts.len() {
188            let (xi, yi) = pts[i];
189            let (xj, yj) = pts[j];
190            if (yi > y) != (yj > y) && x < (xj - xi) * (y - yi) / (yj - yi) + xi { c = !c; }
191            j = i;
192        }
193        c
194    };
195    let frames = (0..N).map(|n| {
196        let th = n as f64 / N as f64 * std::f64::consts::TAU;
197        let (c, sn) = (th.cos(), th.sin());
198        let mut g = vec![vec![' '; W]; H];
199        for (y, row) in g.iter_mut().enumerate().take(21) {
200            for (x, cell) in row.iter_mut().enumerate() {
201                // Cells are about twice as tall as wide: x spans twice the units.
202                let u = (x as f64 + 0.5 - cx) / (2.0 * r_out);
203                let v = (y as f64 + 0.5 - cy) / r_out;
204                if c.abs() < 0.04 { if u.abs() < 0.03 && v.abs() < 1.0 { *cell = '|'; } continue; }
205                let su = u / c;
206                if !inside(su, v) { continue; }
207                // Light from the upper left; brighter towards the centre and facing us.
208                let shade = (0.55 + 0.45 * c.abs()) * (1.0 - 0.35 * (su * su + v * v).sqrt()) + 0.1 * (-su * sn.signum() - v) * 0.5;
209                let k = ((shade.clamp(0.0, 0.999)) * RAMP.len() as f64) as usize;
210                *cell = RAMP[k.min(RAMP.len() - 1)];
211            }
212        }
213        for (i, l) in word.iter().enumerate() {
214            let chars: Vec<char> = l.chars().collect();
215            let x0 = (W - chars.len()) / 2;
216            for (k, ch) in chars.into_iter().enumerate() { g[23 + i][x0 + k] = ch; }
217        }
218        g
219    }).collect();
220    Frames { fps: 12, width: W as u16, height: H as u16, frames, source: "star".into() }
221}
