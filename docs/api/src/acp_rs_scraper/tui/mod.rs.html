1//! Terminal UI (2.0): the default when no flags are given (or with `--tui`).
2//!
3//! Browse every nick and post in the local database offline, open post details
4//! that cross-load linked posts, search in place, and drive scrape / export /
5//! import / maintenance from a menu bar built from the CLI flags. Anything that
6//! writes runs the headless CLI as a child process ([`jobs`]), so it can be
7//! cancelled, and keeps running in the background when you quit.
8//!
9//! Layout: `mod.rs` (event loop), [`state`] (pure state + transitions),
10//! [`input`] (keys/mouse), [`layout`] (geometry), [`widgets`] (drawing),
11//! [`modals`], [`commands`] (menu ↔ flags), [`data`] (read-only DB), [`jobs`],
12//! [`theme`] (styles), [`theming`] (theme files in the app), [`editor`] (Shift-F5).

34/// Messages from background tasks to the event loop.
35enum Msg {
36    SessionChecked(std::result::Result<bool, String>, jobs::JobKind, Vec<String>),
37    LoginNeedCode(u64),
38    LoginDone(u64, std::result::Result<(), String>),
39    AutoSyncChecked(std::result::Result<bool, String>, Vec<String>),
40    ThemeDownloaded(String, bool, std::result::Result<crate::themes::download::Fetched, String>),
41}

45fn restore_terminal() {
46    use crossterm::{event::DisableMouseCapture, execute, terminal::{LeaveAlternateScreen, disable_raw_mode}};
47    let _ = disable_raw_mode();
48    let _ = execute!(std::io::stdout(), LeaveAlternateScreen, DisableMouseCapture, crossterm::cursor::Show);
49}
50
51impl Drop for Terminal {
52    fn drop(&mut self) { restore_terminal(); }
53}

55/// Open the terminal UI. Needs a terminal on stdin and stdout.
56pub async fn run(args: Args) -> Result<()> {
57    use std::io::IsTerminal;
58    if !std::io::stdin().is_terminal() || !std::io::stdout().is_terminal() {
59        bail!("the terminal UI needs a terminal; for headless use pass flags, e.g. --scrape, --nick NICK, --search TEXT, --export FILE (see --help)");
60    }
61    let db = home()?.join(DB_REL_PATH);
62    let prefs = load_prefs();
63    let mut app = App::new(Box::new(data::SqliteStore::new(db)), &args, theme::Theme::reset(),
64        session_path().map(|p| p.exists()).unwrap_or(false));
65    setup_themes(&mut app, &args, &prefs)?;
66    app.email = prefs.email.filter(|e| !e.is_empty());
67    if let Some(s) = &prefs.sync { app.menu.menu.apply_sync_prefs(s); }
68    (app.empty_anim, app.empty_anim_accent) = empty_anim::from_prefs(prefs.ui.as_ref());
69
70    let (log_tx, log_rx) = std::sync::mpsc::channel::<String>();
71    set_log_sink(Some(log_tx));
72    let r = event_loop(&mut app, log_rx).await;
73    set_log_sink(None);
74    let detached = r?;
75    if let Some((kind, pid, log)) = detached {
76        eprintln!("The {kind} keeps running in the background (pid {pid}). Follow it with:\n  tail -f {}", log.display());
77    }
78    Ok(())
79}

81/// Theme files at startup: convert an old `[theme]` table once, select `--theme`,
82/// `[ui] theme` or the default, and on a first run (empty `~/.acp/themes`) start the
83/// background download of the default theme (unless it is turned off).
84fn setup_themes(app: &mut App, args: &Args, prefs: &Prefs) -> Result<()> {
85    let dir = crate::themes::themes_dir()?;
86    let first = !crate::themes::has_themes(&dir);
87    let (migrated, notes) = crate::themes::prepare(&dir, prefs);
88    for n in &notes { app.push_status(n.clone()); }
89    if let Some(m) = &migrated {
90        if let Err(e) = save_ui_theme(m) { app.push_status(format!("warning: could not select {m} in preferences.toml: {e:#}")); }
91        if let Some(n) = notes.iter().find(|n| n.starts_with("Your [theme]")) {
92            app.modal = Some(modals::Modal::Message(modals::MessageModal::info("Theme converted", n)));
93        }
94    }
95    app.themes_dir = Some(dir.clone());
96    app.theme_mode = crate::themes::Mode::parse(prefs.ui.as_ref().and_then(|u| u.theme_mode.as_deref()));
97    app.dark_bg = crate::themes::colorfgbg_dark(std::env::var("COLORFGBG").ok().as_deref());
98    let name = args.theme.clone().or(migrated).or_else(|| prefs.ui.as_ref().and_then(|u| u.theme.clone())).unwrap_or_else(|| DEFAULT_THEME.into());
99    let download = first && !dir.join(format!("{DEFAULT_THEME}.toml")).exists();
100    let enabled = crate::themes::download::auto_enabled_now(prefs);
101    let before = app.modal.take();
102    let ok = app.use_theme(&name, false);
103    if download && enabled {
104        // Not an error yet: the default is on its way (drawn with the terminal's colours until then).
105        if !ok { app.modal = before; }
106        app.start_theme_download(DEFAULT_THEME, false);
107    } else {
108        if ok { app.modal = before; }
109        if download { app.push_status(format!("warning: {}", crate::themes::download::off_notice(&dir))); }
110    }
111    Ok(())
112}

114async fn event_loop(app: &mut App, log_rx: std::sync::mpsc::Receiver<String>) -> Result<Option<(&'static str, u32, PathBuf)>> {
115    use crossterm::{event::EnableMouseCapture, execute, terminal::{EnterAlternateScreen, enable_raw_mode}};
116    let jar = Arc::new(Jar::default());
117    if let Err(e) = load_session(&jar) { app.push_status(format!("warning: ignoring saved session: {e:#}")); }
118    let http = client(jar.clone())?;
119
120    enable_raw_mode()?;
121    execute!(std::io::stdout(), EnterAlternateScreen, EnableMouseCapture)?;
122    let prev = std::panic::take_hook();
123    std::panic::set_hook(Box::new(move |info| { restore_terminal(); prev(info); }));
124    // `theme_mode = "system"` without COLORFGBG: ask the terminal for its background.
125    if app.theme_mode == crate::themes::Mode::System && app.dark_bg.is_none() {
126        app.dark_bg = images::query_background();
127        if app.dark_bg == Some(false) { app.refresh_theme(); }
128    }
129    // Ask the terminal for its graphics protocol before the input thread starts reading.
130    app.images = images::Images::detect(theme::rgb_bytes(app.theme.on_accent), theme::rgb_bytes(app.theme.accent));
131    let osc8 = osc8_supported();
132    let mut term = Terminal(ratatui::Terminal::new(ratatui::backend::CrosstermBackend::new(std::io::stdout()))?);
133
134    // Input thread → channel.
135    let (ev_tx, mut ev_rx) = tokio::sync::mpsc::unbounded_channel();
136    let stop = Arc::new(std::sync::atomic::AtomicBool::new(false));
137    let stop2 = stop.clone();
138    std::thread::spawn(move || {
139        while !stop2.load(std::sync::atomic::Ordering::Relaxed) {
140            if crossterm::event::poll(Duration::from_millis(100)).unwrap_or(false) {
141                match crossterm::event::read() { Ok(e) => { if ev_tx.send(e).is_err() { break; } } Err(_) => break }
142            }
143        }
144    });
145    let (msg_tx, mut msg_rx) = tokio::sync::mpsc::unbounded_channel::<Msg>();
146    let mut job: Option<jobs::Job> = None;
147    // Small writes (bookmarks) that run next to the main job.
148    let mut side: Vec<jobs::Job> = Vec::new();
149    let mut code_tx: Option<tokio::sync::oneshot::Sender<String>> = None;
150    let mut login_task: Option<std::thread::JoinHandle<()>> = None;
151    let mut detached = None;
152    // Generation of the current login attempt; replies from abandoned ones are ignored.
153    let mut login_gen: u64 = 0;
154
155    'outer: loop {
156        let now = Instant::now();
157        app.tick(now);
158        app.sync_tick(now);
159        if let Some(im) = app.images.as_mut() { im.poll(); }
160        term.0.draw(|f| widgets::draw(f, app, now))?;
161        if osc8 { emit_osc8(&app.osc_links, &app.theme); }
162        if let Some(im) = app.images.as_mut()
163            && im.flush(&mut std::io::stdout()) == images::Flush::Repaint {
164                // Sixel / iTerm2 pictures are burnt into the cells: repaint everything, then place them again.
165                term.0.clear()?;
166                term.0.draw(|f| widgets::draw(f, app, now))?;
167                if osc8 { emit_osc8(&app.osc_links, &app.theme); }
168                if let Some(im) = app.images.as_mut() { im.flush(&mut std::io::stdout()); }
169            }
170        let mut wait = Duration::from_millis(if app.anim.is_some() { TUI_FRAME_MS } else { TUI_TICK_MS });
171        // The empty-Body animation: wake exactly when its next frame is due (one draw per frame).
172        if app.empty_anim_frame.is_some() && let Some(fr) = &app.empty_anim {
173            wait = wait.min(fr.until_next(Instant::now().saturating_duration_since(app.empty_anim_start)));
174        }
175        tokio::select! {
176            Some(ev) = ev_rx.recv() => {
177                if matches!(ev, crossterm::event::Event::Resize(..)) && let Some(im) = app.images.as_mut() { im.invalidate(); }
178                input::on_event(app, ev);
179                while let Ok(ev) = ev_rx.try_recv() { input::on_event(app, ev); }
180            }
181            Some(m) = msg_rx.recv() => match m {
182                Msg::SessionChecked(r, k, argv) => app.session_checked(r, k, argv),
183                Msg::LoginNeedCode(g) if g == login_gen => app.login_need_code(),
184                Msg::LoginDone(g, r) if g == login_gen => { code_tx = None; login_task = None; app.login_done(r); }
185                Msg::AutoSyncChecked(r, argv) => app.auto_sync_checked(r, argv),
186                Msg::ThemeDownloaded(name, manual, r) => app.theme_downloaded(&name, manual, r),
187                Msg::LoginNeedCode(_) | Msg::LoginDone(..) => {}
188            },
189            _ = tokio::time::sleep(wait) => {}
190        }
191        while let Ok(l) = log_rx.try_recv() { app.push_status(l); }
192        if let Some(j) = &mut job {
193            let (lines, status) = j.poll();
194            for l in lines { app.push_status(l); }
195            if let Some(st) = status {
196                let tail = jobs::tail(&j.log, 12);
197                app.job_finished(st.success(), st.code(), &tail);
198                job = None;
199            }
200        }
201        let mut k = 0;
202        while k < side.len() {
203            let (_, status) = side[k].poll();
204            if let Some(st) = status {
205                let j = side.remove(k);
206                app.side_job_finished(st.success(), &jobs::tail(&j.log, 3), side.len());
207                if st.success() { let _ = std::fs::remove_file(&j.log); }
208            } else { k += 1; }
209        }
210        for eff in std::mem::take(&mut app.effects) {
211            match eff {
212                Effect::CheckSessionThen(kind, argv) => {
213                    let (http, tx) = (http.clone(), msg_tx.clone());
214                    app.push_status("checking the saved session…");
215                    tokio::spawn(async move {
216                        let r = session_valid(&http).await.map_err(|e| format!("{e:#}"));
217                        let _ = tx.send(Msg::SessionChecked(r, kind, argv));
218                    });
219                }
220                Effect::RunJob(kind, argv) => {
221                    if job.is_some() { app.error("Busy", "another job is still running"); continue; }
222                    match jobs::Job::spawn(kind, &argv) {
223                        Ok(j) => { app.job_started(JobView { kind, pid: j.pid(), log: j.log.clone() }, &j.cmd); job = Some(j); }
224                        Err(e) => app.error("Cannot start", &format!("{e:#}")),
225                    }
226                }
227                Effect::AutoSync(argv) => {
228                    let (http, tx) = (http.clone(), msg_tx.clone());
229                    tokio::spawn(async move {
230                        let r = session_valid(&http).await.map_err(|e| format!("{e:#}"));
231                        let _ = tx.send(Msg::AutoSyncChecked(r, argv));
232                    });
233                }
234                Effect::SaveSyncPrefs(p) => match save_sync_prefs(&p) {
235                    Ok(()) => app.push_status("sync defaults saved to ~/.acp/preferences.toml"),
236                    Err(e) => app.error("Cannot save", &format!("{e:#}")),
237                },
238                Effect::SaveUiTheme(name) => match save_ui_theme(&name) {
239                    Ok(()) => app.push_status(format!("theme {name} selected (saved to ~/.acp/preferences.toml)")),
240                    Err(e) => app.error("Cannot save the theme choice", &format!("{e:#}")),
241                },
242                Effect::ExportTheme(f) => match export_theme(&f) {
243                    Ok(p) => app.modal = Some(modals::Modal::Message(modals::MessageModal::info("Theme exported", &format!("Wrote {}", p.display())))),
244                    Err(e) => app.error("Cannot export the theme", &format!("{e:#}")),
245                },
246                Effect::DownloadTheme(name, manual) => {
247                    let Some(dir) = app.themes_dir.clone() else { continue };
248                    let tx = msg_tx.clone();
249                    tokio::spawn(async move {
250                        let r = crate::themes::download::download(&dir, &name).await.map_err(|e| format!("{e:#}"));
251                        let _ = tx.send(Msg::ThemeDownloaded(name, manual, r));
252                    });
253                }
254                Effect::RunSideJob(kind, argv) => match jobs::Job::spawn(kind, &argv) {
255                    Ok(j) => side.push(j),
256                    Err(e) => app.error("Cannot start", &format!("{e:#}")),
257                },
258                Effect::OpenFile(p) => if let Err(e) = open_file(&p) { app.error("Cannot open", &format!("{e:#}")); },
259                Effect::CancelJob => if let Some(mut j) = job.take() {
260                    if let Err(e) = j.cancel() { app.push_status(format!("error: cancel: {e:#}")); }
261                    app.job_cancelled();
262                },
263                Effect::LoginEmail(email) => {
264                    // The LiveView login future is not Send: run it on its own
265                    // single-threaded runtime. Dropping `code_tx` ends it.
266                    let (ctx, crx) = tokio::sync::oneshot::channel::<String>();
267                    code_tx = Some(ctx);
268                    login_gen += 1;
269                    let g = login_gen;
270                    let (http, jar, tx) = (http.clone(), jar.clone(), msg_tx.clone());
271                    login_task = Some(std::thread::spawn(move || {
272                        let tx2 = tx.clone();
273                        let r = match tokio::runtime::Builder::new_current_thread().enable_all().build() {
274                            Ok(rt) => rt.block_on(login_flow(&http, &jar, Some(email), move || async move {
275                                let _ = tx2.send(Msg::LoginNeedCode(g));
276                                crx.await.map_err(|_| anyhow!("login cancelled"))
277                            })),
278                            Err(e) => Err(e.into()),
279                        };
280                        let _ = tx.send(Msg::LoginDone(g, r.map_err(|e| format!("{e:#}"))));
281                    }));
282                }
283                Effect::LoginCode(c) => match code_tx.take() {
284                    Some(tx) => { let _ = tx.send(c); }
285                    None => app.login_done(Err("the login expired; send a new code".into())),
286                },
287                Effect::LoginCancel => { code_tx = None; login_task = None; login_gen += 1; }
288                Effect::Logout => match logout(&jar) {
289                    Ok(removed) => app.logged_out(removed),
290                    Err(e) => app.error("Logout failed", &format!("{e:#}")),
291                },
292                Effect::Quit { cancel_job } => {
293                    if let Some(mut j) = job.take() {
294                        if cancel_job { let _ = j.cancel(); } else { detached = Some((j.kind.name(), j.pid(), j.log.clone())); }
295                    }
296                    break 'outer;
297                }
298            }
299        }
300    }
301    stop.store(true, std::sync::atomic::Ordering::Relaxed);
302    if let Some(im) = app.images.as_ref() { im.clear_screen_images(&mut std::io::stdout()); }
303    drop((code_tx, login_task));
304    drop(term);
305    Ok(detached)
306}

308/// Write a copy of theme `f` to `~/.acp/ai/data/exports/themes/<name>-<stamp>.toml`.
309pub fn export_theme(f: &crate::themes::ThemeFile) -> Result<PathBuf> {
310    let stamp = chrono::Local::now().format("%Y%m%d-%H%M%S");
311    let p = home()?.join(EXPORTS_REL_PATH).join("themes").join(format!("{}-{stamp}.toml", f.name));
312    crate::themes::atomic_write(&p, crate::themes::to_toml(f).as_bytes()).with_context(|| format!("writing {}", p.display()))?;
313    Ok(p)
314}

316/// Whether to re-emit attachment links as OSC 8 hyperlinks (off for the Linux console,
317/// `TERM=dumb`, or with `ACP_NO_OSC8` set).
318pub fn osc8_supported() -> bool {
319    if std::env::var_os("ACP_NO_OSC8").is_some() { return false; }
320    !matches!(std::env::var("TERM").as_deref(), Ok("linux") | Ok("dumb"))
321}

323/// The OSC 8 sequence for `text` linking to `url` (no control characters pass through).
324pub fn osc8(text: &str, url: &str) -> String {
325    let clean = |s: &str| s.chars().filter(|c| !c.is_control()).collect::<String>();
326    format!("\x1b]8;;{}\x1b\\{}\x1b]8;;\x1b\\", clean(url), clean(text))
327}

329/// Re-print link text as OSC 8 hyperlinks over the drawn frame, in the theme's link colour
330/// on its background so the cells look the same.
331fn emit_osc8(links: &[(u16, u16, String, String)], t: &theme::Theme) {
332    use crossterm::{cursor::MoveTo, queue, style::{Attribute, Color as CColor, Print, SetAttribute, SetBackgroundColor, SetForegroundColor}};
333    if links.is_empty() { return; }
334    let mut out = std::io::stdout();
335    let conv = |c: ratatui::style::Color| crate::themes::rgb(c).map(CColor::from);
336    let fg = ratatui::style::Style::from(&t.link()).fg.and_then(conv);
337    let bg = conv(t.bg);
338    for (x, y, text, url) in links {
339        let _ = queue!(out, MoveTo(*x, *y), SetAttribute(Attribute::Underlined));
340        if let Some(c) = fg { let _ = queue!(out, SetForegroundColor(c)); }
341        if let Some(c) = bg { let _ = queue!(out, SetBackgroundColor(c)); }
342        let _ = queue!(out, Print(osc8(text, url)), SetAttribute(Attribute::Reset));
343    }
344    let _ = out.flush();
345}

347/// Open `p` with the desktop's default application, detached (`xdg-open`; `open` on macOS;
348/// `ACP_OPENER` overrides).
349pub fn open_file(p: &Path) -> Result<()> {
350    let opener = std::env::var("ACP_OPENER").ok().filter(|s| !s.is_empty())
351        .unwrap_or_else(|| if cfg!(target_os = "macos") { "open".into() } else { "xdg-open".into() });
352    let mut c = std::process::Command::new(&opener);
353    c.arg(p).stdin(std::process::Stdio::null()).stdout(std::process::Stdio::null()).stderr(std::process::Stdio::null());
354    #[cfg(unix)]
355    { use std::os::unix::process::CommandExt; c.process_group(0); }
356    c.spawn().with_context(|| format!("starting {opener} {}", p.display()))?;
357    Ok(())
358}
