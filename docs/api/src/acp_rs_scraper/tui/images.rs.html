1//! Pictures in the terminal: the header logo and post images.
2//!
3//! The graphics protocol is found once at start by querying the terminal
4//! (kitty graphics reply, sixel in the DA1 attributes, cell size via
5//! `CSI 16 t`), with iTerm2 inline images picked from the environment (iTerm2,
6//! WezTerm). Anything that doesn't answer — the Linux console, tmux, many SSH,
7//! waypipe and Windows-terminal setups — gets Unicode half blocks, which are
8//! drawn straight into the ratatui buffer and work everywhere.
9//!
10//! Graphics protocols are drawn after ratatui's frame, like the OSC 8 links:
11//! widgets reserve blank cells and record a [`Placement`]; [`Images::flush`]
12//! emits the escapes only when the placements change (kitty deletes and
13//! re-places; sixel / iTerm2 ask for a full repaint first so no fragments stay).
14//! Images decode on a worker thread; AVIF/HEIC (the site's format) are
15//! converted once to a cached PNG with `avifdec`, `magick` or `ffmpeg`.
16//!
17//! This is hand-rolled rather than ratatui-image because that crate's sixel
18//! encoder pulls in a quantiser that needs Rust 1.90, above our MSRV (1.88).
19

26/// The red star logo (`assets/red-star.png`), embedded.

28/// Largest side of a decoded post image, in pixels (keeps scaling cheap).

31pub const IMAGE_ROWS: u16 = 14;
32/// Cell size assumed when the terminal doesn't report one (pixels).
33pub const DEFAULT_CELL: (u16, u16) = (10, 20);

35/// How pictures reach the screen.
36#[derive(Clone, Copy, Debug, PartialEq, Eq)]
37pub enum Protocol {
38    /// Unicode `▀` cells with true-colour fg/bg (works everywhere).
39    Halfblocks,
40    /// Kitty graphics protocol.
41    Kitty,
42    /// DEC sixel.
43    Sixel,
44    /// iTerm2 inline images (OSC 1337), also WezTerm.
45    Iterm2,
46}
47
48impl Protocol {
49    /// Name for the status line and `ACP_IMAGE_PROTOCOL`.
50    pub fn name(self) -> &'static str {
51        match self { Protocol::Halfblocks => "halfblocks", Protocol::Kitty => "kitty", Protocol::Sixel => "sixel", Protocol::Iterm2 => "iterm2" }
52    }
53    /// Parse `ACP_IMAGE_PROTOCOL`.
54    pub fn parse(s: &str) -> Option<Self> {
55        match s.trim().to_ascii_lowercase().as_str() {
56            "halfblocks" | "blocks" => Some(Protocol::Halfblocks), "kitty" => Some(Protocol::Kitty),
57            "sixel" => Some(Protocol::Sixel), "iterm2" | "iterm" => Some(Protocol::Iterm2), _ => None,
58        }
59    }
60}

121    /// Images drawn with `protocol`; `background` decodes on a worker thread (else
122    /// inline, for tests). The logo is laid on a `badge` colour so the red star
123    /// stays visible on the red header.
124    pub fn new(protocol: Protocol, cell: (u16, u16), background: bool, badge: [u8; 3]) -> Self {
125        let logo = logo_on(badge).map(|i| Pic { id: 1, img: i.to_rgba8() });
126        let (tx, rx) = if background {
127            let (tx, rx_req) = std::sync::mpsc::channel::<PathBuf>();
128            let (tx_res, rx) = std::sync::mpsc::channel();
129            std::thread::spawn(move || {
130                while let Ok(p) = rx_req.recv() {
131                    let r = decode(&p).map_err(|e| format!("{e:#}"));
132                    if tx_res.send((p, r)).is_err() { break; }
133                }
134            });
135            (Some(tx), Some(rx))
136        } else { (None, None) };
137        let cell = (cell.0.max(1), cell.1.max(2));
138        Images { protocol, cell, logo, badge, cache: HashMap::new(), next_id: 2, tx, rx, frame: Vec::new(), last: Vec::new(),
139            transmitted: HashSet::new(), scaled: HashMap::new(), encoded: HashMap::new() }
140    }

142    /// Detect the terminal's graphics support (call in raw mode, before the input
143    /// thread reads stdin). `ACP_IMAGES=off` disables pictures;
144    /// `ACP_IMAGE_PROTOCOL=halfblocks|sixel|kitty|iterm2` forces a protocol.
145    pub fn detect(badge: [u8; 3]) -> Option<Self> {
146        if std::env::var("ACP_IMAGES").is_ok_and(|v| matches!(v.as_str(), "off" | "0" | "no")) { return None; }
147        let forced = std::env::var("ACP_IMAGE_PROTOCOL").ok().and_then(|s| Protocol::parse(&s));
148        let in_tmux = std::env::var_os("TMUX").is_some();
149        let reply = if forced == Some(Protocol::Halfblocks) || in_tmux { String::new() } else { query_terminal() };
150        let mut p = parse_reply(&reply);
151        if p.0 == Protocol::Halfblocks && !in_tmux && iterm2_env() { p.0 = Protocol::Iterm2; }
152        let cell = p.1.or_else(cell_from_winsize).unwrap_or(DEFAULT_CELL);
153        Some(Images::new(forced.unwrap_or(p.0), cell, true, badge))
154    }

156    /// The state of `path`, starting its decode on first use.
157    pub fn get(&mut self, path: &Path) -> &mut ImgState {
158        if !self.cache.contains_key(path) {
159            let st = match &self.tx {
160                Some(tx) => { let _ = tx.send(path.to_path_buf()); ImgState::Loading }
161                None => match decode(path) {
162                    Ok(i) => { self.next_id += 1; ImgState::Ready(Pic { id: self.next_id, img: i.to_rgba8() }) }
163                    Err(e) => ImgState::Failed(format!("{e:#}")),
164                },
165            };
166            self.cache.insert(path.to_path_buf(), st);
167        }
168        self.cache.get_mut(path).expect("just inserted")
169    }
170
171    /// Take finished decodes from the worker; true when something changed.
172    pub fn poll(&mut self) -> bool {
173        let Some(rx) = &self.rx else { return false };
174        let mut done = Vec::new();
175        while let Ok(d) = rx.try_recv() { done.push(d); }
176        let changed = !done.is_empty();
177        for (p, r) in done {
178            let st = match r {
179                Ok(i) => { self.next_id += 1; ImgState::Ready(Pic { id: self.next_id, img: i.to_rgba8() }) }
180                Err(e) => ImgState::Failed(e),
181            };
182            self.cache.insert(p, st);
183        }
184        changed
185    }

187    /// Draw the logo into `area`. With half blocks it is drawn in two colours
188    /// with sextant cells instead (2×3 dots per cell), which keeps the star's
189    /// points at header size.
190    pub fn draw_logo(&mut self, buf: &mut Buffer, area: Rect) -> bool {
191        let Some(pic) = self.logo.take() else { return false };
192        if self.protocol == Protocol::Halfblocks {
193            let (cols, rows) = fit(pic.img.width(), pic.img.height(), self.cell, area.width, area.height);
194            let rect = Rect { x: area.x + (area.width - cols) / 2, y: area.y + (area.height - rows) / 2, width: cols, height: rows };
195            let small = self.scaled.entry((pic.id, cols, rows)).or_insert_with(|| {
196                image::imageops::resize(&pic.img, u32::from(cols) * 2, u32::from(rows) * 3, image::imageops::FilterType::Triangle)
197            });
198            let b = self.badge;
199            let star = |p: &image::Rgba<u8>| {
200                let d = |i: usize| (i32::from(p.0[i]) - i32::from(b[i])).abs();
201                d(0) + d(1) + d(2) > 120
202            };
203            sextants(buf, rect, small, star, Color::Rgb(193, 22, 22), Color::Rgb(b[0], b[1], b[2]));
204        } else {
205            self.place(buf, area, &pic, true, true);
206        }
207        self.logo = Some(pic);
208        true
209    }

211    /// Draw the picture at `path` into `area` (left-aligned, aspect kept).
212    /// `graphics = false` forces half blocks (e.g. under a modal). Err = what to show instead.
213    pub fn draw_file(&mut self, buf: &mut Buffer, area: Rect, path: &Path, graphics: bool) -> std::result::Result<(), String> {
214        let pic = match self.get(path) {
215            ImgState::Loading => return Err("… loading picture".into()),
216            ImgState::Failed(e) => return Err(format!("▣ picture unavailable: {e}")),
217            ImgState::Ready(_) => match self.cache.remove(path) { Some(ImgState::Ready(p)) => p, _ => unreachable!() },
218        };
219        self.place(buf, area, &pic, graphics, false);
220        self.cache.insert(path.to_path_buf(), ImgState::Ready(pic));
221        Ok(())
222    }

224    fn place(&mut self, buf: &mut Buffer, area: Rect, pic: &Pic, graphics: bool, centre: bool) {
225        let (cols, rows) = fit(pic.img.width(), pic.img.height(), self.cell, area.width, area.height);
226        if cols == 0 || rows == 0 { return; }
227        let x = if centre { area.x + (area.width - cols) / 2 } else { area.x };
228        let y = if centre { area.y + (area.height - rows) / 2 } else { area.y };
229        let rect = Rect { x, y, width: cols, height: rows };
230        if graphics && self.protocol != Protocol::Halfblocks {
231            for cy in rect.top()..rect.bottom() { for cx in rect.left()..rect.right() {
232                if let Some(c) = buf.cell_mut((cx, cy)) { c.set_symbol(" "); }
233            } }
234            self.frame.push(Placement { id: pic.id, rect });
235            return;
236        }
237        let key = (pic.id, cols, rows);
238        let small = self.scaled.entry(key).or_insert_with(|| {
239            image::imageops::resize(&pic.img, u32::from(cols), u32::from(rows) * 2, image::imageops::FilterType::Triangle)
240        });
241        halfblocks(buf, rect, small);
242    }

244    /// Emit this frame's graphics (after ratatui drew). Writes nothing when the
245    /// placements didn't change.
246    pub fn flush(&mut self, out: &mut impl Write) -> Flush {
247        let frame = std::mem::take(&mut self.frame);
248        if frame == self.last { return Flush::Done; }
249        if matches!(self.protocol, Protocol::Sixel | Protocol::Iterm2) && !self.last.is_empty() {
250            self.last.clear();
251            return Flush::Repaint;
252        }
253        let mut s = String::new();
254        if self.protocol == Protocol::Kitty { s.push_str("\x1b_Ga=d,d=a,q=2\x1b\\"); }
255        for (n, p) in frame.iter().enumerate() {
256            s.push_str(&format!("\x1b7\x1b[{};{}H", p.rect.y + 1, p.rect.x + 1));
257            match self.protocol {
258                Protocol::Kitty => {
259                    if !self.transmitted.contains(&p.id)
260                        && let Some(img) = self.pixels(p.id) {
261                            s.push_str(&kitty_transmit(p.id, &png_bytes(img)));
262                            self.transmitted.insert(p.id);
263                        }
264                    s.push_str(&format!("\x1b_Ga=p,i={},p={},c={},r={},C=1,q=2\x1b\\", p.id, n + 1, p.rect.width, p.rect.height));
265                }
266                Protocol::Sixel => {
267                    let key = (p.id, p.rect.width, p.rect.height);
268                    if !self.encoded.contains_key(&key) {
269                        let (w, h) = (u32::from(p.rect.width) * u32::from(self.cell.0), u32::from(p.rect.height) * u32::from(self.cell.1));
270                        if let Some(img) = self.pixels(p.id) {
271                            let small = image::imageops::resize(img, w, h, image::imageops::FilterType::Triangle);
272                            self.encoded.insert(key, sixel(&small));
273                        }
274                    }
275                    if let Some(e) = self.encoded.get(&key) { s.push_str(e); }
276                }
277                Protocol::Iterm2 => {
278                    if let Some(img) = self.pixels(p.id) {
279                        let png = png_bytes(img);
280                        s.push_str(&format!("\x1b]1337;File=inline=1;size={};width={};height={};preserveAspectRatio=1;doNotMoveCursor=1:{}\x07",
281                            png.len(), p.rect.width, p.rect.height, base64(&png)));
282                    }
283                }
284                Protocol::Halfblocks => {}
285            }
286            s.push_str("\x1b8");
287        }
288        let _ = out.write_all(s.as_bytes());
289        let _ = out.flush();
290        self.last = frame;
291        Flush::Done
292    }

308/// Cells (cols, rows) a `w`×`h` picture fills inside `max_cols`×`max_rows`, keeping its aspect.
309pub fn fit(w: u32, h: u32, cell: (u16, u16), max_cols: u16, max_rows: u16) -> (u16, u16) {
310    if w == 0 || h == 0 || max_cols == 0 || max_rows == 0 { return (0, 0); }
311    let (cw, ch) = (f64::from(cell.0), f64::from(cell.1));
312    let scale = (f64::from(max_cols) * cw / f64::from(w)).min(f64::from(max_rows) * ch / f64::from(h));
313    let cols = ((f64::from(w) * scale / cw).round() as u16).clamp(1, max_cols);
314    let rows = ((f64::from(h) * scale / ch).round() as u16).clamp(1, max_rows);
315    (cols, rows)
316}

318/// Draw `img` (cols × 2·rows pixels) as `▀` / `▄` cells; transparent pixels leave the cell's background.
319pub fn halfblocks(buf: &mut Buffer, rect: Rect, img: &RgbaImage) {
320    let px = |x: u32, y: u32| img.get_pixel_checked(x, y).filter(|p| p.0[3] >= 128).map(|p| Color::Rgb(p.0[0], p.0[1], p.0[2]));
321    for row in 0..rect.height {
322        for col in 0..rect.width {
323            let (top, bot) = (px(u32::from(col), u32::from(row) * 2), px(u32::from(col), u32::from(row) * 2 + 1));
324            let Some(c) = buf.cell_mut((rect.x + col, rect.y + row)) else { continue };
325            match (top, bot) {
326                (Some(t), Some(b)) => { c.set_symbol("▀").set_fg(t).set_bg(b); }
327                (Some(t), None) => { c.set_symbol("▀").set_fg(t); }
328                (None, Some(b)) => { c.set_symbol("▄").set_fg(b); }
329                (None, None) => {}
330            }
331        }
332    }
333}

335/// The sextant character for 6 dots (bit 0 top-left, 1 top-right, 2 middle-left, …, 5 bottom-right).
336pub fn sextant(bits: u8) -> char {
337    match bits & 63 {
338        0 => ' ',
339        63 => '█',
340        21 => '▌',
341        42 => '▐',
342        n => {
343            let off = if n < 21 { n - 1 } else if n < 42 { n - 2 } else { n - 3 };
344            char::from_u32(0x1FB00 + u32::from(off)).unwrap_or('█')
345        }
346    }
347}

349/// Draw `img` (2·cols × 3·rows pixels) in two colours: `on` pixels in `fg`, the rest `bg`.
350pub fn sextants(buf: &mut Buffer, rect: Rect, img: &RgbaImage, on: impl Fn(&image::Rgba<u8>) -> bool, fg: Color, bg: Color) {
351    for row in 0..rect.height {
352        for col in 0..rect.width {
353            let mut bits = 0u8;
354            for dy in 0..3 { for dx in 0..2 {
355                let (x, y) = (u32::from(col) * 2 + dx, u32::from(row) * 3 + dy);
356                if img.get_pixel_checked(x, y).is_some_and(&on) { bits |= 1 << (dy * 2 + dx); }
357            } }
358            if let Some(c) = buf.cell_mut((rect.x + col, rect.y + row)) {
359                c.set_char(sextant(bits)).set_fg(fg).set_bg(bg);
360            }
361        }
362    }
363}

365/// Standard base64 (no line breaks).
366pub fn base64(data: &[u8]) -> String {
367    const T: &[u8; 64] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
368    let mut s = String::with_capacity(data.len().div_ceil(3) * 4);
369    for ch in data.chunks(3) {
370        let b = [ch[0], *ch.get(1).unwrap_or(&0), *ch.get(2).unwrap_or(&0)];
371        let n = (u32::from(b[0]) << 16) | (u32::from(b[1]) << 8) | u32::from(b[2]);
372        for i in 0..4 {
373            if i <= ch.len() { s.push(T[((n >> (18 - 6 * i)) & 63) as usize] as char); } else { s.push('='); }
374        }
375    }
376    s
377}

379fn png_bytes(img: &RgbaImage) -> Vec<u8> {
380    let mut v = std::io::Cursor::new(Vec::new());
381    let _ = DynamicImage::ImageRgba8(img.clone()).write_to(&mut v, image::ImageFormat::Png);
382    v.into_inner()
383}

385/// Kitty: transmit a PNG as image `id`, in 4096-byte chunks.
386pub fn kitty_transmit(id: u32, png: &[u8]) -> String {
387    let b = base64(png);
388    let chunks: Vec<&[u8]> = b.as_bytes().chunks(4096).collect();
389    let mut s = String::new();
390    for (i, c) in chunks.iter().enumerate() {
391        let more = u8::from(i + 1 < chunks.len());
392        let c = std::str::from_utf8(c).unwrap_or_default();
393        if i == 0 { s.push_str(&format!("\x1b_Ga=t,f=100,i={id},q=2,m={more};{c}\x1b\\")); }
394        else { s.push_str(&format!("\x1b_Gm={more};{c}\x1b\\")); }
395    }
396    s
397}

399/// Encode RGBA as sixel (6×6×6 colour cube; transparent pixels left alone).
400pub fn sixel(img: &RgbaImage) -> String {
401    let (w, h) = img.dimensions();
402    let q = |v: u8| (u16::from(v) * 5 + 127) / 255;
403    let idx = |x: u32, y: u32| -> Option<u16> {
404        let p = img.get_pixel(x, y).0;
405        (p[3] >= 128).then(|| q(p[0]) * 36 + q(p[1]) * 6 + q(p[2]))
406    };
407    let mut s = format!("\x1bP0;1;0q\"1;1;{w};{h}");
408    let mut used = [false; 216];
409    for y in 0..h { for x in 0..w { if let Some(i) = idx(x, y) { used[i as usize] = true; } } }
410    for (i, u) in used.iter().enumerate() {
411        if *u { let c = |k: usize| k * 100 / 5; s.push_str(&format!("#{i};2;{};{};{}", c(i / 36), c(i / 6 % 6), c(i % 6))); }
412    }
413    let mut band = 0;
414    while band < h {
415        let mut cols: Vec<u16> = Vec::new();
416        let mut bits = vec![0u8; (w as usize) * 216];
417        for dy in 0..6 {
418            let y = band + dy;
419            if y >= h { break; }
420            for x in 0..w {
421                if let Some(i) = idx(x, y) {
422                    let b = &mut bits[i as usize * w as usize + x as usize];
423                    if *b == 0 && !cols.contains(&i) { cols.push(i); }
424                    *b |= 1 << dy;
425                }
426            }
427        }
428        for (n, c) in cols.iter().enumerate() {
429            if n > 0 { s.push('$'); }
430            s.push_str(&format!("#{c}"));
431            let row = &bits[*c as usize * w as usize..(*c as usize + 1) * w as usize];
432            let mut x = 0;
433            while x < row.len() {
434                let v = row[x];
435                let mut run = 1;
436                while x + run < row.len() && row[x + run] == v { run += 1; }
437                let ch = (63 + v) as char;
438                if run > 3 { s.push_str(&format!("!{run}{ch}")); } else { for _ in 0..run { s.push(ch); } }
439                x += run;
440            }
441        }
442        s.push('-');
443        band += 6;
444    }
445    s.push_str("\x1b\\");
446    s
447}

449/// Read what the terminal says to: a kitty graphics query, the cell-size
450/// report (`CSI 16 t`) and the primary device attributes (DA1, answered by
451/// everything, so it ends the wait). Gives up after 400 ms.
452#[cfg(unix)]
453fn query_terminal() -> String {
454    let q = b"\x1b_Gi=31,s=1,v=1,a=q,t=d,f=24;AAAA\x1b\\\x1b[16t\x1b[c";
455    let mut out = std::io::stdout();
456    if out.write_all(q).and_then(|_| out.flush()).is_err() { return String::new(); }
457    let deadline = std::time::Instant::now() + std::time::Duration::from_millis(400);
458    let mut got = Vec::new();
459    loop {
460        let left = deadline.saturating_duration_since(std::time::Instant::now());
461        if left.is_zero() { break; }
462        let mut fds = libc::pollfd { fd: libc::STDIN_FILENO, events: libc::POLLIN, revents: 0 };
463        // SAFETY: one valid pollfd for the duration of the call.
464        let r = unsafe { libc::poll(&mut fds, 1, left.as_millis() as libc::c_int) };
465        if r <= 0 { break; }
466        let mut b = [0u8; 256];
467        // SAFETY: reading into a local buffer of the given length.
468        let n = unsafe { libc::read(libc::STDIN_FILENO, b.as_mut_ptr().cast(), b.len()) };
469        if n <= 0 { break; }
470        got.extend_from_slice(&b[..n as usize]);
471        let s = String::from_utf8_lossy(&got);
472        if let Some(i) = s.find("\x1b[?") && s[i..].contains('c') { break; }
473    }
474    String::from_utf8_lossy(&got).into_owned()
475}

480/// The protocol and cell size a terminal's replies announce.
481pub fn parse_reply(r: &str) -> (Protocol, Option<(u16, u16)>) {
482    let cell = r.find("\x1b[6;").and_then(|i| {
483        let rest = &r[i + 4..];
484        let end = rest.find('t')?;
485        let mut it = rest[..end].split(';');
486        let h: u16 = it.next()?.parse().ok()?;
487        let w: u16 = it.next()?.parse().ok()?;
488        (w > 0 && h > 0).then_some((w, h))
489    });
490    if r.contains("\x1b_Gi=31;OK") { return (Protocol::Kitty, cell); }
491    let sixel = r.find("\x1b[?").and_then(|i| {
492        let rest = &r[i + 3..];
493        let end = rest.find('c')?;
494        Some(rest[..end].split(';').any(|p| p == "4"))
495    }).unwrap_or(false);
496    (if sixel { Protocol::Sixel } else { Protocol::Halfblocks }, cell)
497}

509/// The logo composited on a square of `badge` colour (with a small margin).
510pub fn logo_on(badge: [u8; 3]) -> Option<DynamicImage> {
511    let star = image::load_from_memory(LOGO_PNG).ok()?.to_rgba8();
512    let m = star.width() / 10;
513    let side = star.width().max(star.height()) + 2 * m;
514    let mut bg = RgbaImage::from_pixel(side, side, image::Rgba([badge[0], badge[1], badge[2], 255]));
515    image::imageops::overlay(&mut bg, &star, i64::from(m), i64::from(m));
516    Some(DynamicImage::ImageRgba8(bg))
517}

526/// Decode `path` (converting AVIF/HEIC first), shrunk to [`MAX_IMAGE_PX`].
527pub fn decode(path: &Path) -> Result<DynamicImage> {
528    if !path.exists() { bail!("file missing"); }
529    let src = if matches!(ext(path).as_str(), "avif" | "heic" | "heif") { convert_to_png(path)? } else { path.to_path_buf() };
530    let img = image::ImageReader::open(&src)?.with_guessed_format()?.decode()
531        .with_context(|| format!("decoding {}", src.display()))?;
532    Ok(if img.width() > MAX_IMAGE_PX || img.height() > MAX_IMAGE_PX { img.thumbnail(MAX_IMAGE_PX, MAX_IMAGE_PX) } else { img })
533}

535/// Cached PNG for a format the image crate can't read: `~/.acp/cache/thumbs/<hash>.png`.
536pub fn convert_to_png(path: &Path) -> Result<PathBuf> {
537    let mut h = DefaultHasher::new();
538    path.hash(&mut h);
539    std::fs::metadata(path).ok().and_then(|m| m.modified().ok()).hash(&mut h);
540    let dir = home()?.join(THUMBS_REL_DIR);
541    std::fs::create_dir_all(&dir)?;
542    let out = dir.join(format!("{:016x}.png", h.finish()));
543    if out.exists() { return Ok(out); }
544    let tmp = out.with_extension("tmp.png");
545    let size = format!("{MAX_IMAGE_PX}x{MAX_IMAGE_PX}>");
546    let scale = format!("scale='min({MAX_IMAGE_PX},iw)':-2");
547    let tries: [(&str, Vec<&std::ffi::OsStr>); 3] = [
548        ("avifdec", vec![path.as_os_str(), tmp.as_os_str()]),
549        ("magick", vec![path.as_os_str(), "-resize".as_ref(), size.as_ref(), tmp.as_os_str()]),
550        ("ffmpeg", vec!["-loglevel".as_ref(), "error".as_ref(), "-y".as_ref(), "-i".as_ref(), path.as_os_str(), "-vf".as_ref(), scale.as_ref(), "-frames:v".as_ref(), "1".as_ref(), tmp.as_os_str()]),
551    ];
552    let mut tried = Vec::new();
553    for (cmd, args) in tries {
554        let r = std::process::Command::new(cmd).args(&args)
555            .stdin(std::process::Stdio::null()).stdout(std::process::Stdio::null()).stderr(std::process::Stdio::null()).status();
556        match r {
557            Ok(s) if s.success() && tmp.exists() => { std::fs::rename(&tmp, &out)?; return Ok(out); }
558            Ok(_) => tried.push(format!("{cmd} failed")),
559            Err(_) => tried.push(format!("{cmd} not found")),
560        }
561    }
562    let _ = std::fs::remove_file(&tmp);
563    bail!("cannot convert {} to PNG ({}); install libavif (avifdec), ImageMagick or ffmpeg", path.display(), tried.join(", "))
564}
