11/// The base layout's areas.
12#[derive(Clone, Copy, Debug, PartialEq, Eq)]
13pub struct Areas {
14    /// Top bar (title).
15    pub top: Rect,
16    /// Menu bar.
17    pub menu: Rect,
18    /// Left nick bar.
19    pub nicks: Rect,
20    /// Body.
21    pub body: Rect,
22    /// Status / log footer.
23    pub status: Rect,
24}

26/// Split the screen: top bar, menu bar, nicks | body, status (`log_rows` tall).
27pub fn areas(r: Rect, log_rows: u16) -> Areas {
28    let row = |y: u16, h: u16| Rect { x: r.x, y, width: r.width, height: h };
29    let top = row(r.y, 1.min(r.height));
30    let menu = row(r.y + 1, 1.min(r.height.saturating_sub(1)));
31    let status_h = log_rows.max(1).min(r.height.saturating_sub(4));
32    let main_y = r.y + 2;
33    let main_h = r.height.saturating_sub(2 + status_h);
34    let status = row(main_y + main_h, status_h);
35    let nw = NICK_BAR_WIDTH.min(r.width / 3);
36    Areas { top, menu,
37        nicks: Rect { x: r.x, y: main_y, width: nw, height: main_h },
38        body: Rect { x: r.x + nw, y: main_y, width: r.width.saturating_sub(nw), height: main_h },
39        status }
40}

42/// The post list with an open detail: compact rows above and below the detail.
43#[derive(Clone, Debug, PartialEq, Eq)]
44pub struct DetailSplit {
45    /// Compact rows above: (post index, rect).
46    pub above: Vec<(usize, Rect)>,
47    /// The detail box.
48    pub detail: Rect,
49    /// Compact rows below.
50    pub below: Vec<(usize, Rect)>,
51}
52
53/// Lay out the list around the detail of post `cursor` (of `n`), `p` = slide progress.
54/// Fully open, the list keeps [`DETAIL_RESIDUAL_ROWS`] rows: one above (when there
55/// is a previous post) and the rest below; while sliding, the detail grows from
56/// the selected row and the rows below are pushed down and out.
57pub fn detail_split(area: Rect, n: usize, cursor: usize, p: f32) -> DetailSplit {
58    let residual = DETAIL_RESIDUAL_ROWS.min(area.height / 3);
59    let full = area.height.saturating_sub(residual);
60    let h = ((full as f32) * p.clamp(0.0, 1.0)).round() as u16;
61    let h = h.clamp(3.min(area.height), full.max(3.min(area.height)));
62    let above_n = if cursor > 0 { 1.min(residual) } else { 0 } as usize;
63    let mut above = Vec::new();
64    let mut y = area.y;
65    for k in 0..above_n {
66        let i = cursor - above_n + k;
67        above.push((i, Rect { x: area.x, y, width: area.width, height: 1 }));
68        y += 1;
69    }
70    let detail = Rect { x: area.x, y, width: area.width, height: h.min(area.y + area.height - y) };
71    y += detail.height;
72    let mut below = Vec::new();
73    let mut i = cursor + 1;
74    while y < area.y + area.height && i < n {
75        below.push((i, Rect { x: area.x, y, width: area.width, height: 1 }));
76        y += 1;
77        i += 1;
78    }
79    DetailSplit { above, detail, below }
80}

82/// Where each frame of a detail stack goes inside the root detail box (`inner`).
83#[derive(Clone, Debug, PartialEq, Eq)]
84pub struct StackLayout {
85    /// (depth, rect, collapsed) for every frame, root first.
86    pub frames: Vec<(usize, Rect, bool)>,
87    /// Border of the nested box holding frames 1.. (none with a single frame).
88    pub nested: Option<Rect>,
89}

91fn lerp(a: u16, b: u16, p: f32) -> u16 { (a as f32 + (b as f32 - a as f32) * p.clamp(0.0, 1.0)).round() as u16 }
92
93/// Stack layout for `n` frames; `push` = progress of the latest push (1.0 = settled).
94/// Frame 0 (the post opened from the list) fills `inner` alone; with more frames
95/// it collapses to a [`DETAIL_HEADER_ROWS`] header and the rest nest in a box
96/// below it, where every frame but the top one is a header too. A push animates
97/// the previous top frame from its full height down to a header.
98pub fn stack_layout(inner: Rect, n: usize, push: f32) -> StackLayout {
99    if n == 0 { return StackLayout { frames: Vec::new(), nested: None }; }
100    if n == 1 { return StackLayout { frames: vec![(0, inner, false)], nested: None }; }
101    let hdr = DETAIL_HEADER_ROWS;
102    let bottom = inner.y + inner.height;
103    let clip = |y: u16, h: u16| h.min(bottom.saturating_sub(y));
104    let mut frames = Vec::new();
105    // The frame that was on top before the push shrinks from "everything left" to a header.
106    let prev_full = |avail: u16| lerp(avail, hdr, push).max(hdr.min(avail));
107    let h0 = if n == 2 { prev_full(inner.height.saturating_sub(3)) } else { hdr };
108    frames.push((0, Rect { height: clip(inner.y, h0), ..inner }, true));
109    let ny = inner.y + frames[0].1.height;
110    let nested = Rect { x: inner.x + 1, y: ny, width: inner.width.saturating_sub(2), height: bottom.saturating_sub(ny) };
111    let ni = Rect { x: nested.x + 1, y: nested.y + 1, width: nested.width.saturating_sub(2), height: nested.height.saturating_sub(2) };
112    let nb = ni.y + ni.height;
113    let mut y = ni.y;
114    for d in 1..n {
115        let top = d == n - 1;
116        let h = if top { nb.saturating_sub(y) }
117            else if d == n - 2 { prev_full(nb.saturating_sub(y).saturating_sub(2)) }
118            else { hdr };
119        let h = h.min(nb.saturating_sub(y));
120        frames.push((d, Rect { x: ni.x, y, width: ni.width, height: h }, !top));
121        y += h;
122    }
123    StackLayout { frames, nested: Some(nested) }
124}

126/// Centered rect of at most `w`×`h` inside `r` with a 1–2 cell margin.
127pub fn modal_rect(r: Rect, w: u16, h: u16) -> Rect {
128    let mw = r.width.saturating_sub(4).min(w).max(1);
129    let mh = r.height.saturating_sub(2).min(h).max(1);
130    Rect { x: r.x + (r.width - mw) / 2, y: r.y + (r.height - mh) / 2, width: mw, height: mh }
131}

133/// Word-wrap `text` to `width` columns, keeping its newlines (blank lines stay).
134pub fn wrap(text: &str, width: usize) -> Vec<String> {
135    use unicode_width::UnicodeWidthStr;
136    let width = width.max(1);
137    let mut out = Vec::new();
138    for para in text.split('\n') {
139        let mut line = String::new();
140        let mut w = 0;
141        for word in para.split(' ') {
142            let ww = word.width();
143            if w > 0 && w + 1 + ww > width {
144                out.push(std::mem::take(&mut line));
145                w = 0;
146            }
147            if ww > width {
148                // Hard-break words longer than the line.
149                for ch in word.chars() {
150                    let cw = unicode_width::UnicodeWidthChar::width(ch).unwrap_or(0);
151                    if w + cw > width { out.push(std::mem::take(&mut line)); w = 0; }
152                    line.push(ch);
153                    w += cw;
154                }
155                continue;
156            }
157            if w > 0 { line.push(' '); w += 1; }
158            line.push_str(word);
159            w += ww;
160        }
161        out.push(line);
162    }
163    out
164}

166/// Cut `s` to at most `width` columns, adding `…` when cut.
167pub fn ellipsize(s: &str, width: usize) -> String {
168    use unicode_width::UnicodeWidthChar;
169    let mut w = 0;
170    let mut out = String::new();
171    let total: usize = s.chars().map(|c| c.width().unwrap_or(0)).sum();
172    if total <= width { return s.to_string(); }
173    for c in s.chars() {
174        let cw = c.width().unwrap_or(0);
175        if w + cw + 1 > width { break; }
176        out.push(c);
177        w += cw;
178    }
179    out.push('…');
180    out
181}
