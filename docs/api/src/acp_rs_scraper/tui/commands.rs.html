1//! The menu bar's command catalog and the menu → command-line bridge.
2//!
3//! Every menu command maps to CLI flags (see SPECS "Menu bar ↔ CLI flag map").
4//! Actions that change data (scrape, export, import, maintenance) are turned into
5//! the exact argv the headless CLI takes, validated by clap ([`parse_with_mode`]),
6//! and run as a child process, so the TUI and the CLI share one code path.
7

15/// Menu bar items, left to right.
16#[derive(Clone, Copy, Debug, PartialEq, Eq)]
17pub enum CommandId {
18    /// Scrape options (Start Scanning runs it).
19    Scrape,
20    /// Offline search into the Body.
21    Search,
22    /// Dump the DB / selection to a file.
23    Export,
24    /// Import an export into the DB.
25    Import,
26    /// Refresh the nick bar.
27    List,
28    /// Remove nicks / clear everything.
29    Maintain,
30    /// Login / logout.
31    Session,
32    /// Bookmarks: show, toggle, sync.
33    Bookmarks,
34    /// Periodic sync with the site.
35    Sync,
36    /// Log level for jobs started from the TUI.
37    Debug,
38    /// Usage modal.
39    Help,
40    /// Leave the TUI.
41    Quit,
42}

89/// The input type shown right of an option's label.
90#[derive(Clone, Debug, PartialEq)]
91pub enum Widget {

94    /// Known values.
95    Choice {
96        /// Values.
97        choices: &'static [&'static str],

107/// One option under a menu command.
108#[derive(Clone, Debug, PartialEq)]
109pub struct Opt {
110    /// Stable key used by the argv builders.
111    pub key: &'static str,
112    /// Label.
113    pub label: &'static str,
114    /// CLI flag(s) it maps to (shown as a hint).
115    pub flag: &'static str,
116    /// Marked `*`: must be set for the command's action.
117    pub required: bool,
118    /// Input type.
119    pub widget: Widget,
120}

122/// One menu bar item.
123#[derive(Clone, Debug, PartialEq)]
124pub struct MenuCommand {
125    /// Which command.
126    pub id: CommandId,
127    /// Label in the bar.
128    pub label: &'static str,
129    /// Hotkey letter (underlined; Alt+letter or the letter while the menu is focused).
130    pub hint: char,
131    /// Options in its dropdown (empty: activating runs `action`).
132    pub opts: Vec<Opt>,
133    /// Action of an option-less item.
134    pub action: Option<Action>,
135}
136
137fn opt(key: &'static str, label: &'static str, flag: &'static str, required: bool, widget: Widget) -> Opt {
138    Opt { key, label, flag, required, widget }
139}
140fn check(v: bool) -> Widget { Widget::Check(v) }
141fn text(v: &str) -> Widget { Widget::Text(TextField::new(v)) }
142fn choice(choices: &'static [&'static str]) -> Widget { Widget::Choice { choices, idx: 0 } }
143fn button(a: Action) -> Widget { Widget::Button(a) }

238    pub fn get(&self, id: CommandId) -> &MenuCommand { self.items.iter().find(|c| c.id == id).expect("menu id") }
239    /// Option by command and key.
240    pub fn opt_mut(&mut self, id: CommandId, key: &str) -> &mut Opt {
241        self.items.iter_mut().find(|c| c.id == id).expect("menu id").opts.iter_mut().find(|o| o.key == key).expect("opt key")
242    }
243    fn opt(&self, id: CommandId, key: &str) -> &Opt { self.get(id).opts.iter().find(|o| o.key == key).expect("opt key") }
244    /// Checkbox value.
245    pub fn check(&self, id: CommandId, key: &str) -> bool { matches!(self.opt(id, key).widget, Widget::Check(true)) }
246    /// Text value, trimmed.
247    pub fn text(&self, id: CommandId, key: &str) -> String {
248        match &self.opt(id, key).widget { Widget::Text(t) => t.value.trim().to_string(), _ => String::new() }
249    }
250    /// Selected choice.
251    pub fn choice(&self, id: CommandId, key: &str) -> &'static str {
252        match &self.opt(id, key).widget { Widget::Choice { choices, idx } => choices[*idx], _ => "" }
253    }
254    /// Set a checkbox.

258        if let Widget::Text(t) = &mut self.opt_mut(id, key).widget { t.set(v); }
259    }
260    /// Select a choice by value (ignored when unknown).
261    pub fn set_choice(&mut self, id: CommandId, key: &str, v: &str) {
262        if let Widget::Choice { choices, idx } = &mut self.opt_mut(id, key).widget
263            && let Some(i) = choices.iter().position(|c| *c == v) { *idx = i; }

281    /// argv for Start Scanning (headless scrape of the chosen nicks).
282    pub fn scrape_argv(&self, selected: &[String]) -> std::result::Result<Vec<String>, String> {
283        use CommandId::Scrape as S;
284        let t = self.text(S, "nicks");
285        let nicks = if t.is_empty() { selected.to_vec() } else { parse_nicks(&t)?.0 };
286        if nicks.is_empty() { return Err("Scrape needs at least one nick: type them in Scrape › Nicks or select nicks on the left".into()); }
287        let mut v = vec![ARGV0.to_string(), "--nicks".into(), nicks.join(",")];
288        if self.choice(S, "mode") == "Trending" { v.push("--trending".into()); } else { v.push("--latest".into()); }
289        if !self.check(S, "mentions") { v.push("--no-mentions".into()); }
290        if self.check(S, "said_by") { v.push("--said-by".into()); }
291        let cl = self.text(S, "crawl_limit");
292        if !cl.is_empty() {
293            let n: usize = cl.parse().map_err(|_| format!("Crawl limit must be a number, not {cl:?}"))?;
294            if n != DEFAULT_CRAWL_LIMIT { v.extend(["--crawl-limit".into(), n.to_string()]); }
295        }
296        if !self.check(S, "attachments") { v.push("--no-attachments".into()); }
297        if self.check(S, "full") { v.push("--full".into()); }
298        if self.check(S, "refetch") { v.push("--refetch-truncated".into()); }
299        let out = self.text(S, "output");
300        if !out.is_empty() { v.extend(["--format".into(), self.choice(S, "format").into(), "--output".into(), out]); }
301        v.extend(self.debug_args());
302        validate(v)
303    }

305    /// argv for Export. `selected` = selected nicks; `view` = the current search, if any.
306    pub fn export_argv(&self, selected: &[String], view: Option<&SearchFilter>) -> std::result::Result<Vec<String>, String> {
307        use CommandId::Export as E;
308        let path = self.text(E, "path");
309        if path.is_empty() { return Err("Export needs a Path".into()); }
310        let mut v = vec![ARGV0.to_string(), "--export".into(), path];
311        let nicks_arg = |v: &mut Vec<String>, n: &[String]| if !n.is_empty() { v.extend(["--nicks".into(), n.join(",")]); };
312        match self.choice(E, "scope") {
313            "Selected nicks" => {
314                if selected.is_empty() { return Err("Export › From is \"Selected nicks\" but no nick is selected".into()); }
315                nicks_arg(&mut v, selected);
316            }
317            "Current view" => match view {
318                Some(f) => {
319                    if !f.keyword.trim().is_empty() { v.extend(["--search".into(), f.keyword.trim().into()]); }
320                    nicks_arg(&mut v, &f.nicks);
321                }
322                None => nicks_arg(&mut v, selected),
323            },
324            _ => {}
325        }
326        v.extend(["--format".into(), self.choice(E, "format").into()]);
327        let name = self.text(E, "name");
328        if !name.is_empty() { v.extend(["--name".into(), name]); }
329        if self.check(E, "compress") { v.extend(["--compress".into(), "--tar-gz".into()]); }
330        v.extend(self.debug_args());
331        validate(v)
332    }

384    /// argv for a sync (Sync now / auto sync): `--sync` with the nicks (field, else `selected`).
385    pub fn sync_argv(&self, selected: &[String]) -> std::result::Result<Vec<String>, String> {
386        use CommandId::Sync as Y;
387        let t = self.text(Y, "nicks");
388        let nicks = if t.is_empty() { selected.to_vec() } else { parse_nicks(&t)?.0 };
389        let bm = self.check(Y, "bookmarks");
390        if nicks.is_empty() && !bm { return Err("nothing to sync: tick Sync › Bookmarks, or type nicks in Sync › Nicks / select nicks".into()); }
391        let mut v = vec![ARGV0.to_string(), "--sync".into()];
392        if !nicks.is_empty() { v.extend(["--nicks".into(), nicks.join(",")]); if !self.check(Y, "mentions") { v.push("--no-mentions".into()); } }
393        if !bm { v.push("--no-bookmarks".into()); }
394        v.extend(self.debug_args());
395        validate(v)
396    }
397    /// Sync › Interval as a duration (at least a minute).
398    pub fn sync_interval(&self) -> std::result::Result<std::time::Duration, String> {
399        let t = self.text(CommandId::Sync, "interval");
400        if t.is_empty() { return Ok(std::time::Duration::from_secs(TUI_SYNC_DEFAULT_MINUTES * 60)); }
401        match t.parse::<u64>() {
402            Ok(m) if m >= 1 => Ok(std::time::Duration::from_secs(m * 60)),
403            _ => Err(format!("Sync › Interval must be a whole number of minutes (1 or more), not {t:?}")),
404        }
405    }
406    /// The Sync menu as `[sync]` preferences.
407    pub fn sync_prefs(&self) -> SyncPrefs {
408        use CommandId::Sync as Y;
409        SyncPrefs { auto: Some(self.check(Y, "auto")), interval_minutes: self.sync_interval().ok().map(|d| d.as_secs() / 60),
410            nicks: Some(self.text(Y, "nicks")), mentions: Some(self.check(Y, "mentions")), bookmarks: Some(self.check(Y, "bookmarks")) }
411    }
412    /// Fill the Sync menu from `[sync]` preferences.
413    pub fn apply_sync_prefs(&mut self, p: &SyncPrefs) {
414        use CommandId::Sync as Y;
415        if let Some(v) = p.auto { self.set_check(Y, "auto", v); }
416        if let Some(v) = p.interval_minutes { self.set_text(Y, "interval", &v.to_string()); }
417        if let Some(v) = &p.nicks { self.set_text(Y, "nicks", v); }
418        if let Some(v) = p.mentions { self.set_check(Y, "mentions", v); }
419        if let Some(v) = p.bookmarks { self.set_check(Y, "bookmarks", v); }
420    }

433/// Check `argv` with clap exactly as the CLI would and make sure it runs headless.
434pub fn validate(argv: Vec<String>) -> std::result::Result<Vec<String>, String> {
435    match parse_with_mode(&argv) {
436        Ok((_, false)) => Ok(argv),
437        Ok((_, true)) => Err("these options would open the terminal UI instead of running".into()),
438        Err(e) => Err(e.to_string().lines().next().unwrap_or("invalid options").trim_start_matches("error: ").to_string()),
439    }
440}

442/// Shell-style rendering of an argv (for the status line and Help).
443pub fn shell_line(argv: &[String]) -> String {
444    argv.iter().map(|a| if a.is_empty() || a.contains(|c: char| c.is_whitespace() || "\"'$\\".contains(c)) {
445        format!("'{}'", a.replace('\'', "'\\''"))
446    } else { a.clone() }).collect::<Vec<_>>().join(" ")
