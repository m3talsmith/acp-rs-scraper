24/// How a modal closed.
25#[derive(Clone, Copy, Debug, PartialEq, Eq)]
26pub enum ModalResult {
27    /// `CLOSED_CANCELLED`: discard the draft state.
28    ClosedCancelled,
29    /// `CLOSED_SUBMITTED`: apply it.
30    ClosedSubmitted,
31}
32
33/// A clickable element inside a modal.
34#[derive(Clone, Copy, Debug, PartialEq, Eq)]
35pub enum ModalHit {
36    /// The n-th button.
37    Button(usize),
38    /// The n-th field.
39    Field(usize),
40    /// The n-th list row.
41    Item(usize),
42}
43
44/// Behaviour every modal shares.
45pub trait ModalState {
46    /// Title in the modal's border.
47    fn title(&self) -> String;
48    /// A key; `Some` closes the modal with that result.
49    fn on_key(&mut self, k: KeyEvent) -> Option<ModalResult>;
50    /// A click on one of its elements; `Some` closes it.
51    fn on_click(&mut self, h: ModalHit) -> Option<ModalResult>;
52    /// Mouse wheel (positive: down).
53    fn on_scroll(&mut self, _delta: i32) {}
54}
55
56/// The open modal.
57#[derive(Debug)]
58pub enum Modal {
59    /// Email + one-time code.
60    Login(LoginModal),
61    /// Usage and keys.
62    Help(HelpModal),
63    /// Confirm a destructive / consequential action.
64    Confirm(ConfirmModal),
65    /// Error or status.
66    Message(MessageModal),
67    /// Advanced search filters.
68    Search(SearchModal),
69    /// Command palette.
70    Palette(PaletteModal),
71    /// Pick / create a research project (and add to it).
72    Research(ResearchModal),
73    /// Pick a theme (F5).
74    ThemePicker(Box<ThemePickerModal>),
75}
76
77impl Modal {
78    /// The shared behaviour.
79    pub fn state(&mut self) -> &mut dyn ModalState {
80        match self {
81            Modal::Login(m) => m, Modal::Help(m) => m, Modal::Confirm(m) => m,
82            Modal::Message(m) => m, Modal::Search(m) => m, Modal::Palette(m) => m, Modal::Research(m) => m, Modal::ThemePicker(m) => m.as_mut(),
83        }
84    }
85}

87/// Move a focus index by `d` within `0..n`, wrapping.
88pub fn cycle(i: usize, n: usize, d: i32) -> usize {
89    if n == 0 { return 0; }
90    ((i as i64 + d as i64).rem_euclid(n as i64)) as usize
