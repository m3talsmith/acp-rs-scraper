1//! Help modal: the real `--help` text (generated from clap, so it cannot drift)
2//! plus the TUI key bindings.
3
4use super::*;
5use crossterm::event::KeyCode;
6
7/// Scrollable help text.
8#[derive(Debug, Clone, PartialEq)]
9pub struct HelpModal {
10    /// Lines.
11    pub lines: Vec<String>,
12    /// First visible line.
13    pub scroll: usize,
14}
15
16/// Help text: key bindings, then `acp-rs-scraper --help`.
17pub fn help_text() -> Vec<String> {
18    use clap::CommandFactory;
19    let mut v = vec!["Terminal UI keys".to_string(), String::new()];
20    for (k, d) in crate::config::TUI_KEYBINDINGS { v.push(format!("  {k:<22} {d}")); }
21    v.extend(["".into(), "Menu commands map to these flags; Start Scanning runs the same scrape the CLI does.".into(), "".into()]);
22    let help = crate::cli::Args::command().render_long_help().to_string();
23    v.extend(help.lines().map(str::to_string));
24    v
25}
26
27impl HelpModal {
28    /// Help with the generated text.
29    pub fn new() -> Self { HelpModal { lines: help_text(), scroll: 0 } }
30    fn scroll_by(&mut self, d: i32) {
31        let max = self.lines.len().saturating_sub(1) as i64;
32        self.scroll = (self.scroll as i64 + d as i64).clamp(0, max) as usize;
33    }
34}
35
36impl Default for HelpModal { fn default() -> Self { Self::new() } }
37
38impl ModalState for HelpModal {
39    fn title(&self) -> String { "Help".into() }
40    fn on_key(&mut self, k: KeyEvent) -> Option<ModalResult> {
41        match k.code {
42            KeyCode::Esc | KeyCode::Char('q') | KeyCode::Char('?') | KeyCode::F(1) => return Some(ModalResult::ClosedCancelled),
43            KeyCode::Enter => return Some(ModalResult::ClosedSubmitted),
44            KeyCode::Down | KeyCode::Char('j') => self.scroll_by(1),
45            KeyCode::Up | KeyCode::Char('k') => self.scroll_by(-1),
46            KeyCode::PageDown | KeyCode::Char(' ') => self.scroll_by(15),
47            KeyCode::PageUp => self.scroll_by(-15),
48            KeyCode::Home => self.scroll = 0,
49            KeyCode::End => self.scroll_by(i32::MAX / 2),
50            _ => {}
51        }
52        None
53    }
54    fn on_click(&mut self, h: ModalHit) -> Option<ModalResult> {
55        matches!(h, ModalHit::Button(_)).then_some(ModalResult::ClosedSubmitted)
56    }
57    fn on_scroll(&mut self, d: i32) { self.scroll_by(d * 3); }
58}
