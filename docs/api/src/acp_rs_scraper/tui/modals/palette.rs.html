1//! Ctrl+Space command palette: type to filter every command, Enter runs it.
2
3use super::*;
4use crate::tui::commands::Action;
5use crate::tui::widgets::text::TextField;
6use crossterm::event::KeyCode;
7
8/// What a palette entry does.
9#[derive(Debug, Clone, Copy, PartialEq, Eq)]
10pub enum PaletteCmd {
11    /// Run an action.
12    Action(Action),
13    /// Open the n-th menu item's dropdown.
14    OpenMenu(usize),
15}
16
17/// Palette state.
18#[derive(Debug, Clone, PartialEq)]
19pub struct PaletteModal {
20    /// Every entry.
21    pub entries: Vec<(String, PaletteCmd)>,
22    /// Filter text.
23    pub query: TextField,
24    /// Selected row among the filtered ones.
25    pub cursor: usize,
26}
27
28impl PaletteModal {
29    /// Palette over `entries`.
30    pub fn new(entries: Vec<(String, PaletteCmd)>) -> Self { PaletteModal { entries, query: TextField::default(), cursor: 0 } }
31    /// Entries matching the filter (every word, case-insensitive). An entry whose label
32    /// ends in `[name]` takes one extra word after the words that match it: its argument
33    /// (`theme starter mine`).
34    pub fn filtered(&self) -> Vec<&(String, PaletteCmd)> {
35        let q = self.query.value.to_lowercase();
36        let words: Vec<&str> = q.split_whitespace().collect();
37        self.entries.iter().filter(|(l, _)| {
38            let l = l.to_lowercase();
39            if words.iter().all(|w| l.contains(w)) { return true; }
40            l.ends_with("[name]") && words.len() >= 2 && words[..words.len() - 1].iter().all(|w| l.contains(w))
41        }).collect()
42    }
43    /// The argument typed after an `[name]` entry's words (`theme starter mine` → `mine`).
44    pub fn argument(&self) -> Option<String> {
45        let (label, _) = self.filtered().get(self.cursor).copied()?;
46        let l = label.to_lowercase();
47        if !l.ends_with("[name]") { return None; }
48        let words: Vec<&str> = self.query.value.split_whitespace().collect();
49        let last = words.last()?;
50        (words.len() >= 2 && !l.contains(&last.to_lowercase())).then(|| last.to_string())
51    }
52    /// The entry Enter would run.
53    pub fn chosen(&self) -> Option<PaletteCmd> { self.filtered().get(self.cursor).map(|e| e.1) }
54}
55
56impl ModalState for PaletteModal {
57    fn title(&self) -> String { "Commands".into() }
58    fn on_key(&mut self, k: KeyEvent) -> Option<ModalResult> {
59        let n = self.filtered().len();
60        match k.code {
61            KeyCode::Esc => return Some(ModalResult::ClosedCancelled),
62            KeyCode::Enter => return (n > 0).then_some(ModalResult::ClosedSubmitted),
63            KeyCode::Down | KeyCode::Tab => self.cursor = cycle(self.cursor, n, 1),
64            KeyCode::Up | KeyCode::BackTab => self.cursor = cycle(self.cursor, n, -1),
65            _ => { if self.query.on_key(k) { self.cursor = 0; } }
66        }
67        None
68    }
69    fn on_click(&mut self, h: ModalHit) -> Option<ModalResult> {
70        match h { ModalHit::Item(i) => { self.cursor = i; Some(ModalResult::ClosedSubmitted) } _ => None }
71    }
72    fn on_scroll(&mut self, d: i32) { let n = self.filtered().len(); self.cursor = cycle(self.cursor, n, d.signum()); }
73}
