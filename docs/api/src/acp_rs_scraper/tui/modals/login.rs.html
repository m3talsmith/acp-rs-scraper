1//! Login modal: email, then the emailed one-time code (or the login link).
2//! The modal only collects input; the app runs the login and reports back.
3
4use super::*;
5use crate::tui::widgets::text::TextField;
6use crossterm::event::KeyCode;
7
8/// Which step the login is at.
9#[derive(Debug, Clone, PartialEq)]
10pub enum LoginStep {
11    /// Typing the email.
12    Email,
13    /// Waiting for the site (the text says what for).
14    Busy(String),
15    /// Typing the code from the email.
16    Code,
17}
18
19/// What the modal asks the app to do.
20#[derive(Debug, Clone, PartialEq)]
21pub enum LoginRequest {
22    /// Start the login for this email (sends the code).
23    SendEmail(String),
24    /// Submit this code / link.
25    SubmitCode(String),
26}
27
28/// Login state.
29#[derive(Debug, Clone, PartialEq)]
30pub struct LoginModal {
31    /// Step.
32    pub step: LoginStep,
33    /// Email field.
34    pub email: TextField,
35    /// Code field.
36    pub code: TextField,
37    /// Last error.
38    pub error: Option<String>,
39    /// Request for the app (taken after each event).
40    pub request: Option<LoginRequest>,
41    /// Shown above the fields (e.g. "Log in to start the scan").
42    pub reason: Option<String>,
43}
44
45impl LoginModal {
46    /// Login with `email` pre-filled.
47    pub fn new(email: Option<String>, reason: Option<String>) -> Self {
48        LoginModal { step: LoginStep::Email, email: TextField::new(email.as_deref().unwrap_or("")),
49            code: TextField::default(), error: None, request: None, reason }
50    }
51    /// The site sent the code: ask for it.
52    pub fn need_code(&mut self) { self.step = LoginStep::Code; self.error = None; }
53    /// The login failed: back to the step that can retry.
54    pub fn failed(&mut self, e: &str) {
55        self.error = Some(e.to_string());
56        self.code.set("");
57        self.step = LoginStep::Email;
58    }
59    fn submit(&mut self) {
60        match self.step {
61            LoginStep::Email => {
62                let e = self.email.value.trim().to_string();
63                if !e.contains('@') { self.error = Some("enter the email you log in to acp.us with".into()); return; }
64                self.error = None;
65                self.step = LoginStep::Busy(format!("Requesting a login code for {e}…"));
66                self.request = Some(LoginRequest::SendEmail(e));
67            }
68            LoginStep::Code => {
69                let c = self.code.value.trim().to_string();
70                if c.is_empty() { self.error = Some("paste the code (or the login link) from the email".into()); return; }
71                self.error = None;
72                self.step = LoginStep::Busy("Verifying…".into());
73                self.request = Some(LoginRequest::SubmitCode(c));
74            }
75            LoginStep::Busy(_) => {}
76        }
77    }
78}
79
80impl ModalState for LoginModal {
81    fn title(&self) -> String { "Log in to acp.us".into() }
82    fn on_key(&mut self, k: KeyEvent) -> Option<ModalResult> {
83        match k.code {
84            KeyCode::Esc => return Some(ModalResult::ClosedCancelled),
85            KeyCode::Enter => self.submit(),
86            _ => match self.step {
87                LoginStep::Email => { self.email.on_key(k); }
88                LoginStep::Code => { self.code.on_key(k); }
89                LoginStep::Busy(_) => {}
90            },
91        }
92        None
93    }
94    fn on_click(&mut self, h: ModalHit) -> Option<ModalResult> {
95        match h {
96            ModalHit::Button(0) => { self.submit(); None }
97            ModalHit::Button(_) => Some(ModalResult::ClosedCancelled),
98            _ => None,
99        }
100    }
101}
