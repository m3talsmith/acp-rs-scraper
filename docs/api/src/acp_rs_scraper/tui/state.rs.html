100    /// Start.
101    pub start: Instant,

103
104impl Anim {
105    /// 0.0 → 1.0 over [`TUI_ANIM_MS`] (eased).
106    pub fn progress(&self, now: Instant) -> f32 {
107        let t = (now.saturating_duration_since(self.start).as_millis() as f32 / TUI_ANIM_MS as f32).clamp(0.0, 1.0);
108        1.0 - (1.0 - t) * (1.0 - t)
109    }
110    /// Finished at `now`.
111    pub fn done(&self, now: Instant) -> bool { now.saturating_duration_since(self.start) >= Duration::from_millis(TUI_ANIM_MS) }
112}

255    /// State over `store`; `args` seeds the menu, `--nick(s)` the selection.
256    pub fn new(store: Box<dyn Store>, args: &Args, theme: Theme, session_saved: bool) -> Self {
257        let mut app = App {
258            store, nicks: Vec::new(), nick_cursor: 0, nick_offset: 0, selected: Vec::new(),
259            base: BodyView::default(), search: None, focus: Focus::Nicks,
260            menu: MenuState { menu: Menu::new(args), item: 0, open: None, opt: 0, editing: false },
261            modal: None, status: VecDeque::new(), show_log: false,
262            session: if session_saved { SessionState::Saved } else { SessionState::LoggedOut },
263            job: None, pending_scan: None, anim: None, hits: Vec::new(), theme, effects: Vec::new(),
264            page_rows: 5, email: None,
265        };
266        app.refresh_nicks();
267        let want = args.nick_list().unwrap_or_default();
268        if !want.is_empty() {
269            app.selected = want.iter().filter_map(|w| app.nicks.iter().find(|n| n.nick.eq_ignore_ascii_case(w)).map(|n| n.nick.clone())).collect();
270            if let Some(i) = app.selected.first().and_then(|s| app.nicks.iter().position(|n| &n.nick == s)) { app.nick_cursor = i; }
271            app.menu.menu.set_text(CommandId::Scrape, "nicks", &want.join(","));
272            app.load_base();
273        }
274        app
275    }

277    /// Add a status line (kept to [`TUI_STATUS_KEEP`]).
278    pub fn push_status(&mut self, s: impl Into<String>) {
279        let s = s.into();
280        for l in s.lines() {
281            self.status.push_back(l.to_string());
282            while self.status.len() > TUI_STATUS_KEEP { self.status.pop_front(); }
283        }
284    }

288        self.modal = Some(Modal::Message(MessageModal::error(title, text)));

291    // ------------------------------------------------------------ nicks
292    /// Re-read the nick bar; keep the selection that still exists.
293    pub fn refresh_nicks(&mut self) {
294        match self.store.nicks() {
295            Ok(n) => self.nicks = n,
296            Err(e) => { self.push_status(format!("error: reading nicks: {e:#}")); self.nicks.clear(); }
297        }
298        let keep: Vec<String> = self.selected.iter()
299            .filter_map(|s| self.nicks.iter().find(|n| n.nick.eq_ignore_ascii_case(s)).map(|n| n.nick.clone())).collect();
300        self.selected = keep;
301        self.nick_cursor = self.nick_cursor.min(self.nicks.len().saturating_sub(1));
302    }
303    /// Select only the n-th nick.
304    pub fn select_only(&mut self, i: usize) {
305        let Some(n) = self.nicks.get(i) else { return };
306        self.nick_cursor = i;
307        self.selected = vec![n.nick.clone()];
308        self.selection_changed();
309    }
310    /// Add / remove the n-th nick from the selection.
311    pub fn toggle_nick(&mut self, i: usize) {
312        let Some(n) = self.nicks.get(i).map(|n| n.nick.clone()) else { return };
313        self.nick_cursor = i;
314        if let Some(p) = self.selected.iter().position(|s| *s == n) { self.selected.remove(p); } else { self.selected.push(n); }
315        self.selection_changed();
316    }
317    /// Move the cursor by `d`, adding the passed nicks to the selection (Shift-↑/↓).
318    pub fn extend_selection(&mut self, d: i32) {
319        if self.nicks.is_empty() { return; }
320        let cur = self.nicks[self.nick_cursor].nick.clone();
321        if !self.selected.contains(&cur) { self.selected.push(cur); }
322        self.move_nick_cursor(d);
323        let n = self.nicks[self.nick_cursor].nick.clone();
324        if !self.selected.contains(&n) { self.selected.push(n); }
325        self.selection_changed();
326    }
327    /// Select every nick.
328    pub fn select_all(&mut self) {
329        self.selected = self.nicks.iter().map(|n| n.nick.clone()).collect();
330        self.selection_changed();
331    }
332    /// Move the nick cursor.
333    pub fn move_nick_cursor(&mut self, d: i32) {
334        if self.nicks.is_empty() { return; }
335        self.nick_cursor = (self.nick_cursor as i64 + d as i64).clamp(0, self.nicks.len() as i64 - 1) as usize;
336    }
337    fn selection_changed(&mut self) {
338        self.menu.menu.set_text(CommandId::Scrape, "nicks", &self.selected.join(","));
339        self.search = None;
340        self.load_base();
341    }
342    /// Reload the nick posts view for the selection.
343    pub fn load_base(&mut self) {
344        let posts = match self.store.posts_for(&self.selected) {
345            Ok(p) => p,
346            Err(e) => { self.push_status(format!("error: reading posts: {e:#}")); Vec::new() }
347        };
348        let who = self.selected.iter().map(|n| format!("@{n}")).collect::<Vec<_>>().join(", ");
349        self.base = BodyView { title: who, posts, ..BodyView::default() };
350        self.anim = None;
351    }

358    /// Replace the Body with the results of `f`.
359    pub fn run_search(&mut self, f: SearchFilter) {
360        if f.is_empty() { self.clear_search(); return; }
361        match self.store.search(&f) {
362            Ok(posts) => {
363                self.push_status(format!("search {}: {} posts", f.describe(), posts.len()));
364                self.search = Some(BodyView { title: format!("Search: {}", f.describe()), posts, filter: Some(f), ..BodyView::default() });
365                self.focus = Focus::Body;
366                self.anim = None;
367            }
368            Err(e) => self.error("Search failed", &format!("{e:#}")),
369        }
370    }

382    /// Move the post cursor by `d` (detail closed).
383    pub fn move_post_cursor(&mut self, d: i32) {
384        let rows = self.page_rows.max(1);
385        let v = self.view_mut();
386        if v.posts.is_empty() { return; }
387        v.cursor = (v.cursor as i64 + d as i64).clamp(0, v.posts.len() as i64 - 1) as usize;
388        if v.cursor < v.offset { v.offset = v.cursor; }
389        if v.cursor >= v.offset + rows { v.offset = v.cursor + 1 - rows; }
390    }
391    /// Scroll the post list without moving the cursor past the screen.
392    pub fn scroll_posts(&mut self, d: i32) {
393        let rows = self.page_rows.max(1);
394        let v = self.view_mut();
395        let max = v.posts.len().saturating_sub(1) as i64;
396        v.offset = (v.offset as i64 + d as i64).clamp(0, max) as usize;
397        v.cursor = v.cursor.clamp(v.offset, (v.offset + rows - 1).min(max as usize));
398    }

400    // ------------------------------------------------------------ detail stack
401    /// Build a detail frame for `post` (attachments, links resolved against the store).
402    pub fn frame_for(&self, post: Post) -> DetailFrame {
403        let s = &self.store;
404        let mut links = Vec::new();
405        let mut seen: HashSet<String> = HashSet::from([post.post_id.clone()]);
406        let mut add = |kind, id: &str, links: &mut Vec<Link>| {
407            if id.is_empty() || !seen.insert(id.to_string()) { return; }
408            links.push(Link { kind, id: id.into(), target: s.post(id).ok().flatten() });
409        };
410        let direct = if post.in_reply_to_id.is_empty() { post.parent_post_id.clone() } else { post.in_reply_to_id.clone() };
411        add(LinkKind::ReplyTo, &direct, &mut links);
412        add(LinkKind::Thread, &post.parent_post_id, &mut links);
413        for id in crate::tui::data::linked_ids(&post.content) { add(LinkKind::Mentioned, &id, &mut links); }
414        for r in s.replies(&post.post_id).unwrap_or_default() {
415            if seen.insert(r.post_id.clone()) { links.push(Link { kind: LinkKind::Reply, id: r.post_id.clone(), target: Some(r) }); }
416        }
417        DetailFrame { attachments: s.attachments(&post.post_id).unwrap_or_default(), post, links,
418            link_cursor: 0, scroll: 0, follow_link: false, notice: None }
419    }
420    /// Open the n-th post of the view as the root detail (replacing any open stack).
421    pub fn open_post(&mut self, i: usize) {
422        let Some(p) = self.view().posts.get(i).cloned() else { return };
423        let f = self.frame_for(p);
424        let v = self.view_mut();
425        v.cursor = i;
426        v.stack = vec![f];
427        self.focus = Focus::Body;
428        self.anim = Some(Anim { kind: AnimKind::Open, start: Instant::now() });
429    }
430    /// Top frame.
431    pub fn top(&self) -> Option<&DetailFrame> { self.view().stack.last() }
432    /// Mutable top frame.
433    pub fn top_mut(&mut self) -> Option<&mut DetailFrame> { self.view_mut().stack.last_mut() }
434    /// Open the n-th link of the top frame on top of the stack.
435    pub fn open_link(&mut self, i: usize) {
436        let depth = self.view().stack.len();
437        let Some(link) = self.top().and_then(|f| f.links.get(i)).cloned() else { return };
438        if let Some(f) = self.top_mut() { f.link_cursor = i; }
439        let Some(target) = link.target else {
440            if let Some(f) = self.top_mut() { f.notice = Some(format!("{} is not in the database", link.id)); }
441            self.push_status(format!("{} is not in the database (scrape its thread to read it)", link.id));
442            return;
443        };
444        if depth >= MAX_DETAIL_DEPTH {
445            self.push_status(format!("at most {MAX_DETAIL_DEPTH} nested posts: press Esc to close one first"));
446            if let Some(f) = self.top_mut() { f.notice = Some(format!("nesting limit ({MAX_DETAIL_DEPTH}) reached")); }
447            return;
448        }
449        let f = self.frame_for(target);
450        self.view_mut().stack.push(f);
451        self.anim = Some(Anim { kind: AnimKind::Push, start: Instant::now() });
452    }
453    /// Close the top frame; returns whether one was open.
454    pub fn pop_detail(&mut self) -> bool {
455        let v = self.view_mut();
456        let r = v.stack.pop().is_some();
457        self.anim = None;
458        r
459    }
460    /// Keep frames `0..=depth` (click on a collapsed header).
461    pub fn pop_to(&mut self, depth: usize) {
462        self.view_mut().stack.truncate(depth + 1);
463        self.anim = None;
464    }
465    /// Move the top frame's link cursor.
466    pub fn move_link(&mut self, d: i32) {
467        if let Some(f) = self.top_mut() && !f.links.is_empty() {
468            f.link_cursor = cycle(f.link_cursor, f.links.len(), d);
469            f.follow_link = true;
470        }
471    }
472    /// Scroll the top frame.
473    pub fn scroll_detail(&mut self, d: i32) {
474        if let Some(f) = self.top_mut() { f.scroll = (f.scroll as i32 + d).max(0) as u16; f.follow_link = false; }
475    }
476    /// Advance / finish the animation.
477    pub fn tick(&mut self, now: Instant) {
478        if self.anim.is_some_and(|a| a.done(now)) { self.anim = None; }
479    }
480    /// Animation progress (1.0 when none runs).
481    pub fn anim_progress(&self, now: Instant) -> (Option<AnimKind>, f32) {
482        match self.anim { Some(a) => (Some(a.kind), a.progress(now)), None => (None, 1.0) }
483    }

485    // ------------------------------------------------------------ actions
486    /// Run a menu / key / palette action.
487    pub fn act(&mut self, a: Action) {
488        let sel = self.selected.clone();
489        match a {
490            Action::StartScan => {
491                if self.job.is_some() { self.effects.push(Effect::CancelJob); return; }
492                match self.menu.menu.scrape_argv(&sel) {
493                    Err(e) => self.error("Cannot start scanning", &e),
494                    Ok(argv) => {
495                        self.close_menu();
496                        if self.session == SessionState::Valid { self.effects.push(Effect::RunJob(JobKind::Scan, argv)); }
497                        else { self.session = SessionState::Checking; self.effects.push(Effect::CheckSessionThenScan(argv)); }
498                    }
499                }
500            }
501            Action::CancelJob => if self.job.is_some() { self.effects.push(Effect::CancelJob); },
502            Action::RunSearch => match self.menu.menu.search_filter() {
503                Ok(f) => { self.close_menu(); self.run_search(f); }
504                Err(e) => self.error("Search", &e),
505            },
506            Action::OpenSearchModal => { self.close_menu(); self.modal = Some(Modal::Search(SearchModal::new(&self.current_filter()))); }
507            Action::ClearSearch => { self.close_menu(); self.clear_search(); }
508            Action::RunExport => {
509                let view = self.search.as_ref().and_then(|v| v.filter.clone());
510                match self.menu.menu.export_argv(&sel, view.as_ref()) {
511                    Ok(argv) => { self.close_menu(); self.start_job(JobKind::Export, argv); }
512                    Err(e) => self.error("Cannot export", &e),
513                }
514            }
515            Action::RunImport => match self.menu.menu.import_argv() {
516                Ok(argv) => { self.close_menu(); self.start_job(JobKind::Import, argv); }
517                Err(e) => self.error("Cannot import", &e),
518            },
519            Action::RefreshNicks => { self.close_menu(); self.refresh_nicks(); self.load_base_keep(); self.push_status(format!("{} nicks", self.nicks.len())); }
520            Action::RemoveNicks => {
521                let m = self.menu.menu.clone();
522                match (m.remove_nicks(&sel), m.remove_argv(&sel)) {
523                    (Ok(n), Ok(argv)) => {
524                        self.close_menu();
525                        let skip = m.check(CommandId::Maintain, "yes") || m.check(CommandId::Maintain, "dry_run");
526                        if skip { self.start_job(JobKind::Maintain, argv); return; }
527                        let who = n.iter().map(|n| format!("@{n}")).collect::<Vec<_>>().join(", ");
528                        let mut lines = vec![format!("Delete {who}: their posts, replies in their threads, attachments, mentions and profiles?")];
529                        if m.check(CommandId::Maintain, "keep_replies") { lines.push("Other people's replies in their threads are kept.".into()); }
530                        lines.push("Tick Maintain › Dry run first to preview the counts.".into());
531                        self.modal = Some(Modal::Confirm(ConfirmModal::new("Remove nicks", lines, "Remove", ConfirmAction::Job(JobKind::Maintain, argv))));
532                    }
533                    (Err(e), _) | (_, Err(e)) => self.error("Cannot remove", &e),
534                }
535            }
536            Action::ClearAll => match self.menu.menu.clear_all_argv() {
537                Ok(argv) => {
538                    self.close_menu();
539                    let m = &self.menu.menu;
540                    if m.check(CommandId::Maintain, "dry_run") || m.check(CommandId::Maintain, "yes") { self.start_job(JobKind::Maintain, argv); return; }
541                    let mut c = ConfirmModal::new("Clear all", vec![
542                        "Delete the whole database and every downloaded attachment?".into(),
543                        "preferences.toml, the saved session and exports are kept.".into(),
544                        "Type yes to confirm.".into()], "Delete everything", ConfirmAction::Job(JobKind::Maintain, argv));
545                    c.strict = true;
546                    c.focus = 0;
547                    self.modal = Some(Modal::Confirm(c));
548                }
549                Err(e) => self.error("Cannot clear", &e),
550            },
551            Action::Login => { self.close_menu(); self.open_login(None); }
552            Action::Logout => { self.close_menu(); self.effects.push(Effect::Logout); }
553            Action::Help => { self.close_menu(); self.modal = Some(Modal::Help(HelpModal::new())); }
554            Action::ToggleLog => { self.show_log = !self.show_log; }
555            Action::Quit => {
556                self.close_menu();
557                if let Some(j) = &self.job {
558                    let mut c = ConfirmModal::new("Quit", vec![
559                        format!("A {} is running (pid {}).", j.kind.name(), j.pid),
560                        "Keep it running in the background, or cancel it and quit?".into(),
561                        format!("Its log: {}", j.log.display())], "Keep running & quit", ConfirmAction::Quit);
562                    c.buttons = vec!["Keep running & quit".into(), "Cancel it & quit".into(), "Stay".into()];
563                    c.focus = 0;
564                    self.modal = Some(Modal::Confirm(c));
565                } else {
566                    self.effects.push(Effect::Quit { cancel_job: false });
567                }
568            }
569        }
570    }
571    fn load_base_keep(&mut self) {
572        let (c, o) = (self.base.cursor, self.base.offset);
573        self.load_base();
574        self.base.cursor = c.min(self.base.posts.len().saturating_sub(1));
575        self.base.offset = o.min(self.base.cursor);
576    }
577    fn start_job(&mut self, kind: JobKind, argv: Vec<String>) {
578        if self.job.is_some() { self.error("Busy", "another job is still running; wait for it or cancel it (F5)"); return; }
579        self.effects.push(Effect::RunJob(kind, argv));
580    }
581    /// Open the Login modal; `then_scan` runs after a successful login.
582    pub fn open_login(&mut self, then_scan: Option<Vec<String>>) {
583        let reason = then_scan.as_ref().map(|_| "Log in to start the scan.".to_string());
584        self.pending_scan = then_scan;
585        self.modal = Some(Modal::Login(LoginModal::new(self.email.clone(), reason)));
586    }
587    /// The session check before a scan came back.
588    pub fn session_checked(&mut self, r: std::result::Result<bool, String>, argv: Vec<String>) {
589        match r {
590            Ok(true) => { self.session = SessionState::Valid; self.effects.push(Effect::RunJob(JobKind::Scan, argv)); }
591            Ok(false) => { self.session = SessionState::LoggedOut; self.open_login(Some(argv)); }
592            Err(e) => { self.session = SessionState::Saved; self.error("Cannot reach acp.us", &e); }
593        }
594    }
595    /// The site sent the login code.
596    pub fn login_need_code(&mut self) {
597        if let Some(Modal::Login(m)) = &mut self.modal { m.need_code(); }
598    }
599    /// The login finished.
600    pub fn login_done(&mut self, r: std::result::Result<(), String>) {
601        let Some(Modal::Login(m)) = &mut self.modal else { return };
602        match r {
603            Ok(()) => {
604                self.session = SessionState::Valid;
605                self.email = Some(m.email.value.trim().to_string()).filter(|e| !e.is_empty()).or(self.email.take());
606                self.close_modal(ModalResult::ClosedSubmitted);
607            }
608            Err(e) => m.failed(&e),
609        }
610    }
611    /// Logged out.
612    pub fn logged_out(&mut self, removed: bool) {
613        self.session = SessionState::LoggedOut;
614        self.push_status(if removed { "logged out: saved session removed" } else { "logged out (no saved session)" });
615        self.open_login(None);
616    }
617    /// A job started.
618    pub fn job_started(&mut self, j: JobView, cmd: &str) {
619        self.push_status(format!("started {} (pid {}): {cmd}", j.kind.name(), j.pid));
620        self.job = Some(j);
621    }
622    /// The job ended (`ok` = exit 0); `tail` = its last log lines.
623    pub fn job_finished(&mut self, ok: bool, code: Option<i32>, tail: &[String]) {
624        let Some(j) = self.job.take() else { return };
625        self.refresh_nicks();
626        self.load_base_keep();
627        if ok {
628            self.push_status(format!("{} finished", j.kind.name()));
629        } else {
630            let code = code.map(|c| c.to_string()).unwrap_or_else(|| "killed".into());
631            let mut text = format!("The {} stopped (exit {code}). Last lines of {}:\n", j.kind.name(), j.log.display());
632            for l in tail { text.push_str(&format!("{l}\n")); }
633            self.error(&format!("{} failed", j.kind.name()), text.trim_end());
634        }
635    }
636    /// The job was cancelled.
637    pub fn job_cancelled(&mut self) {
638        if let Some(j) = self.job.take() { self.push_status(format!("{} cancelled (pid {})", j.kind.name(), j.pid)); }
639        self.refresh_nicks();
640        self.load_base_keep();
641    }

644    /// Close the dropdown and leave the menu.
645    pub fn close_menu(&mut self) {
646        self.menu.open = None;
647        self.menu.editing = false;
648        if self.focus == Focus::Menu { self.focus = if self.view().posts.is_empty() { Focus::Nicks } else { Focus::Body }; }
649    }
650    /// Highlight / open the n-th menu item (option-less items run at once).
651    pub fn open_menu(&mut self, i: usize) {
652        let Some(c) = self.menu.menu.items.get(i) else { return };
653        self.focus = Focus::Menu;
654        self.menu.item = i;
655        self.menu.editing = false;
656        if let Some(a) = c.action { self.menu.open = None; self.act(a); return; }
657        if self.menu.open == Some(i) { self.menu.open = None; } else { self.menu.open = Some(i); self.menu.opt = 0; }
658    }
659    /// Activate the highlighted option of the open dropdown (Enter / click).
660    pub fn activate_opt(&mut self, i: usize) {
661        use crate::tui::commands::Widget;
662        let Some(item) = self.menu.open else { return };
663        self.menu.opt = i;
664        let Some(o) = self.menu.menu.items[item].opts.get_mut(i) else { return };
665        match &mut o.widget {
666            Widget::Check(v) => *v = !*v,
667            Widget::Choice { choices, idx } => *idx = (*idx + 1) % choices.len(),
668            Widget::Text(_) => self.menu.editing = !self.menu.editing,
669            Widget::Button(a) => { let a = *a; self.act(a); }
670        }
671    }
672    /// Close the modal with `r` and apply it.
673    pub fn close_modal(&mut self, r: ModalResult) {
674        let Some(m) = self.modal.take() else { return };
675        match (m, r) {
676            (Modal::Login(_), ModalResult::ClosedSubmitted) => {
677                self.push_status("logged in");
678                if let Some(argv) = self.pending_scan.take() { self.effects.push(Effect::RunJob(JobKind::Scan, argv)); }
679            }
680            (Modal::Login(m), ModalResult::ClosedCancelled) => {
681                self.pending_scan = None;
682                if self.session == SessionState::Checking { self.session = SessionState::LoggedOut; }
683                let _ = m;
684                self.effects.push(Effect::LoginCancel);
685            }
686            (Modal::Confirm(c), ModalResult::ClosedSubmitted) => match c.action {
687                ConfirmAction::Job(k, argv) => self.start_job(k, argv),
688                ConfirmAction::Quit => self.effects.push(Effect::Quit { cancel_job: c.chosen == 1 }),
689            },
690            (Modal::Search(s), ModalResult::ClosedSubmitted) => {
691                if s.cleared { self.clear_search(); }
692                else if let Ok(f) = s.filter() { self.run_search(f); }
693            }
694            (Modal::Palette(p), ModalResult::ClosedSubmitted) => match p.chosen() {
695                Some(PaletteCmd::Action(a)) => self.act(a),
696                Some(PaletteCmd::OpenMenu(i)) => self.open_menu(i),
697                None => {}
698            },
699            _ => {}
700        }
701    }
702    /// Every command for the palette.
703    pub fn palette_entries(&self) -> Vec<(String, PaletteCmd)> {
704        use crate::tui::commands::Widget;
705        let mut v = vec![
706            ("Start Scanning (F5)".to_string(), PaletteCmd::Action(Action::StartScan)),
707            ("Search… (Shift-S)".into(), PaletteCmd::Action(Action::OpenSearchModal)),
708            ("Clear search (Ctrl-S)".into(), PaletteCmd::Action(Action::ClearSearch)),
709            ("Toggle log panel (l)".into(), PaletteCmd::Action(Action::ToggleLog)),
710        ];
711        for (i, c) in self.menu.menu.items.iter().enumerate() {
712            if c.opts.is_empty() { v.push((c.label.to_string(), PaletteCmd::OpenMenu(i))); continue; }
713            v.push((format!("{} › options", c.label), PaletteCmd::OpenMenu(i)));
714            for o in &c.opts {
715                if let Widget::Button(a) = o.widget { v.push((format!("{} › {}", c.label, o.label), PaletteCmd::Action(a))); }
716            }
717        }
718        v
719    }
