115impl Anim {
116    /// 0.0 → 1.0 over [`TUI_ANIM_MS`] (eased).
117    pub fn progress(&self, now: Instant) -> f32 {
118        let t = (now.saturating_duration_since(self.start).as_millis() as f32 / TUI_ANIM_MS as f32).clamp(0.0, 1.0);
119        1.0 - (1.0 - t) * (1.0 - t)
120    }
121    /// Finished at `now`.
122    pub fn done(&self, now: Instant) -> bool { now.saturating_duration_since(self.start) >= Duration::from_millis(TUI_ANIM_MS) }
123}

125/// How the Body lays out posts (View menu, `v`, or the nav bar's `[View: …]`).
126#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
127pub enum ViewMode {
128    /// Cards; a detail slides in and linked posts nest inside it.
129    #[default]
130    Nested,
131    /// One endless column of full posts; a detail replaces the list (no nesting).
132    Stack,
133    /// The stack on top and the clicked post in a content pane below.
134    Pane,
135    /// One post at a time, side to side; wraps around at the ends.
136    CarouselInfinite,
137    /// One post at a time, side to side; stops at the ends.
138    CarouselFixed,
139    /// A grid of cards; a post opens in a modal with a Close button.
140    Grid,
141}
142
143impl ViewMode {
144    /// Every mode, in `v` order.
145    pub const ALL: [ViewMode; 6] = [ViewMode::Nested, ViewMode::Stack, ViewMode::Pane, ViewMode::CarouselInfinite, ViewMode::CarouselFixed, ViewMode::Grid];
146    /// Label.
147    pub fn label(self) -> &'static str {
148        match self {
149            ViewMode::Nested => "Nested", ViewMode::Stack => "Stack", ViewMode::Pane => "Stack + pane",
150            ViewMode::CarouselInfinite => "Carousel ∞", ViewMode::CarouselFixed => "Carousel", ViewMode::Grid => "Grid",
151        }
152    }
153    /// The next mode (`d` = ±1).
154    pub fn cycle(self, d: i32) -> ViewMode {
155        let i = Self::ALL.iter().position(|m| *m == self).unwrap_or(0);
156        Self::ALL[cycle(i, Self::ALL.len(), d)]
157    }
158    /// Posts move with ←/→.
159    pub fn horizontal(self) -> bool { matches!(self, ViewMode::CarouselInfinite | ViewMode::CarouselFixed | ViewMode::Grid) }
160}

400    pub fn refresh_nicks(&mut self) {
401        match self.store.nicks() {
402            Ok(n) => self.all_nicks = n,
403            Err(e) => { self.push_status(format!("error: reading nicks: {e:#}")); self.all_nicks.clear(); }
404        }
405        // Selected nicks hidden by the filter stay selected.
406        let keep: Vec<String> = self.selected.iter()
407            .filter_map(|s| self.all_nicks.iter().find(|n| n.nick.eq_ignore_ascii_case(s)).map(|n| n.nick.clone())).collect();
408        self.selected = keep;
409        let cur = self.nicks.get(self.nick_cursor).map(|n| n.nick.clone());
410        self.nicks = crate::tui::data::filter_nicks(&self.all_nicks, &self.nick_filter.value);
411        self.nick_cursor = cur.and_then(|c| self.nicks.iter().position(|n| n.nick == c))
412            .unwrap_or(self.nick_cursor).min(self.nicks.len().saturating_sub(1));
413    }

562        let in_search = self.search.as_ref().is_some_and(|s| s.filter.is_some());
563        let label = |p: &Post| format!("This {} {} · @{}", if in_search { "search result" } else { kind(p) }, p.post_id, p.nick);
564        if let Some(f) = self.top() {
565            if !f.post.post_id.starts_with("research:") { v.push((label(&f.post), ResearchSource::Posts(vec![f.post.post_id.clone()]))); }
566            let mut seen: HashSet<String> = HashSet::new();
567            for l in &f.links {
568                match l.kind {
569                    LinkKind::Attachment => {
570                        let p = Path::new(&l.id);
571                        if p.exists() && seen.insert(l.id.clone()) {
572                            let name = p.file_name().map(|n| n.to_string_lossy().into_owned()).unwrap_or_default();
573                            v.push((format!("Attachment file {name}"), ResearchSource::File(l.id.clone())));
574                        }
575                    }
576                    _ => if let Some(t) = &l.target && seen.insert(t.post_id.clone()) && t.post_id != f.post.post_id {
577                        let what = match l.kind { LinkKind::ReplyTo => "Replied-to", LinkKind::Thread => "Thread", LinkKind::Reply => "Reply", _ => "Linked" };
578                        v.push((format!("{what} {} {} · @{}", kind(t), t.post_id, t.nick), ResearchSource::Posts(vec![t.post_id.clone()])));
579                    }
580                }
581            }
582            for u in web_links(&f.post.content) {
583                if seen.insert(u.clone()) { v.push((format!("Web page {u}"), ResearchSource::Url(u))); }
584            }
585        } else {
586            let view = self.view();
587            if let Some(p) = view.posts.get(view.cursor).filter(|p| !p.post_id.starts_with("research:")) {
588                v.push((label(p), ResearchSource::Posts(vec![p.post_id.clone()])));
589            }
590        }
591        if let Some(all) = self.view_source() { v.push(all); }

687    /// Start an automatic sync when Sync › Auto sync is on and one is due. Skips (and
688    /// reschedules) while a job runs or when logged out; never opens a modal.
689    pub fn sync_tick(&mut self, now: Instant) {
690        if !self.menu.menu.check(CommandId::Sync, "auto") { self.sync_next = None; return; }
691        if self.sync_next.is_some_and(|t| now < t) { return; }
692        let every = match self.menu.menu.sync_interval() {
693            Ok(d) => d,
694            Err(e) => { self.menu.menu.set_check(CommandId::Sync, "auto", false); self.push_status(format!("auto sync off: {e}")); return; }
695        };
696        self.sync_next = Some(now + every);
697        if self.job.is_some() { self.push_status("auto sync skipped: a job is running"); return; }
698        if self.session == SessionState::LoggedOut { self.push_status("auto sync skipped: logged out (Session › Log in)"); return; }
699        match self.menu.menu.sync_argv(&self.selected.clone()) {
700            Ok(argv) => { self.push_status("auto sync…"); self.effects.push(Effect::AutoSync(argv)); }
701            Err(e) => self.push_status(format!("auto sync skipped: {e}")),
702        }
703    }

710            Err(e) => self.push_status(format!("auto sync skipped: offline ({})", e.lines().next().unwrap_or(""))),

794    /// Build a detail frame for `post` (attachments, links resolved against the store).
795    pub fn frame_for(&self, post: Post) -> DetailFrame {
796        let s = &self.store;
797        let mut links = Vec::new();
798        let mut seen: HashSet<String> = HashSet::from([post.post_id.clone()]);
799        let mut add = |kind, id: &str, links: &mut Vec<Link>| {
800            if id.is_empty() || !seen.insert(id.to_string()) { return; }
801            links.push(Link { kind, id: id.into(), target: s.post(id).ok().flatten() });
802        };
803        let direct = if post.in_reply_to_id.is_empty() { post.parent_post_id.clone() } else { post.in_reply_to_id.clone() };
804        add(LinkKind::ReplyTo, &direct, &mut links);
805        add(LinkKind::Thread, &post.parent_post_id, &mut links);
806        for id in crate::tui::data::linked_ids(&post.content) { add(LinkKind::Mentioned, &id, &mut links); }
807        for r in s.replies(&post.post_id).unwrap_or_default() {
808            if seen.insert(r.post_id.clone()) { links.push(Link { kind: LinkKind::Reply, id: r.post_id.clone(), target: Some(r) }); }
809        }
810        let attachments = s.attachments(&post.post_id).unwrap_or_default();
811        for a in &attachments {
812            links.push(Link { kind: LinkKind::Attachment, id: a.local_path.display().to_string(), target: None });
813        }
814        DetailFrame { attachments, post, links,
815            link_cursor: 0, scroll: 0, follow_link: false, notice: None }
816    }

832    /// Open the n-th link of the top frame on top of the stack.
833    pub fn open_link(&mut self, i: usize) {
834        let depth = self.view().stack.len();
835        let Some(link) = self.top().and_then(|f| f.links.get(i)).cloned() else { return };
836        if let Some(f) = self.top_mut() { f.link_cursor = i; }
837        if link.kind == LinkKind::Attachment {
838            let p = PathBuf::from(&link.id);
839            if p.exists() { self.push_status(format!("opening {}", p.display())); self.effects.push(Effect::OpenFile(p)); }
840            else {
841                if let Some(f) = self.top_mut() { f.notice = Some(format!("{} is missing (re-scrape to download it)", link.id)); }
842                self.push_status(format!("{} is missing", link.id));
843            }
844            return;
845        }
846        let Some(target) = link.target else {
847            if let Some(f) = self.top_mut() { f.notice = Some(format!("{} is not in the database", link.id)); }
848            self.push_status(format!("{} is not in the database (scrape its thread to read it)", link.id));
849            return;
850        };
851        let max = if self.view_mode == ViewMode::Nested { MAX_DETAIL_DEPTH } else { MAX_FLAT_DEPTH };
852        if depth >= max {
853            self.push_status(format!("at most {max} open posts here: press Esc to close one first"));
854            let what = if max == MAX_DETAIL_DEPTH { "nesting" } else { "depth" };
855            if let Some(f) = self.top_mut() { f.notice = Some(format!("{what} limit ({max}) reached")); }
856            return;
857        }
858        let f = self.frame_for(target);
859        let v = self.view_mut();
860        v.stack.push(f);
861        v.forward.clear();
862        self.anim = Some(Anim { kind: AnimKind::Push, start: Instant::now() });
863    }

1219/// http(s) links in a body, except links to posts on the site itself.
1220pub fn web_links(text: &str) -> Vec<String> {
1221    use std::sync::LazyLock;
1222    static RE: LazyLock<regex::Regex> = LazyLock::new(|| regex::Regex::new(r"https?://[^\s<>()\[\]\x22']+").expect("url re"));
1223    let mut seen = HashSet::new();
1224    RE.find_iter(text).map(|m| m.as_str().trim_end_matches(['.', ',', ';', ':', '!', '?']).to_string())
1225        .filter(|u| !u.contains("/rs/post/"))
1226        .filter(|u| seen.insert(u.clone())).collect()
1227}

1229/// A research file or web page shown as a card: its note, where it came from, and its text.
1230pub fn research_post(it: &ResearchItem) -> Post {
1231    let mut body = String::new();
1232    if !it.note.trim().is_empty() { body.push_str(&format!("> {}\n\n", it.note.trim())); }
1233    body.push_str(&it.r#ref);
1234    if let Some(s) = it.meta["size"].as_u64() { body.push_str(&format!(" · {s} bytes")); }
1235    if let Some(m) = it.meta["mime_type"].as_str().or(it.meta["content_type"].as_str()) { body.push_str(&format!(" · {m}")); }
1236    if !it.content.trim().is_empty() {
1237        let text: String = it.content.chars().take(6000).collect();
1238        body.push_str("\n\n");
1239        body.push_str(&text);
1240        if it.content.chars().count() > 6000 { body.push_str("\n\n…"); }
1241    }
1242    let nick = match it.kind { ItemKind::WebUrl => "web", ItemKind::LocalFile => "file", _ => "missing post" };
1243    let content = if it.kind.is_post() { format!("{} is not in the database any more.\n\n{body}", it.r#ref) } else { body };
1244    Post { post_id: format!("research:{}", it.id), parent_post_id: String::new(), in_reply_to_id: String::new(), nick: nick.into(),
1245        title: it.title.clone(), content, posted_at: it.added_at.clone(), media: Vec::new(), truncated: false }
