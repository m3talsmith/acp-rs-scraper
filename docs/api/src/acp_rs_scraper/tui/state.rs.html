101/// A running animation.
102#[derive(Clone, Copy, Debug, PartialEq)]
103pub struct Anim {
104    /// Kind.
105    pub kind: AnimKind,
106    /// Start.
107    pub start: Instant,
108}
109
110impl Anim {
111    /// 0.0 → 1.0 over [`TUI_ANIM_MS`] (eased).
112    pub fn progress(&self, now: Instant) -> f32 {
113        let t = (now.saturating_duration_since(self.start).as_millis() as f32 / TUI_ANIM_MS as f32).clamp(0.0, 1.0);
114        1.0 - (1.0 - t) * (1.0 - t)
115    }
116    /// Finished at `now`.
117    pub fn done(&self, now: Instant) -> bool { now.saturating_duration_since(self.start) >= Duration::from_millis(TUI_ANIM_MS) }
118}

120/// How the Body lays out posts (View menu, `v`, or the nav bar's `[View: …]`).
121#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
122pub enum ViewMode {
123    /// Cards; a detail slides in and linked posts nest inside it.
124    #[default]
125    Nested,
126    /// One endless column of full posts; a detail replaces the list (no nesting).
127    Stack,
128    /// The stack on top and the clicked post in a content pane below.
129    Pane,
130    /// One post at a time, side to side; wraps around at the ends.
131    CarouselInfinite,
132    /// One post at a time, side to side; stops at the ends.
133    CarouselFixed,
134    /// A grid of cards; a post opens in a modal with a Close button.
135    Grid,
136}
137
138impl ViewMode {
139    /// Every mode, in `v` order.
140    pub const ALL: [ViewMode; 6] = [ViewMode::Nested, ViewMode::Stack, ViewMode::Pane, ViewMode::CarouselInfinite, ViewMode::CarouselFixed, ViewMode::Grid];
141    /// Label.
142    pub fn label(self) -> &'static str {
143        match self {
144            ViewMode::Nested => "Nested", ViewMode::Stack => "Stack", ViewMode::Pane => "Stack + pane",
145            ViewMode::CarouselInfinite => "Carousel ∞", ViewMode::CarouselFixed => "Carousel", ViewMode::Grid => "Grid",
146        }
147    }
148    /// The next mode (`d` = ±1).
149    pub fn cycle(self, d: i32) -> ViewMode {
150        let i = Self::ALL.iter().position(|m| *m == self).unwrap_or(0);
151        Self::ALL[cycle(i, Self::ALL.len(), d)]
152    }
153    /// Posts move with ←/→.
154    pub fn horizontal(self) -> bool { matches!(self, ViewMode::CarouselInfinite | ViewMode::CarouselFixed | ViewMode::Grid) }
155}

344    /// State over `store`; `args` seeds the menu, `--nick(s)` the selection.
345    pub fn new(store: Box<dyn Store>, args: &Args, theme: Theme, session_saved: bool) -> Self {
346        let mut app = App {
347            store, nicks: Vec::new(), all_nicks: Vec::new(), nick_filter: Default::default(), filter_editing: false, nick_cursor: 0, nick_offset: 0, selected: Vec::new(),
348            base: BodyView::default(), search: None, focus: Focus::Nicks,
349            menu: MenuState { menu: Menu::new(args), item: 0, open: None, opt: 0, editing: false },
350            modal: None, status: VecDeque::new(), show_log: false,
351            session: if session_saved { SessionState::Saved } else { SessionState::LoggedOut },
352            job: None, pending_job: None, marks: HashSet::new(), view_mode: ViewMode::Nested, grid_cols: 1, sync_next: None, auto_running: false, anim: None, hits: Vec::new(), theme, effects: Vec::new(), osc_links: Vec::new(), images: None,
353            page_rows: 5, email: None,
354        };
355        app.refresh_nicks();
356        app.reload_marks();
357        let want = args.nick_list().unwrap_or_default();
358        if !want.is_empty() {
359            app.selected = want.iter().filter_map(|w| app.nicks.iter().find(|n| n.nick.eq_ignore_ascii_case(w)).map(|n| n.nick.clone())).collect();
360            if let Some(i) = app.selected.first().and_then(|s| app.nicks.iter().position(|n| &n.nick == s)) { app.nick_cursor = i; }
361            app.menu.menu.set_text(CommandId::Scrape, "nicks", &want.join(","));
362            app.load_base();
363        }
364        app
365    }

369        let s = s.into();
370        for l in s.lines() {
371            self.status.push_back(l.to_string());
372            while self.status.len() > TUI_STATUS_KEEP { self.status.pop_front(); }

381    // ------------------------------------------------------------ nicks
382    /// Re-read the nick bar; keep the selection that still exists.
383    pub fn refresh_nicks(&mut self) {
384        match self.store.nicks() {
385            Ok(n) => self.all_nicks = n,
386            Err(e) => { self.push_status(format!("error: reading nicks: {e:#}")); self.all_nicks.clear(); }
387        }
388        // Selected nicks hidden by the filter stay selected.
389        let keep: Vec<String> = self.selected.iter()
390            .filter_map(|s| self.all_nicks.iter().find(|n| n.nick.eq_ignore_ascii_case(s)).map(|n| n.nick.clone())).collect();
391        self.selected = keep;
392        let cur = self.nicks.get(self.nick_cursor).map(|n| n.nick.clone());
393        self.nicks = crate::tui::data::filter_nicks(&self.all_nicks, &self.nick_filter.value);
394        self.nick_cursor = cur.and_then(|c| self.nicks.iter().position(|n| n.nick == c))
395            .unwrap_or(self.nick_cursor).min(self.nicks.len().saturating_sub(1));
396    }
397    /// Re-apply the filter after it changed: the cursor stays on its nick when still shown, else the first match.
398    pub fn apply_nick_filter(&mut self) {
399        let cur = self.nicks.get(self.nick_cursor).map(|n| n.nick.clone());
400        self.nicks = crate::tui::data::filter_nicks(&self.all_nicks, &self.nick_filter.value);
401        self.nick_cursor = cur.and_then(|c| self.nicks.iter().position(|n| n.nick == c)).unwrap_or(0);
402        self.nick_offset = 0;
403    }

426    /// Add / remove the n-th nick from the selection.
427    pub fn toggle_nick(&mut self, i: usize) {
428        let Some(n) = self.nicks.get(i).map(|n| n.nick.clone()) else { return };
429        self.nick_cursor = i;

433    /// Move the cursor by `d`, adding the passed nicks to the selection (Shift-↑/↓).
434    pub fn extend_selection(&mut self, d: i32) {
435        if self.nicks.is_empty() { return; }
436        let cur = self.nicks[self.nick_cursor].nick.clone();
437        if !self.selected.contains(&cur) { self.selected.push(cur); }
438        self.move_nick_cursor(d);
439        let n = self.nicks[self.nick_cursor].nick.clone();
440        if !self.selected.contains(&n) { self.selected.push(n); }
441        self.selection_changed();
442    }

444    pub fn select_all(&mut self) {
445        for n in &self.nicks { if !self.selected.contains(&n.nick) { self.selected.push(n.nick.clone()); } }
446        self.selection_changed();
447    }

495    /// Bookmark / unbookmark `id`: shown at once, stored by a side job (works offline).
496    pub fn toggle_mark(&mut self, id: &str) {
497        let on = !self.marks.contains(id);
498        match self.menu.menu.bookmark_argv(id, on) {
499            Ok(argv) => {
500                if on { self.marks.insert(id.to_string()); } else { self.marks.remove(id); }
501                self.push_status(format!("{} {id}", if on { "★ bookmarked" } else { "☆ unbookmarked" }));

507    /// A side job ended; `pending` = side jobs still running.
508    pub fn side_job_finished(&mut self, ok: bool, tail: &[String], pending: usize) {
509        if !ok { self.push_status(format!("error: bookmark not saved: {}", tail.last().map(String::as_str).unwrap_or("?"))); }
510        // Re-read only once every write landed, so quick toggles don't flicker back.
511        if pending == 0 { self.reload_marks(); }
512    }
513    /// Run `argv` (a network job) after making sure we're logged in.
514    fn start_online_job(&mut self, kind: JobKind, argv: Vec<String>) {
515        if self.job.is_some() { self.error("Busy", "another job is still running; wait for it or cancel it (F5)"); return; }
516        self.close_menu();
517        if self.session == SessionState::Valid { self.effects.push(Effect::RunJob(kind, argv)); }
518        else { self.session = SessionState::Checking; self.effects.push(Effect::CheckSessionThen(kind, argv)); }
519    }

521    // ------------------------------------------------------------ sync
522    /// Start an automatic sync when Sync › Auto sync is on and one is due. Skips (and
523    /// reschedules) while a job runs or when logged out; never opens a modal.
524    pub fn sync_tick(&mut self, now: Instant) {
525        if !self.menu.menu.check(CommandId::Sync, "auto") { self.sync_next = None; return; }
526        if self.sync_next.is_some_and(|t| now < t) { return; }
527        let every = match self.menu.menu.sync_interval() {
528            Ok(d) => d,
529            Err(e) => { self.menu.menu.set_check(CommandId::Sync, "auto", false); self.push_status(format!("auto sync off: {e}")); return; }
530        };
531        self.sync_next = Some(now + every);
532        if self.job.is_some() { self.push_status("auto sync skipped: a job is running"); return; }
533        if self.session == SessionState::LoggedOut { self.push_status("auto sync skipped: logged out (Session › Log in)"); return; }
534        match self.menu.menu.sync_argv(&self.selected.clone()) {
535            Ok(argv) => { self.push_status("auto sync…"); self.effects.push(Effect::AutoSync(argv)); }
536            Err(e) => self.push_status(format!("auto sync skipped: {e}")),
537        }
538    }
539    /// The quiet session check of an automatic sync came back.
540    pub fn auto_sync_checked(&mut self, r: std::result::Result<bool, String>, argv: Vec<String>) {
541        match r {
542            Ok(true) if self.job.is_none() => { self.session = SessionState::Valid; self.auto_running = true; self.effects.push(Effect::RunJob(JobKind::Sync, argv)); }
543            Ok(true) => self.push_status("auto sync skipped: a job is running"),
544            Ok(false) => { self.session = SessionState::LoggedOut; self.push_status("auto sync skipped: logged out (Session › Log in)"); }
545            Err(e) => self.push_status(format!("auto sync skipped: offline ({})", e.lines().next().unwrap_or(""))),
546        }
547    }

610    /// Move the post cursor by `d` (detail closed).
611    pub fn move_post_cursor(&mut self, d: i32) {
612        let rows = self.page_rows.max(1);
613        let v = self.view_mut();
614        if v.posts.is_empty() { return; }
615        v.cursor = (v.cursor as i64 + d as i64).clamp(0, v.posts.len() as i64 - 1) as usize;
616        if v.cursor < v.offset { v.offset = v.cursor; }
617        if v.cursor >= v.offset + rows { v.offset = v.cursor + 1 - rows; }
618    }
619    /// Scroll the post list without moving the cursor past the screen.
620    pub fn scroll_posts(&mut self, d: i32) {
621        let rows = self.page_rows.max(1);
622        let v = self.view_mut();
623        let max = v.posts.len().saturating_sub(1) as i64;
624        v.offset = (v.offset as i64 + d as i64).clamp(0, max) as usize;
625        v.cursor = v.cursor.clamp(v.offset, (v.offset + rows - 1).min(max as usize));
626    }

629    /// Build a detail frame for `post` (attachments, links resolved against the store).
630    pub fn frame_for(&self, post: Post) -> DetailFrame {
631        let s = &self.store;
632        let mut links = Vec::new();
633        let mut seen: HashSet<String> = HashSet::from([post.post_id.clone()]);
634        let mut add = |kind, id: &str, links: &mut Vec<Link>| {
635            if id.is_empty() || !seen.insert(id.to_string()) { return; }
636            links.push(Link { kind, id: id.into(), target: s.post(id).ok().flatten() });
637        };
638        let direct = if post.in_reply_to_id.is_empty() { post.parent_post_id.clone() } else { post.in_reply_to_id.clone() };
639        add(LinkKind::ReplyTo, &direct, &mut links);
640        add(LinkKind::Thread, &post.parent_post_id, &mut links);
641        for id in crate::tui::data::linked_ids(&post.content) { add(LinkKind::Mentioned, &id, &mut links); }
642        for r in s.replies(&post.post_id).unwrap_or_default() {
643            if seen.insert(r.post_id.clone()) { links.push(Link { kind: LinkKind::Reply, id: r.post_id.clone(), target: Some(r) }); }
644        }
645        let attachments = s.attachments(&post.post_id).unwrap_or_default();
646        for a in &attachments {
647            links.push(Link { kind: LinkKind::Attachment, id: a.local_path.display().to_string(), target: None });
648        }
649        DetailFrame { attachments, post, links,
650            link_cursor: 0, scroll: 0, follow_link: false, notice: None }
651    }
652    /// Open the n-th post of the view as the root detail (replacing any open stack).
653    pub fn open_post(&mut self, i: usize) {
654        let Some(p) = self.view().posts.get(i).cloned() else { return };
655        let f = self.frame_for(p);
656        let v = self.view_mut();
657        v.cursor = i;
658        v.stack = vec![f];
659        v.forward.clear();
660        self.focus = Focus::Body;
661        self.anim = Some(Anim { kind: AnimKind::Open, start: Instant::now() });
662    }
663    /// Top frame.
664    pub fn top(&self) -> Option<&DetailFrame> { self.view().stack.last() }
665    /// Mutable top frame.
666    pub fn top_mut(&mut self) -> Option<&mut DetailFrame> { self.view_mut().stack.last_mut() }
667    /// Open the n-th link of the top frame on top of the stack.
668    pub fn open_link(&mut self, i: usize) {
669        let depth = self.view().stack.len();
670        let Some(link) = self.top().and_then(|f| f.links.get(i)).cloned() else { return };
671        if let Some(f) = self.top_mut() { f.link_cursor = i; }
672        if link.kind == LinkKind::Attachment {
673            let p = PathBuf::from(&link.id);
674            if p.exists() { self.push_status(format!("opening {}", p.display())); self.effects.push(Effect::OpenFile(p)); }
675            else {
676                if let Some(f) = self.top_mut() { f.notice = Some(format!("{} is missing (re-scrape to download it)", link.id)); }
677                self.push_status(format!("{} is missing", link.id));
678            }
679            return;
680        }
681        let Some(target) = link.target else {
682            if let Some(f) = self.top_mut() { f.notice = Some(format!("{} is not in the database", link.id)); }
683            self.push_status(format!("{} is not in the database (scrape its thread to read it)", link.id));
684            return;
685        };
686        let max = if self.view_mode == ViewMode::Nested { MAX_DETAIL_DEPTH } else { MAX_FLAT_DEPTH };
687        if depth >= max {
688            self.push_status(format!("at most {max} open posts here: press Esc to close one first"));
689            let what = if max == MAX_DETAIL_DEPTH { "nesting" } else { "depth" };
690            if let Some(f) = self.top_mut() { f.notice = Some(format!("{what} limit ({max}) reached")); }
691            return;
692        }
693        let f = self.frame_for(target);
694        let v = self.view_mut();
695        v.stack.push(f);
696        v.forward.clear();
697        self.anim = Some(Anim { kind: AnimKind::Push, start: Instant::now() });
698    }
699    /// Close the top frame; returns whether one was open.
700    pub fn pop_detail(&mut self) -> bool {
701        let v = self.view_mut();
702        let r = match v.stack.pop() { Some(f) => { v.forward.push(f); true } None => false };
703        self.anim = None;
704        r
705    }
706    /// Keep frames `0..=depth` (click on a collapsed header); the closed ones can be reopened with next.
707    pub fn pop_to(&mut self, depth: usize) {
708        let v = self.view_mut();
709        while v.stack.len() > depth + 1 { if let Some(f) = v.stack.pop() { v.forward.push(f); } }
710        self.anim = None;
711    }
712    /// Previous post of the detail stack (like Esc: closes the top one, remembering it).
713    pub fn stack_prev(&mut self) -> bool { self.pop_detail() }
714    /// Next post of the detail stack: reopen the last one closed with prev.
715    pub fn stack_next(&mut self) -> bool {
716        let v = self.view_mut();
717        let Some(f) = v.forward.pop() else { return false };
718        if v.stack.is_empty() && let Some(i) = v.posts.iter().position(|p| p.post_id == f.post.post_id) { v.cursor = i; }
719        v.stack.push(f);
720        self.focus = Focus::Body;
721        self.anim = None;
722        true
723    }
724    /// Position in the stack history: (open frames, open + reopenable).
725    pub fn stack_pos(&self) -> (usize, usize) { let v = self.view(); (v.stack.len(), v.stack.len() + v.forward.len()) }
726    /// Move the top frame's link cursor.

729            f.link_cursor = cycle(f.link_cursor, f.links.len(), d);

747    /// Run a menu / key / palette action.
748    pub fn act(&mut self, a: Action) {
749        let sel = self.selected.clone();
750        match a {
751            Action::StartScan => {
752                if self.job.is_some() { self.effects.push(Effect::CancelJob); return; }
753                match self.menu.menu.scrape_argv(&sel) {
754                    Err(e) => self.error("Cannot start scanning", &e),
755                    Ok(argv) => self.start_online_job(JobKind::Scan, argv),
756                }
757            }
758            Action::CancelJob => if self.job.is_some() { self.effects.push(Effect::CancelJob); },
759            Action::RunSearch => match self.menu.menu.search_filter() {
760                Ok(f) => { self.close_menu(); self.run_search(f); }
761                Err(e) => self.error("Search", &e),
762            },
763            Action::OpenSearchModal => { self.close_menu(); self.modal = Some(Modal::Search(SearchModal::new(&self.current_filter()))); }
764            Action::ClearSearch => { self.close_menu(); self.clear_search(); }
765            Action::RunExport => {
766                let view = self.search.as_ref().and_then(|v| v.filter.clone());
767                match self.menu.menu.export_argv(&sel, view.as_ref()) {
768                    Ok(argv) => { self.close_menu(); self.start_job(JobKind::Export, argv); }
769                    Err(e) => self.error("Cannot export", &e),
770                }
771            }
772            Action::RunImport => match self.menu.menu.import_argv() {
773                Ok(argv) => { self.close_menu(); self.start_job(JobKind::Import, argv); }
774                Err(e) => self.error("Cannot import", &e),
775            },
776            Action::RefreshNicks => { self.close_menu(); self.refresh_nicks(); self.load_base_keep(); self.push_status(format!("{} nicks", self.nicks.len())); }
777            Action::RemoveNicks => {
778                let m = self.menu.menu.clone();
779                match (m.remove_nicks(&sel), m.remove_argv(&sel)) {
780                    (Ok(n), Ok(argv)) => {
781                        self.close_menu();
782                        let skip = m.check(CommandId::Maintain, "yes") || m.check(CommandId::Maintain, "dry_run");
783                        if skip { self.start_job(JobKind::Maintain, argv); return; }
784                        let who = n.iter().map(|n| format!("@{n}")).collect::<Vec<_>>().join(", ");
785                        let mut lines = vec![format!("Delete {who}: their posts, replies in their threads, attachments, mentions and profiles?")];
786                        if m.check(CommandId::Maintain, "keep_replies") { lines.push("Other people's replies in their threads are kept.".into()); }
787                        lines.push("Tick Maintain › Dry run first to preview the counts.".into());
788                        self.modal = Some(Modal::Confirm(ConfirmModal::new("Remove nicks", lines, "Remove", ConfirmAction::Job(JobKind::Maintain, argv))));
789                    }
790                    (Err(e), _) | (_, Err(e)) => self.error("Cannot remove", &e),
791                }
792            }
793            Action::ClearAll => match self.menu.menu.clear_all_argv() {
794                Ok(argv) => {
795                    self.close_menu();
796                    let m = &self.menu.menu;
797                    if m.check(CommandId::Maintain, "dry_run") || m.check(CommandId::Maintain, "yes") { self.start_job(JobKind::Maintain, argv); return; }
798                    let mut c = ConfirmModal::new("Clear all", vec![
799                        "Delete the whole database and every downloaded attachment?".into(),
800                        "preferences.toml, the saved session and exports are kept.".into(),
801                        "Type yes to confirm.".into()], "Delete everything", ConfirmAction::Job(JobKind::Maintain, argv));
802                    c.strict = true;
803                    c.focus = 0;
804                    self.modal = Some(Modal::Confirm(c));
805                }
806                Err(e) => self.error("Cannot clear", &e),
807            },
808            Action::Login => { self.close_menu(); self.open_login(None); }
809            Action::Logout => { self.close_menu(); self.effects.push(Effect::Logout); }
810            Action::Help => { self.close_menu(); self.modal = Some(Modal::Help(HelpModal::new())); }
811            Action::ToggleLog => { self.show_log = !self.show_log; }
812            Action::StackPrev => { if !self.stack_prev() { self.push_status("no previous post: this is the list"); } }
813            Action::StackNext => { if !self.stack_next() { self.push_status("no next post: open one, or go back with prev first"); } }
814            Action::CycleView => { self.close_menu(); let m = self.view_mode.cycle(1); self.set_view_mode(m); }
815            Action::SetView(m) => { self.close_menu(); self.set_view_mode(m); }
816            Action::ShowBookmarks => { self.close_menu(); self.open_bookmarks(); }
817            Action::ToggleBookmark => {
818                self.close_menu();
819                match self.current_post_id() { Some(id) => self.toggle_mark(&id), None => self.push_status("no post to bookmark: open or highlight one first") }
820            }
821            Action::SyncNow => match self.menu.menu.sync_argv(&sel) {
822                Ok(argv) => { self.sync_next = None; self.start_online_job(JobKind::Sync, argv); }
823                Err(e) => self.error("Cannot sync", &e),
824            },
825            Action::SaveSyncPrefs => { self.close_menu(); self.effects.push(Effect::SaveSyncPrefs(self.menu.menu.sync_prefs())); }
826            Action::SyncBookmarks => match self.menu.menu.sync_bookmarks_argv() {
827                Ok(argv) => self.start_online_job(JobKind::Sync, argv),
828                Err(e) => self.error("Cannot sync", &e),
829            },
830            Action::Quit => {
831                self.close_menu();
832                if let Some(j) = &self.job {
833                    let mut c = ConfirmModal::new("Quit", vec![
834                        format!("A {} is running (pid {}).", j.kind.name(), j.pid),
835                        "Keep it running in the background, or cancel it and quit?".into(),
836                        format!("Its log: {}", j.log.display())], "Keep running & quit", ConfirmAction::Quit);
837                    c.buttons = vec!["Keep running & quit".into(), "Cancel it & quit".into(), "Stay".into()];
838                    c.focus = 0;
839                    self.modal = Some(Modal::Confirm(c));
840                } else {
841                    self.effects.push(Effect::Quit { cancel_job: false });
842                }
843            }
844        }
845    }

942        let Some(item) = self.menu.open else { return };
943        self.menu.opt = i;
944        let Some(o) = self.menu.menu.items[item].opts.get_mut(i) else { return };
945        match &mut o.widget {
946            Widget::Check(v) => *v = !*v,
947            Widget::Choice { choices, idx } => *idx = (*idx + 1) % choices.len(),
948            Widget::Text(_) => self.menu.editing = !self.menu.editing,
949            Widget::Button(a) => { let a = *a; self.act(a); }

952    /// Close the modal with `r` and apply it.
953    pub fn close_modal(&mut self, r: ModalResult) {
954        let Some(m) = self.modal.take() else { return };
955        match (m, r) {
956            (Modal::Login(_), ModalResult::ClosedSubmitted) => {
957                self.push_status("logged in");
958                if let Some((k, argv)) = self.pending_job.take() { self.effects.push(Effect::RunJob(k, argv)); }
959            }
960            (Modal::Login(m), ModalResult::ClosedCancelled) => {
961                self.pending_job = None;
962                if self.session == SessionState::Checking { self.session = SessionState::LoggedOut; }
963                let _ = m;
964                self.effects.push(Effect::LoginCancel);
965            }
966            (Modal::Confirm(c), ModalResult::ClosedSubmitted) => match c.action {
967                ConfirmAction::Job(k, argv) => self.start_job(k, argv),
968                ConfirmAction::Quit => self.effects.push(Effect::Quit { cancel_job: c.chosen == 1 }),
969            },
970            (Modal::Search(s), ModalResult::ClosedSubmitted) => {
971                if s.cleared { self.clear_search(); }
972                else if let Ok(f) = s.filter() { self.run_search(f); }
973            }
974            (Modal::Palette(p), ModalResult::ClosedSubmitted) => match p.chosen() {
975                Some(PaletteCmd::Action(a)) => self.act(a),
976                Some(PaletteCmd::OpenMenu(i)) => self.open_menu(i),
977                None => {}
978            },
979            _ => {}
980        }
981    }
