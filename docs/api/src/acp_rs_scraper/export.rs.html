49fn ext_from_mime(m: &str) -> Option<&'static str> {
50    Some(match m.split(';').next()?.trim() {
51        "image/jpeg" => "jpg",
52        "image/png" => "png",
53        "image/gif" => "gif",
54        "image/webp" => "webp",
55        "image/avif" => "avif",
56        "image/svg+xml" => "svg",
57        "video/mp4" => "mp4",
58        "video/webm" => "webm",
59        "video/quicktime" => "mov",
60        "audio/mpeg" => "mp3",
61        "audio/ogg" => "ogg",
62        "audio/wav" | "audio/x-wav" => "wav",
63        "audio/mp4" => "m4a",
64        "application/pdf" => "pdf",
65        _ => return None,
66    })
67}

69fn mime_from_ext(e: &str) -> &'static str {
70    match e {
71        "jpg" | "jpeg" => "image/jpeg",
72        "png" => "image/png",
73        "gif" => "image/gif",
74        "webp" => "image/webp",
75        "avif" => "image/avif",
76        "svg" => "image/svg+xml",
77        "mp4" | "m4v" => "video/mp4",
78        "webm" => "video/webm",
79        "mov" => "video/quicktime",
80        "mp3" => "audio/mpeg",
81        "ogg" | "oga" => "audio/ogg",
82        "wav" => "audio/wav",
83        "m4a" => "audio/mp4",
84        "pdf" => "application/pdf",
85        _ => "application/octet-stream",
86    }
87}

89fn ext_from_url(u: &str) -> Option<String> {
90    let path = url::Url::parse(u).ok()?.path().to_string();
91    let e = path.rsplit('/').next()?.rsplit_once('.')?.1.to_ascii_lowercase();
92    (e.len() <= 5 && e.chars().all(|c| c.is_ascii_alphanumeric())).then_some(e)
93}

95fn url_key(u: &str) -> String {
96    let mut h = DefaultHasher::new();
97    u.hash(&mut h);
98    format!("{:016x}", h.finish())
99}

101/// Download every media URL once (dedupe by URL), skipping failures.
102pub async fn download_all(http: &reqwest::Client, posts: &[Post], dir: &Path) -> Vec<Attachment> {
103    let _ = std::fs::create_dir_all(dir);
104    let mut done: HashMap<String, Option<(PathBuf, String)>> = HashMap::new();
105    let mut out = Vec::new();
106    let mut seen_pair = HashSet::new();
107    for p in posts {
108        for m in &p.media {
109            if !seen_pair.insert((p.post_id.clone(), m.url.clone())) { continue; }
110            if !done.contains_key(&m.url) {
111                let r = fetch_one_pub(http, &m.url, dir).await;
112                if let Err(e) = &r { crate::log_error(format_args!("warning: attachment {} skipped: {e:#}", m.url)); }
113                done.insert(m.url.clone(), r.ok());
114            }
115            if let Some(Some((path, mime))) = done.get(&m.url) {
116                out.push(Attachment {
117                    post_id: p.post_id.clone(), url: m.url.clone(), kind: m.kind.clone(),
118                    local_path: path.clone(), mime_type: mime.clone(),
119                });
120            }
121        }
122    }
123    crate::log_info(format_args!("attachments: {} files for {} references", done.values().flatten().count(), out.len()));
124    out
125}

127pub async fn fetch_one_pub(http: &reqwest::Client, u: &str, dir: &Path) -> Result<(PathBuf, String)> {
128    let key = url_key(u);
129    // Reuse an existing download.
130    if let Ok(rd) = std::fs::read_dir(dir) {
131        for e in rd.flatten() {
132            let n = e.file_name().to_string_lossy().to_string();
133            if n.starts_with(&key) {
134                let ext = n.rsplit_once('.').map(|x| x.1).unwrap_or("");
135                return Ok((e.path(), mime_from_ext(ext).to_string()));
136            }
137        }
138    }
139    let resp = http.get(u)
140        .header("User-Agent", FIREFOX_UA)
141        .header("Accept", "image/avif,image/webp,video/*,audio/*,*/*;q=0.8")
142        .header("Sec-Fetch-Dest", "image")
143        .header("Sec-Fetch-Mode", "no-cors")
144        .header("Referer", "https://acp.us/")
145        .send().await?.error_for_status()?;
146    let ct = resp.headers().get(reqwest::header::CONTENT_TYPE)
147        .and_then(|v| v.to_str().ok()).unwrap_or("").to_string();
148    let ext = ext_from_mime(&ct).map(str::to_string).or_else(|| ext_from_url(u)).unwrap_or_else(|| "bin".into());
149    let mime = if ct.is_empty() { mime_from_ext(&ext).to_string() } else { ct.split(';').next().unwrap_or("").trim().to_string() };
150    let bytes = resp.bytes().await?;
151    let path = dir.join(format!("{key}.{ext}"));
152    std::fs::write(&path, &bytes)?;
153    Ok((path, mime))
154}

156pub fn save_attachments(db: &Path, atts: &[Attachment]) -> Result<()> {
157    let mut c = Connection::open(db)?;
158    c.execute_batch(
159        "CREATE TABLE IF NOT EXISTS attachments (
160            id INTEGER PRIMARY KEY AUTOINCREMENT,
161            post_id TEXT NOT NULL,
162            url TEXT NOT NULL,
163            local_path TEXT NOT NULL,
164            mime_type TEXT NOT NULL
165        );
166        CREATE UNIQUE INDEX IF NOT EXISTS attachments_post_url_uq ON attachments(post_id, url);",
167    )?;
168    let tx = c.transaction()?;
169    {
170        let mut st = tx.prepare(
171            "INSERT INTO attachments (post_id,url,local_path,mime_type) VALUES (?1,?2,?3,?4)
172             ON CONFLICT(post_id,url) DO UPDATE SET local_path=excluded.local_path, mime_type=excluded.mime_type")?;
173        for a in atts {
174            st.execute(params![a.post_id, a.url, a.local_path.to_string_lossy(), a.mime_type])?;
175        }
176    }
177    tx.commit()?;
178    Ok(())
179}

183/// `rel`: map attachment url -> path written into the export.
184fn build_json(nick: &str, posts: &[Post], atts: &[Attachment], profiles: &[Profile], hits: &[MentionHit], rel: &dyn Fn(&Attachment) -> String) -> Value {
185    let mut by_post: HashMap<&str, Vec<Value>> = HashMap::new();
186    for a in atts {
187        by_post.entry(a.post_id.as_str()).or_default().push(json!({
188            "url": a.url, "kind": a.kind, "mime_type": a.mime_type, "local_path": rel(a),
189        }));
190    }
191    let ids: HashSet<&str> = posts.iter().map(|p| p.post_id.as_str()).collect();
192    let mut kids: HashMap<&str, Vec<&Post>> = HashMap::new();
193    let mut roots = Vec::new();
194    for p in posts {
195        if p.parent_post_id.is_empty() || !ids.contains(p.parent_post_id.as_str()) {
196            roots.push(p);
197        } else {
198            kids.entry(p.parent_post_id.as_str()).or_default().push(p);
199        }
200    }
201    fn node(p: &Post, kids: &HashMap<&str, Vec<&Post>>, by: &HashMap<&str, Vec<Value>>) -> Value {
202        let mut ch: Vec<&Post> = kids.get(p.post_id.as_str()).cloned().unwrap_or_default();
203        ch.sort_by(|a, b| a.posted_at.cmp(&b.posted_at));
204        json!({
205            "post_id": p.post_id, "parent_post_id": p.parent_post_id, "nick": p.nick,
206            "title": p.title, "content": p.content, "posted_at": p.posted_at,
207            "attachments": by.get(p.post_id.as_str()).cloned().unwrap_or_default(),
208            "children": ch.iter().map(|c| node(c, kids, by)).collect::<Vec<_>>(),
209        })
210    }
211    roots.sort_by(|a, b| b.posted_at.cmp(&a.posted_at));
212    let profiles_v: Vec<Value> = profiles.iter().map(|p| json!({
213        "nick": p.nick, "user_id": p.user_id, "display_name": p.display_name, "bio": p.bio,
214        "avatar_url": p.avatar_url, "avatar_path": p.avatar_path,
215        "banner_url": p.banner_url, "banner_path": p.banner_path,
216        "location": p.location, "role": p.role, "joined_at": p.joined_at,
217        "followers": p.followers, "following": p.following, "posts_count": p.posts_count,
218        "scraped_at": p.scraped_at,
219    })).collect();
220    let mentions_v: Vec<Value> = hits.iter().map(|h| json!({
221        "post_id": h.post_id, "mentioned_nick": h.mentioned_nick, "match_type": h.match_type.as_str(),
222    })).collect();
223    json!({
224        "nick": nick,
225        "format": EXPORT_FORMAT,
226        "format_version": EXPORT_FORMAT_VERSION,
227        "exported_at": chrono::Local::now().to_rfc3339(),
228        "profiles": profiles_v,
229        "mentions": mentions_v,
230        "posts": roots.iter().map(|p| node(p, &kids, &by_post)).collect::<Vec<_>>(),
231    })
232}

234fn esc(s: &str) -> String {
235    s.replace('&', "&amp;").replace('<', "&lt;").replace('>', "&gt;").replace('"', "&quot;")
236}

238fn html_post(v: &Value, out: &mut String, depth: usize) {
239    let g = |k: &str| v[k].as_str().unwrap_or_default().to_string();
240    out.push_str(&format!("<article class=\"post d{}\" id=\"p-{}\">", depth.min(6), esc(&g("post_id"))));
241    if !g("title").is_empty() { out.push_str(&format!("<h2>{}</h2>", esc(&g("title")))); }
242    out.push_str(&format!("<div class=\"meta\"><b>@{}</b> · <time>{}</time></div>", esc(&g("nick")), esc(&g("posted_at"))));
243    out.push_str(&format!("<div class=\"body\">{}</div>", esc(&g("content")).replace('\n', "<br>")));
244    for a in v["attachments"].as_array().into_iter().flatten() {
245        let p = esc(a["local_path"].as_str().unwrap_or_default());
246        let m = a["mime_type"].as_str().unwrap_or_default();
247        if a["kind"] == "thumbnail" { continue; }
248        out.push_str("<div class=\"att\">");
249        if m.starts_with("image/") { out.push_str(&format!("<img src=\"{p}\" loading=\"lazy\">")); }
250        else if m.starts_with("video/") { out.push_str(&format!("<video src=\"{p}\" controls preload=\"metadata\"></video>")); }
251        else if m.starts_with("audio/") { out.push_str(&format!("<audio src=\"{p}\" controls></audio>")); }
252        else { out.push_str(&format!("<a href=\"{p}\">{p}</a>")); }
253        out.push_str("</div>");
254    }
255    for c in v["children"].as_array().into_iter().flatten() { html_post(c, out, depth + 1); }
256    out.push_str("</article>");
257}

259fn build_html(doc: &Value) -> String {
260    let title = esc(doc["title"].as_str().or(doc["nick"].as_str()).unwrap_or_default());
261    let nick = title.clone();
262    let mut body = String::new();
263    for pr in doc["profiles"].as_array().into_iter().flatten() {
264        let dn = pr["display_name"].as_str().unwrap_or("");
265        let un = pr["nick"].as_str().unwrap_or("");
266        let bio = pr["bio"].as_str().unwrap_or("");
267        let av = pr["avatar_path"].as_str().or(pr["avatar_url"].as_str()).unwrap_or("");
268        body.push_str("<header class=\"profile\">");
269        if !av.is_empty() { body.push_str(&format!("<img class=\"avatar\" src=\"{}\" alt=\"\">", esc(av))); }
270        body.push_str(&format!("<div><h2>{} <span class=\"meta\">@{}</span></h2>", esc(dn), esc(un)));
271        if !bio.is_empty() { body.push_str(&format!("<p>{}</p>", esc(bio))); }
272        let bits: Vec<String> = ["location","role","joined_at"].iter().filter_map(|k| pr[*k].as_str().filter(|s| !s.is_empty()).map(|s| format!("{k}: {}", esc(s)))).collect();
273        if !bits.is_empty() { body.push_str(&format!("<p class=\"meta\">{}</p>", bits.join(" · "))); }
274        body.push_str("</div></header>");
275    }
276    body.push_str("<h2>Posts</h2>");
277    for p in doc["posts"].as_array().into_iter().flatten() { html_post(p, &mut body, 0); }
278    let mentions = doc["mentions"].as_array().cloned().unwrap_or_default();
279    if !mentions.is_empty() {
280        body.push_str("<h2>Mentions</h2><ul class=\"mentions\">");
281        for h in &mentions {
282            body.push_str(&format!(
283                "<li><a href=\"#p-{}\">{}</a> → @{} <span class=\"meta\">({})</span></li>",
284                esc(h["post_id"].as_str().unwrap_or("")),
285                esc(h["post_id"].as_str().unwrap_or("")),
286                esc(h["mentioned_nick"].as_str().unwrap_or("")),
287                esc(h["match_type"].as_str().unwrap_or("")),
288            ));
289        }
290        body.push_str("</ul>");
291    }
292    format!(r#"<!DOCTYPE html><html lang="en"><head><meta charset="utf-8">
293<meta name="viewport" content="width=device-width,initial-scale=1"><title>{nick}</title>
294<style>
295body{{font:16px/1.5 system-ui,sans-serif;max-width:800px;margin:2em auto;padding:0 1em;color:#18181b;background:#fafafa}}
296.post{{background:#fff;border:1px solid #e4e4e7;border-radius:8px;padding:.8em 1em;margin:.8em 0}}
297.post .post{{margin-left:1.2em;border-left:3px solid #dc2626;border-radius:0 8px 8px 0}}
298h2{{margin:.2em 0;font-size:1.25em}} .meta{{color:#71717a;font-size:.85em}} .body{{margin:.5em 0;white-space:normal}}
299.att img,.att video{{max-width:100%;border-radius:6px}} .att{{margin:.4em 0}}
300.profile{{display:flex;gap:1em;align-items:flex-start;background:#fff;border:1px solid #e4e4e7;border-radius:8px;padding:1em;margin:1em 0}}
301.profile .avatar{{width:72px;height:72px;border-radius:50%;object-fit:cover}}
302.mentions{{padding-left:1.2em}}
303</style><script type="application/json" id="{data_id}">{data}</script></head><body><h1>{nick}</h1><p class="meta">Exported {}</p>{body}</body></html>"#,
304        esc(doc["exported_at"].as_str().unwrap_or_default()),
305        data_id = HTML_DATA_ID,
306        data = doc.to_string().replace('<', "\\u003c"))
307}

309fn render(f: Format, doc: &Value) -> Result<String> {
310    Ok(match f {
311        Format::Json => serde_json::to_string_pretty(doc)?,
312        Format::Html => build_html(doc),
313    })

317pub fn export(args: &Args, name: &str, nick: &str, posts: &[Post], atts: &[Attachment], profiles: &[Profile], hits: &[MentionHit], exports_dir: &Path) -> Result<()> {
318    let ext = match args.format { Format::Json => "json", Format::Html => "html" };
319    let base = name.to_string();
320
321    // stdout: no bundle; attachment paths are the absolute persisted ones.
322    if args.output.is_none() && args.compression().is_none() {
323        let mut doc = build_json(nick, posts, atts, profiles, hits, &|a| a.local_path.to_string_lossy().into_owned());
324        doc["title"] = json!(name);
325        let mut so = std::io::stdout().lock();
326        so.write_all(render(args.format, &doc)?.as_bytes())?;
327        so.write_all(b"\n")?;
328        return Ok(());
329    }
330
331    // Export dir: the file plus attachments/ beside it.
332    let (dir, file, tar_path) = if let Some(c) = args.compression() {
333        let tp = args.output.clone().unwrap_or_else(|| exports_dir.join(format!("{base}.{}", c.ext())));
334        let stem = tp.file_name().map(|n| n.to_string_lossy().trim_end_matches(".tar.gz").trim_end_matches(".tgz").to_string())
335            .unwrap_or(base.clone());
336        let dir = tp.parent().map(Path::to_path_buf).unwrap_or_default().join(&stem);
337        let file = dir.join(format!("{stem}.{ext}"));
338        (dir, file, Some((tp, stem)))
339    } else {
340        let f = args.output.clone().unwrap();
341        // A directory as --output: write <name>.<ext> inside it.
342        let f = if f.is_dir() { f.join(format!("{base}.{ext}")) } else { f };
343        (f.parent().map(Path::to_path_buf).filter(|p| !p.as_os_str().is_empty()).unwrap_or_else(|| ".".into()), f, None)
344    };
345    let att_dir = dir.join("attachments");
346    std::fs::create_dir_all(&att_dir)?;
347    for a in atts {
348        if let Some(n) = a.local_path.file_name() {
349            let dst = att_dir.join(n);
350            if !dst.exists() { std::fs::copy(&a.local_path, &dst).with_context(|| format!("copy {}", a.local_path.display()))?; }
351        }
352    }
353    let mut doc = build_json(nick, posts, atts, profiles, hits, &|a| {
354        format!("attachments/{}", a.local_path.file_name().unwrap_or_default().to_string_lossy())
355    });
356    doc["title"] = json!(name);
357    std::fs::write(&file, render(args.format, &doc)?)?;
358    crate::log_info(format_args!("wrote {}", file.display()));
359
360    if let Some((tp, stem)) = tar_path {
361        if let Some(parent) = tp.parent().filter(|p| !p.as_os_str().is_empty()) { std::fs::create_dir_all(parent)?; }
362        // Only tar-gz today; match on args.compression() here when adding formats.
363        let f = std::fs::File::create(&tp)?;
364        let gz = flate2::write::GzEncoder::new(f, flate2::Compression::default());
365        let mut t = tar::Builder::new(gz);
366        t.append_dir_all(&stem, &dir)?;
367        t.into_inner()?.finish()?;
368        crate::log_info(format_args!("bundle {}", tp.display()));
369    }
370    Ok(())
371}

373/// Load posts matching `query` (case-insensitive LIKE on title + content),
374/// optionally restricted to `nick`, plus context: each match's parent and
375/// its direct children. Attachments come from the attachments table.
376pub fn search_db(db: &Path, query: Option<&str>, nicks: &[String]) -> Result<(Vec<Post>, Vec<Attachment>, usize)> {
377    let c = Connection::open(db).with_context(|| format!("open {}", db.display()))?;
378    let like = query.map(|q| format!("%{}%", q.replace('\\', "\\\\").replace('%', "\\%").replace('_', "\\_")));
379    let cols = "post_id,parent_post_id,nick,title,content,posted_at";
380    let row = |r: &rusqlite::Row| -> rusqlite::Result<Post> {
381        Ok(Post { post_id: r.get(0)?, parent_post_id: r.get(1)?, nick: r.get(2)?, title: r.get(3)?,
382                  content: r.get(4)?, posted_at: r.get(5)?, media: Vec::new() })
383    };
384    let mut st = c.prepare(&format!(
385        "SELECT {cols} FROM posts WHERE (?1 IS NULL OR title LIKE ?1 ESCAPE '\\' OR content LIKE ?1 ESCAPE '\\')
386         ORDER BY posted_at"))?;
387    let want: HashSet<String> = nicks.iter().map(|n| n.to_lowercase()).collect();
388    let matches: Vec<Post> = st.query_map(params![like], row)?
389        .filter(|p| p.as_ref().map_or(true, |p| want.is_empty() || want.contains(&p.nick.to_lowercase())))
390        .collect::<rusqlite::Result<_>>()?;
391    let n = matches.len();
392    let mut seen: HashSet<String> = matches.iter().map(|p| p.post_id.clone()).collect();
393    let mut out = matches;
394    let mut by_id = c.prepare(&format!("SELECT {cols} FROM posts WHERE post_id = ?1"))?;
395    let mut kids = c.prepare(&format!("SELECT {cols} FROM posts WHERE parent_post_id = ?1 ORDER BY posted_at"))?;
396    // Context (parent + children) only for text searches; a full/nick dump has it already.
397    let base: Vec<(String, String)> = if query.is_none() { Vec::new() } else { out.iter().map(|p| (p.post_id.clone(), p.parent_post_id.clone())).collect() };
398    for (id, parent) in base {
399        if !parent.is_empty() {
400            for p in by_id.query_map(params![parent], row)? { let p = p?; if seen.insert(p.post_id.clone()) { out.push(p); } }
401        }
402        for p in kids.query_map(params![id], row)? { let p = p?; if seen.insert(p.post_id.clone()) { out.push(p); } }
403    }
404    let mut atts = Vec::new();
405    let has_att: bool = c.query_row("SELECT count(*) FROM sqlite_master WHERE type='table' AND name='attachments'", [], |r| r.get::<_, i64>(0))? > 0;
406    if has_att {
407        let mut st = c.prepare("SELECT post_id,url,local_path,mime_type FROM attachments WHERE post_id = ?1")?;
408        for p in &out {
409            for a in st.query_map(params![p.post_id], |r| Ok(Attachment {
410                post_id: r.get(0)?, url: r.get(1)?, kind: "file".into(),
411                local_path: PathBuf::from(r.get::<_, String>(2)?), mime_type: r.get(3)?,
412            }))? {
413                let a = a?;
414                if a.local_path.exists() { atts.push(a); } else { crate::log_error(format_args!("warning: missing attachment file {}", a.local_path.display())); }
415            }
416        }
417    }
418    Ok((out, atts, n))
419}

421pub fn slug(s: &str) -> String {
422    let mut o = String::new();
423    for ch in s.chars() {
424        if ch.is_ascii_alphanumeric() { o.push(ch.to_ascii_lowercase()); }
425        else if !o.ends_with('-') { o.push('-'); }
426    }
427    let o = o.trim_matches('-').chars().take(40).collect::<String>();
428    if o.is_empty() { "query".into() } else { o }
429}

433/// Import an export (.json, .html with embedded #acp-data, or .tar.gz/.tgz).
434/// Returns (posts seen, attachments seen) for reporting.
435#[allow(clippy::type_complexity)]
436pub fn import(file: &Path, compress: Option<Compress>, att_dir: &Path, tmp_root: &Path) -> Result<(Vec<Post>, Vec<Attachment>, Vec<Profile>, Vec<MentionHit>)> {
437    // Compression comes from the flags; magic bytes only produce a helpful error.
438    let mut magic = [0u8; 2];
439    {
440        use std::io::Read;
441        let mut f = std::fs::File::open(file).with_context(|| format!("open {}", file.display()))?;
442        let _ = f.read(&mut magic)?;
443    }
444    let gz = magic == [0x1f, 0x8b];
445    match (compress, gz) {
446        (None, true) => anyhow::bail!(
447            "{} looks gzip-compressed; add --compress --tar-gz to import a .tar.gz archive", file.display()),
448        (Some(Compress::TarGz), false) => anyhow::bail!(
449            "{} is not gzip-compressed; drop --compress --tar-gz to import a plain .json/.html export", file.display()),
450        _ => {}
451    }
452    let mut _tmp_guard = None;
453    let doc_path = if let Some(Compress::TarGz) = compress {
454        let tmp = tmp_root.join(format!("import-{}", chrono::Local::now().format("%Y%m%d-%H%M%S%f")));
455        std::fs::create_dir_all(&tmp)?;
456        let f = std::fs::File::open(file).with_context(|| format!("open {}", file.display()))?;
457        tar::Archive::new(flate2::read::GzDecoder::new(f)).unpack(&tmp)?;
458        _tmp_guard = Some(TmpDir(tmp.clone()));
459        find_doc(&tmp).with_context(|| "no .json/.html export inside archive")?
460    } else {
461        file.to_path_buf()
462    };
463    let text = std::fs::read_to_string(&doc_path).with_context(|| format!("read {}", doc_path.display()))?;
464    // JSON vs HTML: decided by content, not file extension.
465    let doc: Value = if text.trim_start().starts_with('<') {
466        let h = scraper::Html::parse_document(&text);
467        let sel = scraper::Selector::parse(&format!("script#{HTML_DATA_ID}")).unwrap();
468        let el = h.select(&sel).next().context("HTML has no embedded acp-data JSON (not an acp-rs export?)")?;
469        serde_json::from_str(&el.text().collect::<String>())?
470    } else {
471        serde_json::from_str(&text)?
472    };
473    if doc["format"].as_str().is_some_and(|f| f != EXPORT_FORMAT) {
474        anyhow::bail!("unknown export format {}", doc["format"]);
475    }
476    let v = doc["format_version"].as_u64().unwrap_or(1);
477    if v > EXPORT_FORMAT_VERSION as u64 {
478        anyhow::bail!("export format_version {v} is newer than supported {EXPORT_FORMAT_VERSION}");
479    }
480    // v1 (no profiles/mentions) and v2 both accepted.
481    let base = doc_path.parent().map(Path::to_path_buf).unwrap_or_default();
482    std::fs::create_dir_all(att_dir)?;
483    let mut posts = Vec::new();
484    let mut atts = Vec::new();
485    let mut seen = HashSet::new();
486    fn walk(n: &Value, parent: &str, base: &Path, att_dir: &Path, posts: &mut Vec<Post>, atts: &mut Vec<Attachment>, seen: &mut HashSet<(String, String)>) {
487        let g = |k: &str| n[k].as_str().unwrap_or_default().to_string();
488        let id = g("post_id");
489        if id.is_empty() { return; }
490        let pp = n["parent_post_id"].as_str().map(str::to_string).unwrap_or_else(|| parent.to_string());
491        posts.push(Post { post_id: id.clone(), parent_post_id: pp, nick: g("nick"), title: g("title"),
492            content: g("content"), posted_at: g("posted_at"), media: Vec::new() });
493        for a in n["attachments"].as_array().into_iter().flatten() {
494            let url = a["url"].as_str().unwrap_or_default().to_string();
495            if url.is_empty() || !seen.insert((id.clone(), url.clone())) { continue; }
496            let lp = PathBuf::from(a["local_path"].as_str().unwrap_or_default());
497            let src = if lp.is_absolute() { lp } else { base.join(lp) };
498            let Some(fname) = src.file_name() else { continue };
499            let dst = att_dir.join(fname);
500            if !dst.exists() && let Err(e) = std::fs::copy(&src, &dst) {
501                crate::log_error(format_args!("warning: attachment {} not imported: {e}", src.display()));
502                continue;
503            }
504            atts.push(Attachment { post_id: id.clone(), url, kind: a["kind"].as_str().unwrap_or("file").into(),
505                local_path: dst, mime_type: a["mime_type"].as_str().unwrap_or_default().into() });
506        }
507        for c in n["children"].as_array().into_iter().flatten() { walk(c, &id, base, att_dir, posts, atts, seen); }
508    }
509    for p in doc["posts"].as_array().into_iter().flatten() {
510        walk(p, "", &base, att_dir, &mut posts, &mut atts, &mut seen);
511    }
512        let mut profiles = Vec::new();
513    for pr in doc["profiles"].as_array().into_iter().flatten() {
514        let g = |k: &str| pr[k].as_str().map(str::to_string);
515        profiles.push(Profile {
516            nick: g("nick").unwrap_or_default(),
517            user_id: g("user_id"), display_name: g("display_name"), bio: g("bio"),
518            avatar_url: g("avatar_url"), avatar_path: g("avatar_path"),
519            banner_url: g("banner_url"), banner_path: g("banner_path"),
520            location: g("location"), role: g("role"), joined_at: g("joined_at"),
521            followers: pr["followers"].as_i64(), following: pr["following"].as_i64(),
522            posts_count: pr["posts_count"].as_i64(),
523            raw_json: pr.to_string(), scraped_at: g("scraped_at").unwrap_or_default(),
524        });
525    }
526    let mut hits = Vec::new();
527    for h in doc["mentions"].as_array().into_iter().flatten() {
528        let mt = h["match_type"].as_str().and_then(MatchType::parse);
529        if let Some(match_type) = mt {
530            hits.push(MentionHit {
531                post_id: h["post_id"].as_str().unwrap_or("").into(),
532                mentioned_nick: h["mentioned_nick"].as_str().unwrap_or("").into(),
533                match_type,
534            });
535        }
536    }
537    Ok((posts, atts, profiles, hits))
538}

540struct TmpDir(PathBuf);
541impl Drop for TmpDir {
542    fn drop(&mut self) { let _ = std::fs::remove_dir_all(&self.0); }
543}

545/// First .json, else .html, in the archive (shallowest first).
546fn find_doc(dir: &Path) -> Option<PathBuf> {
547    let mut found: Vec<(usize, u8, PathBuf)> = Vec::new();
548    let mut stack = vec![(dir.to_path_buf(), 0usize)];
549    while let Some((d, depth)) = stack.pop() {
550        for e in std::fs::read_dir(&d).ok()?.flatten() {
551            let p = e.path();
552            if p.is_dir() {
553                if p.file_name().is_some_and(|n| n != "attachments") { stack.push((p, depth + 1)); }
554            } else if let Some(x) = p.extension().and_then(|x| x.to_str()) {
555                match x.to_ascii_lowercase().as_str() {
556                    "json" => found.push((depth, 0, p)),
557                    "html" | "htm" => found.push((depth, 1, p)),
558                    _ => {}
559                }
560            }
561        }
562    }
563    found.sort();
564    found.into_iter().next().map(|x| x.2)
565}
