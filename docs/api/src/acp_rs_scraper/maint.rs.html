1//! Offline maintenance: `--remove-nick` and `--clear-all`.
2//!
3//! Removal runs in a single SQLite transaction. The same code path computes the
4//! preview (`--dry-run` and the confirmation counts) by rolling the transaction
5//! back instead of committing, so the preview always matches what would happen.
6//! Files on disk are only touched after a successful commit, and only files that
7//! live inside the attachments directory and are no longer referenced by any
8//! remaining `attachments` row or profile avatar/banner path.
9
10use anyhow::{Context, Result};
11use rusqlite::{Connection, OptionalExtension};
12use std::{
13    collections::BTreeSet,
14    path::{Path, PathBuf},
15};
16
17/// What a `--remove-nick` run removed (or would remove, for a preview).
18#[derive(Debug, Default, Clone, PartialEq, Eq)]
19pub struct RemoveReport {
20    /// Posts authored by the removed nicks.
21    pub own_posts: usize,
22    /// Other people's replies in the removed nicks' threads (0 with `--keep-replies`).
23    pub child_replies: usize,
24    /// `attachments` rows deleted.
25    pub attachment_rows: usize,
26    /// `mentions` rows deleted (mentioned_nick matches, or the post was removed).
27    pub mention_rows: usize,
28    /// `profiles` rows deleted.
29    pub profile_rows: usize,
30    /// Files no longer referenced by anything, which are (or would be) deleted.
31    pub files: Vec<PathBuf>,
32    /// Nicks that had nothing at all in the database.
33    pub unknown_nicks: Vec<String>,
34}
35
36impl RemoveReport {
37    /// True if nothing would be removed.
38    pub fn is_empty(&self) -> bool {
39        self.own_posts + self.child_replies + self.attachment_rows + self.mention_rows + self.profile_rows == 0
40            && self.files.is_empty()
41    }
42
43    /// Human-readable multi-line summary.
44    pub fn summary(&self, verb: &str) -> String {
45        let mut s = format!(
46            "{verb}: {} posts by the nick(s), {} replies in their threads, {} attachment rows, \
47             {} mention rows, {} profiles, {} files",
48            self.own_posts, self.child_replies, self.attachment_rows, self.mention_rows,
49            self.profile_rows, self.files.len());
50        if !self.unknown_nicks.is_empty() {
51            s.push_str(&format!("\nnot in database: {}", self.unknown_nicks.join(", ")));
52        }
53        s
54    }
55}

57fn table_exists(c: &Connection, t: &str) -> Result<bool> {
58    Ok(c.query_row("SELECT 1 FROM sqlite_master WHERE type='table' AND name=?1", [t], |_| Ok(()))
59        .optional()?.is_some())
60}

62/// Remove nicks from the database at `db`. With `commit == false` nothing is
63/// changed (the transaction is rolled back and no files are deleted), which is
64/// how `--dry-run` and the confirmation preview work.
65pub fn remove_nicks(db: &Path, attach_dir: &Path, nicks: &[String], keep_replies: bool, commit: bool) -> Result<RemoveReport> {
66    let mut rep = RemoveReport::default();
67    if !db.exists() {
68        rep.unknown_nicks = nicks.to_vec();
69        return Ok(rep);
70    }
71    let mut c = Connection::open(db).with_context(|| format!("open {}", db.display()))?;
72    let has_posts = table_exists(&c, "posts")?;
73    let has_atts = table_exists(&c, "attachments")?;
74    let has_mentions = table_exists(&c, "mentions")?;
75    let has_profiles = table_exists(&c, "profiles")?;
76    let tx = c.transaction()?;
77    tx.execute_batch(
78        "CREATE TEMP TABLE IF NOT EXISTS rm_nicks (nick TEXT PRIMARY KEY);
79         CREATE TEMP TABLE IF NOT EXISTS rm_posts (post_id TEXT PRIMARY KEY, own INTEGER NOT NULL);
80         CREATE TEMP TABLE IF NOT EXISTS rm_files (path TEXT PRIMARY KEY);
81         DELETE FROM rm_nicks; DELETE FROM rm_posts; DELETE FROM rm_files;")?;
82    for n in nicks {
83        tx.execute("INSERT OR IGNORE INTO rm_nicks VALUES (lower(?1))", [n])?;
84    }
85    for n in nicks {
86        let mut found = false;
87        if has_posts {
88            found |= tx.query_row("SELECT 1 FROM posts WHERE lower(nick)=lower(?1) LIMIT 1", [n], |_| Ok(())).optional()?.is_some();
89        }
90        if has_profiles {
91            found |= tx.query_row("SELECT 1 FROM profiles WHERE lower(nick)=lower(?1)", [n], |_| Ok(())).optional()?.is_some();
92        }
93        if has_mentions {
94            found |= tx.query_row("SELECT 1 FROM mentions WHERE lower(mentioned_nick)=lower(?1) LIMIT 1", [n], |_| Ok(())).optional()?.is_some();
95        }
96        if !found { rep.unknown_nicks.push(n.clone()); }
97    }
98    if has_posts {
99        rep.own_posts = tx.execute(
100            "INSERT OR IGNORE INTO rm_posts SELECT post_id, 1 FROM posts WHERE lower(nick) IN (SELECT nick FROM rm_nicks)", [])?;
101        if !keep_replies {
102            rep.child_replies = tx.execute(
103                "INSERT OR IGNORE INTO rm_posts SELECT post_id, 0 FROM posts
104                 WHERE parent_post_id <> '' AND parent_post_id IN
105                   (SELECT p.post_id FROM posts p JOIN rm_posts r ON r.post_id = p.post_id AND r.own = 1
106                    WHERE p.parent_post_id = '')", [])?;
107        }
108    }
109    if has_atts {
110        tx.execute("INSERT OR IGNORE INTO rm_files SELECT local_path FROM attachments WHERE post_id IN (SELECT post_id FROM rm_posts)", [])?;
111        rep.attachment_rows = tx.execute("DELETE FROM attachments WHERE post_id IN (SELECT post_id FROM rm_posts)", [])?;
112    }
113    if has_mentions {
114        rep.mention_rows = tx.execute(
115            "DELETE FROM mentions WHERE lower(mentioned_nick) IN (SELECT nick FROM rm_nicks)
116                OR post_id IN (SELECT post_id FROM rm_posts)", [])?;
117    }
118    if has_profiles {
119        tx.execute(
120            "INSERT OR IGNORE INTO rm_files SELECT avatar_path FROM profiles
121               WHERE lower(nick) IN (SELECT nick FROM rm_nicks) AND avatar_path IS NOT NULL AND avatar_path <> ''", [])?;
122        tx.execute(
123            "INSERT OR IGNORE INTO rm_files SELECT banner_path FROM profiles
124               WHERE lower(nick) IN (SELECT nick FROM rm_nicks) AND banner_path IS NOT NULL AND banner_path <> ''", [])?;
125        rep.profile_rows = tx.execute("DELETE FROM profiles WHERE lower(nick) IN (SELECT nick FROM rm_nicks)", [])?;
126    }
127    if has_posts {
128        tx.execute("DELETE FROM posts WHERE post_id IN (SELECT post_id FROM rm_posts)", [])?;
129    }
130    // Keep only files nothing else still references.
131    if has_atts {
132        tx.execute("DELETE FROM rm_files WHERE path IN (SELECT local_path FROM attachments)", [])?;
133    }
134    if has_profiles {
135        tx.execute("DELETE FROM rm_files WHERE path IN (SELECT avatar_path FROM profiles WHERE avatar_path IS NOT NULL)
136                    OR path IN (SELECT banner_path FROM profiles WHERE banner_path IS NOT NULL)", [])?;
137    }
138    let candidates: Vec<String> = {
139        let mut st = tx.prepare("SELECT path FROM rm_files ORDER BY path")?;
140        st.query_map([], |r| r.get(0))?.collect::<rusqlite::Result<_>>()?
141    };
142    let root = attach_dir.canonicalize().ok();
143    let files: BTreeSet<PathBuf> = candidates.iter().map(PathBuf::from)
144        .filter(|p| p.is_file())
145        .filter(|p| match (&root, p.canonicalize()) {
146            (Some(r), Ok(cp)) => cp.starts_with(r),
147            _ => false,
148        })
149        .collect();
150    rep.files = files.into_iter().collect();
151    if commit {
152        tx.commit()?;
153        for f in &rep.files {
154            if let Err(e) = std::fs::remove_file(f) {
155                crate::log_error(format_args!("warning: could not delete {}: {e}", f.display()));
156            }
157        }
158    } else {
159        tx.rollback()?;
160    }
161    Ok(rep)
162}

164/// Paths `--clear-all` deletes, filtered to those that exist: the database
165/// (plus any `-wal`/`-shm`/`-journal` sidecars) and the attachments directory.
166pub fn clear_all_targets(db: &Path, attach_dir: &Path) -> Vec<PathBuf> {
167    let mut v = vec![db.to_path_buf()];
168    for ext in ["-wal", "-shm", "-journal"] {
169        let mut s = db.as_os_str().to_owned();
170        s.push(ext);
171        v.push(PathBuf::from(s));
172    }
173    v.push(attach_dir.to_path_buf());
174    v.into_iter().filter(|p| p.exists()).collect()
175}
176
177/// Delete everything from [`clear_all_targets`]. Returns what was removed.
178pub fn clear_all(db: &Path, attach_dir: &Path) -> Result<Vec<PathBuf>> {
179    let t = clear_all_targets(db, attach_dir);
180    for p in &t {
181        if p.is_dir() {
182            std::fs::remove_dir_all(p).with_context(|| format!("remove {}", p.display()))?;
183        } else {
184            std::fs::remove_file(p).with_context(|| format!("remove {}", p.display()))?;
185        }
186    }
187    Ok(t)
188}

190/// Count files (recursively) and their total size under `dir`.
191pub fn dir_usage(dir: &Path) -> (usize, u64) {
192    let mut n = 0;
193    let mut bytes = 0;
194    if let Ok(rd) = std::fs::read_dir(dir) {
195        for e in rd.flatten() {
196            let p = e.path();
197            if p.is_dir() {
198                let (a, b) = dir_usage(&p);
199                n += a;
200                bytes += b;
201            } else if let Ok(m) = e.metadata() {
202                n += 1;
203                bytes += m.len();
204            }
205        }
206    }
207    (n, bytes)
208}

210/// Ask for confirmation on stderr. `--yes` skips the prompt; a non-interactive
211/// stdin without `--yes` is refused. With `strict`, only the word `yes` counts.
212pub fn confirm(question: &str, strict: bool, assume_yes: bool) -> Result<bool> {
213    use std::io::IsTerminal;
214    if assume_yes { return Ok(true); }
215    if !std::io::stdin().is_terminal() {
216        anyhow::bail!("refusing to run without confirmation: stdin is not a terminal (pass --yes)");
217    }
218    let hint = if strict { "type 'yes' to continue" } else { "y/N" };
219    let a = crate::prompt(&format!("{question} [{hint}]: "))?.to_lowercase();
220    Ok(if strict { a == "yes" } else { a == "y" || a == "yes" })
221}

228    /// A throwaway HOME-like directory with the real layout under it.
229    struct TempHome(PathBuf);
230    impl TempHome {
231        fn new(tag: &str) -> Self {
232            let p = std::env::temp_dir().join(format!("acp-maint-{tag}-{}-{:?}", std::process::id(),
233                std::time::SystemTime::now().duration_since(std::time::UNIX_EPOCH).unwrap().as_nanos()));
234            std::fs::create_dir_all(p.join(crate::ATTACH_REL_PATH)).unwrap();
235            TempHome(p)
236        }
237        fn db(&self) -> PathBuf { self.0.join(crate::DB_REL_PATH) }
238        fn att(&self) -> PathBuf { self.0.join(crate::ATTACH_REL_PATH) }
239        fn file(&self, name: &str) -> String {
240            let p = self.att().join(name);
241            std::fs::write(&p, b"x").unwrap();
242            p.to_string_lossy().into_owned()
243        }
244    }
245    impl Drop for TempHome {
246        fn drop(&mut self) { let _ = std::fs::remove_dir_all(&self.0); }
247    }

249    fn seed(h: &TempHome) -> Connection {
250        let c = Connection::open(h.db()).unwrap();
251        c.execute_batch(
252            "CREATE TABLE posts (id INTEGER PRIMARY KEY AUTOINCREMENT, post_id TEXT NOT NULL, parent_post_id TEXT NOT NULL,
253               nick TEXT NOT NULL, title TEXT NOT NULL, content TEXT NOT NULL, posted_at TEXT NOT NULL);
254             CREATE TABLE attachments (id INTEGER PRIMARY KEY AUTOINCREMENT, post_id TEXT NOT NULL, url TEXT NOT NULL,
255               local_path TEXT NOT NULL, mime_type TEXT NOT NULL);
256             CREATE TABLE mentions (id INTEGER PRIMARY KEY AUTOINCREMENT, post_id TEXT NOT NULL,
257               mentioned_nick TEXT NOT NULL, match_type TEXT NOT NULL);
258             CREATE TABLE profiles (nick TEXT PRIMARY KEY, avatar_path TEXT, banner_path TEXT,
259               raw_json TEXT NOT NULL DEFAULT '{}', scraped_at TEXT NOT NULL DEFAULT '');").unwrap();
260        let shared = h.file("shared.png");
261        let own = h.file("alice-only.png");
262        let reply_img = h.file("bob-reply.png");
263        let avatar = h.file("alice-avatar.png");
264        let bob_avatar = h.file("bob-avatar.png");
265        for (id, parent, nick) in [("A1", "", "Alice"), ("A2", "B1", "alice"), ("R1", "A1", "bob"),
266                                   ("B1", "", "bob"), ("R2", "B1", "carol")] {
267            c.execute("INSERT INTO posts (post_id,parent_post_id,nick,title,content,posted_at) VALUES (?1,?2,?3,'','','')",
268                params![id, parent, nick]).unwrap();
269        }
270        for (pid, path) in [("A1", &own), ("A1", &shared), ("B1", &shared), ("R1", &reply_img)] {
271            c.execute("INSERT INTO attachments (post_id,url,local_path,mime_type) VALUES (?1,?2,?3,'image/png')",
272                params![pid, format!("u/{pid}/{path}"), path]).unwrap();
273        }
274        for (pid, n) in [("B1", "alice"), ("R2", "bob"), ("A1", "bob")] {
275            c.execute("INSERT INTO mentions (post_id,mentioned_nick,match_type) VALUES (?1,?2,'nick')", params![pid, n]).unwrap();
276        }
277        c.execute("INSERT INTO profiles (nick,avatar_path) VALUES ('alice',?1)", [&avatar]).unwrap();
278        c.execute("INSERT INTO profiles (nick,avatar_path) VALUES ('bob',?1)", [&bob_avatar]).unwrap();
279        c
280    }

285    fn dry_run_changes_nothing() {
286        let h = TempHome::new("dry");
287        let c = seed(&h);
288        let r = remove_nicks(&h.db(), &h.att(), &["alice".into()], false, false).unwrap();

295    #[test]
296    fn removes_nick_replies_and_unreferenced_files() {
297        let h = TempHome::new("rm");
298        let c = seed(&h);
299        let r = remove_nicks(&h.db(), &h.att(), &["ALICE".into(), "ghost".into()], false, true).unwrap();
300        assert_eq!(r.unknown_nicks, vec!["ghost".to_string()]);
301        // Alice's two posts plus bob's reply R1 under her root A1. B1/R2 stay.
302        let left: Vec<String> = c.prepare("SELECT post_id FROM posts ORDER BY post_id").unwrap()
303            .query_map([], |r| r.get(0)).unwrap().map(|x| x.unwrap()).collect();
304        assert_eq!(left, ["B1", "R2"]);
305        assert_eq!(count(&c, "SELECT count(*) FROM attachments"), 1);
306        // mentions of alice and mentions attached to removed posts are gone; R2->bob stays.
307        assert_eq!(count(&c, "SELECT count(*) FROM mentions"), 1);
308        assert_eq!(count(&c, "SELECT count(*) FROM profiles"), 1);
309        // shared.png is still used by B1; bob's avatar untouched.
310        assert!(h.att().join("shared.png").exists());
311        assert!(h.att().join("bob-avatar.png").exists());
312        for gone in ["alice-only.png", "bob-reply.png", "alice-avatar.png"] {
313            assert!(!h.att().join(gone).exists(), "{gone} should be deleted");
314        }
315        assert_eq!(r.files.len(), 3);
316    }
