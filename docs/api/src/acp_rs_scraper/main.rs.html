87fn load_email() -> Result<String> {
88    let path = home()?.join(PREFS_REL_PATH);
89    if path.exists() {
90        let p: Prefs = toml::from_str(&std::fs::read_to_string(&path)?)?;
91        if let Some(e) = p.email.filter(|e| !e.is_empty()) {
92            return Ok(e);
93        }
94    }
95    std::fs::create_dir_all(path.parent().unwrap())?;
96    let email = prompt("Email: ")?;
97    std::fs::write(&path, toml::to_string(&Prefs { email: Some(email.clone()) })?)?;
98    Ok(email)
99}

149// ---------------------------------------------------------------- liveview diff rendering
150/// Merge a LiveView diff into the rendered tree (LV rule: a subtree that
151/// carries its own statics `s` replaces, otherwise merge key by key).
152fn lv_merge(base: &mut Value, diff: &Value) {
153    match (base.as_object_mut(), diff.as_object()) {
154        (Some(b), Some(d)) => {
155            for (k, v) in d {
156                match b.get_mut(k) {
157                    Some(old) if v.is_object() && v.get("s").is_none() && old.is_object() => lv_merge(old, v),
158                    _ => { b.insert(k.clone(), v.clone()); }
159                }
160            }
161        }
162        _ => *base = diff.clone(),
163    }
164}
165
166fn lv_statics<'a>(node: &'a Value, root: &'a Value, comps: &'a Value) -> Vec<String> {
167    let strs = |v: &Value| v.as_array().map(|a| a.iter().map(|x| x.as_str().unwrap_or_default().to_string()).collect::<Vec<_>>());
168    match &node["s"] {
169        Value::Array(_) => strs(&node["s"]).unwrap_or_default(),
170        Value::Number(n) => {
171            let i = n.as_i64().unwrap_or(0);
172            if i < 0 {
173                // shared statics of another component
174                lv_statics(&comps[(-i).to_string()], root, comps)
175            } else {
176                strs(&root["p"][i.to_string()]).unwrap_or_default()
177            }
178        }
179        _ => vec![],
180    }
181}
182
183fn lv_render(node: &Value, root: &Value, comps: &Value, out: &mut String) {
184    match node {
185        Value::String(s) => out.push_str(s),
186        Value::Number(n) => {
187            // component reference
188            if let Some(c) = comps.get(n.to_string()) { lv_render(c, root, comps, out); }
189        }
190        Value::Object(_) => {
191            let st = lv_statics(node, root, comps);
192            if let Some(rows) = node.get("d").and_then(Value::as_array) {
193                for row in rows {
194                    for (i, s) in st.iter().enumerate() {
195                        out.push_str(s);
196                        if let Some(v) = row.get(i) { lv_render(v, root, comps, out); }
197                    }
198                }
199            } else {
200                for (i, s) in st.iter().enumerate() {
201                    out.push_str(s);
202                    if let Some(v) = node.get(i.to_string()) { lv_render(v, root, comps, out); }
203                }
204            }
205        }
206        _ => {}
207    }
208}

275impl Lv {
276    async fn connect(p: &LvPage, jar: &Jar) -> Result<Self> {
277        let u = format!(
278            "wss://acp.us/live/websocket?_csrf_token={}&vsn=2.0.0",
279            urlenc(&p.csrf)
280        );
281        let mut req = u.into_client_request()?;
282        let h = req.headers_mut();
283        h.insert("User-Agent", FIREFOX_UA.parse()?);
284        h.insert("Origin", BASE.parse()?);
285        if let Some(c) = jar.cookies(&Url::parse(BASE)?) {
286            h.insert("Cookie", c);
287        }
288        let (ws, _) = tokio_tungstenite::connect_async(req).await?;
289        let mut lv = Lv { ws, topic: format!("lv:{}", p.id), r: 0, rendered: json!({}) };
290        let reply = lv
291            .push("phx_join", json!({
292                "url": p.url,
293                "params": {"_csrf_token": p.csrf, "_mounts": 0},
294                "session": p.session,
295                "static": p.statik,
296            }))
297            .await?;
298        if reply["status"] != "ok" {
299            bail!("LiveView join failed: {reply}");
300        }
301        lv.rendered = reply["response"]["rendered"].clone();
302        Ok(lv)
303    }
304
305    async fn push(&mut self, event: &str, payload: Value) -> Result<Value> {
306        self.r += 1;
307        let r = self.r.to_string();
308        let msg = json!(["1", r, self.topic, event, payload]);
309        dbg_log("push", &msg.to_string());
310        self.ws.send(Message::text(msg.to_string())).await?;
311        while let Some(m) = self.ws.next().await {
312            if let Message::Text(t) = m? {
313                let v: Value = serde_json::from_str(&t)?;
314                dbg_log("recv", &t);
315                if v[3] == "diff" {
316                    lv_merge(&mut self.rendered, &strip_meta(&v[4]));
317                }
318                if v[1] == r.as_str() && v[3] == "phx_reply" {
319                    if let Some(d) = v[4].pointer("/response/diff") {
320                        lv_merge(&mut self.rendered, &strip_meta(d));
321                    }
322                    return Ok(v[4].clone());
323                }
324            }
325        }
326        bail!("websocket closed")
327    }
328
329    /// Render the current LiveView tree (root plus every component) to HTML.
330    fn html(&self) -> String {
331        let mut out = String::new();
332        let comps = self.rendered.get("c").cloned().unwrap_or(json!({}));
333        lv_render(&self.rendered, &self.rendered, &comps, &mut out);
334        if let Some(m) = comps.as_object() {
335            for c in m.values() { lv_render(c, &self.rendered, &comps, &mut out); }
336        }
337        out
338    }

340    async fn form(&mut self, event: &str, field: &str, val: &str, cid: Option<u64>) -> Result<Value> {
341        let mut p = json!({
342            "type": "form",
343            "event": event,
344            "value": format!("{}={}", urlenc(field), urlenc(val)),
345        });
346        if let Some(c) = cid {
347            p["cid"] = json!(c);
348        }
349        self.push("event", p).await
350    }

367/// Discover the code form's event / field / cid from the rendered LiveView
368/// (falls back to the raw reply text, then to constants).
369fn detect_code_form(html: &str, reply: &Value) -> (String, String, Option<u64>) {
370    let raw = format!("{html}{}", reply.to_string().replace("\\\"", "\""));
371    let ev = regex::Regex::new(r#"phx-submit="([^"]+)""#).unwrap()
372        .captures_iter(&raw).map(|c| c[1].to_string())
373        .find(|e| e != EMAIL_EVENT)
374        .unwrap_or_else(|| CODE_EVENT_FALLBACK.into());
375    let doc = Html::parse_fragment(html);
376    let field = doc.select(&Selector::parse("form input").unwrap())
377        .filter(|i| !matches!(i.value().attr("type"), Some("hidden" | "email")))
378        .filter_map(|i| i.value().attr("name").map(str::to_string))
379        .find(|n| n != EMAIL_FIELD && !n.starts_with('_'))
380        .unwrap_or_else(|| CODE_FIELD_FALLBACK.into());
381    let cid = Selector::parse(&format!("form[phx-submit='{ev}']")).ok()
382        .and_then(|sel| doc.select(&sel).next()
383            .and_then(|f| f.value().attr("phx-target")?.parse().ok()))
384        .or(Some(EMAIL_FORM_CID));
385    (ev, field, cid)
386}

388/// `phx-trigger-action`: LiveView asks the client to do a real HTTP POST of
389/// the form (action="/auth/user/otp/sign_in") so the server can set the
390/// authenticated session cookie.
391async fn trigger_action_post(
392    http: &reqwest::Client, html: &str, csrf: &str, email: &str, code_field: &str, code: &str,
393) -> Result<()> {
394    let doc = Html::parse_fragment(html);
395    let form = doc.select(&Selector::parse("form[phx-trigger-action]").unwrap()).next()
396        .ok_or_else(|| anyhow!("no redirect and no phx-trigger-action form after code"))?;
397    let action = form.value().attr("action").unwrap_or(OTP_SIGN_IN_FALLBACK);
398    let mut fields: Vec<(String, String)> = Vec::new();
399    for i in form.select(&Selector::parse("input[name]").unwrap()) {
400        let name = i.value().attr("name").unwrap_or_default().to_string();
401        let mut val = i.value().attr("value").unwrap_or_default().to_string();
402        if val.is_empty() {
403            if name == EMAIL_FIELD { val = email.into(); }
404            else if name == code_field { val = code.into(); }
405        }
406        fields.push((name, val));
407    }
408    if !fields.iter().any(|(n, _)| n == "_csrf_token") {
409        fields.push(("_csrf_token".into(), csrf.into()));
410    }
411    let names: Vec<&str> = fields.iter().map(|(n, _)| n.as_str()).collect();
412    verbose!("POST {action} with fields {names:?}");
413    let resp = http.post(Url::parse(BASE)?.join(action)?)
414        .header("Origin", BASE)
415        .header("Referer", format!("{BASE}/auth"))
416        .header("Sec-Fetch-Mode", "navigate")
417        .form(&fields)
418        .send().await?;
419    if !resp.status().is_success() {
420        bail!("sign-in POST failed: {}", resp.status());
421    }
422    Ok(())
423}

425async fn login(http: &reqwest::Client, jar: &Arc<Jar>) -> Result<()> {
426    let resp = http.get(START_URL).send().await?;
427    let url = resp.url().to_string();
428    let html = resp.text().await?;
429    if !url.contains("/auth") {
430        info!("Already logged in.");
431        return Ok(());
432    }
433    let page = parse_lv(&url, &html)?;
434    let email = load_email()?;
435    let mut lv = Lv::connect(&page, jar).await?;
436    let r = lv.form(EMAIL_EVENT, EMAIL_FIELD, &email, Some(EMAIL_FORM_CID)).await?;
437    if r["status"] != "ok" {
438        bail!("email submit failed: {r}");
439    }
440    info!("Login email requested for {email}.");
441    let code = prompt("Verification code (or paste the login link): ")?;
442    if code.starts_with("http") {
443        http.get(&code).send().await?.error_for_status()?;
444    } else {
445        let (ev, field, cid) = detect_code_form(&lv.html(), &r);
446        let r2 = lv.form(&ev, &field, &code, cid).await?;
447        if let Some(to) = find_redirect(&r2) {
448            http.get(Url::parse(BASE)?.join(&to)?).send().await?.error_for_status()?;
449        } else {
450            trigger_action_post(http, &lv.html(), &page.csrf, &email, &field, &code).await?;
451        }
452    }
453    let check = http.get(START_URL).send().await?;
454    if check.url().path().starts_with("/auth") {
455        bail!("still not logged in after verification");
456    }
457    info!("Logged in.");
458    Ok(())
459}

480/// Verified (PostMedia-*.js): `post.media` = [{id, type, url, thumb_url, width, height}],
481/// `type == "video"` renders `<video src=url poster=thumb_url>`; other types render `<img>`.
482/// Body (ProseMirror) has no image node in the editor bundle; nodes found: embed
483/// {provider,id,kind} (third-party iframes, not downloaded), postRef, mention.
484/// GUESSED: any other node attr named src/url/href pointing at the site's media hosts.
485fn collect_media(v: &Value, out: &mut Vec<MediaRef>) {
486    if let Some(ms) = v.get("media").and_then(Value::as_array) {
487        for m in ms {
488            let kind = m.get("type").and_then(Value::as_str).unwrap_or("image").to_string();
489            if let Some(u) = m.get("url").and_then(Value::as_str) {
490                out.push(MediaRef { url: u.into(), kind: kind.clone() });
491            }
492            if let Some(u) = m.get("thumb_url").and_then(Value::as_str) {
493                out.push(MediaRef { url: u.into(), kind: "thumbnail".into() });
494            }
495        }
496    }
497    for k in ["attachments", "files", "images"] {
498        if let Some(ms) = v.get(k).and_then(Value::as_array) {
499            for m in ms {
500                if let Some(u) = m.get("url").or(m.get("src")).and_then(Value::as_str) {
501                    let kind = m.get("type").and_then(Value::as_str).unwrap_or("file").to_string();
502                    out.push(MediaRef { url: u.into(), kind });
503                }
504            }
505        }
506    }
507    fn walk(n: &Value, out: &mut Vec<MediaRef>) {
508        if let Some(a) = n.get("attrs").and_then(Value::as_object) {
509            for k in ["src", "url", "href"] {
510                if let Some(u) = a.get(k).and_then(Value::as_str)
511                    && MEDIA_HOSTS.iter().any(|h| u.contains(h))
512                {
513                    let kind = n.get("type").and_then(Value::as_str).unwrap_or("file").to_string();
514                    out.push(MediaRef { url: u.into(), kind });
515                }
516            }
517        }
518        for key in ["content", "marks"] {
519            if let Some(kids) = n.get(key).and_then(Value::as_array) { kids.iter().for_each(|k| walk(k, out)); }
520        }
521    }
522    if let Some(b) = v.get("body") {
523        let b = match b.as_str() { Some(r) => serde_json::from_str(r).unwrap_or(Value::Null), None => b.clone() };
524        walk(&b, out);
525    }
526}

531/// Flatten a ProseMirror doc (`{type, text?, content:[...]}`) to plain text.
532fn prosemirror_text(v: &Value, out: &mut String) {
533    if let Some(t) = v.get("text").and_then(Value::as_str) {
534        out.push_str(t);
535    }
536    let ty = v.get("type").and_then(Value::as_str).unwrap_or("");
537    if ty == "hard_break" || ty == "hardBreak" {
538        out.push('\n');
539    }
540    if let Some(kids) = v.get("content").and_then(Value::as_array) {
541        for k in kids {
542            prosemirror_text(k, out);
543        }
544    }
545    if matches!(ty, "paragraph" | "heading" | "blockquote" | "list_item" | "listItem" | "code_block" | "codeBlock") {
546        out.push('\n');
547    }
548}

554fn to_post(v: &Value, parent: &str, fallback_nick: &str) -> Post {

560        };

571        parent_post_id: parent.to_string(),
572        nick,
573        title: s(v, "title"),
574        content,
575        posted_at: s(v, "inserted_at"),
576        media: { let mut m = Vec::new(); collect_media(v, &mut m); m },
577    }

595/// Collect every object having an `id` and `inserted_at` from any `replies`
596/// array in a JSON value, including JSON hidden in strings (diffs / data-props).
597fn harvest_replies(v: &Value, out: &mut Vec<Value>) {
598    match v {
599        Value::Object(m) => {
600            if let Some(Value::Array(rs)) = m.get("replies") {
601                out.extend(rs.iter().filter(|r| r.get("inserted_at").is_some()).cloned());
602            }
603            m.values().for_each(|x| harvest_replies(x, out));
604        }
605        Value::Array(a) => a.iter().for_each(|x| harvest_replies(x, out)),
606        Value::String(st) if st.contains("replies") => {
607            let st = st.replace("&quot;", "\"").replace("&amp;", "&");
608            if let Some(i) = st.find('{')
609                && let Ok(j) = serde_json::from_str::<Value>(&st[i..st.rfind('}').unwrap_or(i) + 1]) {
610                    harvest_replies(&j, out);
611                }
612        }
613        _ => {}
614    }
615}

633async fn fetch_feed(
634    http: &reqwest::Client, jar: &Jar, url: &str, limit: Option<usize>,
635    known: Option<&HashSet<String>>,
636    allow_incremental: bool, // false for Trending: no early stop on all-known pages
637) -> Result<FeedPage> {
638    let resp = http.get(url).send().await?.error_for_status()?;
639    let final_url = resp.url().to_string();
640    let html = resp.text().await?;
641    let feed = props(&html, FEED_PROPS_SEL).with_context(|| format!("no forum/Feed data-props on {url}"))?;
642    let mut items: Vec<Value> = feed["posts"].as_array().cloned().unwrap_or_default();
643    let mut at_end = feed["at_end"].as_bool().unwrap_or(true);
644    let cursor = feed["next_cursor"].as_str().map(str::to_string);
645    let hit_limit = |n: usize| limit.is_some_and(|l| n >= l);
646    let mut stopped_incremental = false;
647    let mut pages: u32 = 1;
648    info!("feed {url}: {} posts from initial page", items.len());
649    if allow_incremental
650        && let Some(k) = known
651        && page_all_known(&items, k) {
652            info!("feed: initial page all known — stopping (use --full to re-crawl)");
653            stopped_incremental = true;
654            at_end = true;
655        }
656
657    if !at_end && cursor.is_some() && !hit_limit(items.len()) && !stopped_incremental {
658        let page = parse_lv(&final_url, &html)?;
659        let mut lv = Lv::connect(&page, jar).await?;
660        let (stop, extra) = page_rest(&mut lv, &mut items, cursor, limit, known, allow_incremental).await?;
661        stopped_incremental = stop;
662        pages += extra;
663    }
664    if let Some(l) = limit { items.truncate(l); }
665    Ok(FeedPage { items, html, stopped_incremental, pages })
666}

668/// Page a feed that is open on `lv` to its end with `page-next {cursor}`,
669/// appending new (id-deduped) posts to `items`. Returns
670/// `(stopped_incremental, extra_pages)`. Shared by profile, site and
671/// "Posts & replies" feeds so incremental/limit rules are identical.
672async fn page_rest(
673    lv: &mut Lv, items: &mut Vec<Value>, mut cursor: Option<String>, limit: Option<usize>,
674    known: Option<&HashSet<String>>, allow_incremental: bool,
675) -> Result<(bool, u32)> {
676    let hit_limit = |n: usize| limit.is_some_and(|l| n >= l);
677    let mut ids: HashSet<String> = items.iter().map(|x| s(x, "id")).collect();
678    let mut stale = 0;
679    let mut pages = 0;
680    let mut at_end = false;
681    let mut stopped_incremental = false;
682    for _ in 0..MAX_PAGES {
683        let Some(c) = cursor.clone() else { at_end = true; break };
684        let r = lv.push("event", json!({"type": "hook", "event": FEED_NEXT_EVENT, "value": {"cursor": c}})).await?;
685        if r["status"] != "ok" { err!("warning: page-next failed: {r}"); break; }
686        let body = reply_body(&r);
687        let body = if body["posts"].is_array() { body.clone() }
688            else { props(&lv.html(), FEED_PROPS_SEL).unwrap_or(Value::Null) };
689        let ps = body["posts"].as_array().cloned().unwrap_or_default();
690        pages += 1;
691        if allow_incremental
692            && let Some(k) = known
693            && page_all_known(&ps, k) {
694                info!("feed: page {} all known — stopping (use --full to re-crawl)", pages + 1);
695                stopped_incremental = true;
696                break;
697            }
698        let new: Vec<Value> = ps.into_iter().filter(|x| ids.insert(s(x, "id"))).collect();
699        verbose!("feed: +{} new posts", new.len());
700        items.extend(new.iter().cloned());
701        at_end = body["at_end"].as_bool().unwrap_or(new.is_empty());
702        cursor = body["next_cursor"].as_str().map(str::to_string).or(cursor);
703        if new.is_empty() { stale += 1; } else { stale = 0; }
704        if hit_limit(items.len()) { info!("feed: reached limit {}", limit.unwrap_or(0)); break; }
705        if at_end || stale >= 2 || (new.is_empty() && cursor.as_deref() == Some(c.as_str())) { break; }
706    }
707    if !at_end && !hit_limit(items.len()) && !stopped_incremental {
708        err!("warning: feed paging stopped before at_end (run with --debug)");
709    }
710    Ok((stopped_incremental, pages))
711}

713/// How the profile's "Posts & replies" tab switches the feed, as found in the
714/// rendered profile HTML (the tab is server-rendered and only visible when
715/// logged in, so it is discovered at runtime rather than hard-coded).
716#[derive(Debug, Clone, PartialEq)]
717enum RepliesTab {
718    /// A `live_patch`/`navigate`/plain link: load this URL as its own feed.
719    Link(String),
720    /// A `phx-click` event pushed on the profile LiveView, with its
721    /// `phx-value-*` params (or JS `push` value) and optional `phx-target` cid.
722    Event { event: String, value: serde_json::Map<String, Value>, cid: Option<u64> },
723}

725/// True for a tab label like "Posts & replies" / "Posts and replies".
726fn is_replies_label(text: &str) -> bool {
727    let t = text.split_whitespace().collect::<Vec<_>>().join(" ").to_lowercase();
728    let t = t.replace("&amp;", "&");
729    t.contains("replies") && t.contains("posts") && (t.contains('&') || t.contains(" and "))
730}

732/// Find the "Posts & replies" tab on a profile page. Matches by visible label,
733/// falling back to attributes that name a replies tab (`phx-value-tab=replies`,
734/// `href` ending in `/replies` or containing `tab=replies`).
735fn find_replies_tab(html: &str, page_url: &str) -> Option<RepliesTab> {
736    let doc = Html::parse_document(html);
737    let sel = Selector::parse("a, button, [phx-click], [role=tab]").ok()?;
738    let attr_says_replies = |e: &scraper::node::Element| {
739        e.attrs().any(|(k, v)| {
740            let v = v.to_lowercase();
741            (k.starts_with("phx-value-") && v.contains("repl"))
742                || (k == "href" && (v.ends_with("/replies") || v.contains("tab=replies") || v.contains("with_replies")))
743        })
744    };
745    let el = doc.select(&sel)
746        .filter(|e| { let v = e.value(); v.attr("href").is_some() || v.attr("phx-click").is_some() })
747        .find(|e| is_replies_label(&e.text().collect::<String>()))
748        .or_else(|| doc.select(&sel).find(|e| attr_says_replies(e.value())))?;
749    let v = el.value();
750    let abs = |h: &str| Url::parse(page_url).ok().and_then(|b| b.join(h).ok()).map(|u| u.to_string());
751    let cid = v.attr("phx-target").and_then(|t| t.parse::<u64>().ok());
752    if let Some(click) = v.attr("phx-click") {
753        let click = click.trim();
754        if click.starts_with('[') {
755            // Phoenix.LiveView.JS command list, e.g. [["push",{"event":"tab","value":{...}}]]
756            let cmds: Value = serde_json::from_str(click).ok()?;
757            for c in cmds.as_array()? {
758                let (op, args) = (c[0].as_str().unwrap_or(""), &c[1]);
759                match op {
760                    "push" => {
761                        let value = args["value"].as_object().cloned().unwrap_or_default();
762                        let cid = args["target"].as_u64().or(cid);
763                        return Some(RepliesTab::Event { event: args["event"].as_str()?.to_string(), value, cid });
764                    }
765                    "patch" | "navigate" => return abs(args["href"].as_str()?).map(RepliesTab::Link),
766                    _ => {}
767                }
768            }
769        } else if !click.is_empty() {
770            let value = v.attrs()
771                .filter_map(|(k, val)| k.strip_prefix("phx-value-").map(|k| (k.to_string(), json!(val))))
772                .collect();
773            return Some(RepliesTab::Event { event: click.to_string(), value, cid });
774        }
775    }
776    v.attr("href").filter(|h| !h.is_empty() && *h != "#").and_then(abs).map(RepliesTab::Link)
777}

779/// Fetch the profile's "Posts & replies" feed (paged to the end), using the
780/// tab found by [`find_replies_tab`]. `None` when the profile has no such tab.
781async fn fetch_replies_feed(
782    http: &reqwest::Client, jar: &Jar, profile_url: &str, profile_html: &str,
783    known: Option<&HashSet<String>>,
784) -> Result<Option<Vec<Value>>> {
785    let Some(tab) = find_replies_tab(profile_html, profile_url) else {
786        verbose!("profile: no 'Posts & replies' tab found");
787        return Ok(None);
788    };
789    dbg_log("profile", &format!("posts & replies tab: {tab:?}"));
790    match tab {
791        RepliesTab::Link(url) => {
792            verbose!("profile: 'Posts & replies' tab -> {url}");
793            Ok(Some(fetch_feed(http, jar, &url, None, known, true).await?.items))
794        }
795        RepliesTab::Event { event, value, cid } => {
796            verbose!("profile: 'Posts & replies' tab -> phx-click {event} {value:?}");
797            let page = parse_lv(profile_url, profile_html)?;
798            let mut lv = Lv::connect(&page, jar).await?;
799            let mut ev = json!({"type": "click", "event": event, "value": Value::Object(value)});
800            if let Some(c) = cid { ev["cid"] = json!(c); }
801            let r = lv.push("event", ev).await?;
802            if r["status"] != "ok" { bail!("posts & replies tab event failed: {r}"); }
803            let body = reply_body(&r);
804            let feed = if body["posts"].is_array() { body.clone() }
805                else { props(&lv.html(), FEED_PROPS_SEL).context("no forum/Feed props after tab switch")? };
806            let mut items: Vec<Value> = feed["posts"].as_array().cloned().unwrap_or_default();
807            info!("feed posts & replies: {} posts from initial page", items.len());
808            let at_end = feed["at_end"].as_bool().unwrap_or(true);
809            let cursor = feed["next_cursor"].as_str().map(str::to_string);
810            if known.is_some_and(|k| page_all_known(&items, k)) {
811                info!("feed: initial page all known — stopping (use --full to re-crawl)");
812            } else if !at_end && cursor.is_some() {
813                page_rest(&mut lv, &mut items, cursor, None, known, true).await?;
814            }
815            Ok(Some(items))
816        }
817    }
818}

829/// Thread page props plus all replies (paged via the InfiniteScroll hook).
830async fn fetch_thread(http: &reqwest::Client, jar: &Jar, tid: &str) -> Result<(Value, Vec<Value>)> {
831    let url = THREAD_URL.replace("{id}", tid);
832    let resp = http.get(&url).send().await?;
833    let final_url = resp.url().to_string();
834    let html = resp.text().await?;
835    let th = props(&html, THREAD_PROPS_SEL).with_context(|| format!("no forum/Thread props for {tid}"))?;
836    let mut replies: Vec<Value> = th["replies"].as_array().cloned().unwrap_or_default();
837    // Thread.js has no load-more event; more replies come from the
838    // InfiniteScroll hook (`pushEventTo(this.el, "next-page", {})`),
839    // whose target cid is the element's phx-target (none = root view).
840    // New replies arrive as a diff re-rendering forum/Thread data-props.
841    let (hook, cid) = {
842        let doc = Html::parse_document(&html);
843        let sel = Selector::parse("[phx-hook=InfiniteScroll],[data-phx-hook=InfiniteScroll]").unwrap();
844        let el = doc.select(&sel).next();
845        (el.is_some(), el.and_then(|e| e.value().attr("phx-target").and_then(|t| t.parse::<u64>().ok())))
846    };
847    let total = th.pointer("/root/replies_count").and_then(Value::as_u64).unwrap_or(0) as usize;
848    if (hook || total > replies.len())
849        && let Ok(page) = parse_lv(&final_url, &html)
850        && let Ok(mut lv) = Lv::connect(&page, jar).await
851    {
852        let mut ids: HashSet<String> = replies.iter().map(|x| s(x, "id")).collect();
853        for _ in 0..MAX_PAGES {
854            let mut ev = json!({"type": "hook", "event": THREAD_NEXT_EVENT, "value": {}});
855            if let Some(c) = cid { ev["cid"] = json!(c); }
856            let Ok(r) = lv.push("event", ev).await else { break };
857            let mut got = Vec::new();
858            harvest_replies(reply_body(&r), &mut got);
859            if let Some(p) = props(&lv.html(), THREAD_PROPS_SEL) { harvest_replies(&p, &mut got); }
860            let new: Vec<Value> = got.into_iter().filter(|x| ids.insert(s(x, "id"))).collect();
861            if new.is_empty() { break; }
862            replies.extend(new);
863        }
864    }
865    Ok((th, replies))
866}

868/// Scrape a nick: their feed (root posts + their replies elsewhere) and all
869/// replies in their own threads. Also returns the profile page HTML.
870#[derive(Default)]
871struct ScrapeStats {
872    feed_items: usize,
873    new_posts: usize,
874    skipped_known: usize,
875    threads_fetched: usize,
876    threads_skipped: usize,
877    /// Items only found via the profile's "Posts & replies" tab.
878    replies_tab_items: usize,
879}
880
881async fn scrape(
882    http: &reqwest::Client, jar: &Jar, nick: &str,
883    known: &HashSet<String>, reply_counts: &HashMap<String, usize>, full: bool,
884) -> Result<(Vec<Post>, String, ScrapeStats)> {
885    let nick = nick.trim_start_matches('@');
886    let known_opt = if full { None } else { Some(known) };
887    // Profile feed is always Latest (/rs/@nick); Trending does not apply to profiles.
888    let profile_url = PROFILE_URL.replace("{nick}", nick);
889    let mut feed = fetch_feed(http, jar, &profile_url, None, known_opt, true).await?;
890    // Also walk the "Posts & replies" tab (the nick's replies in other threads).
891    let mut replies_tab_added = 0;
892    match fetch_replies_feed(http, jar, &profile_url, &feed.html, known_opt).await {
893        Ok(Some(extra)) => {
894            replies_tab_added = merge_feed_items(&mut feed.items, extra);
895            info!("@{nick}: posts & replies tab added {replies_tab_added} items");
896        }
897        Ok(None) => {}
898        Err(e) => err!("warning: posts & replies tab: {e:#}"),
899    }
900    let mut seen = HashSet::new();
901    let mut out = Vec::new();
902    let mut roots: Vec<(String, usize)> = Vec::new(); // (id, replies_count from feed)
903    let mut stats = ScrapeStats { feed_items: feed.items.len(), replies_tab_items: replies_tab_added, ..Default::default() };
904    for it in &feed.items {
905        let id = s(it, "id");
906        if id.is_empty() || !seen.insert(id.clone()) { continue; }
907        let thread = s(it, "thread_id");
908        let is_root = thread.is_empty() || thread == id;
909        let already = known.contains(&id);
910        if already && !full {
911            stats.skipped_known += 1;
912        } else {
913            stats.new_posts += 1;
914        }
915        if is_root {
916            let rc = it.get("replies_count").and_then(Value::as_u64).unwrap_or(0) as usize;
917            roots.push((id.clone(), rc));
918            if !already || full {
919                out.push(to_post(it, "", nick));
920            }
921        } else if !already || full {
922            out.push(to_post(it, &thread, nick));
923        }
924    }
925    for (tid, feed_rc) in roots {
926        let db_rc = reply_counts.get(&tid).copied().unwrap_or(0);
927        let need = full || !known.contains(&tid) || feed_rc > db_rc;
928        if !need {
929            stats.threads_skipped += 1;
930            continue;
931        }
932        stats.threads_fetched += 1;
933        let (_, replies) = match fetch_thread(http, jar, &tid).await {
934            Ok(x) => x,
935            Err(e) => { err!("warning: {e:#}"); continue; }
936        };
937        let mut n = 0;
938        for rp in &replies {
939            let id = s(rp, "id");
940            if id.is_empty() || id == tid || !seen.insert(id.clone()) { continue; }
941            if known.contains(&id) && !full {
942                stats.skipped_known += 1;
943                continue;
944            }
945            out.push(to_post(rp, &tid, nick));
946            stats.new_posts += 1;
947            n += 1;
948        }
949        verbose!("thread {tid}: {n} new replies (feed says {feed_rc}, db had {db_rc})");
950    }
951    Ok((out, feed.html, stats))
952}

961        |r| r.get::<_, i64>(0)).unwrap_or(0) > 0;

963    let mut st = c.prepare("SELECT post_id FROM posts")?;
964    for id in st.query_map([], |r| r.get::<_, String>(0))? { out.insert(id?); }
965    Ok(out)
966}

968fn db_reply_counts(db: &std::path::Path) -> Result<HashMap<String, usize>> {

979        let (k, v) = row?;

995fn list_nicks_db(db: &std::path::Path) -> Result<Vec<NickInfo>> {
996    let mut map: HashMap<String, NickInfo> = HashMap::new();
997    if !db.exists() { return Ok(Vec::new()); }
998    let c = Connection::open(db)?;
999    let has_posts: bool = c.query_row(
1000        "SELECT count(*) FROM sqlite_master WHERE type='table' AND name='posts'", [],
1001        |r| r.get::<_, i64>(0)).unwrap_or(0) > 0;
1002    if has_posts {
1003        let mut st = c.prepare(
1004            "SELECT nick, COUNT(*), MAX(posted_at) FROM posts GROUP BY lower(nick)")?;
1005        for row in st.query_map([], |r| Ok((r.get::<_, String>(0)?, r.get::<_, i64>(1)?, r.get::<_, Option<String>>(2)?)))? {
1006            let (nick, cnt, latest) = row?;
1007            let key = nick.to_lowercase();
1008            map.insert(key, NickInfo {
1009                nick, display_name: None, post_count: cnt, latest,
1010                profile_scraped: false, scraped_at: None,
1011            });
1012        }
1013    }
1014    let has_p: bool = c.query_row(
1015        "SELECT count(*) FROM sqlite_master WHERE type='table' AND name='profiles'", [],
1016        |r| r.get::<_, i64>(0)).unwrap_or(0) > 0;
1017    if has_p {
1018        let mut st = c.prepare("SELECT nick, display_name, scraped_at FROM profiles")?;
1019        for row in st.query_map([], |r| Ok((r.get::<_, String>(0)?, r.get::<_, Option<String>>(1)?, r.get::<_, Option<String>>(2)?)))? {
1020            let (nick, dn, sa) = row?;
1021            let key = nick.to_lowercase();
1022            let e = map.entry(key).or_insert_with(|| NickInfo {
1023                nick: nick.clone(), display_name: None, post_count: 0, latest: None,
1024                profile_scraped: false, scraped_at: None,
1025            });
1026            e.nick = nick;
1027            e.display_name = dn.or(e.display_name.clone());
1028            e.profile_scraped = true;
1029            e.scraped_at = sa;
1030        }
1031    }
1032    let mut v: Vec<_> = map.into_values().collect();
1033    v.sort_by_key(|a| a.nick.to_lowercase());
1034    Ok(v)
1035}

1186#[allow(clippy::too_many_arguments)]
1187async fn crawl_mentions(
1188    http: &reqwest::Client, jar: &Jar, targets: &[mentions::MentionTarget],
1189    limit: usize, seen: &mut HashSet<String>, known: &HashSet<String>, full: bool,
1190    mode: FeedMode,
1191) -> Result<(Vec<Post>, Vec<mentions::MentionHit>)> {
1192    let allow_inc = mode.allow_incremental_stop() && !full;
1193    let known_opt = if allow_inc { Some(known) } else { None };
1194    let url = mode.site_feed_url();
1195    info!("mentions feed: {url} (mode={mode})");
1196    let feed = fetch_feed(http, jar, url, Some(limit), known_opt, allow_inc).await?;
1197    let mut posts = Vec::new();
1198    let mut hits = Vec::new();
1199    let mut checked = HashSet::new();
1200    for it in &feed.items {
1201        let id = s(it, "id");
1202        let tid = { let t = s(it, "thread_id"); if t.is_empty() { id.clone() } else { t } };
1203        let mut any = false;
1204        for t in targets {
1205            if !mentions::match_post(it, t).is_empty() { any = true; break; }
1206        }
1207        if !any { continue; }
1208        if !checked.insert(tid.clone()) { continue; }
1209        let (th, replies) = match fetch_thread(http, jar, &tid).await {
1210            Ok(x) => x,
1211            Err(e) => { err!("warning: mention thread {tid}: {e:#}"); continue; }
1212        };
1213        let root = th.get("root").cloned().unwrap_or(th.clone());
1214        let mut thread_posts = vec![root];
1215        thread_posts.extend(replies);
1216        for tp in &thread_posts {
1217            let pid = s(tp, "id");
1218            if pid.is_empty() { continue; }
1219            for t in targets {
1220                for mt in mentions::match_post(tp, t) {
1221                    hits.push(mentions::MentionHit {
1222                        post_id: pid.clone(), mentioned_nick: t.nick.clone(), match_type: mt,
1223                    });
1224                }
1225            }
1226            if !seen.insert(pid.clone()) { continue; }
1227            let parent = {
1228                let thid = s(tp, "thread_id");
1229                if thid.is_empty() || thid == pid { String::new() } else { thid }
1230            };
1231            let author = tp.pointer("/author/username").and_then(Value::as_str).unwrap_or("");
1232            posts.push(to_post(tp, &parent, author));
1233        }
1234    }
1235    Ok((posts, hits))
1236}

1412/// "@a, b,,@A" -> ["a", "b"]: strip @, trim, drop empties, dedupe (case-insensitive, first wins).
1413fn parse_nicks(s: &str) -> std::result::Result<NickList, String> {
1414    let mut seen = HashSet::new();
1415    let v: Vec<String> = s.split(',')
1416        .map(|n| n.trim().trim_start_matches('@').trim().to_string())
1417        .filter(|n| !n.is_empty() && seen.insert(n.to_lowercase()))
1418        .collect();
1419    if v.is_empty() { return Err("expected at least one nick".into()); }
1420    if let Some(bad) = v.iter().find(|n| n.contains(char::is_whitespace) || n.contains('/')) {
1421        return Err(format!("invalid nick {bad:?}"));
1422    }
1423    Ok(NickList(v))
1424}

1435        if let Some(n) = &self.nicks { return Ok(n.0.clone()); }
1436        match &self.nick { Some(n) => Ok(parse_nicks(n).map_err(|e| anyhow!(e))?.0), None => Ok(Vec::new()) }
1437    }
1438
1439    /// Selected archive format (clap guarantees --compress <=> exactly one format flag).
1440    fn compression(&self) -> Option<export::Compress> {
1441        if !self.compress { return None; }
1442        if self.tar_gz { Some(export::Compress::TarGz) } else { None }
1443    }
1444    fn want_mentions(&self) -> bool { self.mentions }
1445    /// Default is Latest (our default), even though the site lands on Trending.
1446    fn feed_mode(&self) -> FeedMode {
1447        if self.trending { FeedMode::Trending } else { FeedMode::Latest }
1448    }
1449}

1451/// `--remove-nick`: preview (rolled-back transaction), confirm, then commit.
1452fn run_remove_nick(args: &Args, nicks: &[String]) -> Result<()> {
1453    let db = home()?.join(DB_REL_PATH);
1454    let att = home()?.join(ATTACH_REL_PATH);
1455    let preview = maint::remove_nicks(&db, &att, nicks, args.keep_replies, false)?;
1456    let who = nicks.iter().map(|n| format!("@{n}")).collect::<Vec<_>>().join(", ");
1457    eprintln!("{}", preview.summary(&format!("remove {who} would delete")));
1458    if args.keep_replies { eprintln!("--keep-replies: other people's replies in their threads are kept"); }
1459    if args.dry_run {
1460        for f in &preview.files { eprintln!("  file: {}", f.display()); }
1461        eprintln!("dry run: nothing changed");
1462        return Ok(());
1463    }
1464    if preview.is_empty() {
1465        eprintln!("nothing to remove");
1466        return Ok(());
1467    }
1468    if !maint::confirm("Proceed?", false, args.yes)? {
1469        eprintln!("aborted: nothing changed");
1470        return Ok(());
1471    }
1472    let done = maint::remove_nicks(&db, &att, nicks, args.keep_replies, true)?;
1473    eprintln!("{}", done.summary("removed"));
1474    Ok(())
1475}

1477/// `--clear-all`: delete the database and attachments dir; prefs and exports stay.
1478fn run_clear_all(args: &Args) -> Result<()> {
1479    let db = home()?.join(DB_REL_PATH);
1480    let att = home()?.join(ATTACH_REL_PATH);
1481    let targets = maint::clear_all_targets(&db, &att);
1482    let (n, bytes) = maint::dir_usage(&att);
1483    if targets.is_empty() {
1484        eprintln!("nothing to clear ({} and {} do not exist)", db.display(), att.display());
1485        return Ok(());
1486    }
1487    eprintln!("clear-all will delete:");
1488    for t in &targets {
1489        if t.is_dir() { eprintln!("  {}/ ({n} files, {bytes} bytes)", t.display()); } else { eprintln!("  {}", t.display()); }
1490    }
1491    eprintln!("kept: {} and {}", home()?.join(PREFS_REL_PATH).display(), home()?.join(EXPORTS_REL_PATH).display());
1492    if args.dry_run {
1493        eprintln!("dry run: nothing changed");
1494        return Ok(());
1495    }
1496    if !maint::confirm("Delete all scraped data?", true, args.yes)? {
1497        eprintln!("aborted: nothing changed");
1498        return Ok(());
1499    }
1500    let gone = maint::clear_all(&db, &att)?;
1501    eprintln!("cleared: {} paths removed", gone.len());
1502    Ok(())
1503}

1591    info!("== mode: {} (site crawl); profiles always use Latest", args.feed_mode());
1592    info!("== targets");

1599        } else {
1600            info!("  @{nick}: new");
1601        }
1602    }
1603    let mut posts: Vec<Post> = Vec::new();
1604    let mut profiles: Vec<mentions::Profile> = Vec::new();
1605    let mut mention_hits: Vec<mentions::MentionHit> = Vec::new();
1606    let mut targets: Vec<mentions::MentionTarget> = Vec::new();
1607    let mut seen = HashSet::new();
1608    let mut summary = Vec::new();
1609    let mut ok_nicks = Vec::new();

1611        info!("== scraping @{nick}");

1613            Ok((ps, html, st)) => {

1655            }
1656            Err(e) => {

1659            }
1660        }

1665            Ok((extra, hits)) => {
1666                let inserted = save(&extra)?;

1668                info!("mentions: {} posts, {} hits, {inserted} new DB rows", extra.len(), hits.len());
1669                posts.extend(extra);
1670                mention_hits.extend(hits);
1671            }
1672            Err(e) => err!("warning: mentions crawl failed: {e:#}"),
1673        }
1674    }
1675    info!("== summary");
1676    for l in &summary { info!("  {l}"); }
1677    if ok_nicks.is_empty() { bail!("all nicks failed"); }
1678    let atts = if args.no_attachments { Vec::new() } else {
1679        let for_dl: Vec<Post> = if args.full {
1680            posts.clone()
1681        } else {
1682            // Only download media for posts we didn't already have.
1683            posts.iter().filter(|p| !known.contains(&p.post_id)).cloned().collect()
1684        };
1685        info!("attachments: downloading for {} posts ({} skipped as known)",

1690    };
1691    let name = clean(args.name.clone().unwrap_or_else(|| nicks_name(&ok_nicks, &stamp)));
1692    let label = ok_nicks.iter().map(|n| format!("@{n}")).collect::<Vec<_>>().join(", ");
1693    export::export(&args, &name, &label, &posts, &atts, &profiles, &mention_hits, &home()?.join(EXPORTS_REL_PATH))?;
1694
1695    Ok(())
1696}
