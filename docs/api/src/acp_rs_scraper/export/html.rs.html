5/// Escape text for HTML element content and attribute values.
6pub fn esc(s: &str) -> String {
7    s.replace('&', "&amp;").replace('<', "&lt;").replace('>', "&gt;").replace('"', "&quot;")
8}

10/// Every `post_id` in an export's post tree.
11pub fn export_ids(posts: &Value, out: &mut HashSet<String>) {
12    for p in posts.as_array().into_iter().flatten() {
13        if let Some(id) = p["post_id"].as_str().filter(|x| !x.is_empty()) { out.insert(id.to_string()); }
14        export_ids(&p["children"], out);
15    }
16}

18/// Link target for post `id` (in thread `root`): its `#post-{id}` anchor when
19/// it is in this export, else its page on the site (`/rs/post/{root}/{id}`
20/// for replies, `/rs/post/{id}` for thread roots, as the site links them).
21pub fn post_href(id: &str, root: &str, ids: &HashSet<String>) -> String {
22    if ids.contains(id) { format!("#post-{id}") }
23    else if root.is_empty() || root == id { format!("{}/rs/post/{id}", base_url()) }
24    else { format!("{}/rs/post/{root}/{id}", base_url()) }
25}

27/// "Replying to" line for a reply: the immediate parent, plus the thread root when that differs.
28pub fn reply_links(root: &str, direct: &str, ids: &HashSet<String>) -> String {
29    if root.is_empty() { return String::new(); }
30    let direct = if direct.is_empty() { root } else { direct };
31    let a = |id: &str, r: &str, cls: &str| format!("<a class=\"{cls}\" href=\"{}\">{}</a>", esc(&post_href(id, r, ids)), esc(id));
32    let mut s = format!("<div class=\"meta reply-to\">Replying to {}", a(direct, root, "parent"));
33    if direct != root { s.push_str(&format!(" · thread {}", a(root, root, "thread"))); }
34    s.push_str("</div>");
35    s
36}

38/// (top-level posts, replies) over a whole export post tree.
39pub fn count_kinds(posts: &Value) -> (usize, usize) {
40    let mut n = (0, 0);
41    for p in posts.as_array().into_iter().flatten() {
42        if p["parent_post_id"].as_str().unwrap_or_default().is_empty() { n.0 += 1 } else { n.1 += 1 }
43        let (a, b) = count_kinds(&p["children"]);
44        n.0 += a; n.1 += b;
45    }
46    n
47}

49/// Append one post (and, recursively, its replies) as nested HTML.
50pub fn html_post(v: &Value, out: &mut String, depth: usize, ids: &HashSet<String>) {
51    let g = |k: &str| v[k].as_str().unwrap_or_default().to_string();
52    let kind = if g("parent_post_id").is_empty() { "post" } else { "reply" };
53    out.push_str(&format!("<article class=\"post kind-{kind} d{}\" id=\"post-{}\">", depth.min(6), esc(&g("post_id"))));
54    if !g("title").is_empty() { out.push_str(&format!("<h2>{}</h2>", esc(&g("title")))); }
55    out.push_str(&format!("<div class=\"meta\"><span class=\"badge {kind}\">{}</span> <b>@{}</b> · <time>{}</time></div>", if kind == "post" { "Post" } else { "Reply" }, esc(&g("nick")), esc(&g("posted_at"))));
56    out.push_str(&reply_links(&g("parent_post_id"), &g("in_reply_to_id"), ids));
57    out.push_str(&format!("<div class=\"body\">{}</div>", paragraphs_html(&g("content"))));
58    for a in v["attachments"].as_array().into_iter().flatten() {
59        let p = esc(a["local_path"].as_str().unwrap_or_default());
60        let m = a["mime_type"].as_str().unwrap_or_default();
61        if a["kind"] == "thumbnail" { continue; }
62        out.push_str("<div class=\"att\">");
63        if m.starts_with("image/") { out.push_str(&format!("<img src=\"{p}\" loading=\"lazy\">")); }
64        else if m.starts_with("video/") { out.push_str(&format!("<video src=\"{p}\" controls preload=\"metadata\"></video>")); }
65        else if m.starts_with("audio/") { out.push_str(&format!("<audio src=\"{p}\" controls></audio>")); }
66        else { out.push_str(&format!("<a href=\"{p}\">{p}</a>")); }
67        out.push_str("</div>");
68    }
69    for c in v["children"].as_array().into_iter().flatten() { html_post(c, out, depth + 1, ids); }
70    out.push_str("</article>");
71}

73/// Render an export document as one self-contained HTML page, embedding the JSON for re-import.
74pub fn build_html(doc: &Value) -> String {
75    let title = esc(doc["title"].as_str().or(doc["nick"].as_str()).unwrap_or_default());
76    let nick = title.clone();
77    let mut body = String::new();
78    for pr in doc["profiles"].as_array().into_iter().flatten() {
79        let dn = pr["display_name"].as_str().unwrap_or("");
80        let un = pr["nick"].as_str().unwrap_or("");
81        let bio = pr["bio"].as_str().unwrap_or("");
82        let av = pr["avatar_path"].as_str().or(pr["avatar_url"].as_str()).unwrap_or("");
83        body.push_str("<header class=\"profile\">");
84        if !av.is_empty() { body.push_str(&format!("<img class=\"avatar\" src=\"{}\" alt=\"\">", esc(av))); }
85        body.push_str(&format!("<div><h2>{} <span class=\"meta\">@{}</span></h2>", esc(dn), esc(un)));
86        if !bio.is_empty() { body.push_str(&format!("<p>{}</p>", esc(bio))); }
87        let bits: Vec<String> = ["location","role","joined_at"].iter().filter_map(|k| pr[*k].as_str().filter(|s| !s.is_empty()).map(|s| format!("{k}: {}", esc(s)))).collect();
88        if !bits.is_empty() { body.push_str(&format!("<p class=\"meta\">{}</p>", bits.join(" · "))); }
89        body.push_str("</div></header>");
90    }
91    let mut ids = HashSet::new();
92    export_ids(&doc["posts"], &mut ids);
93    // Thread roots (top-level posts) and replies in separate sections, so a
94    // nick's own posts are never lost among their replies.
95    let top: Vec<&Value> = doc["posts"].as_array().into_iter().flatten().collect();
96    let (roots, orphans): (Vec<&Value>, Vec<&Value>) = top.into_iter()
97        .partition(|p| p["parent_post_id"].as_str().unwrap_or_default().is_empty());
98    if doc["research"].is_object() { body.push_str(&research_html(&doc["research"], &ids)); }
99    let (np, nr) = count_kinds(&doc["posts"]);
100    body.push_str(&format!("<p class=\"meta counts\">{np} posts · {nr} replies</p>"));
101    body.push_str(&format!("<h2 id=\"posts\">Posts ({})</h2>", roots.len()));
102    if roots.is_empty() { body.push_str("<p class=\"meta\">No top-level posts in this export.</p>"); }
103    for p in roots { html_post(p, &mut body, 0, &ids); }
104    if !orphans.is_empty() {
105        body.push_str(&format!("<h2 id=\"replies\">Replies in other threads ({})</h2>", orphans.len()));
106        for p in orphans { html_post(p, &mut body, 0, &ids); }
107    }
108    let mentions = doc["mentions"].as_array().cloned().unwrap_or_default();
109    if !mentions.is_empty() {
110        body.push_str("<h2>Mentions</h2><ul class=\"mentions\">");
111        for h in &mentions {
112            body.push_str(&format!(
113                "<li><a href=\"#post-{}\">{}</a> → @{} <span class=\"meta\">({})</span></li>",
114                esc(h["post_id"].as_str().unwrap_or("")),
115                esc(h["post_id"].as_str().unwrap_or("")),
116                esc(h["mentioned_nick"].as_str().unwrap_or("")),
117                esc(h["match_type"].as_str().unwrap_or("")),
118            ));
119        }
120        body.push_str("</ul>");
121    }
122    format!(r#"<!DOCTYPE html><html lang="en"><head><meta charset="utf-8">
123<meta name="viewport" content="width=device-width,initial-scale=1"><title>{nick}</title>
124<style>
125body{{font:16px/1.5 system-ui,sans-serif;max-width:800px;margin:2em auto;padding:0 1em;color:#18181b;background:#fafafa}}
126.post{{background:#fff;border:1px solid #e4e4e7;border-radius:8px;padding:.8em 1em;margin:.8em 0}}
127.post .post{{margin-left:1.2em;border-left:3px solid #dc2626;border-radius:0 8px 8px 0}}
128h2{{margin:.2em 0;font-size:1.25em}} .meta{{color:#71717a;font-size:.85em}} .body{{margin:.5em 0;white-space:normal}} .body p{{margin:0 0 .7em}}
129.att img,.att video{{max-width:100%;border-radius:6px}} .att{{margin:.4em 0}}
130.profile{{display:flex;gap:1em;align-items:flex-start;background:#fff;border:1px solid #e4e4e7;border-radius:8px;padding:1em;margin:1em 0}}
131.profile .avatar{{width:72px;height:72px;border-radius:50%;object-fit:cover}}
132.badge{{display:inline-block;font-size:.75em;font-weight:600;padding:0 .45em;border-radius:4px;color:#fff;background:#71717a}} .badge.post{{background:#dc2626}}
133.post.kind-post.d0{{border-left:4px solid #dc2626}}
134.research{{background:#fff;border:1px solid #e4e4e7;border-radius:8px;padding:.8em 1em;margin:1em 0}} .research-items li{{margin:.5em 0}}
135.note{{margin:.3em 0 .3em .5em;padding-left:.7em;border-left:3px solid #dc2626;color:#3f3f46}}
136.mentions{{padding-left:1.2em}} .reply-to a{{color:#dc2626}}
137</style><script type="application/json" id="{data_id}">{data}</script></head><body><h1>{nick}</h1><p class="meta">Exported {}</p>{body}</body></html>"#,
138        esc(doc["exported_at"].as_str().unwrap_or_default()),
139        data_id = HTML_DATA_ID,
140        data = doc.to_string().replace('<', "\\u003c"))
141}

143/// The research section: description, then every item with its kind, title, note and source.
144pub fn research_html(r: &Value, ids: &HashSet<String>) -> String {
145    let s = |v: &Value, k: &str| v[k].as_str().unwrap_or_default().to_string();
146    let items = r["items"].as_array().cloned().unwrap_or_default();
147    let mut h = format!("<section class=\"research\"><h2 id=\"research\">Research: {} <span class=\"meta\">({} items)</span></h2>",
148        esc(&s(r, "name")), items.len());
149    if !s(r, "description").is_empty() { h.push_str(&paragraphs_html(&s(r, "description"))); }
150    h.push_str("<ol class=\"research-items\">");
151    for i in &items {
152        let kind = s(i, "kind");
153        let (title, rf) = (s(i, "title"), s(i, "ref"));
154        h.push_str(&format!("<li class=\"item kind-{}\"><span class=\"badge\">{}</span> ", esc(&kind), esc(&kind)));
155        match kind.as_str() {
156            "post" | "reply" | "search-result" => {
157                let href = if ids.contains(&rf) { format!("#post-{}", esc(&rf)) } else { format!("{}/rs/post/{}", esc(base_url()), esc(&rf)) };
158                h.push_str(&format!("<a href=\"{href}\">{}</a> <span class=\"meta\">{}</span>", esc(&title), esc(&rf)));
159                if let Some(q) = i["meta"]["query"].as_str() { h.push_str(&format!(" <span class=\"meta\">search: {}</span>", esc(q))); }
160            }
161            "web-url" => {
162                h.push_str(&format!("<a href=\"{}\">{}</a> <span class=\"meta\">{}</span>", esc(&rf), esc(&title), esc(&rf)));
163                if let Some(f) = i["file"].as_str() { h.push_str(&format!(" · <a href=\"{}\">snapshot</a>", esc(f))); }
164            }
165            _ => {
166                let link = i["file"].as_str().map(str::to_string).unwrap_or_else(|| format!("file://{rf}"));
167                h.push_str(&format!("<a href=\"{}\">{}</a> <span class=\"meta\">{}</span>", esc(&link), esc(&title), esc(&rf)));
168            }
169        }
170        h.push_str(&format!(" <span class=\"meta\">added {}</span>", esc(&s(i, "added_at"))));
171        if !s(i, "note").is_empty() { h.push_str(&format!("<blockquote class=\"note\">{}</blockquote>", paragraphs_html(&s(i, "note")))); }
172        let content = s(i, "content");
173        if !content.trim().is_empty() {
174            h.push_str(&format!("<details><summary>Text ({} characters)</summary><div class=\"body\">{}</div></details>",
175                content.chars().count(), paragraphs_html(&content)));
176        }
177        h.push_str("</li>");
178    }
179    h.push_str("</ol></section>");
180    h
181}

183/// Body text as HTML paragraphs: blank lines split `<p>`s, single newlines
184/// inside a paragraph become `<br>`.
185pub fn paragraphs_html(text: &str) -> String {
186    text.split("\n\n").map(str::trim).filter(|p| !p.is_empty())
187        .map(|p| format!("<p>{}</p>", esc(p).replace('\n', "<br>")))
188        .collect()
189}

195    #[test]
196    fn posts_and_replies_get_separate_sections_and_badges() {
197        let doc = json!({"nick": "@haz", "posts": [
198            {"post_id": "R1", "parent_post_id": "", "nick": "haz", "content": "my post", "children": [
199                {"post_id": "R1a", "parent_post_id": "R1", "nick": "haz", "content": "self reply", "children": []}]},
200            {"post_id": "C1", "parent_post_id": "T9", "nick": "haz", "content": "comment", "children": []},
201        ]});
202        assert_eq!(count_kinds(&doc["posts"]), (1, 2));
203        let h = build_html(&doc);
204        let posts = h.find("<h2 id=\"posts\">Posts (1)</h2>").expect("posts section");
205        let replies = h.find("<h2 id=\"replies\">Replies in other threads (1)</h2>").expect("replies section");
206        let r1 = h.find("id=\"post-R1\"").unwrap();
207        let c1 = h.find("id=\"post-C1\"").unwrap();
208        assert!(posts < r1 && r1 < replies && replies < c1, "posts before replies");
209        assert!(h.contains("class=\"post kind-post d0\" id=\"post-R1\""));
210        assert!(h.contains("class=\"post kind-reply d0\" id=\"post-C1\""));
211        assert!(h.contains("1 posts · 2 replies"));
212    }

214    #[test]
215    fn reply_links_prefer_in_export_anchors() {
216        let ids: HashSet<String> = ["c1".to_string()].into();
217        let h = reply_links("t1", "c1", &ids);
218        assert!(h.contains("class=\"parent\" href=\"#post-c1\">c1</a>"), "{h}");
219        assert!(h.contains(&format!("class=\"thread\" href=\"{}/rs/post/t1\">t1</a>", base_url())), "{h}");
220        // Parent not exported: link to the reply on the site; root-only reply has no thread link.
221        let h = reply_links("t1", "c2", &HashSet::new());
222        assert!(h.contains(&format!("href=\"{}/rs/post/t1/c2\"", base_url())), "{h}");
223        assert!(!reply_links("t1", "t1", &ids).contains("thread"));
224        assert_eq!(reply_links("", "", &ids), "");
225    }
