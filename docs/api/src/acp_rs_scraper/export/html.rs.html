5/// Escape text for HTML element content and attribute values.
6pub fn esc(s: &str) -> String {
7    s.replace('&', "&amp;").replace('<', "&lt;").replace('>', "&gt;").replace('"', "&quot;")
8}
9
10/// Every `post_id` in an export's post tree.
11pub fn export_ids(posts: &Value, out: &mut HashSet<String>) {
12    for p in posts.as_array().into_iter().flatten() {
13        if let Some(id) = p["post_id"].as_str().filter(|x| !x.is_empty()) { out.insert(id.to_string()); }
14        export_ids(&p["children"], out);
15    }
16}

18/// Link target for post `id` (in thread `root`): its `#post-{id}` anchor when
19/// it is in this export, else its page on the site (`/rs/post/{root}/{id}`
20/// for replies, `/rs/post/{id}` for thread roots, as the site links them).
21pub fn post_href(id: &str, root: &str, ids: &HashSet<String>) -> String {
22    if ids.contains(id) { format!("#post-{id}") }
23    else if root.is_empty() || root == id { format!("{}/rs/post/{id}", base_url()) }
24    else { format!("{}/rs/post/{root}/{id}", base_url()) }
25}
26
27/// "Replying to" line for a reply: the immediate parent, plus the thread root when that differs.
28pub fn reply_links(root: &str, direct: &str, ids: &HashSet<String>) -> String {
29    if root.is_empty() { return String::new(); }
30    let direct = if direct.is_empty() { root } else { direct };
31    let a = |id: &str, r: &str, cls: &str| format!("<a class=\"{cls}\" href=\"{}\">{}</a>", esc(&post_href(id, r, ids)), esc(id));
32    let mut s = format!("<div class=\"meta reply-to\">Replying to {}", a(direct, root, "parent"));
33    if direct != root { s.push_str(&format!(" · thread {}", a(root, root, "thread"))); }
34    s.push_str("</div>");
35    s
36}

38/// Append one post (and, recursively, its replies) as nested HTML.
39pub fn html_post(v: &Value, out: &mut String, depth: usize, ids: &HashSet<String>) {
40    let g = |k: &str| v[k].as_str().unwrap_or_default().to_string();
41    out.push_str(&format!("<article class=\"post d{}\" id=\"post-{}\">", depth.min(6), esc(&g("post_id"))));
42    if !g("title").is_empty() { out.push_str(&format!("<h2>{}</h2>", esc(&g("title")))); }
43    out.push_str(&format!("<div class=\"meta\"><b>@{}</b> · <time>{}</time></div>", esc(&g("nick")), esc(&g("posted_at"))));
44    out.push_str(&reply_links(&g("parent_post_id"), &g("in_reply_to_id"), ids));
45    out.push_str(&format!("<div class=\"body\">{}</div>", esc(&g("content")).replace('\n', "<br>")));
46    for a in v["attachments"].as_array().into_iter().flatten() {
47        let p = esc(a["local_path"].as_str().unwrap_or_default());
48        let m = a["mime_type"].as_str().unwrap_or_default();
49        if a["kind"] == "thumbnail" { continue; }
50        out.push_str("<div class=\"att\">");
51        if m.starts_with("image/") { out.push_str(&format!("<img src=\"{p}\" loading=\"lazy\">")); }
52        else if m.starts_with("video/") { out.push_str(&format!("<video src=\"{p}\" controls preload=\"metadata\"></video>")); }
53        else if m.starts_with("audio/") { out.push_str(&format!("<audio src=\"{p}\" controls></audio>")); }
54        else { out.push_str(&format!("<a href=\"{p}\">{p}</a>")); }
55        out.push_str("</div>");
56    }
57    for c in v["children"].as_array().into_iter().flatten() { html_post(c, out, depth + 1, ids); }
58    out.push_str("</article>");
59}

61/// Render an export document as one self-contained HTML page, embedding the JSON for re-import.
62pub fn build_html(doc: &Value) -> String {
63    let title = esc(doc["title"].as_str().or(doc["nick"].as_str()).unwrap_or_default());
64    let nick = title.clone();
65    let mut body = String::new();
66    for pr in doc["profiles"].as_array().into_iter().flatten() {
67        let dn = pr["display_name"].as_str().unwrap_or("");
68        let un = pr["nick"].as_str().unwrap_or("");
69        let bio = pr["bio"].as_str().unwrap_or("");
70        let av = pr["avatar_path"].as_str().or(pr["avatar_url"].as_str()).unwrap_or("");
71        body.push_str("<header class=\"profile\">");
72        if !av.is_empty() { body.push_str(&format!("<img class=\"avatar\" src=\"{}\" alt=\"\">", esc(av))); }
73        body.push_str(&format!("<div><h2>{} <span class=\"meta\">@{}</span></h2>", esc(dn), esc(un)));
74        if !bio.is_empty() { body.push_str(&format!("<p>{}</p>", esc(bio))); }
75        let bits: Vec<String> = ["location","role","joined_at"].iter().filter_map(|k| pr[*k].as_str().filter(|s| !s.is_empty()).map(|s| format!("{k}: {}", esc(s)))).collect();
76        if !bits.is_empty() { body.push_str(&format!("<p class=\"meta\">{}</p>", bits.join(" · "))); }
77        body.push_str("</div></header>");
78    }
79    body.push_str("<h2>Posts</h2>");
80    let mut ids = HashSet::new();
81    export_ids(&doc["posts"], &mut ids);
82    for p in doc["posts"].as_array().into_iter().flatten() { html_post(p, &mut body, 0, &ids); }
83    let mentions = doc["mentions"].as_array().cloned().unwrap_or_default();
84    if !mentions.is_empty() {
85        body.push_str("<h2>Mentions</h2><ul class=\"mentions\">");
86        for h in &mentions {
87            body.push_str(&format!(
88                "<li><a href=\"#post-{}\">{}</a> → @{} <span class=\"meta\">({})</span></li>",
89                esc(h["post_id"].as_str().unwrap_or("")),
90                esc(h["post_id"].as_str().unwrap_or("")),
91                esc(h["mentioned_nick"].as_str().unwrap_or("")),
92                esc(h["match_type"].as_str().unwrap_or("")),
93            ));
94        }
95        body.push_str("</ul>");
96    }
97    format!(r#"<!DOCTYPE html><html lang="en"><head><meta charset="utf-8">
98<meta name="viewport" content="width=device-width,initial-scale=1"><title>{nick}</title>
99<style>
100body{{font:16px/1.5 system-ui,sans-serif;max-width:800px;margin:2em auto;padding:0 1em;color:#18181b;background:#fafafa}}
101.post{{background:#fff;border:1px solid #e4e4e7;border-radius:8px;padding:.8em 1em;margin:.8em 0}}
102.post .post{{margin-left:1.2em;border-left:3px solid #dc2626;border-radius:0 8px 8px 0}}
103h2{{margin:.2em 0;font-size:1.25em}} .meta{{color:#71717a;font-size:.85em}} .body{{margin:.5em 0;white-space:normal}}
104.att img,.att video{{max-width:100%;border-radius:6px}} .att{{margin:.4em 0}}
105.profile{{display:flex;gap:1em;align-items:flex-start;background:#fff;border:1px solid #e4e4e7;border-radius:8px;padding:1em;margin:1em 0}}
106.profile .avatar{{width:72px;height:72px;border-radius:50%;object-fit:cover}}
107.mentions{{padding-left:1.2em}} .reply-to a{{color:#dc2626}}
108</style><script type="application/json" id="{data_id}">{data}</script></head><body><h1>{nick}</h1><p class="meta">Exported {}</p>{body}</body></html>"#,
109        esc(doc["exported_at"].as_str().unwrap_or_default()),
110        data_id = HTML_DATA_ID,
111        data = doc.to_string().replace('<', "\\u003c"))
112}

114#[cfg(test)]
115mod tests {
116    use super::*;
117
118    #[test]
119    fn reply_links_prefer_in_export_anchors() {
120        let ids: HashSet<String> = ["c1".to_string()].into();
121        let h = reply_links("t1", "c1", &ids);
122        assert!(h.contains("class=\"parent\" href=\"#post-c1\">c1</a>"), "{h}");
123        assert!(h.contains(&format!("class=\"thread\" href=\"{}/rs/post/t1\">t1</a>", base_url())), "{h}");
124        // Parent not exported: link to the reply on the site; root-only reply has no thread link.
125        let h = reply_links("t1", "c2", &HashSet::new());
126        assert!(h.contains(&format!("href=\"{}/rs/post/t1/c2\"", base_url())), "{h}");
127        assert!(!reply_links("t1", "t1", &ids).contains("thread"));
128        assert_eq!(reply_links("", "", &ids), "");
129    }
