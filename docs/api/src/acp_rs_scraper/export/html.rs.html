5/// Escape text for HTML element content and attribute values.
6pub fn esc(s: &str) -> String {
7    s.replace('&', "&amp;").replace('<', "&lt;").replace('>', "&gt;").replace('"', "&quot;")
8}
9
10/// Append one post (and, recursively, its replies) as nested HTML.
11pub fn html_post(v: &Value, out: &mut String, depth: usize) {
12    let g = |k: &str| v[k].as_str().unwrap_or_default().to_string();
13    out.push_str(&format!("<article class=\"post d{}\" id=\"p-{}\">", depth.min(6), esc(&g("post_id"))));
14    if !g("title").is_empty() { out.push_str(&format!("<h2>{}</h2>", esc(&g("title")))); }
15    out.push_str(&format!("<div class=\"meta\"><b>@{}</b> · <time>{}</time></div>", esc(&g("nick")), esc(&g("posted_at"))));
16    out.push_str(&format!("<div class=\"body\">{}</div>", esc(&g("content")).replace('\n', "<br>")));
17    for a in v["attachments"].as_array().into_iter().flatten() {
18        let p = esc(a["local_path"].as_str().unwrap_or_default());
19        let m = a["mime_type"].as_str().unwrap_or_default();
20        if a["kind"] == "thumbnail" { continue; }
21        out.push_str("<div class=\"att\">");
22        if m.starts_with("image/") { out.push_str(&format!("<img src=\"{p}\" loading=\"lazy\">")); }
23        else if m.starts_with("video/") { out.push_str(&format!("<video src=\"{p}\" controls preload=\"metadata\"></video>")); }
24        else if m.starts_with("audio/") { out.push_str(&format!("<audio src=\"{p}\" controls></audio>")); }
25        else { out.push_str(&format!("<a href=\"{p}\">{p}</a>")); }
26        out.push_str("</div>");
27    }
28    for c in v["children"].as_array().into_iter().flatten() { html_post(c, out, depth + 1); }
29    out.push_str("</article>");
30}

32/// Render an export document as one self-contained HTML page, embedding the JSON for re-import.
33pub fn build_html(doc: &Value) -> String {
34    let title = esc(doc["title"].as_str().or(doc["nick"].as_str()).unwrap_or_default());
35    let nick = title.clone();
36    let mut body = String::new();
37    for pr in doc["profiles"].as_array().into_iter().flatten() {
38        let dn = pr["display_name"].as_str().unwrap_or("");
39        let un = pr["nick"].as_str().unwrap_or("");
40        let bio = pr["bio"].as_str().unwrap_or("");
41        let av = pr["avatar_path"].as_str().or(pr["avatar_url"].as_str()).unwrap_or("");
42        body.push_str("<header class=\"profile\">");
43        if !av.is_empty() { body.push_str(&format!("<img class=\"avatar\" src=\"{}\" alt=\"\">", esc(av))); }
44        body.push_str(&format!("<div><h2>{} <span class=\"meta\">@{}</span></h2>", esc(dn), esc(un)));
45        if !bio.is_empty() { body.push_str(&format!("<p>{}</p>", esc(bio))); }
46        let bits: Vec<String> = ["location","role","joined_at"].iter().filter_map(|k| pr[*k].as_str().filter(|s| !s.is_empty()).map(|s| format!("{k}: {}", esc(s)))).collect();
47        if !bits.is_empty() { body.push_str(&format!("<p class=\"meta\">{}</p>", bits.join(" · "))); }
48        body.push_str("</div></header>");
49    }
50    body.push_str("<h2>Posts</h2>");
51    for p in doc["posts"].as_array().into_iter().flatten() { html_post(p, &mut body, 0); }
52    let mentions = doc["mentions"].as_array().cloned().unwrap_or_default();
53    if !mentions.is_empty() {
54        body.push_str("<h2>Mentions</h2><ul class=\"mentions\">");
55        for h in &mentions {
56            body.push_str(&format!(
57                "<li><a href=\"#p-{}\">{}</a> → @{} <span class=\"meta\">({})</span></li>",
58                esc(h["post_id"].as_str().unwrap_or("")),
59                esc(h["post_id"].as_str().unwrap_or("")),
60                esc(h["mentioned_nick"].as_str().unwrap_or("")),
61                esc(h["match_type"].as_str().unwrap_or("")),
62            ));
63        }
64        body.push_str("</ul>");
65    }
66    format!(r#"<!DOCTYPE html><html lang="en"><head><meta charset="utf-8">
67<meta name="viewport" content="width=device-width,initial-scale=1"><title>{nick}</title>
68<style>
69body{{font:16px/1.5 system-ui,sans-serif;max-width:800px;margin:2em auto;padding:0 1em;color:#18181b;background:#fafafa}}
70.post{{background:#fff;border:1px solid #e4e4e7;border-radius:8px;padding:.8em 1em;margin:.8em 0}}
71.post .post{{margin-left:1.2em;border-left:3px solid #dc2626;border-radius:0 8px 8px 0}}
72h2{{margin:.2em 0;font-size:1.25em}} .meta{{color:#71717a;font-size:.85em}} .body{{margin:.5em 0;white-space:normal}}
73.att img,.att video{{max-width:100%;border-radius:6px}} .att{{margin:.4em 0}}
74.profile{{display:flex;gap:1em;align-items:flex-start;background:#fff;border:1px solid #e4e4e7;border-radius:8px;padding:1em;margin:1em 0}}
75.profile .avatar{{width:72px;height:72px;border-radius:50%;object-fit:cover}}
76.mentions{{padding-left:1.2em}}
77</style><script type="application/json" id="{data_id}">{data}</script></head><body><h1>{nick}</h1><p class="meta">Exported {}</p>{body}</body></html>"#,
78        esc(doc["exported_at"].as_str().unwrap_or_default()),
79        data_id = HTML_DATA_ID,
80        data = doc.to_string().replace('<', "\\u003c"))
81}
