1//! Export document (format version 2) as JSON.
2
3use crate::prelude::*;
4
5/// `rel`: map attachment url -> path written into the export.
6pub fn build_json(nick: &str, posts: &[Post], atts: &[Attachment], profiles: &[Profile], hits: &[MentionHit], rel: &dyn Fn(&Attachment) -> String) -> Value {
7    let mut by_post: HashMap<&str, Vec<Value>> = HashMap::new();
8    for a in atts {
9        by_post.entry(a.post_id.as_str()).or_default().push(json!({
10            "url": a.url, "kind": a.kind, "mime_type": a.mime_type, "local_path": rel(a),
11        }));
12    }
13    let ids: HashSet<&str> = posts.iter().map(|p| p.post_id.as_str()).collect();
14    let mut kids: HashMap<&str, Vec<&Post>> = HashMap::new();
15    let mut roots = Vec::new();
16    for p in posts {
17        if p.parent_post_id.is_empty() || !ids.contains(p.parent_post_id.as_str()) {
18            roots.push(p);
19        } else {
20            kids.entry(p.parent_post_id.as_str()).or_default().push(p);
21        }
22    }
23    fn node(p: &Post, kids: &HashMap<&str, Vec<&Post>>, by: &HashMap<&str, Vec<Value>>) -> Value {
24        let mut ch: Vec<&Post> = kids.get(p.post_id.as_str()).cloned().unwrap_or_default();
25        ch.sort_by(|a, b| a.posted_at.cmp(&b.posted_at));
26        json!({
27            "post_id": p.post_id, "parent_post_id": p.parent_post_id, "nick": p.nick,
28            "title": p.title, "content": p.content, "posted_at": p.posted_at,
29            "attachments": by.get(p.post_id.as_str()).cloned().unwrap_or_default(),
30            "children": ch.iter().map(|c| node(c, kids, by)).collect::<Vec<_>>(),
31        })
32    }
33    roots.sort_by(|a, b| b.posted_at.cmp(&a.posted_at));
34    let profiles_v: Vec<Value> = profiles.iter().map(|p| json!({
35        "nick": p.nick, "user_id": p.user_id, "display_name": p.display_name, "bio": p.bio,
36        "avatar_url": p.avatar_url, "avatar_path": p.avatar_path,
37        "banner_url": p.banner_url, "banner_path": p.banner_path,
38        "location": p.location, "role": p.role, "joined_at": p.joined_at,
39        "followers": p.followers, "following": p.following, "posts_count": p.posts_count,
40        "scraped_at": p.scraped_at,
41    })).collect();
42    let mentions_v: Vec<Value> = hits.iter().map(|h| json!({
43        "post_id": h.post_id, "mentioned_nick": h.mentioned_nick, "match_type": h.match_type.as_str(),
44    })).collect();
45    json!({
46        "nick": nick,
47        "format": EXPORT_FORMAT,
48        "format_version": EXPORT_FORMAT_VERSION,
49        "exported_at": chrono::Local::now().to_rfc3339(),
50        "profiles": profiles_v,
51        "mentions": mentions_v,
52        "posts": roots.iter().map(|p| node(p, &kids, &by_post)).collect::<Vec<_>>(),
53    })
54}
