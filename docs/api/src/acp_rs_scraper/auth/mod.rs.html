8/// Look for a redirect target anywhere in a LiveView reply.
9pub fn find_redirect(v: &Value) -> Option<String> {
10    for k in ["redirect", "live_redirect", "navigate"] {
11        if let Some(to) = v.pointer(&format!("/response/{k}/to")).and_then(Value::as_str) {
12            return Some(to.into());
13        }
14    }
15    None
16}

18/// Discover the code form's event / field / cid from the rendered LiveView
19/// (falls back to the raw reply text, then to constants).
20pub fn detect_code_form(html: &str, reply: &Value) -> (String, String, Option<u64>) {
21    let raw = format!("{html}{}", reply.to_string().replace("\\\"", "\""));
22    let ev = regex::Regex::new(r#"phx-submit="([^"]+)""#).unwrap()
23        .captures_iter(&raw).map(|c| c[1].to_string())
24        .find(|e| e != EMAIL_EVENT)
25        .unwrap_or_else(|| CODE_EVENT_FALLBACK.into());
26    let doc = Html::parse_fragment(html);
27    let field = doc.select(&Selector::parse("form input").unwrap())
28        .filter(|i| !matches!(i.value().attr("type"), Some("hidden" | "email")))
29        .filter_map(|i| i.value().attr("name").map(str::to_string))
30        .find(|n| n != EMAIL_FIELD && !n.starts_with('_'))
31        .unwrap_or_else(|| CODE_FIELD_FALLBACK.into());
32    let cid = Selector::parse(&format!("form[phx-submit='{ev}']")).ok()
33        .and_then(|sel| doc.select(&sel).next()
34            .and_then(|f| f.value().attr("phx-target")?.parse().ok()))
35        .or(Some(EMAIL_FORM_CID));
36    (ev, field, cid)
37}

39/// `phx-trigger-action`: LiveView asks the client to do a real HTTP POST of
40/// the form (action="/auth/user/otp/sign_in") so the server can set the
41/// authenticated session cookie.
42pub async fn trigger_action_post(
43    http: &reqwest::Client, html: &str, csrf: &str, email: &str, code_field: &str, code: &str,
44) -> Result<()> {
45    let doc = Html::parse_fragment(html);
46    let form = doc.select(&Selector::parse("form[phx-trigger-action]").unwrap()).next()
47        .ok_or_else(|| anyhow!("no redirect and no phx-trigger-action form after code"))?;
48    let action = form.value().attr("action").unwrap_or(OTP_SIGN_IN_FALLBACK);
49    let mut fields: Vec<(String, String)> = Vec::new();
50    for i in form.select(&Selector::parse("input[name]").unwrap()) {
51        let name = i.value().attr("name").unwrap_or_default().to_string();
52        let mut val = i.value().attr("value").unwrap_or_default().to_string();
53        if val.is_empty() {
54            if name == EMAIL_FIELD { val = email.into(); }
55            else if name == code_field { val = code.into(); }
56        }
57        fields.push((name, val));
58    }
59    if !fields.iter().any(|(n, _)| n == "_csrf_token") {
60        fields.push(("_csrf_token".into(), csrf.into()));
61    }
62    let names: Vec<&str> = fields.iter().map(|(n, _)| n.as_str()).collect();
63    verbose!("POST {action} with fields {names:?}");
64    let resp = http.post(Url::parse(base_url())?.join(action)?)
65        .header("Origin", base_url())
66        .header("Referer", format!("{}/auth", base_url()))
67        .header("Sec-Fetch-Mode", "navigate")
68        .form(&fields)
69        .send().await?;
70    if !resp.status().is_success() {
71        bail!("sign-in POST failed: {}", resp.status());
72    }
73    Ok(())
74}

76/// Log in: open `/rs` (redirects to `/auth` when logged out), submit the saved email over the LiveView, prompt for the one-time code (or login link), submit it and finish with the sign-in POST. Fails if `/rs` still redirects to `/auth`.
77pub async fn login(http: &reqwest::Client, jar: &Arc<Jar>) -> Result<()> {
78    let resp = http.get(start_url()).send().await?;
79    let url = resp.url().to_string();
80    let html = resp.text().await?;
81    if !url.contains("/auth") {
82        info!("Already logged in.");
83        return Ok(());
84    }
85    let page = parse_lv(&url, &html)?;
86    let email = load_email()?;
87    let mut lv = Lv::connect(&page, jar).await?;
88    let r = lv.form(EMAIL_EVENT, EMAIL_FIELD, &email, Some(EMAIL_FORM_CID)).await?;
89    if r["status"] != "ok" {
90        bail!("email submit failed: {r}");
91    }
92    info!("Login email requested for {email}.");
93    let code = prompt("Verification code (or paste the login link): ")?;
94    if code.starts_with("http") {
95        http.get(&code).send().await?.error_for_status()?;
96    } else {
97        let (ev, field, cid) = detect_code_form(&lv.html(), &r);
98        let r2 = lv.form(&ev, &field, &code, cid).await?;
99        if let Some(to) = find_redirect(&r2) {
100            http.get(Url::parse(base_url())?.join(&to)?).send().await?.error_for_status()?;
101        } else {
102            trigger_action_post(http, &lv.html(), &page.csrf, &email, &field, &code).await?;
103        }
104    }
105    let check = http.get(start_url()).send().await?;
106    if check.url().path().starts_with("/auth") {
107        bail!("still not logged in after verification");
108    }
109    info!("Logged in.");
110    Ok(())
111}
